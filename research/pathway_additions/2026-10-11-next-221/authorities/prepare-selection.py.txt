def save(name, data):
    with (ROOT / name).open('x') as stream:
        json.dump(data, stream, indent=2, ensure_ascii=False)
        stream.write('\n')


def capture(path, expected=None):
    path = Path(path).absolute()
    data = path.read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    if expected is not None and digest != expected:
        raise ValueError(f'Input hash mismatch: {path}')
    row = {'path': str(path), 'resolved_path': str(path.resolve(strict=True)),
           'sha256': digest, 'bytes': len(data)}
    if str(path) in bindings and bindings[str(path)] != row:
        raise ValueError(f'Input changed during capture: {path}')
    bindings[str(path)] = row
    return data


def read(path, expected=None):
    return json.loads(capture(path, expected))


def recheck():
    for row in list(bindings.values()):
        path = Path(row['path'])
        if str(path.resolve(strict=True)) != row['resolved_path']:
            raise ValueError('Input path changed: ' + str(path))
        if hashlib.sha256(path.read_bytes()).hexdigest() != row['sha256']:
            raise ValueError('Input bytes changed: ' + str(path))

def nodes(record):
    return [record, *(n for key in ['taxa', 'participants', 'reactions', 'gene_clusters']
                      for n in record.get(key, []))]

for prefix, filename in [('GO', 'go'), ('CHEBI', 'chebi'), ('NCBITaxon', 'ncbitaxon'), ('RHEA', 'rhea')]:
    path = CACHE / (filename + '.db')
    resolved = path.resolve(strict=True)
    keys = [k for k in ontology['sources'] if k.startswith(prefix.lower()+'-') and k != 'rhea-rdf-142']
    assert len(keys) == 1
    owner = keys[0]
    metadata = ontology['sources'][owner]
    stat = resolved.stat()
    lock = {'path': str(path), 'resolved_path': str(resolved), 'source_key': owner,
            'metadata': metadata, 'size': stat.st_size, 'mtime_ns': stat.st_mtime_ns,
            'digest_status': 'Pinned expected whole-database hash; no fresh whole-database streaming in this preparation'}
    database_locks.append(lock)
    with sqlite3.connect(resolved.as_uri()+'?mode=ro', uri=True) as connection:
        versions = list(connection.execute("SELECT subject,object FROM statements WHERE predicate='owl:versionIRI'"))
        assert len(versions) == 1
        details = list(connection.execute("SELECT predicate,object,value FROM statements WHERE subject=? AND predicate IN ('owl:versionInfo','dcterms:license')", (versions[0][0],)))
        assert any(p == 'owl:versionInfo' and (o or v) == metadata['version'] for p,o,v in details)
        terms, missing = extract_terms(connection, sorted(i for i in ont_ids if i.startswith(prefix+':')), owner)
        assert not missing
        lock['queried_metadata'] = {'version_iri_rows': versions, 'release_license_rows': details}
        sql_terms.update(terms)

    expected = proposal['terms'][identifier]

        reuse.append({'id': identifier, 'existing_term_preserved': existing,

    original = by_key[expected['source']]
    raw = Path(original['path'])

    proof = [(p, index, r) for p, index, r in acquisitions

    if identifier.startswith('MetaCyc:'):

        pgdb = xml.find('metadata/PGDB')
        protein = xml.find('Protein')
        assert pgdb.attrib == {'orgid': 'META', 'version': '27.0'}
        assert protein.attrib['frameid'] == identifier.split(':')[1]
        entry['kind'], entry['version'] = 'metacyc-protein-xml', pgdb.attrib['version']
        entry['path'] = 'ethylmalonyl-' + raw.stem + '-protein-27.0-2026-10-11.xml'
        source_facts = {'top_level_attributes': protein.attrib, 'pgdb_attributes': pgdb.attrib,
                        'direct_common_name': protein.findtext('common-name'),
