"""Write native append-only history for the actual PWY-1361 integration."""
from pathlib import Path
import datetime
import hashlib
import importlib.util
import json
import subprocess
import sys

W = Path('/private/tmp/pathwaymech-next-265-20261015')
C = Path('/private/tmp/pathwaymech-next-265-ci')
OUT = C / 'integration-history-scaffolds.json'
assert not OUT.exists(), 'Inspect an existing partial receipt; never repeat history implicitly.'
integration = json.loads((C / 'record-and-authority-integration.json').read_bytes())
assert integration['maintained_integration_performed']
assert hashlib.sha256((W / 'data/pathways/benzoyl-coa-degradation-i-aerobic.yaml').read_bytes()).hexdigest() == integration['record_sha256']
spec = importlib.util.spec_from_file_location('gate_environment', C / 'run_initial_checks.py')
gates = importlib.util.module_from_spec(spec)
spec.loader.exec_module(gates)

jobs = [
    [
        '--kind', 'record', '--slug', 'benzoyl-coa-degradation-i-aerobic',
        '--target-root', 'data/pathways', '--event', 'CREATE', '--outcome', 'changed',
        '--summary', 'Add the complete seven-step aerobic benzoyl-CoA degradation pathway',
        '--details',
        'Integrate the independently accepted exact PWY-1361 draft: seven reactions, '
        '35 participants, four taxon contexts, 67 edges and 68 references. Preserve '
        'all 261 prior records. Retain the putative fourth step without an assigned catalyst, original '
        'experimental preparations, lower-route catalyst exclusions, generic cofactor '
        'scope, CIB regulatory context, and explicit full-original access gaps. '
        'Apply the independently accepted category and evidence-locator corrections '
        'for issues 577-580 without rewriting frozen original findings. The actual '
        'merged-baseline duplicate assessment and exact authority union are bound in '
        'research/pathway_additions/2026-10-15-next-265/. Production rendering, full '
        'repository gates and canonical maintained-record reviews follow this '
        'integration; this event does not claim those pending results.',
    ],
    [
        '--kind', 'mapping', '--slug', 'benzoyl-coa-selected-authorities',
        '--path', 'conf/identifier_sources.json', '--event', 'EDIT', '--outcome', 'changed',
        '--summary', 'Add the independently verified selected PWY-1361 identifier authorities',
        '--details',
        'Copy the three exact accepted owner proposals in conf/identifier_sources.json '
        'and data/identifier_authorities/{sources,ontology}.json. Add 17 source terms, '
        '15 ontology terms and 16 source owners/manifest rows; preserve every prior '
        'term, owner, and the ordered 1306-row manifest prefix. The independently '
        'verified exact final YAML passes the full 262-record identifier consumer. '
        'Selected source and SQL provenance remain distinct from scientific evidence; '
        'no policy/parser change, historical missing-cache repair, or full historical '
        'rebuild is claimed. Production corpus gates remain pending.',
    ],
    [
        '--kind', 'report', '--slug', 'next265-benzoyl-coa-research',
        '--path', 'research/pathway_additions/2026-10-15-next-265/README.md',
        '--event', 'CREATE', '--outcome', 'changed',
        '--summary', 'Retain pathway research, original review findings and explicit corrections',
        '--details',
        'Create the PWY-1361 research report and identifier/reference register. Retain '
        'the exact source packages through lossless maps and separately retain '
        'independent assessments and correction successors. The initial 24-package '
        'scope plus 19 snapshots reconstructs all 9629 source occurrences from the '
        'actual maintained storage. Issue 581 corrects compact audit coverage to '
        '5415 pairs; issue 582 distinguishes eight KGX owners from nine combined '
        'HTML/KGX owners. Preserve original findings, failed attempts, access gaps '
        'and finite search boundaries. Later verification and publication results '
        'must be appended from actual execution, not inferred from this event.',
    ],
]

receipts = []
for arguments in jobs:
    command = [sys.executable, '-B', 'scripts/new_history_record.py',
               '--actor-name', 'Codex', '--actor-type', 'ai_agent', '--agent-tool', 'codex',
               *arguments]
    for number in range(577, 583):
        command += ['--issue', str(number)]
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    result = subprocess.run(command, cwd=W, env=gates.ENV, capture_output=True, text=True)
    receipt = {'command': command, 'started_utc': started, 'exit_code': result.returncode,
               'stdout': result.stdout, 'stderr': result.stderr}
    if result.returncode == 0:
        path = Path(result.stdout.strip().splitlines()[-1])
        if not path.is_absolute():
            path = W / path
        assert path.is_relative_to(W / 'history') and path.is_file()
        receipt.update({'path': str(path.relative_to(W)),
                        'sha256': hashlib.sha256(path.read_bytes()).hexdigest()})
    receipts.append(receipt)
    OUT.write_text(json.dumps(receipts, indent=2) + '\n')
    print(json.dumps(receipt), flush=True)
    assert result.returncode == 0
