"""Adopt only the independently accepted issue583 path/hash successor."""

import argparse
import datetime
import hashlib
import json
import os
import subprocess
from pathlib import Path

WORKTREE = Path('/private/tmp/pathwaymech-next-265-20261015')
SOURCE = Path('/private/tmp/pathwaymech-next-265-helper-formatting-review')
CI = Path('/private/tmp/pathwaymech-next-265-ci')
REPORT = WORKTREE / 'research/pathway_additions/2026-10-15-next-265'
SOURCE_MANIFEST = 'af184586b5df4dba49f9962d3c7c542301f70efbbcd08a282cb08d3105d4a319'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def safe(root, relative):
    name = Path(relative)
    assert not name.is_absolute() and '..' not in name.parts
    path = root / name
    assert path.resolve().is_relative_to(root.resolve())
    assert not path.is_symlink()
    return path


def verify_inventory(root, manifest):
    rows = json.loads(manifest.read_bytes())['files']
    expected = {row['path'] for row in rows}
    assert len(expected) == len(rows)
    paths = list(root.rglob('*'))
    assert not any(path.is_symlink() for path in paths)
    assert {p.relative_to(root).as_posix() for p in paths if p.is_file()} == (
        expected | {'artifact-manifest.json'}
    )
    for row in rows:
        path = safe(root, row['path'])
        assert sha(path) == row['sha256'] and path.stat().st_size == row['bytes']
    return rows


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--execute', action='store_true')
    args = parser.parse_args()
    suffix = 'integration' if args.execute else 'preflight'
    receipt = CI / f'helper-formatting-{suffix}.json'
    assert not receipt.exists()
    assert sha(SOURCE / 'artifact-manifest.json') == SOURCE_MANIFEST
    verify_inventory(SOURCE, SOURCE / 'artifact-manifest.json')
    proposal = json.loads((SOURCE / 'proposed-adoption.json').read_bytes())
    rows = proposal['exact_write_set']
    assert len(rows) == 20 and len({row['destination'] for row in rows}) == 20
    assert subprocess.check_output(
        ['git', 'rev-parse', 'HEAD'], cwd=WORKTREE, text=True
    ).strip() == '39b84c79eb9997fb6785ef040f571bad498846e7'
    protected = [WORKTREE / 'data/pathways/benzoyl-coa-degradation-i-aerobic.yaml',
                 WORKTREE / 'data/identifier_authorities/sources.json',
                 WORKTREE / 'data/identifier_authorities/ontology.json',
                 WORKTREE / 'conf/identifier_sources.json', WORKTREE / 'pyproject.toml']
    protected_before = {str(path): sha(path) for path in protected}
    for package in proposal['packages']:
        root = REPORT / package['package']
        manifest = root / 'artifact-manifest.json'
        assert sha(manifest) == package['original_manifest_sha256']
        verify_inventory(root, manifest)
    for row in rows:
        target = safe(WORKTREE, row['destination'])
        assert target.is_relative_to(REPORT / 'evidence-retention') or (
            target.is_relative_to(REPORT / 'evidence-extension')
        )
        source = Path(row['external_source'])
        assert source.is_relative_to(SOURCE / 'proposed') and not source.is_symlink()
        assert sha(source) == row['new_sha256'] and source.stat().st_size == row['new_bytes']
        if row['operation'] == 'create_new_no_overwrite':
            assert not target.exists()
        else:
            assert row['operation'] == 'replace_exact_pinned_current_bytes'
            assert sha(target) == row['expected_current_sha256']
    acceptance_digest = None
    if args.execute:
        acceptance_path = CI / 'accepted-helper-formatting.json'
        acceptance = json.loads(acceptance_path.read_bytes())
        assert acceptance['status'] == 'accepted_for_exact_helper_successor'
        assert acceptance['source_manifest_sha256'] == SOURCE_MANIFEST
        assert sha(Path(acceptance['review_manifest'])) == acceptance['review_manifest_sha256']
        assert sha(Path(acceptance['review_acceptance'])) == acceptance['review_acceptance_sha256']
        acceptance_digest = sha(acceptance_path)
        # Preserve every historical owner before replacing any live byte.
        ordered = [r for r in rows if r['operation'] == 'create_new_no_overwrite']
        ordered += [r for r in rows if r['operation'] != 'create_new_no_overwrite'
                    and not r['destination'].endswith('/artifact-manifest.json')]
        ordered += [r for r in rows if r['operation'] != 'create_new_no_overwrite'
                    and r['destination'].endswith('/artifact-manifest.json')]
        for row in ordered:
            target = safe(WORKTREE, row['destination'])
            body = Path(row['external_source']).read_bytes()
            target.parent.mkdir(parents=True, exist_ok=True)
            if row['operation'] == 'create_new_no_overwrite':
                with target.open('xb') as stream:
                    stream.write(body)
            else:
                assert sha(target) == row['expected_current_sha256']
                temporary = target.with_name('.' + target.name + '.owned-formatting-staging')
                with temporary.open('xb') as stream:
                    stream.write(body)
                os.replace(temporary, target)
            assert sha(target) == row['new_sha256']
        for package in proposal['packages']:
            root = REPORT / package['package']
            assert sha(root / 'artifact-manifest.json') == (
                package['prospective_current_manifest_sha256']
            )
            verify_inventory(root, root / 'artifact-manifest.json')
            bridge = json.loads((root / 'helper-formatting/successor.json').read_bytes())
            original_manifest = root / bridge['original_manifest_historical_owner']
            assert sha(original_manifest) == package['original_manifest_sha256']
            original_rows = json.loads(original_manifest.read_bytes())['files']
            changed = {row['original_package_path']: row['historical_owner']
                       for row in bridge['changed_original_rows']}
            for row in original_rows:
                historical = safe(root, changed.get(row['path'], row['path']))
                assert sha(historical) == row['sha256']
                assert historical.stat().st_size == row['bytes']
    assert protected_before == {str(path): sha(path) for path in protected}
    result = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
              'status': 'adopted_and_verified' if args.execute else 'preflight_passed',
              'source_manifest_sha256': SOURCE_MANIFEST,
              'source_adoption_sha256': sha(SOURCE / 'proposed-adoption.json'),
              'root_acceptance_sha256': acceptance_digest, 'exact_write_set': rows,
              'protected_unchanged': protected_before,
              'packages': proposal['packages'],
              'scope': 'Exact bounded formatting/documentation ownership successor only; '
                       'source objects, scientific record and policy unchanged. '
                       'Production checks and publication remain separate.'}
    receipt.write_text(json.dumps(result, indent=2) + '\n')
    print(json.dumps({k: v for k, v in result.items() if k != 'exact_write_set'}, indent=2))


if __name__ == '__main__':
    main()
