            subject = el.get('{'+RDF+'}about')
            if subject in needed:
                values = triples.setdefault(subject, defaultdict(list))
                for child in el:
                    pred = ET.QName(child).localname
                    val = child.get('{'+RDF+'}resource') or child.text
                    if val not in values[pred]: values[pred].append(val)
                    if pred in {'substrates','products','compound'} or re.fullmatch(r'contains\d+',pred):
                        subjects.add(val)
            el.clear()
            while el.getprevious() is not None: del el.getparent()[0]

db = sqlite3.connect('file:'+str(C/'chebi.db')+'?mode=ro', uri=True)
def chemical(cid):
    rows = db.execute('select predicate,value from statements where subject=? and value is not null',(cid,)).fetchall()
    return {k:v for k,v in rows}
def one(subject, prop):
    values = triples[subject][prop]
    assert len(values)==1, (subject,prop,values)
    return values[0]
def side(subject):
    out=[]
    for pred,values in triples[subject].items():
        if not re.fullmatch(r'contains\d+',pred): continue
        for participant in values:
            cu = one(participant,'compound')
            cid = 'CHEBI:'+re.search(r'CHEBI[_:](\d+)',one(cu,'chebi'))[1]
            c = chemical(cid)
            out.append({'id':cid,'coefficient':int(pred[8:]),'side':subject,'participant':participant,'compound':cu,'formula':c['chemrof:generalized_empirical_formula'],'charge':int(c['chemrof:charge'])})
    return out
