"""Independent role-aware assignment from original native and ChEBI InChIs."""
from pathlib import Path
from datetime import datetime, timezone
import json
from rdkit import Chem, rdBase

O=Path(__file__).resolve().parent
results=[]
for row in json.loads((O/'stereo-inputs.json').read_text()):
    values=[]
    for scope,key in [('native','native_inchi'),('current','current_inchi')]:
        molecule=Chem.MolFromInchi(row[key]);assert molecule is not None
        Chem.AssignStereochemistry(molecule,force=True,cleanIt=True)
        centers=dict(Chem.FindMolChiralCenters(molecule,includeUnassigned=True))
        ornithine=[i for i in centers if any(a.GetSymbol()=='N' for a in molecule.GetAtomWithIdx(i).GetNeighbors())]
        assert len(ornithine)==1
        o=ornithine[0];citryl=[]
        for i,cip in centers.items():
            if i==o:continue
            assert any(a.GetSymbol()=='O' for a in molecule.GetAtomWithIdx(i).GetNeighbors())
            length=len(Chem.GetShortestPath(molecule,o,i))-1
            assert length in (4,7)
            citryl.append({'role':'N2-citryl' if length==4 else 'N5-citryl','cip':cip,'bond_distance_to_ornithine_alpha':length})
        values.append({'scope':scope,'ornithine_alpha':centers[o],'citryl_centers':sorted(citryl,key=lambda r:r['role'])})
    results.append({'native_id':row['native'],'current_id':row['current'],'structures':values})
assert results[0]['structures'][0]['ornithine_alpha']=='S'
assert results[1]['structures'][0]['ornithine_alpha']=='R'
assert results[2]['structures'][0]['citryl_centers']==[{'role':'N5-citryl','cip':'?','bond_distance_to_ornithine_alpha':7}]
assert results[2]['structures'][1]['citryl_centers']==[{'role':'N5-citryl','cip':'S','bond_distance_to_ornithine_alpha':7}]
assert [x['cip'] for x in results[3]['structures'][0]['citryl_centers']]==['?','?']
assert [x['cip'] for x in results[3]['structures'][1]['citryl_centers']]==['R','S']
result={'status':'passed_independent_structure_interpretation','at':datetime.now(timezone.utc).isoformat(),'rdkit':rdBase.rdkitVersion,'results':results,'primary_visual_observation':'Original PMC2856993 Figure2A shows N2-citryl R, ornithine alpha R, N5-citryl S. Panels B/C are receptor-bound ferric ligand density, not free-ion or intermediate assay.','limit':'Assignments use atom connectivity and CIP rules, not arbitrary InChI sign interpretation.'}
(O/'stereo.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result))
