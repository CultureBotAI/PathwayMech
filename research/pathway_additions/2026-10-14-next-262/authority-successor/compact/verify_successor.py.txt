def sha(p):
 h=hashlib.sha256()
 with Path(p).open('rb')as stream:
  for block in iter(lambda:stream.read(1024*1024),b''):h.update(block)
 return h.hexdigest()
def load(p):return json.loads(Path(p).read_text())

 return {'path':str(p),'sha256':sha(p),'bytes':p.stat().st_size}
def bound(p):return {'path':str(p),'sha256':sha(p),'bytes':p.stat().st_size}

def flatten(value,path=()):
 """Independent complete tree projection, including structural markers."""
 rows={path+('@type',):type(value).__name__}
 if isinstance(value,dict):
  rows[path+('@keys',)]=tuple(sorted(value))
  for k,v in value.items():rows.update(flatten(v,path+(k,)))
 elif isinstance(value,list):
  rows[path+('@length',)]=len(value)
  for i,v in enumerate(value):rows.update(flatten(v,path+(i,)))
 else:rows[path+('@value',)]=value
 return rows
