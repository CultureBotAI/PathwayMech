def sha(data):
    return hashlib.sha256(data).hexdigest()


def load(path):
    return json.loads(path.read_text())


def verify_package(root, expected, count):
    raw = (root / 'artifact-manifest.json').read_bytes()
    assert sha(raw) == expected
    rows = json.loads(raw)['files']
    assert len(rows) == count
    index = {}
    for row in rows:
        path = root / row['path']
        assert path.resolve().is_relative_to(root)
        data = path.read_bytes()
        assert len(data) == row['bytes'] and sha(data) == row['sha256'], path
        assert row['path'] not in index
        index[row['path']] = row
    actual = {str(p.relative_to(root)) for p in root.rglob('*') if p.is_file()}
    assert actual == set(index) | {'artifact-manifest.json'}
    return index

# Each structure DOI is declared in its complete issuing entry. The reference
# digest is explicitly the entry digest, never a digest of all associated files.
structure_bindings = []
source_indexes = {}
for root in [BIO, PRIMARY]:
    source_indexes[root] = {row['path']: row for row in load(root / 'artifact-manifest.json')['files']}


def bind_original(root, relative, role, reference_id=None):
    row = source_indexes[root][relative]
    path = root / relative
    data = path.read_bytes()
    assert len(data) == row['bytes'] and sha(data) == row['sha256']
    binding = {'path': str(path), 'sha256': sha(data), 'bytes': len(data), 'role': role}
    if reference_id:
        binding['reference_id'] = reference_id
    structure_bindings.append(binding)
    return data


structure_specs = [
    ('8IH6', BIO, 'originals/structure-lead/RCSB-8IH6-entry.json',
     ['originals/exact-follow/8IH6-polymer1.json', 'originals/exact-follow/8IH6-polymer2.json',
      'originals/exact-follow/8IH6-assembly1.json', 'originals/exact-follow/8IH6.cif']),
    ('1BJP', PRIMARY, 'originals/RCSB-1BJP-entry.json',

structure_refs = {}
for entry_id, root, entry_path, companions in structure_specs:
    data = bind_original(root, entry_path, 'complete issuing entry')
    entry = json.loads(data)
    assert entry['rcsb_id'] == entry_id
    doi_rows = [x for x in entry['database_2'] if x.get('database_id') == 'PDB' and x.get('database_code') == entry_id]
    assert len(doi_rows) == 1
    doi = doi_rows[0]['pdbx_DOI']
    assert doi == '10.2210/pdb' + entry_id.lower() + '/pdb'
    reference_id = 'DOI:' + doi
    structure_bindings[-1]['reference_id'] = reference_id
    for companion in companions:
        bind_original(root, companion, 'associated complete polymer/assembly/mmCIF original', reference_id)
    version = entry['rcsb_accession_info']
    reference = {'id': reference_id, 'title': 'RCSB PDB ' + entry_id + ' complete issuing entry and associated structure context',
                 'url': 'https://data.rcsb.org/rest/v1/core/entry/' + entry_id,
                 'source_version': 'Entry revision ' + str(version['major_revision']) + '.' + str(version['minor_revision']) +
                    '; source_sha256 hashes entry JSON only. Exact associated polymer, assembly and mmCIF originals are separately bound in the source dossier.',
                 'source_sha256': sha(data)}
    assert reference_id not in refs
    refs[reference_id] = reference
    structure_refs[entry_id] = reference_id
