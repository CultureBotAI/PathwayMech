def sha(data):
    return hashlib.sha256(data).hexdigest()

def bind(path):
    return {'path': str(path), 'sha256': sha(path.read_bytes()), 'bytes': path.stat().st_size}

def dump(path, payload):
    assert not path.exists(), path
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + '\n')

def difference(a, b, path=''):
    assert type(a) is type(b)
    if isinstance(a, dict):
        assert a.keys() == b.keys()
        return [item for k in a for item in difference(a[k], b[k], path + ('.' if path else '') + k)]
    if isinstance(a, list):
        assert len(a) == len(b)
        return [item for i, (x, y) in enumerate(zip(a, b)) for item in difference(x, y, f'{path}[{i}]')]
    return [] if a == b else [{'field_path': path, 'old': a, 'replacement': b}]
