def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as stream:
        for part in iter(lambda:stream.read(8*1024*1024),b''): h.update(part)
    return h.hexdigest()
def row(path):
    path=Path(path);return {'path':str(path),'bytes':path.stat().st_size,'sha256':sha(path)}
def check(item):
    path=Path(item['path']);assert path.is_file(),path
    if 'bytes' in item: assert path.stat().st_size==item['bytes'],path
    assert sha(path)==item['sha256'],path
def load(path): return json.loads(Path(path).read_bytes())
def write(path,value):
    with Path(path).open('x') as stream: json.dump(value,stream,indent=2,ensure_ascii=False);stream.write('\n')
def bindings(value):
    if isinstance(value,dict):
        if {'path','sha256'}<=set(value):yield value
        for item in value.values():yield from bindings(item)
    elif isinstance(value,list):
        for item in value:yield from bindings(item)

def execute(name,command,commands):
    started=now();tick=time.monotonic()
    with (OUT/(name+'.stdout')).open('xb') as stdout,(OUT/(name+'.stderr')).open('xb') as stderr:
        result=subprocess.run(command,cwd=commands['cwd'],env={**os.environ,**commands['environment']},stdout=stdout,stderr=stderr)
    receipt={'name':name,'command':command,'cwd':commands['cwd'],'environment':commands['environment'],'started_at':started,'finished_at':now(),'duration_seconds':round(time.monotonic()-tick,3),'exit_code':result.returncode,'stdout':row(OUT/(name+'.stdout')),'stderr':row(OUT/(name+'.stderr'))}
    write(OUT/(name+'-execution.json'),receipt);print('BUILDER_TERMINAL '+json.dumps(receipt),flush=True);return receipt

def verify(data,executions):
    assert all(r['exit_code']==0 for r in executions)
    results={}
    for kind,expected in [('sources',data['source_additions']),('ontology',data['ontology_additions'])]:
        old=data['old'][kind];new=load(OUT/(kind+'.json'));assert new['version']==old['version']
        for i,value in old['terms'].items():assert new['terms'].get(i)==value,(kind,i)
        for key,value in old['sources'].items():assert new['sources'].get(key)==value,(kind,key)
        added={i:v for i,v in new['terms'].items() if i not in old['terms']};assert added==expected,(kind,'unexpected term content or set')
        new_metadata={k:v for k,v in new['sources'].items() if k not in old['sources']}
        expected_metadata={entry['key']:{k:entry[k] for k in ['url','version','license','sha256']+(['kind'] if entry['kind']=='metacyc-reaction-xml' else [])} for entry in data['additions']} if kind=='sources' else {}
        assert new_metadata==expected_metadata,(kind,'unexpected metadata')
        results[kind]={'old_terms_preserved':len(old['terms']),'old_metadata_preserved':len(old['sources']),'new_terms':added,'new_metadata':new_metadata,'total_terms':len(new['terms']),'total_metadata':len(new['sources']),'output':row(OUT/(kind+'.json'))}
    assert results['sources']['total_terms']==2384 and results['sources']['total_metadata']==545
    assert results['ontology']['total_terms']==1931 and results['ontology']['total_metadata']==5
    assert load(OUT/'manifest.json')[:682]==data['manifest']
    previous=load(OUT/'baseline/rhea-updates.json');updates=load(OUT/'rhea-updates.json');assert updates['ids'][:15]==previous['ids'] and updates['ids'][15:]==['RHEA:25433']
    assert {k:v for k,v in updates.items() if k!='ids'}=={k:v for k,v in previous.items() if k!='ids'}
    for item in load(OUT/'corpus-bindings.json')['files']:check(item)
    for item in data['original_bindings'].values():check(item)
    for item in data['new_originals']:check(item['source']);check(item['external_copy'])
    ontology=load(OUT/'ontology.json')
    for item in data['dbs']:
        p=Path(item['path']);assert p.stat().st_size==item['bytes'] and p.stat().st_mtime_ns==item['mtime_ns'];assert ontology['sources'][item['source_key']]['sha256']==item['sha256']
    check(data['ontology_builder'])
    for item in data['code_inputs']:check(item)
    for item in data['records']:check(item)
    for item in load(OUT/'baseline-capture.json')['authority_bindings']:check(item)
    result={'status':'full_native_ribose_build_passed_exact_adopted226_preservation','at':now(),'target_sha256':TARGET_SHA,'baseline_observed_head':data['head'],'baseline_scope':'Exact adopted226 bytes; commit and merge bridge remain root owned','actual_records':227,'retention_ids':4284,'results':results,'old_manifest_entries_preserved':682,'manifest_entries':698,'old_rhea_ids_preserved':15,'rhea_ids':16,'input_originals_verified':True,'captured_root_record_authority_code_inputs_unchanged':True,'maintained_writes':False,'shared_cache_writes':False,'git_github_actions':False,'source_exhaustion':False}
    write(OUT/'verification.json',result);print('VERIFIED '+json.dumps({k:v for k,v in result.items() if k!='results'}),flush=True)

def main():
    data=intake();commands=prepare(data)
    with ThreadPoolExecutor(max_workers=2) as pool:
        futures=[pool.submit(execute,name,commands[name],commands) for name in ['source_builder','ontology_builder']];executions=[f.result() for f in futures]
    write(OUT/'execution-ledger.json',{'executions':executions});verify(data,executions);print('READY_FOR_SEAL',flush=True)

if __name__=='__main__':
    try:main()
    except Exception:
        traceback.print_exc()
        if OUT.exists():(OUT/('failure-'+str(time.time_ns())+'.txt')).write_text(traceback.format_exc())
        raise
