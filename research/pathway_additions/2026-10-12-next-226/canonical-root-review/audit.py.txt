def read(path):
    return json.loads(Path(path).read_bytes())

def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(8 * 1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()

def bind(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': digest(path)}

def check(item, base=None):
    path = Path(item['path'])
    if not path.is_absolute():
        assert base is not None and '..' not in path.parts
        path = base / path
    assert path.is_file(), path
    actual = bind(path)
    assert actual['sha256'] == item['sha256'], path
    assert 'bytes' not in item or actual['bytes'] == item['bytes'], path
    checked[str(path)] = actual
    return actual

def walk(value):
    if isinstance(value, dict):
        if 'path' in value and 'sha256' in value:
            yield value
        for child in value.values():
            yield from walk(child)
    elif isinstance(value, list):
        for child in value:
            yield from walk(child)

def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

# Independent read-only SQL queries for six new ChEBI terms.
fresh_ontology = {}
connection = sqlite3.connect('file:/private/tmp/pathwaymech-curation-authorities/chebi.db?mode=ro',uri=True)
try:
    for identifier,term in added['ontology'].items():
        if not identifier.startswith('CHEBI:'):
            continue
        rows = list(connection.execute("SELECT predicate,value FROM statements WHERE subject=? AND predicate IN ('rdfs:label','oio:hasExactSynonym')",(identifier,)))
        labels = {v for k,v in rows if k=='rdfs:label' and v}
        assert len(labels)==1
        label = labels.pop()
        result = {'label':label,'synonyms':sorted({v for k,v in rows if k=='oio:hasExactSynonym' and v and v!=label}),'source':'chebi-255'}
        assert result == term
        fresh_ontology[identifier] = result
finally:
    connection.close()

# A fresh streaming scan verifies the single added Rhea assertion and uniqueness.

matches = []
with gzip.open(P/'originals'/rhea['path'],'rb') as f:
    depth = 0
    for event,node in ET.iterparse(f,events=['start','end']):
        if event=='start':
            depth += 1
            if depth==1: root=node
            continue
        if depth==2:
            if node.get(rdf+'about')=='http://rdf.rhea-db.org/25433':
                matches.append(ET.tostring(node,encoding='unicode'))
            root.clear()
        depth -= 1
assert len(matches)==1
