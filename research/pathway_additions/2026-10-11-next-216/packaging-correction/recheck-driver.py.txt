from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import os
import subprocess
import sys

root = Path('/private/tmp/pathwaymech-next-216-20261011')
out = Path('/private/tmp/pathwaymech-next-216-packaging-rechecks-20261011')
out.mkdir(exist_ok=False)
env = dict(os.environ, PYTHONPATH=str(root / 'src'), UV_NO_SYNC='1', UV_CACHE_DIR='/private/tmp/pathwaymech-curation-uv-cache')
receipts = []
for name, cmd in [('render', ['just', 'render-pages']), ('validate', ['just', 'validate'])]:
    log = out / (name + '.log')
    started = datetime.now(timezone.utc).isoformat()
    with log.open('x') as stream:
        result = subprocess.run(cmd, cwd=root, env=env, stdout=stream, stderr=subprocess.STDOUT)
    receipt = {'name': name, 'command': cmd, 'environment': {k: env[k] for k in ['PYTHONPATH', 'UV_NO_SYNC', 'UV_CACHE_DIR']}, 'started_at': started, 'finished_at': datetime.now(timezone.utc).isoformat(), 'exit_code': result.returncode, 'log': str(log), 'log_sha256': hashlib.sha256(log.read_bytes()).hexdigest()}
    receipts.append(receipt)
    (out / 'execution.json').write_text(json.dumps(receipts, indent=2) + '\n')
    print(json.dumps(receipt), flush=True)
    if result.returncode:
        sys.exit(result.returncode)
