executions = []
for args in tasks:
    start = datetime.now(timezone.utc).isoformat()
    process = subprocess.run(common + args, cwd=root, env={**os.environ, 'PYTHONPATH': 'src', 'UV_NO_SYNC': '1', 'UV_CACHE_DIR': '/private/tmp/pathwaymech-curation-uv-cache'}, text=True, capture_output=True)
