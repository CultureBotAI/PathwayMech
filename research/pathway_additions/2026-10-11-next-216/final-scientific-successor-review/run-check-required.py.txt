from pathlib import Path
from datetime import datetime,UTC
import hashlib
import json
import os
import subprocess

R=Path('/private/tmp/pathwaymech-next-216-20261011')
O=Path('/private/tmp/pathwaymech-next-216-packaging-scientific-review')
now=lambda:datetime.now(UTC).isoformat().replace('+00:00','Z')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
cmd=[str(R/'.venv/bin/python'),'scripts/record_review.py','check','--require-reviews','--base','3fb840bf48d871bc37576cc669d23441c0206e66']
started=now()
p=subprocess.run(cmd,cwd=R,env=dict(os.environ,PYTHONPATH=str(R/'src'),UV_NO_SYNC='1',UV_CACHE_DIR='/private/tmp/pathwaymech-curation-uv-cache'),capture_output=True)
(O/'check-required.stdout').write_bytes(p.stdout)
(O/'check-required.stderr').write_bytes(p.stderr)
receipt={'command':cmd,'started_at':started,'finished_at':now(),'exit_code':p.returncode,'stdout_sha256':sha(O/'check-required.stdout'),'stderr_sha256':sha(O/'check-required.stderr')}
(O/'check-required-execution.json').write_text(json.dumps(receipt,indent=2)+'\n')
assert p.returncode==0,p.stderr.decode()
result=json.loads(p.stdout)
assert result['valid_reviews']==89
for item in json.loads((O/'inspect.json').read_text())['source']['inputs']:
    assert sha(R/item['path'])==item['sha256']
for item in json.loads((O/'input-archive-manifest.json').read_text())['inputs']:
    assert sha(Path(item['external_archive']))==item['sha256']
for item in json.loads((O/'save-result.json').read_text())['saved_files']:
    assert sha(R/item['path'])==item['sha256']
    whitespace=subprocess.run(['git','diff','--no-index','--check','/dev/null',str(R/item['path'])],cwd=R,capture_output=True)
    assert whitespace.returncode in (0,1) and not whitespace.stdout and not whitespace.stderr
print(json.dumps({'status':'passed','valid_reviews':result['valid_reviews'],'receipt_sha256':sha(O/'check-required-execution.json'),'saved_pair_whitespace':'passed','current_and_archived_formal_hashes':'passed'}))
