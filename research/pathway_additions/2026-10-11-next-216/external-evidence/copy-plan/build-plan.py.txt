def sha(p):
    with Path(p).open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()


def read(p):
    return json.loads(Path(p).read_text())


def bind(p):
    p = Path(p).absolute()
    return {'path': str(p), 'sha256': sha(p), 'bytes': p.stat().st_size}


def write_new(p, obj):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('x') as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)
        f.write('\n')


def add(p, dest, role, *, unique=False, maintained_helper=False):
    p = Path(p).absolute()
    if str(p) in paths:
        return paths[str(p)]
    b = bind(p)
    if b['sha256'] in canonical_by_hash and not unique:
        entry = dict(b, disposition='exact-byte-alias', destination=canonical_by_hash[b['sha256']], role=role)
        aliases.append(entry)
        paths[str(p)] = entry
        return entry
    dest = Path(dest)
    if not maintained_helper and '.py' in dest.name and not dest.name.endswith('.txt'):
        dest = dest.with_name(dest.name + '.txt')
    mode = 'exact_bytes'
    copy_source = p
    # JSON string escaping preserves inert transcript whitespace without changing
    # its recoverable bytes or making captured output a maintained source file.
    if p.suffix in {'.log', '.diff', '.tsv'} or dest.suffix in {'.log', '.diff', '.tsv'} or (p.suffix == '.txt' and '.py' not in p.name):
        data = p.read_bytes()
        try:
            content = data.decode('utf-8')
            assert content.encode('utf-8') == data
            payload = dict(format='lossless-console-or-artifact-v1', source=b, encoding='utf-8', content=content)
        except UnicodeDecodeError:
            payload = dict(format='lossless-console-or-artifact-v1', source=b, encoding='base64', content=base64.b64encode(data).decode('ascii'))
        copy_source = HERE / 'payload' / (b['sha256'] + '.json')
        if not copy_source.exists():
            write_new(copy_source, payload)
        dest = dest.with_name(dest.name + '.json')
        mode = 'lossless_json_wrapper'
    assert not dest.is_absolute() and '..' not in dest.parts
    if any(x['destination'] == str(dest) for x in copies):
        raise ValueError('destination collision: ' + str(dest))
    entry = dict(b, disposition='planned-copy', destination=str(dest), copy_source=str(copy_source), destination_sha256=sha(copy_source), mode=mode, role=role)
    copies.append(entry)
    canonical_by_hash.setdefault(b['sha256'], str(dest))
    paths[str(p)] = entry
    return entry


def retained(p, reason):
    p = Path(p).absolute()
    if str(p) in paths:
        return
    entry = dict(bind(p), disposition='retained-external', reason=reason)
    external.append(entry)
    paths[str(p)] = entry


def archived(p, group, relative=None, *, unique=False):
    return add(p, DOSSIER / 'external-evidence' / group / (relative or Path(p).name), 'immutable external observation; not a historical canonical review', unique=unique)

# Census uses pathlib/os.walk rather than ignore-respecting Git searches.
# Never descend original-body directories or symlinked cache/environment trees.
roots = [FINAL, INITIAL, REVIEW, ENDPOINT, PREFLIGHT, IMOD, IMOD411, DEDUP, AUTH]
excluded_trees = []
for root in roots:
    for directory, dirs, files in os.walk(root, followlinks=False):
        directory = Path(directory)
        for n in list(dirs):
            p = directory / n
            if p.is_symlink() or n in {'originals', '__pycache__', 'source-cache'}:
                dirs.remove(n)
                excluded_trees.append({'path': str(p), 'reason': 'External original source cache, interpreter artifact, or symlink tree; no maintained copy. Source bodies are bound separately by original-source manifests.'})
        for name in files:
            p = directory / name
            if p.is_symlink():
                inventory.append({'path': str(p), 'disposition': 'retained-external-symlink', 'target': os.readlink(p)})
                continue
            if str(p) not in paths:
                retained(p, 'Historical support or generated external fixture not selected for maintained copy. Preserved unchanged externally with this exact hash; accepted formal inputs and essential positive/negative proof are separately mapped.')
            inventory.append(paths[str(p)])
