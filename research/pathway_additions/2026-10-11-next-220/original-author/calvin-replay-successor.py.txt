def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for b in iter(lambda: f.read(4 * 1024 * 1024), b""):
            h.update(b)
    return h.hexdigest()


def snap(path):
    p = Path(path).absolute()
    return str(p.resolve(strict=True)), sha(p)


def digest(value):
    return hashlib.sha256(value).hexdigest()

def side(uri):
    result = []
    for key, values in g[uri].items():
        if not re.fullmatch("contains[0-9]+", key):
            continue
        for part in values:
            c = g[g[part]["compound"][0]]
            result.append(
                dict(
                    id=c["accession"][0],
                    coefficient=int(key[8:]),
                    formula=c["formula"][0],
                    charge=int(c["charge"][0]),
                    label=c["name"][0],
                )
            )
    return result


def balance(rows):
    atoms = Counter()
    charge = 0
    for row in rows:
        assert re.fullmatch(r"(?:[A-Z][a-z]?[0-9]*)+", row["formula"])
        for element, count in re.findall(r"([A-Z][a-z]?)([0-9]*)", row["formula"]):
            atoms[element] += int(count or 1) * row["coefficient"]
        charge += row["coefficient"] * row["charge"]
    return dict(atoms), charge

def pointer(value, path):
    for key in path.strip("/").split("/"):
        value = (
            value[int(key)]
            if isinstance(value, list)
            else value[key.replace("~1", "/").replace("~0", "~")]
        )
    return value

        e["subject"] for e in ee if e["predicate"] == "part_of" and e["object"] == doc["id"]
    } == set(rx)
    for rid, (left, right) in equations.items():
        u = RH + rid.split(":")[1]
        assert rx[rid]["label"] == g[u]["equation"][0]
        direction = "left_to_right" if g[u]["substrates"][0].endswith("_L") else "right_to_left"
        assert rx[rid]["direction"] == direction
        for pred, rows in [("consumes", left), ("produces", right)]:
            actual = {}
            for e in ee:
                if e["predicate"] == pred and rid in [e["subject"], e["object"]]:
                    cid = e["subject"] if pred == "consumes" else e["object"]
                    assert cid not in actual
                    actual[cid] = int(re.match(r"Coefficient (\d+)\b", e["description"]).group(1))
                    assert any(x["reference_id"] == rid for x in e["evidence"])
            assert actual == {x["id"]: x["coefficient"] for x in rows}
    for e in ee:
        if e["predicate"] == "catalyzes":
            nrefs = [
                x["reference_id"]
                for x in e["evidence"]

                x

            }
