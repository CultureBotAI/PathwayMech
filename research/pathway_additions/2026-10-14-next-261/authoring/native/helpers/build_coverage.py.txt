def object_context(frame):
 a=full[frame];kind=a['type']
 if frame in scopes:return scopes[frame]
 if kind=='Gene':return 'Exact reciprocal gene/product, nomenclature, coordinates and reference-strain identity are retained as identity context. The schema has no encodes predicate; no substitute causal relationship is invented.'
 if kind=='Organism':return 'Exact native organism identity, rank, strain and NCBI cross-reference are retained. Parent-pathway species annotations are distinguished from component donors, sequence reference taxa and expression hosts.'
 if kind=='Regulation':
  if frame=='REG-5876':return 'Native negative NADH regulation is rejected as accepted biology because PMID8419288 reports stimulation of complex-associated DmpG. Full native object and mechanism annotation remain conflict context.'
  if frame in selection['retained_native_regulators']:return 'Effect retained only at the exact native ENZRXN activity. Native allosteric or kinetic annotation is not an independently verified binding mechanism.'
  return 'Complete adjacent or explicitly excluded regulation context; the regulated activity is outside the selected five-step graph. Chemical identities and exact native target/effect are retained without transfer to route catalysts.'
 if kind=='Enzymatic-Reaction':
  if 'MetaCyc:'+frame in ids:return 'Source-specific activity interface, evidence codes, preparation and native reaction assignment retained. Quantitative kinetics, alternatives and assay conditions remain located source context and are not generalized to other catalysts.'
  return 'Complete adjacent or explicitly excluded activity context. Its alternative substrates, evidence codes, kinetics and regulator roster are retained without adding its transformation to the five-step route.'
 if kind=='Reaction':
  if frame in formal:return 'Formal chemistry is retained through the exact native/Rhea selection and complete native/ChEBI structure proof. Non-selected pathways, enzyme catalogues and bibliographic metadata remain source context.'
  return 'Adjacent reaction definition retained in full source context; it is not an additional transformation of PWY-5415.'
 if kind=='Compound':return 'Exact full native chemical identity and structure context; selected graph identities follow frozen chemistry or participant-authority proofs. A cross-reference alone does not prove exact structural equivalence.'
 if kind=='Pathway':
  if frame in ['PWY-5415','P183-PWY','PWY-5162']:return 'Native parent/module boundary, formal sequence and pathway prose retained; source species annotations have separate parent-level accounting. Shared terminal chemistry does not merge the oxidative sibling.'
  return 'Oxidative sibling/module definition remains adjacent context and establishes the alternative route boundary; no duplicate aggregate is created.'
 if kind=='Protein':return 'Native physical component or adjacent protein context, with complete component, sequence-reference, preparation and gene-product identity evidence retained.'
 if kind=='cco':return 'Exact cellular-component identity; location claims attach to source proteins and do not locate the entire pathway.'
 raise ValueError((frame,kind))
