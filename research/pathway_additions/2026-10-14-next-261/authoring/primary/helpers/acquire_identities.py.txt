import json,urllib.request,urllib.parse,pathlib,datetime,hashlib,time
R=pathlib.Path('/private/tmp/pathwaymech-next-261-primary-integration')
jobs=[]
for gene in ['tbuS','tbuR','tbuD','tbuE','tbuF','tbuG','tbuH','tbuI']:
 q='gene_exact:'+gene
 jobs.append(('UniProt-search-'+gene+'.json','https://rest.uniprot.org/uniprotkb/search?'+urllib.parse.urlencode({'query':q,'format':'json','size':500})))
for gene in ['tbuS','tbuR','tbuD','tbuE','tbuF','tbuG','tbuH','tbuI']:
 q='('+gene+'[Gene Name]) AND (pickettii[Organism] OR PKO1[All Fields])'
 jobs.append(('NCBI-nucleotide-search-'+gene+'.json','https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esearch.fcgi?'+urllib.parse.urlencode({'db':'nuccore','term':q,'retmode':'json','retmax':100})))
jobs.append(('NCBI-U20258.gb','https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=U20258&rettype=gbwithparts&retmode=text'))
out=[]
for name,url in jobs:
 d={'requested_url':url,'artifact':name,'requested_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech evidence research'}),timeout=40) as f:
   b=f.read();(R/'originals'/name).write_bytes(b);d.update(status=f.status,final_url=f.url,content_type=f.headers.get('Content-Type'),bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
 except Exception as e:d['error']=str(e)
 out.append(d);time.sleep(.4)
 (R/'identity-acquisition-receipts.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps(out,indent=2))
