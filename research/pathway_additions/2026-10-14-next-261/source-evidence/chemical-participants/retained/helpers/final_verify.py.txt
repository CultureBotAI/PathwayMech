from pathlib import Path
import datetime,hashlib,json,sqlite3
ROOT=Path('/private/tmp/pathwaymech-next-261-participant-authorities')
def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as stream:
        while b:=stream.read(8*1024*1024):h.update(b)
    return h.hexdigest()
native=json.loads((ROOT/'retained/native-chemical-dispositions.json').read_text())['endpoints']
claims=json.loads((ROOT/'retained/primary-claim-chemical-dispositions.json').read_text())['claims']

assert len(native)==22 and len({r['native_frame'] for r in native})==22
assert len(claims)==50 and {r['claim_id'] for r in claims}=={r['claim_id'] for r in original}
assert all(c['original_claim']==original[int(c['claim_source_json_pointer'].split('/')[-1])] for c in claims)
bindings={}
for r in native:
    p=Path(r['source_path']);assert sha(p)==r['source_sha256'];bindings[str(p)]=r['source_sha256']
    for role in r['roles']:
        p=Path(role['source_path']);assert sha(p)==role['source_sha256'];bindings[str(p)]=role['source_sha256']
contexts=json.loads((ROOT/'retained/assay-context-dispositions.json').read_text())['entries']
for r in contexts:
    p=Path(r['source']['path']);assert sha(p)==r['source']['sha256'];bindings[str(p)]=r['source']['sha256']
    assert r['source_locator'] and len(r['source_assertion'])<=400
checks=json.loads((ROOT/'retained/additional-chemistry-check-v2.json').read_text());assert checks['all_applicable_checks_pass']
historical=json.loads((ROOT/'retained/historical-form-check.json').read_text());assert historical['all_applicable_checks_pass']
preflight=json.loads((ROOT/'retained/authority-adapter-preflight.json').read_text());assert all(preflight['structure_binding_checks'].values())
plan=json.loads((ROOT/'retained/authority-integration-plan-v2.json').read_text())
assert plan['adapter_preflight']['sha256']==sha(ROOT/'retained/authority-adapter-preflight.json')
pub=next(r for r in plan['native_identity_plan'] if r['id']=='PubChem:90657761')
assert pub['source_original']['path']==preflight['accepted_authority']['path']
ids={r['selected_id'] for r in native}|{c for r in claims for c in r['chemical_identity_or_context_ids']}
dbpath=Path('/private/tmp/pathwaymech-curation-authorities/chebi.db');db=sqlite3.connect('file:'+str(dbpath)+'?mode=ro',uri=True)
identity_checks={}
for cid in sorted(ids):
    if cid.startswith('CHEBI:'):
        labels=[r[0] for r in db.execute("select value from statements where subject=? and predicate='rdfs:label'",(cid,))];assert len(labels)==1
        for row in native:
            if row['selected_id']==cid:assert row['selected_label']==labels[0]
        identity_checks[cid]={'issuing_label':labels[0],'exists':True}
    elif cid=='PubChem:90657761':identity_checks[cid]={'exists':True,'actual_adapter_passed':True}
    elif cid=='MetaCyc:CPD-29':
        owners=json.loads((ROOT/'retained/existing-authority-owners.json').read_text());assert cid in owners['existing_terms']
        identity_checks[cid]={'exists':True,'owner':owners['existing_terms'][cid]['term']['source']}
    else:raise AssertionError(cid)
digest=sha(dbpath);assert digest=='26ad4890cdcf6ee357760d815dfee8f5ba81e0873e62cb379369b04205722b0f'
