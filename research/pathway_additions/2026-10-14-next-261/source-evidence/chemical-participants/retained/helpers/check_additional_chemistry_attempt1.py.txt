def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as s:
        while b:=s.read(8*1024*1024):h.update(b)
    return h.hexdigest()
def describe(m):
    if m is None:return {'parse_error':True}
    n=Chem.Mol(m);Chem.RemoveStereochemistry(n)
    generic=any(a.GetAtomicNum()==0 for a in m.GetAtoms())
    return {'smiles':Chem.MolToSmiles(m),'connectivity_smiles':Chem.MolToSmiles(Chem.RemoveHs(n),isomericSmiles=False),
            'inchi':None if generic else Chem.MolToInchi(m),'key':None if generic else Chem.MolToInchiKey(m),
            'formula':rdMolDescriptors.CalcMolFormula(m),'charge':Chem.GetFormalCharge(m),
            'atoms':dict(Counter(a.GetSymbol() for a in Chem.AddHs(m).GetAtoms())),
            'chiral_centers':Chem.FindMolChiralCenters(m,includeUnassigned=True),'generic_variable_atom':generic}
def cml_mol(cml):
    r=Chem.RWMol();ids={}
    for a in cml.findall('atomArray/atom'):
        raw=a.get('elementType');atom=Chem.Atom(raw[0].upper()+raw[1:].lower());atom.SetFormalCharge(int(a.get('formalCharge','0')));ids[a.get('id')]=r.AddAtom(atom)
    for b in cml.findall('bondArray/bond'):
        a,z=b.get('atomRefs').split();r.AddBond(ids[a],ids[z],{'1':Chem.BondType.SINGLE,'2':Chem.BondType.DOUBLE,'3':Chem.BondType.TRIPLE}[b.get('order')])
    m=r.GetMol();Chem.SanitizeMol(m);return m
