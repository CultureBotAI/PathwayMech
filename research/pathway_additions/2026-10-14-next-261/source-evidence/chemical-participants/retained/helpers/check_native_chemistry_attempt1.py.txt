def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as stream:
        while b:=stream.read(8*1024*1024): h.update(b)
    return h.hexdigest()
def describe(m):
    if m is None:return {'parse_error':True}
    unstereo=Chem.Mol(m);Chem.RemoveStereochemistry(unstereo)

            'inchi':Chem.MolToInchi(m),'key':Chem.MolToInchiKey(m),'formula':rdMolDescriptors.CalcMolFormula(m),
            'charge':Chem.GetFormalCharge(m),'atoms':dict(Counter(a.GetSymbol() for a in Chem.AddHs(m).GetAtoms())),
            'chiral_centers':Chem.FindMolChiralCenters(m,includeUnassigned=True)}
def cml_mol(cml):
    rw=Chem.RWMol();amap={};case=[]
    for a in cml.findall('atomArray/atom'):
        raw=a.get('elementType');symbol=raw[0].upper()+raw[1:].lower()
        if raw!=symbol:case.append({'atom':a.get('id'),'source':raw,'interpreted':symbol})
        atom=Chem.Atom(symbol);atom.SetFormalCharge(int(a.get('formalCharge','0')))
        amap[a.get('id')]=rw.AddAtom(atom)
    for b in cml.findall('bondArray/bond'):
        a,z=b.get('atomRefs').split()
        rw.AddBond(amap[a],amap[z],{'1':Chem.BondType.SINGLE,'2':Chem.BondType.DOUBLE,'3':Chem.BondType.TRIPLE,'A':Chem.BondType.AROMATIC,'1.5':Chem.BondType.AROMATIC}[b.get('order')])
    m=rw.GetMol();Chem.SanitizeMol(m)
    return m,case
