from pathlib import Path
import datetime, hashlib, json, urllib.request, urllib.parse, xml.etree.ElementTree as E

ROOT = Path('/private/tmp/pathwaymech-next-261-chemistry')
OUT = ROOT / 'originals'
OUT.mkdir(exist_ok=True)
receipt = ROOT / 'current-compound-acquisition.json'
assert not receipt.exists()
rows = []
for frame in ['ACETALD', 'ACETYL-COA', 'CO-A', 'OXYGEN-MOLECULE', 'PROTON', 'PYRUVATE', 'WATER']:
    path = OUT / ('META-' + frame + '.xml')
    assert not path.exists()
    url = 'https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=' + urllib.parse.quote('META:' + frame, safe='')
    row = {'frame': frame, 'requested_url': url, 'started_utc': datetime.datetime.now(datetime.timezone.utc).isoformat()}
    try:
        with urllib.request.urlopen(url, timeout=60) as response:
            raw = response.read()
            row.update(status=response.status, final_url=response.url, headers=dict(response.headers))
        path.write_bytes(raw)
        row.update(path=str(path), bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest())
        root = E.fromstring(raw)
        objs = [n for n in root.iter('Compound') if n.get('frameid') == frame and n.get('detail') == 'full']
        assert len(objs) == 1
        row['pgdb'] = dict(root.find('metadata/PGDB').attrib)
        assert row['pgdb']['orgid'] == 'META' and row['pgdb']['version'] == '27.0'
        row['complete_original'] = True
    except Exception as error:
        row['error'] = repr(error)
        rows.append(row)
        receipt.write_text(json.dumps(rows, indent=2) + '\n')
        raise
    rows.append(row)
    receipt.write_text(json.dumps(rows, indent=2) + '\n')
    print(frame, row['status'], row['sha256'], flush=True)
