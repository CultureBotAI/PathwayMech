class IntegrityError(ValueError):
    pass


def require(condition, message):
    if not condition:
        raise IntegrityError(message)


def now():
    return datetime.now(UTC).isoformat().replace("+00:00", "Z")


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def file_hash(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(4 * 1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def unique_pairs(items):
    result = {}
    for key, value in items:
        require(key not in result, "Duplicate JSON key: " + key)
        result[key] = value
    return result


def read_json(raw):
    return json.loads(raw, object_pairs_hook=unique_pairs)


def absolute(path):
    return Path(os.path.abspath(path))


def check_entry(entry):
    require(set(entry) == {"path", "resolved_path", "sha256", "role"}, "Invalid binding fields")
    require(entry["role"] in ROLE_SET, "Unknown binding role")
    require(re.fullmatch(r"[0-9a-f]{64}", entry["sha256"]) is not None, "Invalid input digest")
    path = Path(entry["path"])
    require(path.is_absolute() and str(absolute(path)) == str(path), "Noncanonical declared path")
    require(
        path.resolve(strict=True) == Path(entry["resolved_path"]),
        "Resolved input identity changed: " + str(path),
    )
    require(path.is_file(), "Input is not a file: " + str(path))
    actual = file_hash(path)
    require(actual == entry["sha256"], "Input digest changed: " + str(path))
    require(
        path.resolve(strict=True) == Path(entry["resolved_path"]),
        "Resolved input identity changed during hash: " + str(path),
    )
    return dict(entry)


def capture_bindings(entries):
    require(isinstance(entries, list) and entries, "No declared bindings")
    paths = [x["path"] for x in entries]
    require(len(paths) == len(set(paths)), "Duplicate binding path")
    return [check_entry(entry) for entry in entries]


def inventory(root):
    """Includes dotfiles and Git-ignored files; excludes only interpreter caches."""
    root = Path(root)
    require(root.is_dir(), "Missing inventory root: " + str(root))
    found = []
    for directory, dirs, files in os.walk(root, followlinks=False):
        dirs[:] = [name for name in dirs if name != "__pycache__"]
        for name in dirs:
            require(
                not (Path(directory) / name).is_symlink(),
                "Symlinked inventory directory: " + str(Path(directory) / name),
            )
        for name in files:
            if name.endswith((".pyc", ".pyo")):
                continue
            path = Path(directory) / name
            require(path.is_file(), "Non-file inventory entry: " + str(path))
            found.append(str(absolute(path)))
    return sorted(found)


def check_inventories(specs, bindings):
    paths = {x["path"] for x in bindings}
    for spec in specs:
        require(set(spec) == {"root", "paths"}, "Invalid inventory fields")
        actual = inventory(spec["root"])
        require(actual == spec["paths"], "Input inventory changed: " + spec["root"])
        require(set(actual) <= paths, "Inventory contains unbound inputs: " + spec["root"])


def capture_manifest(path, expected_sha):
    require(
        re.fullmatch(r"[0-9a-f]{64}", expected_sha) is not None, "Invalid binding-manifest digest"
    )
    declared = absolute(path)
    resolved = declared.resolve(strict=True)
    raw = declared.read_bytes()
    require(digest(raw) == expected_sha, "Binding-manifest digest changed")
    document = read_json(raw)
    require(
        document["manifest_identity"] == {"path": str(declared), "resolved_path": str(resolved)},
        "Binding-manifest identity changed",
    )
    require(
        declared.resolve(strict=True) == resolved, "Binding-manifest identity changed during read"
    )
    return document, {
        "path": str(declared),
        "resolved_path": str(resolved),
        "sha256": expected_sha,
        "role": "manifest",
    }


def check_output(path, protected_roots, bound_paths):
    output = absolute(path)
    require(
        not output.exists() and not output.is_symlink(), "Output already exists: " + str(output)
    )
    resolved = output.resolve(strict=False)
    require(output.parent.resolve(strict=True).is_dir(), "Output parent must exist")
    for root in protected_roots:
        require(
            not resolved.is_relative_to(Path(root).resolve()),
            "Output is inside a protected input tree",
        )
    for name in bound_paths:
        require(not Path(name).resolve().is_relative_to(resolved), "Output contains a bound input")
    return output, resolved

def formatting_relation(original, successor):
    """Normalize only the root-declared path name and one unused loop binding."""
    left, right = ast.parse(original), ast.parse(successor)
    left_names = [node for node in ast.walk(left) if isinstance(node, ast.Name)]
    right_names = [node for node in ast.walk(right) if isinstance(node, ast.Name)]
    require(not any(node.id in {"ORIGINAL_AUTHOR", "_unused_accession"} for node in left_names), "Original normalization-name collision")
    original_path_names = [node for node in left_names if node.id == "O"]
    renamed_path_names = [node for node in right_names if node.id == "ORIGINAL_AUTHOR"]
    require(original_path_names and len(original_path_names) == len(renamed_path_names), "Original-author rename coverage differs")
    require(not any(node.id == "O" for node in right_names), "Partial original-author rename")
    unused_names = [node for node in right_names if node.id == "_unused_accession"]
    require(len(unused_names) == 1 and isinstance(unused_names[0].ctx, ast.Store), "Unused rename is not a single write-only binding")
    expected_iter = ast.dump(ast.parse('[(3, "Q59647"), (4, "Q59646")]', mode='eval').body, include_attributes=False)
    loops = [node for node in right.body if isinstance(node, ast.For) and ast.dump(node.iter, include_attributes=False) == expected_iter]
    require(len(loops) == 1, "Renamed CIF loop differs")
    loop = loops[0]
    require(isinstance(loop.target, ast.Tuple) and [node.id for node in loop.target.elts] == ["entity", "_unused_accession"], "Unexpected unused-binding target")
    require(not any(isinstance(node, ast.Name) and node.id in {"acc", "_unused_accession"} for statement in loop.body for node in ast.walk(statement)), "Unused loop binding is read inside loop")
    loop_index = right.body.index(loop)
    require(not any(isinstance(node, ast.Name) and node.id in {"acc", "_unused_accession"} and isinstance(node.ctx, ast.Load) for statement in right.body[loop_index + 1:] for node in ast.walk(statement)), "Renamed loop binding is read later")
    for node in renamed_path_names:
        node.id = "O"
    unused_names[0].id = "acc"
    original_ast = ast.dump(left, include_attributes=False)
    successor_ast = ast.dump(right, include_attributes=False)
    require(original_ast == successor_ast, "Executable AST differs beyond declared name normalizations")
    return {"normalized_ast_equal": True, "normalized_ast_sha256": digest(original_ast.encode()), "original_author_name_occurrences": len(original_path_names), "single_unused_binding_normalized": True, "original_bytes": len(original), "successor_bytes": len(successor)}


def check_formatting_lineage(document, repo, helper):
    bound = {row["path"]: row for row in document["files"]}
    for path in [CORRECTED_HELPER, repo / ORIGINAL_ARCHIVE, helper, FORMATTING_RECEIPT]:
        check_entry(bound[str(path)])
    original = CORRECTED_HELPER.read_bytes()
    archived = (repo / ORIGINAL_ARCHIVE).read_bytes()
    successor = helper.read_bytes()
    require(original == archived and digest(original) == ORIGINAL_HELPER_SHA, "Original corrected helper preservation differs")
    require(digest(successor) == HELPER_SHA, "Active helper differs")
    receipt = read_json(FORMATTING_RECEIPT.read_bytes())
    require(receipt["original_sha256"] == ORIGINAL_HELPER_SHA and receipt["successor_sha256"] == HELPER_SHA and receipt["target_sha256"] == TARGET_SHA, "Root formatting identities differ")
    require(receipt["original_archive"] == ORIGINAL_ARCHIVE, "Original archive ownership differs")
    require(len(receipt["historical_resolution_successor"]) == 4, "Historical alias successor coverage differs")
    for row in receipt["historical_resolution_successor"]:
        require(row["unchanged_byte_owner"] == ORIGINAL_ARCHIVE and row["sha256"] == ORIGINAL_HELPER_SHA and row["original_row"]["sha256"] == ORIGINAL_HELPER_SHA, "Historical original-helper bytes reassigned")
    return formatting_relation(original, successor)


def source_dependencies(helper):
    """Enumerate exact original source and predecessor paths without rewriting them."""
    dependencies = {}

    def add(path, expected):
        name = str(absolute(path))
        require(re.fullmatch(r"[0-9a-f]{64}", expected) is not None, "Invalid dependency digest")
        require(name not in dependencies or dependencies[name] == expected, "Conflicting dependency digest: " + name)
        dependencies[name] = expected

    def adjacent(relative):
        raw = (helper.resolve().parent / relative).read_bytes()
        require(digest(raw) == ADJACENT[relative], "Adjacent input changed before parsing: " + relative)
        return read_json(raw)

    source = adjacent("draft/denitrification-source-manifest.json")
    require(source["record_sha256"] == TARGET_SHA, "Source-manifest target mismatch")
    for item in source["bodies"] + source["issuing_databases"] + source["prior_proof_inputs"]:
        add(item["path"], item["sha256"])
    for item in adjacent("predecessor-preservation.json")["files"]:
        add(item["path"], item["sha256"])
    for item in adjacent("draft/denitrification-authority-proposal.json")["sources"]:
        add(item["path"], item["sha256"])
    for item in adjacent("draft/denitrification-chemical-identities.json")["databases"].values():
        add(item["path"], item["sha256"])
    for name in IMOD_METADATA:
        path = ORIGINAL_AUTHOR / "imod" / name
        require(str(path) in dependencies, "Original iMod metadata absent from predecessor closure: " + name)
        raw = path.read_bytes()
        require(digest(raw) == dependencies[str(path)], "Original iMod metadata changed: " + name)
        if name in {"context-manifest.json", "inventory-manifest.json"}:
            for item in read_json(raw):
                add(item["path"], item["sha256"])
    add(ADAPTER_ROOT / "imodulondb.py", ADAPTER_SHA)
    return dependencies


def check_source_coverage(document, helper):
    bound = {x["path"]: x for x in document["files"]}
    dependencies = source_dependencies(helper)
    require(document["production_authority_sources"], "Missing production source bindings")
    for item in document["production_authority_sources"]:
        name = str(absolute(item["path"]))
        require(name not in dependencies or dependencies[name] == item["sha256"], "Conflicting production source digest")
        dependencies[name] = item["sha256"]
    for name, expected in dependencies.items():
        require(name in bound and bound[name]["sha256"] == expected, "Unbound source/prior-proof input: " + name)
    for name in ["__init__.py", "catalogue.py", "fetch.py"]:
        require(str(ADAPTER_ROOT / name) in bound, "Unbound Claw adapter import dependency: " + name)
    require(len(document["production_whole_sources"]) == 4, "Expected four production SQL databases")
    for item in document["production_whole_sources"]:
        require(item in document["production_authority_sources"], "Production SQL binding not in source closure")
        require(item["path"].endswith(".db"), "Production source is not SQL")

def recheck_formal_inputs(document, own_manifest):
    for entry in document["files"]:
        if entry["role"] not in {"source", "prior_proof"}:
            check_entry(entry)
    check_inventories(document["inventories"], document["files"])
    check_entry(own_manifest)


def final_recheck(document, own_manifest):
    # Slow source reads first; formal inputs are freshly checked last. Never use
    # startup bytes or cached digests as current-file evidence.
    check_entry(own_manifest)
    check_inventories(document["inventories"], document["files"])
    for entry in document["files"]:
        if entry["role"] in {"source", "prior_proof"}:
            check_entry(entry)
    recheck_formal_inputs(document, own_manifest)


def child_environment(repo, output):
    # A fresh external prefix prevents imports from consuming unbound local
    # __pycache__ files. No RDKit child is used by this corrected helper.
    env = dict(
        os.environ,
        PYTHONPATH=str(repo / "src"),
        PYTHONNOUSERSITE="1",
        PYTHONSAFEPATH="1",
        PYTHONDONTWRITEBYTECODE="1",
        PYTHONPYCACHEPREFIX=str(output / ".python-cache"),
        UV_NO_SYNC="1",
        UV_CACHE_DIR="/private/tmp/pathwaymech-curation-uv-cache",
    )
    env.pop("PYTHONOPTIMIZE", None)
    env.pop("PYTHONHOME", None)
    return env

                {

            require(process.returncode == 0, "Child check failed: " + name)
        child_raw = child_output.read_bytes()
