def add(path, role, expected=None, defer=False, resolved=None):
    path = driver.absolute(path)
    driver.require(path.is_file(), 'Missing binding input: ' + str(path))
    actual_resolved = str(path.resolve(strict=True))
    driver.require(resolved is None or actual_resolved == resolved, 'Original resolved identity changed: ' + str(path))
    if defer:
        driver.require(expected is not None, 'Deferred source requires accepted digest')
        actual = expected
        deferred_sources.append(str(path))
    else:
        actual = driver.file_hash(path)
        driver.require(expected is None or actual == expected, 'Accepted input changed: ' + str(path))
    driver.require(str(path.resolve(strict=True)) == actual_resolved, 'Input retargeted during preparation')
    row = {'path': str(path), 'resolved_path': actual_resolved, 'sha256': actual, 'role': role}
    if str(path) in files:
        driver.require(all(files[str(path)][key] == row[key] for key in ['path', 'resolved_path', 'sha256']), 'Conflicting binding')
    else:
        files[str(path)] = row


def accepted(path):
    raw = path.read_bytes()
    driver.require(driver.digest(raw) == PINNED[path], 'Accepted proof changed: ' + str(path))
    return driver.read_json(raw)
