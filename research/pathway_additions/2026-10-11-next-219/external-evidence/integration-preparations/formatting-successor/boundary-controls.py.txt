def write(path, value):
    with path.open("x") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")


def entry(path, role="runtime"):
    return {
        "path": str(path),
        "resolved_path": str(path.resolve()),
        "sha256": d.file_hash(path),
        "role": role,
    }


def reject(name, function, message):
    try:
        function()
    except d.IntegrityError as error:
        d.require(message in str(error), "Unexpected rejection for " + name + ": " + str(error))
        results.append(
            {
                "name": name,
                "passed": True,
                "actual_exception": type(error).__name__,
                "message": str(error),
            }
        )
    else:
        raise RuntimeError("Guard failed to reject: " + name)


def fixture(name, role="runtime"):

    runtime.mkdir()
    path = runtime / "input.txt"
    path.write_text("accepted bytes\n")
    document = {
        "manifest_identity": {
            "path": str(root / "manifest.json"),
            "resolved_path": str(root / "manifest.json"),
        },
        "files": [entry(path, role)],
        "inventories": [{"root": str(runtime), "paths": d.inventory(runtime)}],
    }
    write(root / "manifest.json", document)
    own = entry(root / "manifest.json", "manifest")
    d.capture_bindings(document["files"])
    d.final_recheck(document, own)
    return root, path, document, own
