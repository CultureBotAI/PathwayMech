def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def capture(path, expected=None):
    path = Path(path).absolute()
    resolved = path.resolve(strict=True)
    raw = path.read_bytes()
    row = {'path': str(path), 'resolved_path': str(resolved), 'sha256': sha(raw), 'bytes': len(raw)}
    assert path.resolve(strict=True) == resolved
    if expected:
        assert row['sha256'] == expected, (path, expected, row['sha256'])
    if str(path) in bindings:
        assert bindings[str(path)] == row, ('Changed input', path)
    bindings[str(path)] = row
    return raw

def read(path):
    return json.loads(capture(path))

def write_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x') as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write('\n')

def whitespace(raw):
    return any(line.rstrip(b'\r\n').endswith((b' ', b'\t')) for line in raw.splitlines(keepends=True)) or raw.endswith(b'\n\n')

def safe_relative(relative):
    names = []
    for component in Path(relative).parts:
        assert component not in ['..', '/']
        if component == '__pycache__': component = 'captured-python-cache'
        if component == '.ruff_cache': component = 'captured-ruff-cache'
        if component == '.gitignore': component = 'gitignore-observation.txt'
        names.append(component)
    return Path(*names)

def package(path, destination, role, active=False, force_wrapper=False):
    path = Path(path).absolute()
    if str(path) in dispositions:
        return dispositions[str(path)]
    raw = capture(path)
    original_sha = sha(raw)
    destination = Path(destination)
    assert destination.is_relative_to(DOSSIER) and not destination.is_absolute() and '..' not in destination.parts
    assert destination != DOSSIER / 'README.md'
    assert '__pycache__' not in destination.parts
    try:
        decoded = raw.decode('utf-8')
        text = '\x00' not in decoded
    except UnicodeDecodeError:
        decoded, text = None, False
    wrap = force_wrapper or not text or whitespace(raw) or path.suffix in {'.log', '.stderr', '.stdout', '.jsonl', '.tsv', '.diff'} or '.yaml' in path.name or '.yml' in path.name
    if active:
        assert not wrap, ('Active byte-exact dependency would require wrapper', path)
    elif destination.suffix == '.py':
        destination = destination.with_name(destination.name + '.txt')
    if not active and original_sha in canonical:
        owner = canonical[original_sha]
        row = {**bindings[str(path)], 'disposition': 'content-addressed-exact-byte-alias', 'destination': owner['destination'], 'mode': owner['mode'], 'role': role}
        aliases.append(row)
        dispositions[str(path)] = row
        return row
    mode = 'exact_bytes'
    source = path
    if wrap:
        mode = 'lossless_json_wrapper'
        encoding, content = ('utf-8', decoded) if text else ('base64', base64.b64encode(raw).decode('ascii'))
        wrapper = {'format': 'lossless-immutable-evidence-v1', 'original_sha256': original_sha, 'original_bytes': len(raw), 'encoding': encoding, 'content': content}
        source = OUT / 'payload' / (original_sha + '.json')
        if not source.exists(): write_json(source, wrapper)
        else: assert json.loads(source.read_text()) == wrapper
        destination = destination.with_name(destination.name + '.bytes.json')
    assert not any(row['destination'] == str(destination) for row in copies), destination
    row = {**bindings[str(path)], 'disposition': 'planned-copy', 'destination': str(destination), 'copy_source': str(source), 'destination_sha256': sha(source.read_bytes()), 'mode': mode, 'role': role}
    copies.append(row)
    canonical.setdefault(original_sha, row)
    dispositions[str(path)] = row
    return row

def archive(path, group, relative=None, force_wrapper=False):
    relative = safe_relative(relative or Path(path).name)
    return package(path, DOSSIER / 'external-evidence' / group / relative, 'Immutable historical evidence; no canonical or independent approval is implied', force_wrapper=force_wrapper)

def external(path, reason, expected=None):
    path = Path(path).absolute()
    if str(path) in dispositions:
        if expected: assert dispositions[str(path)]['sha256'] == expected
        return dispositions[str(path)]
    capture(path, expected)
    row = {**bindings[str(path)], 'disposition': 'retained-external', 'reason': reason}
    retained.append(row)
    dispositions[str(path)] = row
    return row

def scan(root, group):
    # No gitignore filtering; includes hidden/cache/fixture files. Never follows links.
    for directory, dirs, files in os.walk(root, followlinks=False):
        directory = Path(directory)
        entries = sorted(dirs + files)
        directories.append({'path': str(directory), 'mode': oct(stat.S_IMODE(directory.lstat().st_mode)), 'entries': entries, 'disposition': 'directory-observation-only'})
        for name in list(dirs) + files:
            p = directory / name
            if p.is_symlink():
                target = os.readlink(p)
                symlinks.append({'path': str(p), 'link_target': target, 'link_target_sha256': sha(os.fsencode(target)), 'resolved_path': str(p.resolve(strict=False)), 'dangling': not p.exists(), 'disposition': 'observation-only-no-installed-link'})
                if name in dirs: dirs.remove(name)
            elif p.is_file():
                capture(p)
                inventory[str(p)] = {'group': group, 'relative': str(p.relative_to(root)), **bindings[str(p)]}
            elif name in files:
                raise AssertionError(('Unsupported special file', p))

def resolve(label, row):
    original_path = Path(row['path']).absolute()
    expected = row['sha256']
    assert re.fullmatch('[0-9a-f]{64}', expected)
    key = str(original_path)
    if key in whole:
        assert whole[key]['sha256'] == expected
        owner = whole[key]
        disposition = 'external-whole-database-declared-digest-not-rehashed'
    else:
        raw = capture(original_path)
        if sha(raw) != expected:
            matches = [x for x in inventory.values() if x['sha256'] == expected]
            assert matches, ('Historical bytes have no preserved owner', label, original_path, expected, sha(raw))
            selected = sorted(matches, key=lambda x: x['path'])[0]
            owner = dispositions[selected['path']]
            disposition = 'historical-digest-resolved-to-preserved-exact-byte-owner'
        else:
            if key not in dispositions:
                resolved = original_path.resolve(strict=True)
                if any(resolved.is_relative_to(root) for root in raw_roots):
                    owner = external(original_path, 'Historical fixture path aliases an external original body through an observed raw-directory symlink.', expected)
                elif '/src/' in key or '/scripts/' in key:
                    rel = Path(original_path.parts[3]) / Path(*original_path.parts[4:])
                    owner = archive(original_path, 'historical-runtime', rel)
                else:
                    owner = archive(original_path, 'external-proof-inputs', Path(expected[:16]) / original_path.name)
            else:
                owner = dispositions[key]
            disposition = owner['disposition']
    resolution = {'set': label, 'path': key, 'sha256': expected, 'resolution': disposition, 'owner_path': owner['path'], 'owner_sha256': owner['sha256'], 'destination': owner.get('destination'), 'mode': owner.get('mode'), 'external': owner['disposition'].startswith('retained-external')}
    formal.append(resolution)
