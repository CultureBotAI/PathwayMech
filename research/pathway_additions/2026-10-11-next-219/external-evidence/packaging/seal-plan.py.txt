from datetime import datetime, timezone

PEER = Path('/private/tmp/pathwaymech-denitrification-copy-plan-bounded-readonly-review.json')
PEER_SUCCESSOR = Path('/private/tmp/pathwaymech-denitrification-copy-plan-bounded-readonly-review-successor.json')
BRIDGE = Path('/private/tmp/pathwaymech-next-219-merged-baseline-bridge.json')
PINNED = {
    OUT / 'copy-plan.json': '14fa939c6ddd6d6529aab4a7acb7ed40f059bb5eeca29f2813ebe5df05d366f9',
    OUT / 'validation.json': 'e9a5e768abf0e7e23ef7495c005469e362eded63c93d9609dabd9e2ada0c4a42',
    PEER: '7fe253d3691ec6610fe94c7ae1c84646f43e4661912d5dafdd3bc34da7c98c26',

started = datetime.now(timezone.utc).isoformat()

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def describe(path):
    path = Path(path).absolute()
    raw = path.read_bytes()
    return {'path': str(path), 'resolved_path': str(path.resolve(strict=True)), 'sha256': digest(raw), 'bytes': len(raw)}

def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

def recovered(row):
    raw = Path(row['copy_source']).read_bytes()
    assert digest(raw) == row['destination_sha256']
    if row['mode'] == 'lossless_json_wrapper':
        wrapper = json.loads(raw)
        assert wrapper['format'] == 'lossless-immutable-evidence-v1'
        if wrapper['encoding'] == 'utf-8':
            raw = wrapper['content'].encode('utf-8')
        else:
            assert wrapper['encoding'] == 'base64'
            raw = base64.b64decode(wrapper['content'], validate=True)
        assert digest(raw) == wrapper['original_sha256']
        assert len(raw) == wrapper['original_bytes']
    else:
        assert row['mode'] == 'exact_bytes'
    assert digest(raw) == row['sha256'] and len(raw) == row['bytes']
    return raw

for path, expected in PINNED.items():
    assert describe(path)['sha256'] == expected, path
plan = json.loads((OUT / 'copy-plan.json').read_bytes())
validation = json.loads((OUT / 'validation.json').read_bytes())
assert plan['validation'] == validation
peer = json.loads(PEER.read_bytes())
successor = json.loads(PEER_SUCCESSOR.read_bytes())
assert peer['plan_sha256'] == PINNED[OUT / 'copy-plan.json']
assert successor['plan_sha256'] == PINNED[OUT / 'copy-plan.json']
assert 'passed' in successor['status']
assert '7fe253d3691ec6610fe94c7ae1c84646f43e4661912d5dafdd3bc34da7c98c26' in json.dumps(successor)
for row in plan['input_bindings']:
    assert describe(row['path']) == row, ('Changed small input', row['path'])
for row in plan['directory_observations']:
    path = Path(row['path'])
    assert sorted(os.listdir(path)) == row['entries']
    assert oct(stat.S_IMODE(path.lstat().st_mode)) == row['mode']
for row in plan['symlink_observations']:
    path = Path(row['path'])
    assert path.is_symlink() and os.readlink(path) == row['link_target']
    assert digest(os.fsencode(os.readlink(path))) == row['link_target_sha256']
    assert str(path.resolve(strict=False)) == row['resolved_path']
    assert (not path.exists()) == row['dangling']
for row in plan['issuing_databases_retained_external']:
    path = Path(row['path'])
    assert str(path.resolve(strict=True)) == row['resolved_path']
    assert path.stat().st_size == row['bytes']
    assert row['fresh_hash_in_this_packaging_run'] is False
destinations = {row['destination']: row for row in plan['copy_entries']}
assert len(destinations) == 226
for row in plan['copy_entries']:
    recovered(row)
    destination = Path(row['destination'])
    assert not destination.is_absolute() and '..' not in destination.parts
    assert destination.is_relative_to(DOSSIER)
    assert destination != DOSSIER / 'README.md'
    assert '__pycache__' not in destination.parts
    assert not os.path.lexists(REPO / destination), ('Occupied destination', destination)
    preview = OUT / 'preview' / destination
    assert preview.is_file() and not preview.is_symlink()
    assert digest(preview.read_bytes()) == row['destination_sha256']
    assert stat.S_IMODE(preview.stat().st_mode) == 0o644
for row in plan['exact_byte_aliases']:

    assert owner['sha256'] == row['sha256']
    assert owner['mode'] == row['mode']
    assert len(recovered(owner)) == row['bytes']
assert len(plan['formal_input_resolution']) == 1611
