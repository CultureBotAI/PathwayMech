def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def checked(row):
    path = Path(row['path'])
    raw = path.read_bytes()
    assert sha(raw) == row['sha256'], path
    if 'resolved_path' in row:
        assert str(path.resolve(strict=True)) == row['resolved_path'], path
    if 'bytes' in row:
        assert len(raw) == row['bytes'], path
    return raw

def write_new(path, raw):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream:
        stream.write(raw)
    path.chmod(0o644)
    assert path.read_bytes() == raw
