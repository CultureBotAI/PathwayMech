def diff(a,b,path=''):
    if type(a)!=type(b):return [{'path':path,'old':a,'new':b}]
    if isinstance(a,dict):
        out=[]
        for k in sorted(a.keys()|b.keys()):
            if k not in a or k not in b:out.append({'path':path+'/'+k,'old':a.get(k),'new':b.get(k)})
            else:out.extend(diff(a[k],b[k],path+'/'+k))
        return out
    if isinstance(a,list):
        out=[]
        for i in range(max(len(a),len(b))):
            if i>=len(a) or i>=len(b):out.append({'path':path+f'/{i}','old':a[i] if i<len(a) else None,'new':b[i] if i<len(b) else None})
            else:out.extend(diff(a[i],b[i],path+f'/{i}'))
        return out
    return [] if a==b else [{'path':path,'old':a,'new':b}]
