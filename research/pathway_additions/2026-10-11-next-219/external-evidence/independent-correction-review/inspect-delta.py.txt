def delta(x,y,path=''):
 if type(x)!=type(y):return [{'path':path,'old':x,'new':y}]
 if isinstance(x,dict):
  return sum((delta(x[k],y[k],path+'/'+k) if k in x and k in y else [{'path':path+'/'+k,'old':x.get(k),'new':y.get(k)}] for k in sorted(set(x)|set(y))),[])
 if isinstance(x,list):return sum((delta(x[i],y[i],path+'/'+str(i)) if i<len(x) and i<len(y) else [{'path':path+'/'+str(i),'old':x[i] if i<len(x) else None,'new':y[i] if i<len(y) else None}] for i in range(max(len(x),len(y)))),[])
 return [] if x==y else [{'path':path,'old':x,'new':y}]
