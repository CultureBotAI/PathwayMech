def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def require(ok, label):
    if not ok:
        raise RuntimeError(label)

for name, expected in pins.items():
    require(sha(ROOT/name)==expected, 'Root pin '+name)
handoff=json.loads((ROOT/'handoff.json').read_bytes())
require(sha(handoff['completion']['path'])==handoff['completion']['sha256'], 'Completion pin')
completion=json.loads(Path(handoff['completion']['path']).read_bytes())
for row in completion['proof_artifacts']:
    p=Path(row['path'])
    require(sha(p)==row['sha256'] and p.stat().st_size==row['bytes'], 'Sealed artifact '+str(p))

spec=importlib.util.spec_from_file_location('denit_driver',ROOT/'integration-replay.py')
d=importlib.util.module_from_spec(spec);spec.loader.exec_module(d)
document,own=d.capture_manifest(ROOT/'expected-bindings.json',pins['expected-bindings.json'])

    require(str(path.resolve(strict=True))==row['resolved_path'], 'All declared path identities')
    if row['role'] not in {'source','prior_proof'}:
        d.check_entry(row);formal.append(row)
    else:
        source_paths.append(row)
require(len(formal)==1564 and len(document['files'])==2652,'Declared binding counts')

require(len(production)==len(document['production_authority_sources']),'Production duplicate paths')
closure=[]
for entry in manifest:
    p=str(Path('/private/tmp/pathwaymech-curation-authorities/sources')/entry['path'])
    require(p in production and p in bound and production[p]['sha256']==bound[p]['sha256']==entry['sha256'],'Current manifest closure '+p)
    closure.append({'key':entry['key'],'path':p,'sha256':entry['sha256']})
require(len(closure)==597,'Complete current source manifest')
whole=document['production_whole_sources']
require(len({row['path'] for row in whole})==4,'Distinct production SQL')
for row in whole:
    require(row in document['production_authority_sources'] and bound[row['path']]['sha256']==row['sha256'],'Production SQL exact closure')

original=d.CORRECTED_HELPER.read_bytes();active=helper.read_bytes()

old_names=[node for node in ast.walk(left) if isinstance(node,ast.Name)]

require(len([x for x in new_names if x.id=='ORIGINAL_AUTHOR'])==9,'Successor name occurrences')

require(len(renamed)==1 and isinstance(renamed[0].ctx,ast.Store),'Only unused store binding')

    if node.id=='ORIGINAL_AUTHOR':node.id='O'
renamed[0].id='acc'

require(normalized==ast.dump(left,include_attributes=False),'Independent full normalized AST')
require(hashlib.sha256(normalized.encode()).hexdigest()==formatting['normalized_ast_sha256'],'Normalization digest')
negative=[]

    ('changed literal source path',active.replace(b'pathwaymech-denitrification-author-20261011',b'pathwaymech-denitrification-author-20261012',1)),
    ('extra executable assertion',active+b'\nassert False\n'),
    ('newly read renamed loop binding',active.replace(b'for entity, _unused_accession in [(3, "Q59647"), (4, "Q59646")]:',b'for entity, _unused_accession in [(3, "Q59647"), (4, "Q59646")]:\n    print(_unused_accession)',1)),
]:
    require(changed!=active,'Negative really changed')
    try:
        d.formatting_relation(original,changed)

        negative.append({'case':label,'rejected':True,'message':str(error)})
    else:
        raise RuntimeError('Accepted bad normalization '+label)

controls_path=ROOT/'controls-attempt-1/results.json'
