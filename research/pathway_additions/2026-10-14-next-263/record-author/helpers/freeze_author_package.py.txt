# Every author file is retained once; gzip only encodes exact bytes.
originals = sorted(f for f in B.rglob('*') if f.is_file())
compact = B / 'compact-retained'
compact.mkdir()
rows = []
for path in originals:
    rel = path.relative_to(B)
    raw = path.read_bytes()
    compress = (len(raw) > 60000 and rel != Path('drafts/catechol-degradation-iii.yaml')) or any(line.rstrip(b' \t') != line for line in raw.splitlines())
    stored_rel = Path('compact-retained') / (str(rel) + '.gz' if compress else rel)
    stored = B / stored_rel
    stored.parent.mkdir(parents=True, exist_ok=True)
    payload = gzip.compress(raw, mtime=0) if compress else raw
    stored.write_bytes(payload)
    assert (gzip.decompress(payload) if compress else payload) == raw
    rows.append({'original_path': str(path), 'original_relative_path': str(rel),
                 'original_sha256': sha(path), 'original_bytes': len(raw),
                 'stored_path': str(stored_rel), 'stored_sha256': sha(stored),
                 'stored_bytes': len(payload), 'encoding': 'gzip' if compress else 'identity'})
write('compact-retention-manifest.json', {'frozen_utc': now(), 'root': str(B),
      'schema': 'exact-original-and-stored-files-v1', 'files': rows,
      'source_packages_are_separately_retained': True})
