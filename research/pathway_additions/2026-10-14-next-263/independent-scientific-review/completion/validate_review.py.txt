import datetime,hashlib,json,pathlib,subprocess,sys
O=pathlib.Path('/private/tmp/pathwaymech-next-263-review-completion'); R=pathlib.Path('/private/tmp/pathwaymech-next-263-20261014')
cmd=[sys.executable,'-B','scripts/record_review.py','validate',str(O/'review-content.yaml')]
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
p=subprocess.run(cmd,cwd=R,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
for name,body in [('validate.stdout.log',p.stdout),('validate.stderr.log',p.stderr)]: (O/name).write_bytes(body)
receipt={'started_at':start,'finished_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'command':cmd,'cwd':str(R),'exit_code':p.returncode,'content_sha256':hashlib.sha256((O/'review-content.yaml').read_bytes()).hexdigest(),'stdout_sha256':hashlib.sha256(p.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(p.stderr).hexdigest()}
(O/'validation-receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(p.stdout.decode(),end='');print(p.stderr.decode(),end='',file=sys.stderr);sys.exit(p.returncode)
