sha=lambda b:hashlib.sha256(b).hexdigest()
def load(p):return json.loads(p.read_text())
def bind(p):return {'path':str(p),'bytes':p.stat().st_size,'sha256':sha(p.read_bytes())}
def emit(p,v):
 assert not p.exists(),p;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v,indent=2,ensure_ascii=False)+'\n')
def descriptors(v):
 if isinstance(v,dict):
  if isinstance(v.get('path'),str) and isinstance(v.get('sha256'),str):yield v
  for x in v.values():yield from descriptors(x)
 elif isinstance(v,list):
  for x in v:yield from descriptors(x)

 old_ids=set(old_source['terms'])|set(old_ontology['terms']);new=selected-old_ids;entries={};source_terms={};source_owners={};ontology_terms={};ontology_owners={}

  if entry:
   original=Path(row['original']['path']);assert sha(original.read_bytes())==entry['sha256']==row['original']['sha256'];assert key not in old_source['sources'];assert entry['key']==key
   assert key not in entries or entries[key]==entry;entries[key]=entry;source_terms[ident]=term;source_owners[key]=row['source_metadata']
  else:
   assert ident.startswith(('CHEBI:','RHEA:','NCBITaxon:','GO:'));ontology_terms[ident]=term

 assert (set(out_source['terms'])|set(out_ontology['terms']))-old_ids==new
 return {'new_ids':sorted(new),'source_entries':[entries[k] for k in sorted(entries)],'sources':out_source,'ontology':out_ontology}
