#!/usr/bin/env python3
"""Freeze this external author package once. No maintained or prior-package writes."""
import gzip,hashlib,io,json,tarfile
from datetime import datetime,timezone
from pathlib import Path
P=Path(__file__).resolve().parents[1]
def h(b):return hashlib.sha256(b).hexdigest()
def row(p):return {'path':str(p.relative_to(P)),'bytes':p.stat().st_size,'sha256':h(p.read_bytes())}
assert not (P/'artifact-manifest.json').exists(),'Frozen package; use a new successor directory.'
assert json.loads((P/'package-verification.json').read_text())['status']=='pass'
out=P/'compact';out.mkdir(exist_ok=False)
archives=[]
groups=['originals','derived','receipts','predecessor-bindings']
for group in groups:
 files=sorted(x for x in (P/group).rglob('*')if x.is_file())
 buf=io.BytesIO();members=[]
 with tarfile.open(fileobj=buf,mode='w',format=tarfile.PAX_FORMAT)as t:
  for p in files:
   b=p.read_bytes();name=str(p.relative_to(P));info=tarfile.TarInfo(name);info.size=len(b);info.mode=0o644;info.mtime=0;info.uid=info.gid=0;info.uname=info.gname=''
   t.addfile(info,io.BytesIO(b));members.append({'path':name,'decoded_bytes':len(b),'decoded_sha256':h(b)})
 dest=out/(group+'.tar.gz');dest.write_bytes(gzip.compress(buf.getvalue(),compresslevel=9,mtime=0))
 with tarfile.open(dest,'r:gz')as t:
  for m in members:assert h(t.extractfile(m['path']).read())==m['decoded_sha256']
 archives.append({**row(dest),'encoding':'deterministic tar with gzip mtime0; member bytes unchanged','members':members})
plain=[]
for p in sorted(P.rglob('*')):
 if p.is_file()and p.relative_to(P).parts[0]not in groups+['compact']and p.name not in ['artifact-manifest.json','compact-retention-manifest.json']:
  plain.append(row(p))
cm={'candidate':'MetaCyc:PWY-5417','role':'External source-preparation compact handoff; no scientific assertion from archival format','inventory_includes_hidden_ignored':True,'archives':archives,'plain_files':plain,'decoded_member_count':sum(len(x['members'])for x in archives),'compact_total_bytes':sum(x['bytes']for x in archives)+sum(x['bytes']for x in plain)}
(P/'compact-retention-manifest.json').write_text(json.dumps(cm,indent=2)+'\n')
files=[row(p)for p in sorted(P.rglob('*'))if p.is_file()and p.name!='artifact-manifest.json']
m={'frozen_utc':datetime.now(timezone.utc).isoformat(),'candidate':'MetaCyc:PWY-5417','root':str(P),'role':'Primary-source author preparation, not independent record review','inventory_includes_hidden_ignored':True,'files':files}
(P/'artifact-manifest.json').write_text(json.dumps(m,indent=2)+'\n')
print(json.dumps({'files':len(files),'full_bytes':sum(x['bytes']for x in files),'handoff_sha256':h((P/'HANDOFF.md').read_bytes()),'artifact_manifest_sha256':h((P/'artifact-manifest.json').read_bytes()),'compact_manifest_sha256':h((P/'compact-retention-manifest.json').read_bytes()),'compact_archives':len(archives),'compact_plain_files':len(plain),'compact_total_bytes':cm['compact_total_bytes']},indent=2))
