import hashlib

from datetime import UTC, datetime

def read(p):
    return json.loads(Path(p).read_text())


def sha(p):
    with Path(p).open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()


def save(n, v):
    (D / n).write_text(json.dumps(v, indent=2) + "\n")


acq = (
    read(S / "source-manifest.json")["attempts"]
    + read(A / "source-manifest.json")["acquisitions"]
    + read(B / "acquisition1.json")
)
accepted = {x["path"]: x for x in acq if x.get("accepted") and x.get("path")}
paths = set(read(D / "staphyloferrin-b-reference-paths.json").values())
extra = [
    "PWY-8008.html",
    "NCBI-Newman-genome.xml",
    "G1I0R-73-MONOMER.xml",
    "G1I0R-74-MONOMER.xml",
    "G1I0R-71-MONOMER.xml",
    "CPD-19754.xml",
    "primary-search.json",
]
paths |= {str(S / "originals" / n) for n in extra}
for source_row in read(S / "native-reaction-projection.json"):
    for side in source_row["sides"].values():
        paths |= {str(S / "originals" / (item["id"] + ".xml")) for item in side}
paths |= {
    str(A / "originals" / n)
    for n in [
        "PubChem-122234488.json",
        "PubChem-122234488-structure.json",
        "PubChem-SID318735782.json",
        "thesis-datacite.json",
    ]
}
paths |= {x["path"] for x in read(B / "authority-inputs.json")["inputs"]}
rows = []
for p in sorted(paths):
    z = accepted.get(p, {})
    if z:
        assert sha(p) == z["sha256"]
    row = {k: z[k] for k in ["url", "retrieved_at", "content_type", "status"] if k in z}
    row.update(
        path=p,
        sha256=sha(p),
        bytes=Path(p).stat().st_size,
        hash_scope="Complete retained original response bytes",
    )
    if p.endswith("Kobylarz2016-embedded.pdf"):
        row.update(
            url=(
                "https://www.collectionscanada.gc.ca/obj/thesescanada/vol2/BVAU/TC-BVAU-57023.pdf"
            ),
            hash_scope=(
                "Exact PDF payload, bytes 822 onward of retained original HTTP-en"
                "velope container; no transcoding"
            ),
            container_path=str(A / "originals/Kobylarz-thesis2016.pdf"),
            container_sha256=sha(A / "originals/Kobylarz-thesis2016.pdf"),
        )
    elif p.endswith("rhea.rdf.gz"):
        row.update(
            url="https://ftp.expasy.org/databases/rhea/rdf/rhea.rdf.gz",
            version="Rhea 142, 2026-09-02",
            hash_scope="Entire original gzip RDF",
        )
    elif p.endswith("chebi.db"):
        row.update(
            url="https://github.com/biopragmatics/obo-db-ingest",
            version="ChEBI 255 Semantic SQL",
            hash_scope="Entire original uncompressed Semantic SQL database",
        )
    assert row.get("url"), p
    rows.append(row)
roles = {
    "native_originals": str(S / "originals"),
    "endpoint_originals": str(A / "originals"),
    "thesis_pdf": str(A / "Kobylarz2016-embedded.pdf"),
    "thesis_text": str(A / "Kobylarz2016-text.txt"),
    "rhea": str(C / "rhea.rdf.gz"),
    "chebi": str(C / "chebi.db"),
}

        "CPD-21433": "142971",
        "CPD-12590": "136993",
    },
)
chem = read(S / "chemistry-replay.json")["chemicals"]
bridge = []
for frame, number in chemical_map.items():
    original = ET.parse(str(S / "originals" / (frame + ".xml"))).find("Compound")
    native = original.findtext("inchi")
    cid = "CHEBI:" + number
    current = chem[cid]["chemrof:inchi_string"]
    kind = (
        "exact InChI"
        if native == current
        else "broader native stereochemistry"
        if frame in ["CPD-21431", "CPD-21432", "CPD-21433", "CPD-12590"]
        else "different protonation"
    )
    bridge.append(
        dict(
            native=frame,
            selected=cid,
            native_inchi=native,
            current_inchi=current,
            identity_relation=kind,
            native_chebi_xrefs=original.xpath('dblink[dblink-db="CHEBI"]/dblink-oid/text()'),
            exact_equivalence_asserted=native == current,
        )
    )
save(
    "staphyloferrin-b-native-chemical-bridges.json",
    dict(
        target_sha256=sha(D / "staphyloferrin-b-biosynthesis.yaml"),
        bridges=bridge,
        scope=(
            "Exact original structures and current chemical authority; no inf"
            "erred exactMatch for broader stereo or protonation alternatives."
            " Legacy ChEBI xrefs can resolve through aliases; strings are ret"
            "ained unchanged."
        ),
    ),
)
