@cache
def node(uri):
    needle = f'<rdf:Description rdf:about="{uri}">'
    start = 0
    facts = {}
    while (pos := rdf.find(needle, start)) >= 0:
        end = rdf.index("</rdf:Description>", pos) + 18
        elem = ET.fromstring((wrapper + rdf[pos:end] + "</root>").encode())[0]
        start = end
        for child in elem:
            key = child.tag.split("}", 1)[1]
            value = child.get("{" + RDF + "}resource") or child.text
            if value not in facts.setdefault(key, []):
                facts[key].append(value)
    assert facts, uri
    return facts


sql = sqlite3.connect("file:" + roles["chebi"] + "?mode=ro", uri=True)


@cache
def cf(cid):
    d = dict(
        sql.execute(
            ("select predicate,value from statements where subject=? and value is not null"), (cid,)
        )
    )
    return {
        k: d[k]
        for k in [
            "rdfs:label",
            "chemrof:charge",
            "chemrof:generalized_empirical_formula",
            "chemrof:inchi_string",
        ]
        if k in d
    }


def side(uri):
    n = node(uri)
    out = []
    for member in n["contains"]:
        keys = [k for k, v in n.items() if re.fullmatch("contains[0-9]+", k) and member in v]
        assert len(keys) == 1
        compound = node(member)["compound"][0]
        cid = "CHEBI:" + re.search(r"CHEBI[_:](\d+)", node(compound)["chebi"][0])[1]
        out.append(
            dict(
                side_uri=uri,
                participant_uri=member,
                compound_uri=compound,
                coefficient_predicate=RH + keys[0],
                coefficient=int(keys[0][8:]),
                id=cid,
                facts=cf(cid),
            )
        )
    return out

            assert any(
                e["reference_id"] == "DOI:10.1021/jacs.5b04557.s001" for e in edge["evidence"]
            )
            assert any(e["reference_id"] == "PubChem:122234488" for e in edge["evidence"])
