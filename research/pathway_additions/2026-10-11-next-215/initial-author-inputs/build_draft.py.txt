def ev(r, t, locator):
    assert len(t) <= 400, (len(t), t)
    return dict(reference_id=r, source_assertion=t, source_locator=locator)


def edge(s, p, o, es, d=None):
    x = dict(
        id=f"sbn-edge-{len(record['mechanistic_edges']) + 1:03d}",
        subject=s,
        predicate=p,
        object=o,
        evidence=es,
    )
    if d:
        x["description"] = d
    record["mechanistic_edges"].append(x)
    return x["id"]
