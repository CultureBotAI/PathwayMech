def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def read(path):
    return json.loads(Path(path).read_text())

def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

copied = []
external = []
mapped = []

def adopt(source, relative):
    source = Path(source)
    dest = DEST / relative
    body = source.read_bytes()
    encoding = 'byte-identical'
    if source.suffix in ['.log', '.diff']:
        dest = dest.with_name(dest.name + '.json')
        save(dest, {'original_path': str(source), 'original_sha256': sha(source), 'encoding': 'lossless UTF-8 text', 'text': body.decode('utf-8')})
        assert read(dest)['text'].encode('utf-8') == body
        encoding = 'lossless UTF-8 text in JSON'
    else:
        if dest.suffix == '.py':
            dest = dest.with_name(dest.name + '.txt')
        dest.parent.mkdir(parents=True, exist_ok=True)
        with dest.open('xb') as stream:
            stream.write(body)
        assert dest.read_bytes() == body
    copied.append({'original_path': str(source), 'original_sha256': sha(source), 'maintained_path': str(dest.relative_to(ROOT)), 'maintained_sha256': sha(dest), 'encoding': encoding})
