def fresh_sha(p):
    with Path(p).open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()


@cache
def sha(p):
    return fresh_sha(p)


def read(p):
    return json.loads(Path(p).read_text())


def norm(t):
    return " ".join(t.split())


def metadata(p):
    s = Path(p).stat()
    return dict(bytes=s.st_size, mode=s.st_mode, mtime_ns=s.st_mtime_ns)


def parse_args():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in [
        "repo-root",
        "record",
        "source-manifest",
        "source-projection",
        "authority-root",
        "output",
        "pdf-python",
        "stereo-python",
    ]:
        parser.add_argument("--" + name, required=True, type=Path)
    return parser.parse_args()

        needle = f'<rdf:Description rdf:about="{uri}">'
        start = 0
        facts = {}

            end = rdf.index("</rdf:Description>", pos) + 18
            elem = ET.fromstring((wrapper + rdf[pos:end] + "</root>").encode())[0]

                key = child.tag.split("}", 1)[1]
                value = child.get("{" + RDF + "}resource") or child.text
                if value not in facts.setdefault(key, []):
                    facts[key].append(value)
        assert facts, uri
        return facts

                    coefficient_predicate=RH + keys[0],
                    coefficient=int(keys[0][8:]),
                    id=cid,
                    facts=cf(cid),
                )
            )
        return out
