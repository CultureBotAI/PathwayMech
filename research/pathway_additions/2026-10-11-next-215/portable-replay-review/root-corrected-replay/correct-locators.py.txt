"""Apply the two reviewed sequence locator corrections; preserve initial evidence."""
from pathlib import Path
from datetime import datetime, timezone
import copy
import hashlib
import json
import yaml

root = Path('/private/tmp/pathwaymech-next-215-20261011')
dossier = root / 'research/pathway_additions/2026-10-11-next-215'
target = root / 'data/pathways/staphyloferrin-b-biosynthesis.yaml'

original_helper = Path('/private/tmp/pathwaymech-staphyloferrin-b-portable-replay-20261011/staphyloferrin-b-replay.py')

def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def save(path, value):
    with path.open('x') as f:
        json.dump(value, f, indent=2)
        f.write('\n')

assert digest(target) == '891ce5a5ba08bfae8fdedbd7709f14097f4bd04749c7755fce6d6a51841c840a'

assert digest(original_helper) == '1d5cd54c95f0116aba72de253ec421db47ec389b091ddec7269f003bd92726f2'
issue = Path('/private/tmp/pathwaymech-next-215-issue393-observation.json')
assert digest(issue) == '686816c4200237eacb1e45c2ee08785187b314233205ad89cf853bf02a735e7f'
assert json.loads(issue.read_text())['observed_issue']['state'] == 'OPEN'
initial_review = root / 'reviews/structured/20261011T144835Z-staphyloferrin-b-initial-independent-review/review.yaml'
assert digest(initial_review) == '91f17d5ec7b5f24ef91d58f85e5352c7569446f34d7eacec37a931cf393a9729'
before = target.read_text()
raw = yaml.safe_load(before)
expected = copy.deepcopy(raw)
changes = []
after = before
for edge, n in [('sbn-edge-056', 0), ('sbn-edge-058', 1)]:
    old = f'/results/{n}/comments (CATALYTIC ACTIVITY; SUBUNIT); /sequence/value'
    new = f'/results/{n}/comments (CATALYTIC ACTIVITY; SUBUNIT); /results/{n}/sequence/value'
    assert after.count(old) == 1
    item = next(e for e in expected['mechanistic_edges'] if e['id'] == edge)['evidence'][1]
    assert item['source_locator'] == old
    item['source_locator'] = new
    after = after.replace(old, new)
    changes.append({'edge': edge, 'field': 'evidence[1].source_locator', 'before': old, 'after': new})
assert yaml.safe_load(after) == expected
initial_projection = json.loads(projection.read_text())
assert initial_projection['target_sha256'] == digest(target)
assert initial_projection['graph_to_source'] == raw['mechanistic_edges']
successor_projection = copy.deepcopy(initial_projection)
successor_projection['target_sha256'] = hashlib.sha256(after.encode()).hexdigest()
successor_projection['graph_to_source'] = expected['mechanistic_edges']
