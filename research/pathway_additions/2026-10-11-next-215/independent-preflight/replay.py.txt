OUT = Path(__file__).parent

E = Path('/private/tmp/pathwaymech-staphyloferrin-b-endpoint-20261011/originals')
start = datetime.now(timezone.utc).isoformat()
def read(p): return json.loads(Path(p).read_text())
def sha(p):
    with Path(p).open('rb') as f: return hashlib.file_digest(f, 'sha256').hexdigest()
doc = yaml.safe_load((A/'staphyloferrin-b-biosynthesis.yaml').read_text())
target_sha = sha(A/'staphyloferrin-b-biosynthesis.yaml')
assert target_sha == '891ce5a5ba08bfae8fdedbd7709f14097f4bd04749c7755fce6d6a51841c840a'
inputs = read(OUT/'inputs.json')
for item in inputs['author_artifacts']: assert sha(item['path']) == item['sha256'], item['path']
projection = read(A/'staphyloferrin-b-source-projection.json')
assert projection['graph_to_source'] == doc['mechanistic_edges']
edges = {e['id']: e for e in doc['mechanistic_edges']}
RH = 'http://rdf.rhea-db.org/'
RDF = 'http://www.w3.org/1999/02/22-rdf-syntax-ns#'
pairs = {'RXN-19819':'59113','RXN-18395':'52385','RXN-18393':'51929','RXN-19820':'59125','RXN-19821':'59121','RXN-19822':'59129','RXN-19823':'59133'}
masters = {str(int(r)-1) for r in pairs.values()}
facts = defaultdict(lambda: defaultdict(set))
with gzip.open('/private/tmp/pathwaymech-curation-authorities/rhea.rdf.gz', 'rb') as stream:
    for _, x in ET.iterparse(stream, events=['end'], tag='{'+RDF+'}Description'):
        uri = x.get('{'+RDF+'}about', '')
        key = uri.removeprefix(RH)
        selected = key in pairs.values() or key in masters or key.startswith('Compound_') or any(key.startswith(m+'_') or key.startswith('Participant_'+m+'_') for m in masters)
        if selected:
            for child in x:
                v = child.get('{'+RDF+'}resource') or child.text
                if v is not None: facts[uri][ET.QName(child).localname].add(v)
        x.clear()
        while x.getprevious() is not None: del x.getparent()[0]
def one(uri, prop):
    values = facts[uri][prop]
    assert len(values)==1, (uri,prop,values)
    return next(iter(values))
sql = sqlite3.connect('file:/private/tmp/pathwaymech-curation-authorities/chebi.db?mode=ro',uri=True)
def chemical(cid):
    rows = dict(sql.execute('select predicate,value from statements where subject=? and value is not null',(cid,)))
    return {k:rows[k] for k in ['rdfs:label','chemrof:generalized_empirical_formula','chemrof:charge','chemrof:inchi_string'] if k in rows}
def vector(cid):
    f = chemical(cid); form=f['chemrof:generalized_empirical_formula']
    atoms=re.findall(r'([A-Z][a-z]?)(\d*)',form)
    assert ''.join(a+b for a,b in atoms)==form
    return {**{a:int(b or 1) for a,b in atoms},'charge':int(f['chemrof:charge'])}
all_sides=[]; rxn_sides={}; net=Counter()
for native, r in pairs.items():
    direction=RH+r; master=RH+str(int(r)-1)
    assert direction in facts[master]['directionalReaction']
    local=Counter(); sides={}
    for role, sign, ending in [('substrates',-1,'L'),('products',1,'R')]:
        side=one(direction,role); assert side==master+'_'+ending
        side_items={}
        for participant in facts[side]['contains']:
            coefficients=[int(k[8:]) for k,v in facts[side].items() if re.fullmatch('contains[0-9]+',k) and participant in v]
            assert len(coefficients)==1
            coefficient=coefficients[0]; compound=one(participant,'compound')
            cid='CHEBI:'+re.search(r'CHEBI[_:](\d+)',one(compound,'chebi'))[1]
            side_items[cid]=coefficient
            triple=(cid,'consumes','RHEA:'+r) if sign==-1 else ('RHEA:'+r,'produces',cid)
            matching=[e for e in edges.values() if (e['subject'],e['predicate'],e['object'])==triple]
            assert len(matching)==1
            edge=matching[0]; first=edge['evidence'][0]
            assert first['reference_id']=='RHEA:'+r
            assert all(s in first['source_locator'] for s in [side,participant,compound,'rh:contains'+str(coefficient)])
            assert f'coefficient {coefficient}' in first['source_assertion']
            for a,count in vector(cid).items(): local[a]+=sign*coefficient*count
            net[cid]+=sign*coefficient*(2 if native in ['RXN-19819','RXN-18395','RXN-18393'] else 1)
            all_sides.append(dict(edge=edge['id'],reaction='RHEA:'+r,role=role,id=cid,coefficient=coefficient,side=side,participant=participant,compound=compound))
        sides[role]=side_items
    assert not {a:n for a,n in local.items() if n}, (r,local)
    rxn_sides['RHEA:'+r]=sides
net={c:n for c,n in net.items() if n}
assert net==projection['formal_net']
assert len(all_sides)==40
assert len([e for e in edges.values() if e['predicate'] in ['consumes','produces']])==40
native=ET.parse(str(N/'PWY-8008.xml')).find('Pathway')
assert set(native.xpath('reaction-list/Reaction/@frameid'))==set(pairs)
native_rows=[]
for n,r in pairs.items():
    x=ET.parse(str(N/(n+'.xml'))).find('Reaction')
    sides={s:[{'id':c[0].get('frameid'),'coefficient':c.findtext('coefficient') or '1'} for c in x.findall(s)] for s in ['left','right']}
    row=next(p for p in projection['source_to_graph'] if p['native']==n)
    assert row['native_sides']==sides
    assert x.findtext('reaction-direction')==row['native_direction']=='PHYSIOL-LEFT-TO-RIGHT'
    native_rows.append({'native':n,'reaction':'RHEA:'+r,'sides':sides})

    assert b['exact_equivalence_asserted']==(b['native_inchi']==b['current_inchi'])
for e in edges.values():
    if e['predicate']=='precedes':
        shared=set(rxn_sides[e['subject']]['products']) & set(rxn_sides[e['object']]['substrates'])
        assert shared
        for v in e['evidence']:

            assert uris and all(u in facts for u in uris)
up=read(N/'UniProt-sbn-NCTC8325.json')
proteins={x.findtext('GBSeq_accession-version'):x for x in ET.parse(str(N/'NCBI-Newman-sbn-proteins.xml')).getroot()}

translations={}
for feature in genome.findall('.//GBFeature'):

    if 'protein_id' in qualifiers: translations[qualifiers['protein_id']]=qualifiers
identity=[]
for e in edges.values():
    if e['predicate']!='enables': continue
    accession=e['subject'].split(':',1)[1]; prot=proteins[accession]
    q=[(n.findtext('GBQualifier_name'),n.findtext('GBQualifier_value')) for n in prot.findall('.//GBQualifier')]
    assert ('strain','Newman') in q and ('db_xref','taxon:426430') in q
    upid=next(v['reference_id'].split(':')[1] for v in e['evidence'] if v['reference_id'].startswith('UniProtKB:'))
    current=next(x for x in up['results'] if x['primaryAccession']==upid)
    sequence=prot.findtext('GBSeq_sequence').upper()
    assert sequence==translations[accession]['translation']==current['sequence']['value']
    assert current['organism']['taxonId']==93061
    activity=next(c for c in current['comments'] if c['commentType']=='CATALYTIC ACTIVITY')
    assert any(p['reactionCrossReference']['id']==e['object'] and p['directionType']=='left-to-right' for p in activity['physiologicalReactions'])
    identity.append({'edge':e['id'],'Newman':accession,'NCTC8325':upid,'length':len(sequence),'native_sequence_equal':True,'construct_identity_claim':False})
assembly=[]
for c,m,u,r in [('CPLX-9312','G1I0R-77-MONOMER','Q2G1M5','RXN-19819'),('CPLX-9313','G1I0R-70-MONOMER','Q2G1N2','RXN-18393')]:
    x=ET.parse(str(N/(c+'.xml'))).find('Protein')
