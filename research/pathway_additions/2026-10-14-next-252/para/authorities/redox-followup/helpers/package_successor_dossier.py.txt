from pathlib import Path
import hashlib, json, shutil, datetime

W = Path('/private/tmp/pathwaymech-next-252-20261014')
D = W / 'research/pathway_additions/2026-10-14-next-252/para'
E = Path('/private/tmp/pathwaymech-next-252-primary/dossier-successor-packaging')
E.mkdir(exist_ok=True)
NOW = datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def write(p, data):
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(data, indent=2) + '\n')
mutable = {'curation.md', 'reproduction.md', 'identifiers.json', 'reference-bindings.json'}
initial = []
for p in sorted(D.rglob('*')):
    if p.is_file():
        rel = p.relative_to(D).as_posix()
        initial.append({'path': rel, 'sha256': sha(p), 'bytes': p.stat().st_size, 'authorized_mutation': rel in mutable})
        if rel in mutable:
            shutil.copyfile(p, E / ('initial-' + rel))
write(E / 'initial-dossier-files.json', {'created_utc': NOW, 'files': initial})

def retain(area, src, items, role, provenance=None):
    dest = D / area
    dest.mkdir(parents=True, exist_ok=True)
    retained = []
    for name, outname in items:
        original = src / name
        target = dest / outname
        target.parent.mkdir(parents=True, exist_ok=True)
        assert not target.exists(), str(target)
        shutil.copyfile(original, target)
        assert original.read_bytes() == target.read_bytes()
        retained.append({'external_path': str(original), 'maintained_path': target.relative_to(W).as_posix(), 'sha256': sha(target), 'bytes': target.stat().st_size, 'unchanged_bytes': True, 'kind': 'helper_source_artifact' if outname.endswith('.py.txt') else 'compact_evidence'})
    obj = {'created_utc': NOW, 'area': area, 'role': role, 'retained': retained}
    if provenance:
        obj['provenance'] = provenance
    write(dest / 'retention.json', obj)
    return len(retained)
