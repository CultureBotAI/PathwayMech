import collections

ROOT = Path(__file__).parent
RAW = ROOT / 'raw'
REPO = Path('/private/tmp/pathwaymech-next-216-20261011')

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def save(name, obj):
    with (ROOT / name).open('x') as f:
        json.dump(obj, f, indent=2)
        f.write('\n')

started = datetime.now(timezone.utc).isoformat()
sources = []
for acquisition in sorted(ROOT.glob('acquisition-*.json')):
    for row in json.loads(acquisition.read_text()):
        assert sha(row['path']) == row['sha256']
        row = dict(row)
        row['acquisition_receipt'] = str(acquisition)
        row['accepted_body'] = row['status'] == 200
        if row['name'].endswith('.pdf') or row['name'].startswith('rhea-'):
            row['accepted_body'] = False
            row['disposition'] = 'Rejected as requested PDF/RDF: response is HTML/access failure, not the requested source format.'
        elif row['name'] == 'PMC276539.xml':
            row['disposition'] = 'Rejected fullTextXML HTTP500 JSON error; no full-text claim.'
        elif row['name'] in ['PMC276539.html', 'PMC206259.html']:
            row['disposition'] = 'Original article landing HTML supplies abstract/metadata and scan links; no complete scanned primary PDF was retrieved.'
        elif row['name'] == 'P00350.json':
            row['disposition'] = 'Inspected contextual gnd protein from existing PPP record; excluded from ED route. Zwf is P0AC53, not P00350.'
        else:
            row['disposition'] = 'Accepted original response at the stated URL; evidence tier and inspected scope remain assertion-specific.'
        sources.append(row)

save('source-manifest.json', {'observed_at': started, 'original_responses': sources, 'existing_originals': source_context})
