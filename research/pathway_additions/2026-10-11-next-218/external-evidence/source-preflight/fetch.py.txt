import concurrent.futures
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

import requests

ROOT = Path(__file__).parent
ITEMS = [
    ('ENTNER-DOUDOROFF-PWY.xml', 'https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=META:ENTNER-DOUDOROFF-PWY'),
    ('ENTNER-DOUDOROFF-PWY.html', 'https://bf2i-artsymbiocyc.insa-lyon.fr/META/NEW-IMAGE?type=PATHWAY&object=ENTNER-DOUDOROFF-PWY'),
    ('M00008.txt', 'https://rest.kegg.jp/get/M00008'),
    ('primary-metadata.json', 'https://www.ebi.ac.uk/europepmc/webservices/rest/search?query=EXT_ID:1624451%20OR%20EXT_ID:5337840%20OR%20EXT_ID:15659677%20OR%20EXT_ID:8344525&format=json&resultType=core'),
    ('PMC276539.xml', 'https://www.ebi.ac.uk/europepmc/webservices/rest/PMC276539/fullTextXML'),
]

def fetch(item):
    name, url = item
    row = {'name': name, 'url': url, 'started_at': datetime.now(timezone.utc).isoformat()}
    try:
        r = requests.get(url, timeout=60)
        p = ROOT / 'raw' / name
        with p.open('xb') as f:
            f.write(r.content)
        row.update(status=r.status_code, final_url=r.url, content_type=r.headers.get('Content-Type'), bytes=len(r.content), sha256=hashlib.sha256(r.content).hexdigest(), path=str(p))
    except Exception as e:
        row['error'] = repr(e)
    row['finished_at'] = datetime.now(timezone.utc).isoformat()
    return row

with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    rows = list(pool.map(fetch, ITEMS))
with (ROOT / 'acquisition-1.json').open('x') as f:
    json.dump(rows, f, indent=2)
print(json.dumps(rows, indent=2))
