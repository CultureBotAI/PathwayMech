files = {}
deferred_sources = []


def add(path, role, expected=None, defer=False):
    path = driver.absolute(path)
    driver.require(path.is_file(), "Missing binding input: " + str(path))
    if defer:
        driver.require(expected is not None, "Deferred source needs accepted digest")
        actual = expected
        deferred_sources.append(str(path))
    else:
        actual = driver.file_hash(path)
        driver.require(
            expected is None or actual == expected, "Accepted input changed: " + str(path)
        )
    entry = {
        "path": str(path),
        "resolved_path": str(path.resolve(strict=True)),
        "sha256": actual,
        "role": role,
    }
    if str(path) in files:
        driver.require(files[str(path)]["sha256"] == actual, "Conflicting expected digest")
    else:
        files[str(path)] = entry

]:
    paths = driver.inventory(root)
    inventories.append({"root": str(root), "paths": paths})
    for path in paths:
        add(path, role)

    add(REPO / relative, role)
add(REPO / driver.TARGET, "record", driver.TARGET_SHA)

        files[str(REPO / relative)]["sha256"] == expected, "Root accepted production input changed"
    )

for item in proposal["source_inputs"] + proposal["verified_existing_source_inputs"]:

# Root acceptance supplies a distinct production authority selection. Capture its
# inputs without rewriting original scientific versions or re-running its builder.
def accepted_proof(path):
    raw = Path(path).read_bytes()
    driver.require(
        driver.digest(raw) == proofs[path], "Root/selection proof changed before parsing"
    )
    return driver.read_json(raw)


review = accepted_proof("/private/tmp/pathwaymech-next-218-authority-root-review.json")
core = accepted_proof("/private/tmp/pathwaymech-next-218-core-adoption.json")

    "/private/tmp/pathwaymech-ed-authority-selection-20261011/selected-successor/selection.json"
)

for item in review["database_checks"] + core["cache_sources"]:
    production_sources[item["path"]] = {"path": item["path"], "sha256": item["sha256"]}
for item in selection["reused_inputs"]:
    if "selected_existing_body" in item:
        production_sources[item["selected_existing_body"]] = {
            "path": item["selected_existing_body"],
            "sha256": item["existing_entry_preserved"]["sha256"],
        }
for item in production_sources.values():
    add(item["path"], "source", item["sha256"], defer=True)

# Stdlib-only subprocess reports environment identity and distribution metadata.
# It imports no pathwaymech, YAML, LinkML, lxml, or RDKIT implementation.
probe = """import importlib.metadata as m,json,sys
from pathlib import Path
distributions=[];metadata=[]
for d in m.distributions():
    distributions.append({"name":d.metadata.get("Name"),"version":d.version})
    for f in d.files or []:
        p=Path(d.locate_file(f))
        if any(x.endswith((".dist-info",".egg-info")) for x in p.parts) and p.is_file():
            metadata.append(str(p.absolute()))
for site in sys.path:
    p=Path(site)
    if p.is_dir() and p.name in ("site-packages","dist-packages"):
        metadata.extend(str(x.absolute()) for x in p.glob("*.pth") if x.is_file())
v=Path(sys.prefix)/"pyvenv.cfg"
if v.is_file():metadata.append(str(v.absolute()))
print(json.dumps({"executable":sys.executable,"version":sys.version,"prefix":sys.prefix,"base_prefix":sys.base_prefix,"sys_path":sys.path,"distributions":sorted(distributions,key=lambda x:(x["name"] or "",x["version"])),"metadata_files":sorted(set(metadata))}))
"""

env.pop("PYTHONHOME", None)

        cwd=REPO,

        check=True,

        executable,
        "environment",
        "ad67715379ea955de1a1f4405673acd056097a261eccdf86f1f220ccd1fdfd98"
        if name == "rdkit_python"
        else None,
    )
    for path in value["metadata_files"]:
        add(path, "environment")
    provenance.append((name, value))
