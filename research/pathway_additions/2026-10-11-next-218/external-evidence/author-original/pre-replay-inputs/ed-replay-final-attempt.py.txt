def fields(raw):
    out = {}
    for line in raw.decode().splitlines():
        if line == "///":
            break
        if line[:12].strip():
            key = line[:12].strip()
            out[key] = []
        out[key].append(line[12:].strip())
    return {k: "\n".join(v) for k, v in out.items()}

def chemical_graph(rec):
    assert {x["id"] for x in rec["reactions"]} == set(expected)
    assert rec["id"] == "KEGG:M00008"
    st = {rid: Counter() for rid in expected}
    for edge in rec["mechanistic_edges"]:
        pred = edge["predicate"]
        if pred not in ["consumes", "produces"]:
            continue
        rid, cid, sign = (
            (edge["object"], edge["subject"], -1)
            if pred == "consumes"
            else (edge["subject"], edge["object"], 1)
        )
        assert rid in st
        coef = int(re.match(r"Coefficient: (\d+)\.", edge["description"])[1])
        st[rid][cid] += sign * coef
    assert {k: dict(v) for k, v in st.items()} == expected
    edges = rec["mechanistic_edges"]
    membership = {(e["subject"], e["object"]) for e in edges if e["predicate"] == "part_of"}
    assert membership == {(rid, "KEGG:M00008") for rid in expected}
    bridges = {(e["subject"], e["object"]) for e in edges if e["predicate"] == "provides_input_for"}
    assert bridges == {
        ("KEGG:R02736", "RHEA:12557"),
        ("KEGG:R10907", "RHEA:12557"),
        ("RHEA:12557", "RHEA:17278"),
        ("RHEA:17278", "RHEA:17090"),
    }
    assert not any(e["predicate"] == "precedes" for e in edges)
    assembly = [e for e in edges if e["predicate"] == "has_part"]
    assert (
        len(assembly) == 1
        and assembly[0]["subject"] == complex_ref
        and assembly[0]["object"] == "UniProtKB:P0A955"
    )
    assert "Stoichiometry: 3" in assembly[0]["description"]
    cofac = [e for e in edges if e["predicate"] == "has_cofactor"]
    assert (
        len(cofac) == 1
        and cofac[0]["subject"] == "UniProtKB:P0ADF6"
        and cofac[0]["object"] == "CHEBI:49883"
    )
    assert "inference tier" in cofac[0]["description"]
    for e in edges:
        if e["subject"] == "UniProtKB:P0AC53" and e["predicate"] == "enables":
            assert "qualified" in json.dumps(e).lower() and "exclusive" in json.dumps(e).lower()
    assert not any(e["predicate"] == "inhibits" for e in edges)
    balances = {}
    for rid, sto in st.items():
        res = Counter()
        for cid, coef in sto.items():
            formula = cv(cid, "chemrof:generalized_empirical_formula")
            for atom, n in re.findall(r"([A-Z][a-z]?)(\d*)", formula):
                res[atom] += coef * int(n or 1)
            res["charge"] += coef * int(cv(cid, "chemrof:charge"))
        assert not any(res.values()), (rid, res)
        balances[rid] = dict(res)
    nets = {}
    for first in ["KEGG:R02736", "KEGG:R10907"]:
        net = Counter()
        for rid in [first, "RHEA:12557", "RHEA:17278", "RHEA:17090"]:
            net.update(st[rid])
        nets[first] = {k: v for k, v in net.items() if v}
        assert (
            nets[first]["CHEBI:15378"] == 2
            and nets[first]["CHEBI:59776"] == nets[first]["CHEBI:15361"] == 1
        )
        assert "CHEBI:15377" not in nets[first]
    return balances, nets
