from pathlib import Path
import urllib.request,urllib.parse,hashlib,json,datetime
p=Path('/private/tmp/pathwaymech-entner-doudoroff-independent-adjudication-20261011')
items=[('PMC4807051.xml','https://www.ebi.ac.uk/europepmc/webservices/rest/PMC4807051/fullTextXML'),('pone0152403.xml','https://journals.plos.org/plosone/article/file?id=10.1371/journal.pone.0152403&type=manuscript'),('metadata22519976.json','https://www.ebi.ac.uk/europepmc/webservices/rest/search?query=EXT_ID:22519976&format=json&resultType=core')]
rows=[]
for n,u in items:
 r={'url':u,'name':n,'at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'PathwayMech-source-review/1.0'}),timeout=35) as f:
   b=f.read();v=urllib.parse.urlsplit(f.url);r.update(status=f.status,final_url=urllib.parse.urlunsplit((v.scheme,v.netloc,v.path,'','')),redirect_query_redacted=bool(v.query),content_type=f.headers.get('Content-Type'))
  with (p/'originals'/n).open('xb') as f:f.write(b)
  r.update(path=str(p/'originals'/n),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
 except Exception as e:r['error']=str(e)
 rows.append(r)
with (p/'acquisition.json').open('x') as f:json.dump(rows,f,indent=2)
print(json.dumps(rows,indent=2))
