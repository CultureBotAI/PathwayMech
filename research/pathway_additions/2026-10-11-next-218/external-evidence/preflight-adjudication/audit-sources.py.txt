def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def fields(path):
    out = {}
    for line in Path(path).read_text().splitlines():
        if line == '///':
            break
        key, value = line[:12].strip(), line[12:].strip()
        if key:
            current = key
            out[current] = []
        out[current].append(value)
    return {k: '\n'.join(v) for k, v in out.items()}

# Independently parse the original complete Rhea RDF, not the prospective JSON.
rdf = '{http://www.w3.org/1999/02/22-rdf-syntax-ns#}'
rh = 'http://rdf.rhea-db.org/'
graph = {}
with gzip.open(CACHE/'rhea.rdf.gz', 'rb') as stream:
    for _, elem in etree.iterparse(stream, events=('end',), tag=rdf+'Description'):
        uri = elem.get(rdf+'about')
        if uri:
            node = graph.setdefault(uri, {})
            for child in elem:
                pred = etree.QName(child).localname
                value = child.get(rdf+'resource') or (child.text or '').strip()
                if value not in node.setdefault(pred, []):
                    node[pred].append(value)
        elem.clear()
        while elem.getprevious() is not None:
            del elem.getparent()[0]
native = {}
for master, directed in [('15841','15842'),('12556','12557'),('17277','17278'),('17089','17090')]:
    assert rh+directed in graph[rh+master]['directionalReaction']
    values = collections.Counter()
    for pred, sign in [('substrates', -1), ('products', 1)]:
        side = graph[rh+directed][pred][0]
        for contains, participants in graph[side].items():
            if re.fullmatch(r'contains\d+', contains):
                for participant in participants:
                    compound = graph[graph[participant]['compound'][0]]
                    cid = compound['accession'][0]
                    values[cid] += sign * int(contains[8:])
                    if cid in terms:
                        assert compound['formula'][0] == terms[cid]['chemrof:generalized_empirical_formula']
                        assert compound['charge'][0] == terms[cid]['chemrof:charge']
    native[directed] = dict(values)
assert native['15842']['CHEBI:61548'] == -1
first = dict(native['15842'])

first['CHEBI:58247'] = -1

eq = {'KEGG:R02736': first, 'KEGG:R10907': nad, **{'RHEA:'+k:v for k,v in native.items() if k!='15842'}}
balances = {}
for name, equation in eq.items():
    residual = collections.Counter()
    for cid, coefficient in equation.items():
        term = terms[cid]
        formula = term['chemrof:generalized_empirical_formula']
        atoms = re.findall(r'([A-Z][a-z]?)(\d*)', formula)
        assert ''.join(a+b for a,b in atoms) == formula
        for atom, count in atoms:
            residual[atom] += coefficient * int(count or 1)
        residual['charge'] += coefficient * int(term['chemrof:charge'])
    assert not any(residual.values()), (name,residual)
    balances[name] = dict(residual)
