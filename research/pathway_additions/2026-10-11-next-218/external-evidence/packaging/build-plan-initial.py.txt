def sha(p):
    with Path(p).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def capture(p, expected=None):
    p = Path(p).absolute()
    rp = p.resolve(strict=True)
    raw = p.read_bytes()
    h = hashlib.sha256(raw).hexdigest()
    if expected is not None and expected != h:
        raise ValueError('Source digest mismatch: ' + str(p))
    if str(p) in bindings and bindings[str(p)] != (str(rp), h):
        raise ValueError('Changed captured input: ' + str(p))
    bindings[str(p)] = (str(rp), h)
    return raw

def read(p):
    return json.loads(capture(p))

def save(p, data):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('x') as stream:
        json.dump(data, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

def whitespace(raw):
    return bool(raw) and (any(line.rstrip(b'\r\n').endswith((b' ', b'\t')) for line in raw.splitlines(keepends=True)) or raw.endswith(b'\n\n'))

def add(p, dest, role, *, wrap=False, active=False, unique=False):
    p = Path(p).absolute()
    if str(p) in paths:
        return paths[str(p)]
    raw = capture(p)
    h = hashlib.sha256(raw).hexdigest()
    dest = Path(dest)
    if not active and dest.suffix in {'.py', '.yaml', '.yml', '.log'}:
        dest = dest.with_name(dest.name + '.txt')
    wrap = wrap or p.suffix in {'.log', '.tsv', '.jsonl', '.diff'} or '.yaml' in p.name or '.yml' in p.name or whitespace(raw)
    if active and wrap:
        raise ValueError('Active input requires byte-changing normalization: ' + str(p))
    mode = 'lossless_json_wrapper' if wrap else 'exact_bytes'
    key = (h, mode)
    if key in canon and not unique:
        row = dict(path=str(p), sha256=h, bytes=len(raw), disposition='exact-byte-alias', destination=canon[key], mode=mode, role=role)
        aliases.append(row)
        paths[str(p)] = row
        return row
    source = p
    if wrap:
        try:
            content = raw.decode('utf-8')
            assert content.encode('utf-8') == raw
            encoding = 'utf-8'
        except UnicodeDecodeError:
            content, encoding = base64.b64encode(raw).decode('ascii'), 'base64'
        source = HERE / 'payload' / (h + '.json')
        if not source.exists():
            save(source, dict(format='lossless-immutable-evidence-v1', original_sha256=h, original_bytes=len(raw), encoding=encoding, content=content))
        dest = dest.with_name(dest.name + '.json')
    if dest.is_absolute() or '..' in dest.parts or any(x['destination'] == str(dest) for x in copies):
        raise ValueError('Invalid/colliding destination: ' + str(dest))
    if dest == DOSSIER / 'README.md':
        raise ValueError('Root owns dossier README.md')
    row = dict(path=str(p), resolved_path=str(p.resolve()), sha256=h, bytes=len(raw), disposition='planned-copy', destination=str(dest), copy_source=str(source), destination_sha256=sha(source), mode=mode, role=role)
    copies.append(row)
    canon.setdefault(key, str(dest))
    paths[str(p)] = row
    return row

def archive(p, group, rel=None, **kw):
    return add(p, DOSSIER / 'external-evidence' / group / (rel or Path(p).name), 'Immutable external observation; not a historical canonical review', **kw)

def external(p, reason, expected=None):
    p = Path(p).absolute()
    if str(p) in paths:
        return paths[str(p)]
    raw = capture(p, expected)
    row = dict(path=str(p), resolved_path=str(p.resolve()), sha256=hashlib.sha256(raw).hexdigest(), bytes=len(raw), disposition='retained-external', reason=reason)
    excluded.append(row)
    paths[str(p)] = row
    return row
