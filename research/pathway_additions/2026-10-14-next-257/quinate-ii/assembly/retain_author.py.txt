from pathlib import Path
import datetime
import hashlib
import json
import shutil

B = Path(__file__).parent
R = Path('/private/tmp/pathwaymech-next-257-20261014')
D = R / 'research/pathway_additions/2026-10-14-next-257/quinate-ii'

def retain(sources, target):
    target.mkdir(parents=True, exist_ok=True)
    rows = []
    for path in sources:
        assert path.is_file()
        name = path.name + '.txt' if path.suffix == '.py' else path.name
        dst = target / name
        if dst.exists():
            assert dst.read_bytes() == path.read_bytes(), ('Changed retained file', dst)
        else:
            shutil.copyfile(path, dst)
        rows.append(dict(original_path=str(path), maintained_path=str(dst.relative_to(R)),
                         bytes=path.stat().st_size, sha256=hashlib.sha256(path.read_bytes()).hexdigest()))
    manifest = target / 'retention.json'
    assert not manifest.exists()
    manifest.write_text(json.dumps(dict(observed_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
        scope='Compact author artifacts, not independent acceptance. Full original downloads remain external and hash-bound in the source manifests.',
        files=rows), indent=2) + '\n')
    return rows

primary = Path('/private/tmp/pathwaymech-next-257-primary/package')
primary_rows = retain(sorted(p for p in primary.iterdir() if p.is_file()), D / 'primary')
integration_rows = retain([B / name for name in [
    'extend_core.py', 'extended-core-record.json', 'assemble_record.py', 'assembled-record.json',
    'integration-dispositions.json', 'prospective-regulatory-and-metal-identities.json', 'retain_author.py']], D / 'assembly')
print(json.dumps({'primary_artifacts': len(primary_rows), 'assembly_artifacts': len(integration_rows)}))
