from pathlib import Path
import ast
import datetime
import hashlib
import json
import shutil

B = Path(__file__).parent
DEST = Path('/private/tmp/pathwaymech-next-257-20261014/research/pathway_additions/2026-10-14-next-257/quinate-ii/preflight')
reports = [B / 'dedup-compact.json', B / 'supplemental-search.json', B / 'late-source-supplement/supplemental-search.json', B / 'final-primary-supplement/supplemental-search.json', B / 'assembled-record-supplement/supplemental-search.json']
all_patterns = set()
passes = []


def bind(path):
    data = path.read_bytes()
    return {'path': str(path), 'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}


for path in reports:
    report = json.loads(path.read_text())
    patterns = Path(report['patterns_file']['path'])
    assert bind(patterns)['sha256'] == report['patterns_file']['sha256']
    all_patterns.update(patterns.read_text().splitlines())
    for root in report['roots']:
        assert root['exit_code'] in (0, 1) and not root['stderr']
        for name in ['full_match_projection', 'complete_counts']:
            assert bind(Path(root[name]['path']))['sha256'] == root[name]['sha256']
    passes.append({'report': bind(path), 'patterns_per_root': report['patterns_per_root'], 'roots': [root['root'] for root in report['roots']]})
baseline = json.loads((B / 'baseline-record-bindings.json').read_text())
assert baseline['count'] == 253
assert not any(row['id'] in ['MetaCyc:PWY-6416', 'MetaCyc:PWY-6419'] for row in baseline['records'])
source_bindings = []
for path in [B / 'search-inputs.json', B / 'supplemental-source-bindings.json', B / 'late-source-supplement/supplemental-source-bindings.json', B / 'final-primary-supplement/supplemental-source-bindings.json', B / 'assembled-record-supplement/supplemental-source-bindings.json']:
    for row in json.loads(path.read_text()):
        actual = bind(Path(row['path']))
        assert actual['sha256'] == row['sha256'], row['path']
        source_bindings.append(row)
unique_sources = {row['path']: row for row in source_bindings}
meta_xml = [path for path in unique_sources if Path(path).name.startswith('META-') and path.endswith('.xml')]
union = {'baseline_commit': baseline['baseline_commit'], 'candidate': 'MetaCyc:PWY-6416', 'baseline_records': 253, 'distinct_fixed_strings': len(all_patterns), 'casefold_distinct_strings': len({value.casefold() for value in all_patterns}), 'passes': passes, 'all_search_source_bindings_verified': len(source_bindings), 'distinct_search_source_paths': len(unique_sources), 'distinct_complete_meta_xml_inputs': len(meta_xml), 'ignored_and_hidden_included': True, 'scope': 'Two current repository roots only, five complete passes per root. Text-search bounds are documented in dedup-compact.json. The authoritative membership claim is from all 253 baseline Git blobs; no provider or source-universe exhaustion claim.', 'created_utc': datetime.datetime.now(datetime.timezone.utc).isoformat()}
(B / 'search-union.json').write_text(json.dumps(union, indent=2) + '\n')

for name in names + helpers:
    path = B / name
    data = path.read_bytes()
    if name.endswith('.py'):
        ast.parse(data)
        assert all(line == line.rstrip() for line in data.decode().splitlines()), name
    dest = DEST / (name + '.txt' if name.endswith('.py') else name)
    dest.parent.mkdir(parents=True, exist_ok=True)
    if dest.exists():
        assert dest.read_bytes() == data, str(dest)
    else:
        dest.write_bytes(data)
    retained.append({'original': bind(path), 'retained': bind(dest), 'byte_identical': dest.read_bytes() == data})
