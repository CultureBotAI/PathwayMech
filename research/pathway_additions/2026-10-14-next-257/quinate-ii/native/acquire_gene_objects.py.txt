from acquire import *
import urllib.parse,time
ids=['4994226','4994327','4994407','4993973']
rows=[]
for a in ids:
 name='NCBI-gene-'+a;p=ORIG/(name+'.xml');url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?'+urllib.parse.urlencode({'db':'gene','id':a,'retmode':'xml'})
 row={'name':name,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url)
  ET.fromstring(b);p.write_bytes(b);row.update(path=str(p),bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
 except Exception as e:row['error']=str(e)
 rows.append(row);time.sleep(.5)
(ROOT/'acquisition-gene-objects.json').write_text(json.dumps(rows,indent=2)+'\n'); print([(r['name'],r.get('bytes',r.get('error'))) for r in rows])
