from acquire import *
import urllib.parse,time
items=[('NCBIProtein-'+a,'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?'+urllib.parse.urlencode({'db':'protein','id':a,'retmode':'xml'}),True) for a in ['BAF53458.1','BAF53459.1','BAF53460.1','WP_003855554.1']]
items.append(('IPG-WP_003855554.1','https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?'+urllib.parse.urlencode({'db':'protein','id':'WP_003855554.1','rettype':'ipg','retmode':'text'}),False))
rows=[]
for name,url,isxml in items:
 p=ORIG/(name+('.xml' if isxml else '.tsv'));row={'name':name,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url)
  if isxml:ET.fromstring(b)
  p.write_bytes(b);row.update(path=str(p),bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
 except Exception as e:row['error']=str(e)
 rows.append(row);time.sleep(.6)
(ROOT/'acquisition-protein-objects.json').write_text(json.dumps(rows,indent=2)+'\n');print([(r['name'],r.get('bytes',r.get('error'))) for r in rows])
