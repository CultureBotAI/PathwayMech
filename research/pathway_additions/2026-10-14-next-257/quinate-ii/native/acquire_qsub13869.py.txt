from acquire import *
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=protein&id=1045036213&retmode=xml'
row={'name':'NCBIProtein-ATCC13869-candidate','url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url)
r=ET.fromstring(b); a=r.findtext('./GBSeq/GBSeq_accession-version');p=ORIG/('NCBIProtein-'+a+'.xml');p.write_bytes(b);row.update(path=str(p),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
(ROOT/'acquisition-qsub13869-protein.json').write_text(json.dumps(row,indent=2)+'\n');print(row)
print(ET.tostring(r,encoding='unicode')[:18000])
