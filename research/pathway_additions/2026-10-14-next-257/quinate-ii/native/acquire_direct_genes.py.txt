from acquire import *
rows=[]
for fid in ['G-11737','G-11738']:
 url='https://bf2i-artsymbiocyc.insa-lyon.fr/gene?orgid=META&id='+fid
 row={'frameid':fid,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url)
  f=ORIG/('META-'+fid+'-gene.html');f.write_bytes(b);row.update(path=str(f),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
 except Exception as e:row['error']=str(e)
 rows.append(row)
(ROOT/'acquisition-direct-genes.json').write_text(json.dumps(rows,indent=2)+'\n');print(rows)
