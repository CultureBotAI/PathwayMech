from acquire import *
items=[('PWY-6416','PATHWAY'),('CPLX-7981','ENZYME'),('CPLX-7980','ENZYME'),('MONOMER-15334','ENZYME'),('G-11737','GENE'),('G-11738','GENE')]
def get(item):
 fid,kind=item;url='https://bf2i-artsymbiocyc.insa-lyon.fr/META/NEW-IMAGE?type='+kind+'&object='+fid
 row={'frameid':fid,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url)
  f=ORIG/('META-'+fid+'.html');f.write_bytes(b);row.update(path=str(f),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
 except Exception as e:row['error']=str(e)
 return row
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:rows=list(ex.map(get,items))
(ROOT/'acquisition-native-html.json').write_text(json.dumps(rows,indent=2)+'\n')
for r in rows:print(r['frameid'],r.get('bytes',r.get('error')))
