from pathlib import Path
import datetime,hashlib,json,shutil,subprocess,sys
import yaml

B=Path(__file__).resolve().parent
R=Path('/private/tmp/pathwaymech-next-257-20261014')
N=Path('/private/tmp/pathwaymech-next-257-native')
OLD=Path('/private/tmp/pathwaymech-next-256-authorities/originals')
BASE='11afa1984bbb0d41c34dc82361bd0f0e1c0b9f37'
PY='/private/tmp/pathwaymech-next-195-20261011/.venv/bin/python'
def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        while b:=f.read(1024*1024):h.update(b)
    return h.hexdigest()
def save(name,obj):(B/name).write_text(json.dumps(obj,indent=2)+'\n')
def git(*args):return subprocess.check_output(['git',*args],cwd=R)
assert git('rev-parse','HEAD').decode().strip()==BASE
for name in ['baseline','originals','records']:(B/name).mkdir(exist_ok=True)
for source,name in [('conf/identifier_sources.json','identifier_sources.json'),('conf/rhea_identifier_updates.json','rhea_identifier_updates.json'),('data/identifier_authorities/sources.json','sources.json'),('data/identifier_authorities/ontology.json','ontology.json')]:
    raw=git('show',f'{BASE}:{source}')
    assert (R/source).read_bytes()==raw,(source,'uncommitted baseline change')
    dst=B/'baseline'/name
    if dst.exists():assert dst.read_bytes()==raw
    else:dst.write_bytes(raw)
for p in OLD.iterdir():
    dst=B/'originals'/p.name
    if not dst.exists():dst.symlink_to(p)
old_manifest=json.loads((B/'baseline/identifier_sources.json').read_text())
digests={};artifacts=[]
def verify_descriptors(obj,where):
    if isinstance(obj,dict):
        if 'path' in obj and 'sha256' in obj:
            p=B/'originals'/obj['path'];assert p.is_file(),str(p)
            if str(p) not in digests:digests[str(p)]=sha(p)
            assert digests[str(p)]==obj['sha256'],(where,str(p))
            artifacts.append({'context':where,'path':str(p),'sha256':digests[str(p)]})
        for k,v in obj.items():verify_descriptors(v,where+'/'+k)
    elif isinstance(obj,list):
        for i,v in enumerate(obj):verify_descriptors(v,where+f'/{i}')
verify_descriptors(old_manifest,'identifier_sources')
verify_descriptors(json.loads((B/'baseline/rhea_identifier_updates.json').read_text()),'rhea_identifier_updates')
retained=set()
for name in ['sources','ontology']:retained.update(json.loads((B/'baseline'/f'{name}.json').read_text())['terms'])
save('retained-authority-identifiers.json',sorted(retained))
paths=git('ls-tree','-r','--name-only',BASE,'data/pathways').decode().splitlines()
paths=[p for p in paths if p.endswith('.yaml')]
record_bindings=[]
for source in paths:
    raw=git('show',f'{BASE}:{source}');dst=B/'records'/Path(source).relative_to('data/pathways');dst.parent.mkdir(parents=True,exist_ok=True)
    if dst.exists():assert dst.read_bytes()==raw
    else:dst.write_bytes(raw)
    record_bindings.append({'git_path':source,'path':str(dst),'sha256':hashlib.sha256(raw).hexdigest()})
save('baseline-record-bindings.json',record_bindings)
save('baseline-original-cache-verification.json',{'baseline_commit':BASE,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'manifest_entries':len(old_manifest),'hash_bound_descriptors':len(artifacts),'distinct_original_paths':len(digests),'all_original_sha256_verified':True,'old_cache':str(OLD),'retained_authority_terms':len(retained),'originals':artifacts})
print(json.dumps({'baseline_records':len(paths),'retained_terms':len(retained),'original_paths_verified':len(digests)},indent=2))
