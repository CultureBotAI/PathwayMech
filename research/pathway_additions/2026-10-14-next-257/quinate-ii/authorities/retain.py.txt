def bind(path):
    data = path.read_bytes()
    return {'path': str(path), 'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}


promotion = json.loads((B / 'preservation-and-promotion.json').read_text())
commands = json.loads((B / 'canonical-build-commands.json').read_text())
assert promotion['final_authority_terms'] == 4947

assert promotion['manifest']['final_entries'] == 1041
for name, rel in [('sources', 'data/identifier_authorities/sources.json'), ('ontology', 'data/identifier_authorities/ontology.json')]:
    assert bind(R / rel)['sha256'] == promotion['snapshots'][name]['after_sha256']
assert bind(R / 'conf/identifier_sources.json')['sha256'] == promotion['manifest']['sha256']
for path in [B / 'prepare_union.stderr.txt', B / 'promote.stderr.txt']:
    assert path.read_text() == ''
execution = {'prepared_utc': commands['created_utc'], 'preparation': {'command': ['/private/tmp/pathwaymech-next-195-20261011/.venv/bin/python', '-B', str(B / 'prepare_union.py')], 'environment': commands['environment'], 'exit_code': 0, 'stdout': dict(bind(B / 'prepare_union.stdout.txt'), contents=(B / 'prepare_union.stdout.txt').read_text()), 'stderr': dict(bind(B / 'prepare_union.stderr.txt'), contents='')}, 'promotion': {'command': ['/usr/local/bin/python3.11', str(B / 'promote.py')], 'exit_code': 0, 'stdout': bind(B / 'promote.stdout.txt'), 'stderr': dict(bind(B / 'promote.stderr.txt'), contents='')}, 'initial_environment_failure': 'attempt-missing-task-pythonpath/receipt.json', 'scope': 'Successful tool exit codes and preserved output bindings; initial pre-staging import failure remains explicit.'}

snapshot_bindings = {'baseline_commit': promotion['baseline_commit'], 'baseline_files': [bind(B / 'baseline' / name) for name in ['identifier_sources.json', 'rhea_identifier_updates.json', 'sources.json', 'ontology.json']], 'promoted_files': [bind(R / rel) for rel in ['conf/identifier_sources.json', 'data/identifier_authorities/sources.json', 'data/identifier_authorities/ontology.json']], 'baseline_record_bindings': bind(B / 'baseline-record-bindings.json'), 'selector': bind(B / 'records/planned-authority-selection.yaml'), 'source_originals_external': str(B / 'originals'), 'large_baseline_and_generated_union_json_not_duplicated_in_research': True}
(B / 'snapshot-bindings.json').write_text(json.dumps(snapshot_bindings, indent=2) + '\n')

names = ['README.md', 'execution-receipts.json', 'snapshot-bindings.json', 'baseline-record-bindings.json', 'baseline-original-cache-verification.json', 'retained-authority-identifiers.json', 'planned-identifiers.json', 'proposed-source-additions.json', 'new-source-original-bindings.json', 'selected-identity-previews.json', 'canonical-build-commands.json', 'source_builder-receipt.json', 'ontology_builder-receipt.json', 'source_builder.log.json', 'ontology_builder.log.json', 'preservation-and-promotion.json', 'final-original-bindings.json', 'planned-authority-labels.json', 'candidate-identity-preview.json', 'union-helper-readiness.json', 'acquisition-fungal-genes.json', 'fungal-gene-authority-candidates.json', 'fungal-gene-identity-audit.json', 'attempt-missing-task-pythonpath/receipt.json']
helpers = ['prepare_baseline.py', 'prepare_union.py', 'run_builds.py', 'promote.py', 'preview_candidates.py', 'acquire_fungal_genes.py', 'verify_fungal_genes.py', 'retain.py']
D.mkdir(parents=True, exist_ok=True)
retained = []
for name in names + helpers:
    source = B / name
    data = source.read_bytes()
    if name.endswith('.py'):
        ast.parse(data)
    if name.endswith('.json'):
        json.loads(data)
    assert all(line == line.rstrip() for line in data.decode().splitlines()), name
    target = D / (name + '.txt' if name.endswith('.py') else name)
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists():
        assert target.read_bytes() == data, str(target)
    else:
        target.write_bytes(data)
    retained.append({'original': bind(source), 'retained': bind(target), 'byte_identical': target.read_bytes() == data})
manifest = {'retained_artifact_count': len(retained), 'retained': retained, 'raw_originals_and_full_baseline_cache_external': True, 'promoted_authority_files': snapshot_bindings['promoted_files']}

receipt = {'created_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'destination': str(D), 'retained_file_count_including_manifest_and_receipt': len(retained) + 2, 'all_retained_bytes_equal': True, 'artifact_manifest': bind(B / 'artifact-manifest.json')}
(B / 'retention.json').write_text(json.dumps(receipt, indent=2) + '\n')
(D / 'retention.json').write_bytes((B / 'retention.json').read_bytes())
