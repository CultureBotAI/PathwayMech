def project_slot(e):
    d={}
    def plain(s):return html.unescape(re.sub('<[^>]*>','',s or '')).strip()
    if e.text and e.text.strip():d['text']=plain(e.text)
    if e.attrib:d['attributes']=e.attrib
    for c in e:
        if c.get('frameid'):
            v={'frameid':c.get('frameid'),'kind':c.tag}
            if c.find('common-name') is not None:v['label']=plain(c.findtext('common-name'))
            if c.tag=='Publication':v.update({n:plain(c.findtext(n)) for n in ['pubmed-id','title','year'] if c.findtext(n)})
            if c.tag=='Evidence-Code':v['code']=c.get('frameid')
        else:v=project_slot(c)
        d.setdefault(c.tag,[]).append(v)
    return d

class Parser(HTMLParser):
    def __init__(self):super().__init__();self.edges=[];self.nodes=[];self.ids=set()
    def handle_starttag(self,tag,attrs):
        a=dict(attrs)
        if a.get('id'):self.ids.add(a['id'])
        cls=a.get('class','').split()
        if tag=='a' and 'graph-edge' in cls:self.edges.append(a)
        if tag=='a' and 'graph-node' in cls:self.nodes.append(a)
