import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT = Path('/private/tmp/pathwaymech-next-253-20261014')
OUT = Path(__file__).parent / sys.argv[1]
OUT.mkdir(parents=True, exist_ok=True)
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONNOUSERSITE='1',
           UV_NO_SYNC='1', UV_CACHE_DIR='/private/tmp/pathwaymech-next-253-uv-cache',
           UV_PROJECT_ENVIRONMENT='/private/tmp/pathwaymech-next-195-20261011/.venv',
           PYTHONPATH=f'{ROOT}/src:{ROOT}')
commands = [['just', 'test'], ['just', 'lint']] if sys.argv[1] == 'core' else [
    ['just', 'render-pages'], ['just', 'validate'], ['just', 'vendored-check'],
    ['git', 'diff', '--check']]
paths = subprocess.check_output(['git', 'ls-files', '-z', 'src', 'tests', 'conf',
    'data', 'pages', 'scripts', 'schema', 'pyproject.toml', 'uv.lock', 'justfile'], cwd=ROOT).split(b'\0')
inputs = {p.decode(): hashlib.sha256((ROOT/p.decode()).read_bytes()).hexdigest()
          for p in paths if p and (ROOT/p.decode()).is_file()}
(OUT/'core-input-hashes.json').write_text(json.dumps(inputs, indent=2)+'\n')
receipts = []
for command in commands:
    log = OUT/('-'.join(command).replace('--','')+'.log')
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    before = time.monotonic()
    with log.open('w') as stream:
        result = subprocess.run(command, cwd=ROOT, env=env, stdout=stream, stderr=subprocess.STDOUT)
    receipt = dict(command=command, cwd=str(ROOT), started=started,
        elapsed_seconds=round(time.monotonic()-before,2), exit_code=result.returncode,
        log_path=str(log), log_sha256=hashlib.sha256(log.read_bytes()).hexdigest())
    receipts.append(receipt)
    (OUT/'receipts.json').write_text(json.dumps(receipts, indent=2)+'\n')
    print(json.dumps(receipt), flush=True)
    if result.returncode:
        print(log.read_text()[-10000:], flush=True)
        raise SystemExit(result.returncode)
assert all(hashlib.sha256((ROOT/p).read_bytes()).hexdigest()==sha for p,sha in inputs.items())
print(f'{len(inputs)} core inputs unchanged', flush=True)
