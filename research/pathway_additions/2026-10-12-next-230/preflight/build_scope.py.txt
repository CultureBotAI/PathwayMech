def h(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def bind(p):p=Path(p);return dict(path=str(p),sha256=h(p),bytes=p.stat().st_size)
def save(n,x):(D/n).write_text(json.dumps(x,indent=2,ensure_ascii=False)+'\n')
def xml(f,xpath):
 p=B/f;t=E.parse(str(p));xs=t.xpath(xpath);assert xs,(f,xpath)
 return dict(**bind(p),selector=xpath,selected_count=len(xs),selected_sha256=hashlib.sha256('\n'.join(E.tostring(x,encoding='unicode') if hasattr(x,'tag') else str(x) for x in xs).encode()).hexdigest(),role='source assertion locator; not a literal quotation')
def article(f,term):
 p=B/f;t=E.parse(str(p));xs=[x for x in t.xpath('//body//p|//body//table-wrap|//body//fig|//passage') if term.casefold() in ' '.join(''.join(x.itertext()).split()).casefold()];assert xs,(f,term)
 return xml(f,'|'.join(t.getpath(x)for x in xs))
def native(f,slots):return xml('raw/'+f,'|'.join('/ptools-xml/*[not(self::metadata)]/'+s for s in slots.split(',')))
def jsource(f,selector='$'):
 p=B/f;json.loads(p.read_text());return dict(**bind(p),selector=selector,role='structured issuing source assertion; not a quotation')
def pdf(f,page,derived):
 p=Path(f);return dict(**bind(p),selector='PDF page '+str(page)+' (1-based)',derived_reader=bind(B/'derived'/derived),role='original PDF with separately bound extraction; visual checks stated separately')
def abstract(pmid):
 matches=[]
 for f in sorted((B/'raw').glob('*search.json')):
  try:d=json.loads(f.read_text())
  except Exception:continue
  for r in d.get('resultList',{}).get('result',[]):
   if r.get('id')==str(pmid):matches.append(jsource(str(f.relative_to(B)),f"resultList.result[id='{pmid}'].abstractText"))
 assert matches,pmid
 return matches[-1]
