checks = []

def now():
    return datetime.datetime.now(datetime.UTC).isoformat().replace('+00:00', 'Z')

def load(path):
    return json.loads(Path(path).read_bytes())

def binding(path):
    path = Path(path)
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': digest.hexdigest()}

def write(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

def files(path):
    return {Path(p) for p in subprocess.check_output(['rg', '--files', '-uu', '-L', str(path)], text=True).splitlines()}

def verify(row):
    actual = binding(row['path'])
    assert actual['sha256'] == row['sha256'], row['path']
    assert actual['bytes'] == row.get('bytes', actual['bytes']), row['path']
    return actual

def passed(name, detail):
    checks.append({'name': name, 'status': 'passed', 'detail': detail})

def rejected(name, function):
    try:
        function()
    except AssertionError as error:
        passed(name, {'rejected': True, 'error': str(error)})
    else:
        raise AssertionError('Unexpected acceptance: ' + name)

def main():

    verify({'path': str(PREP / 'handoff.json'), 'sha256': '69149b1c19a6114319c1b10cad87cbeb474a02e79b94a99a69d0919904a4a159'})
