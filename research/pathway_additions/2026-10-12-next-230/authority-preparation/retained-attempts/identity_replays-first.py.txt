def bind(p):
 p=Path(p)
 with p.open('rb') as f:h=hashlib.file_digest(f,'sha256').hexdigest()
 return {'path':str(p),'bytes':p.stat().st_size,'sha256':h}
def save(n,x):(O/n).write_text(json.dumps(x,indent=2,ensure_ascii=False)+'\n')
def read(p):return json.loads(Path(p).read_bytes())
probes=read(O/'native-candidate-probes.json');fresh={}

 if row['kind']!='uniprot':continue

  key=item['primaryAccession'];assert key not in fresh

base_entry=next(e for e in read(B/'manifest.json') if e['key']=='uniprot-ecoli-reviewed-cofactors');fp=B/'originals'/base_entry['path'];assert bind(fp)['sha256']==base_entry['sha256'];frozen={x['primaryAccession']:x for x in read(fp)['results']};fterms={t[0]:t for t in sb.source_terms('uniprot',fp)}
# Include the comparison-only P69783 in this identity inventory; final graph selectors remain author owned.

for acc,row in sorted(fresh.items()):
 old=frozen[acc];new=row['entry'];ot=fterms['UniProtKB:'+acc];nt=row['native_term']

 annotations={k:{'frozen':[x for x in old.get('comments',[]) if x['commentType']==k],'fresh':[x for x in new.get('comments',[]) if x['commentType']==k]} for k in sorted(oldtypes)}
 comparison.append({'id':'UniProtKB:'+acc,'frozen_original':bind(fp),'frozen_source_entry':base_entry,'fresh_original':row['original'],'fresh_request_url':row['request_url'],'frozen_identity':a,'fresh_identity':b,'identity_facets_equal':a==b,'identity_differences':{k:{'frozen':a[k],'fresh':b[k]} for k in a if a[k]!=b[k]},'common_annotation_changes':{k:v for k,v in annotations.items() if v['frozen']!=v['fresh']},'fresh_entry_audit':new.get('entryAudit'),'frozen_entry_audit_present':'entryAudit' in old,'fresh_fields_not_in_frozen':sorted(set(new)-set(old)),'full_description_equal':old.get('proteinDescription')==new.get('proteinDescription'),'precedence':'Preserve prior whole original and native first-source order. Fresh full scientific entry remains separate evidence; no release refresh or full scientific-content equality asserted.'})
