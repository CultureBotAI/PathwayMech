def load(path):
    return json.loads(Path(path).read_text())

def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda: f.read(8 * 1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

def binding(path):
    path = Path(path)
    return {'path': str(path), 'sha256': sha(path), 'bytes': path.stat().st_size}

def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def utc():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')

def locator_proof(edge_id, evidence, source_path, replacement, detail):
    tree = etree.parse(str(source_path))
    old = evidence['source_locator'].removeprefix('XPath ')
    new = replacement.removeprefix('XPath ')
    old_nodes, new_nodes = tree.xpath(old), tree.xpath(new)
    assert old_nodes and new_nodes
    row = {'evidence_id': f'proof-{edge_id}', 'edge_id': edge_id,
           'source': binding(source_path), 'current_assertion': evidence['source_assertion'],
           'current_locator': evidence['source_locator'], 'replacement_locator': replacement,
           'current_selected_xml': [etree.tostring(n, encoding='unicode') for n in old_nodes],
           'replacement_selected_xml': [etree.tostring(n, encoding='unicode') for n in new_nodes],
           'current_selected_text': [' '.join(''.join(n.itertext()).split()) for n in old_nodes],
           'replacement_selected_text': [' '.join(''.join(n.itertext()).split()) for n in new_nodes],
           'scientific_assessment': detail}
    proofs.append(row)
    return row
