 return e.find(rh+name).get(rdf+'resource')

 for side in ['substrates','products']:
  se=selected[resource(re,side)]
  for c in se:
   tag=etree.QName(c).localname
   if not tag.startswith('contains') or not tag[8:].isdigit():continue
   pe=selected[c.get(rdf+'resource')];ce=selected[resource(pe,'compound')]
   curie=ce.findtext(rh+'accession');n=int(tag[8:])
   triples.append([curie,'consumes','RHEA:'+id] if side=='substrates' else ['RHEA:'+id,'produces',curie])
   chains.append({'reaction':'RHEA:'+id,'side':side,'coefficient':n,'chemical':curie,'formula':ce.findtext(rh+'formula'),'charge':ce.findtext(rh+'charge'),'xml':[etree.tostring(v,encoding='unicode') for v in [re,se,pe,ce]]})
