def binding(path): return {'path': str(path), 'sha256': sha(path), 'bytes': Path(path).stat().st_size}
def save(name, value): (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

        text = ''.join(cell.itertext())
        matrix.append({'xpath': tree.getpath(cell), 'raw_text': text,
                       'unicode_codepoints': [f'U+{ord(c):04X}' for c in text],

footnote = ''.join(tree.xpath(base + '/table-wrap-foot')[0].itertext())
assert 'Blank: not determined; ND, not detected.' in footnote
