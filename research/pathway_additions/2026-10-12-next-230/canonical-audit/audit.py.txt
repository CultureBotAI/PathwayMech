"""Audit only a terminal, sealed ATP native build; never runs either builder."""
import argparse
import datetime
import hashlib
import json
from pathlib import Path

import subprocess
import traceback

import yaml

HERE = Path(__file__).resolve().parent
BUILD = Path('/private/tmp/pathwaymech-next-230-canonical-build-20261012')
PREP = Path('/private/tmp/pathwaymech-next-230-canonical-preparation-20261012')
PREP_AUDIT = Path('/private/tmp/pathwaymech-next-230-canonical-preparation-audit-20261012')
ACCEPTANCE = Path('/private/tmp/pathwaymech-next-230-root-final-successor-and-preparation-20261012.json')
CHECKS = []

def now():
    return datetime.datetime.now(datetime.UTC).isoformat().replace('+00:00', 'Z')

def load(path):
    return json.loads(Path(path).read_bytes())

def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()

def bind(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}

def verify(row):
    actual = bind(row['path'])
    assert actual['sha256'] == row['sha256'], row['path']
    assert actual['bytes'] == row.get('bytes', actual['bytes']), row['path']
    return actual

def write(name, value):
    with (HERE / name).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

def files(path):
    return {Path(value) for value in subprocess.check_output(['rg', '--files', '-uu', '-L', str(path)], text=True).splitlines()}

def passed(name, detail):
    CHECKS.append({'name': name, 'status': 'passed', 'detail': detail})

def packet(directory, digest, originals=None):
    handoff = verify({'path': str(directory / 'handoff.json'), 'sha256': digest})
    data = load(handoff['path'])
    manifest = verify(data['manifest'])
    rows = load(manifest['path'])['artifacts']
    assert len({row['path'] for row in rows}) == len(rows)
    for row in rows:
        assert Path(row['path']).is_relative_to(directory), row['path']
        verify(row)
    declared = {Path(row['path']) for row in rows}
    assert files(directory) == declared | {directory / 'handoff.json', directory / 'artifact-manifest.json'} | set(originals or []), 'closed inventory ' + str(directory)
    return data, rows

def main():

    parser.add_argument('--terminal-intake', type=Path, required=True)
    parser.add_argument('--handoff-sha256', required=True)
    args = parser.parse_args()
    started = now()
    terminal = load(args.terminal_intake)

    assert terminal['build_handoff']['path'] == str(BUILD / 'handoff.json')

        assert all(item['exit_code'] == 0 for item in outer)
        for item in outer:
            assert item['command'] == ['/private/tmp/pathwaymech-next-195-20261011/.venv/bin/python', '-B', str(PREP / 'prepare_and_build.py'), '--' + item['name']]
    verify({'path': str(ACCEPTANCE), 'sha256': '15f4cb472dff67ac81fb8ea81c460f55b69bd31829da3e3c58d512ce57c23860'})
    assert load(ACCEPTANCE)['status'] == 'accepted_for_full_native_execution'
    prep_handoff, _ = packet(PREP, '69149b1c19a6114319c1b10cad87cbeb474a02e79b94a99a69d0919904a4a159')
    packet(PREP_AUDIT, 'ba0529a023a0140e05f1e96342f9696fb4a4bc45576041156fa9bfc05275ec03')
    capture = load(PREP / 'capture.json')
    intake = load(BUILD / 'execution-intake.json')
    original_paths = set()
    original_proof = []
    assert len(intake['source_originals_freshly_hashed']) == len(capture['source_originals']) == 756
    for original, executed in zip(capture['source_originals'], intake['source_originals_freshly_hashed']):
        assert {key: value for key, value in executed.items() if key != 'fresh_execution_binding'} == original
        assert executed['fresh_execution_binding'] == {key: original[key] for key in ('path', 'bytes', 'sha256')}
        original_path = BUILD / 'originals' / original['cache_relative_path']
        assert original_path.is_symlink()
        assert original_path.resolve() == Path(original['path']).resolve() == Path(original['resolved_path'])
        actual = verify({**original, 'path': str(original_path)})
        original_paths.add(original_path)
        original_proof.append({'cache_relative_path': original['cache_relative_path'], 'actual_symlink': actual, 'resolved_path': str(original_path.resolve())})
    handoff, build_rows = packet(BUILD, args.handoff_sha256, originals=original_paths)
    passed('terminal_and_closed_build', {'build_artifacts': len(build_rows), 'original_symlinks': len(original_paths), 'hidden_ignored_followed': True, 'root_terminal': terminal['actual_terminal_result']})
    assert handoff['status'] == 'full_original_backed_build_completed_pending_independent_audit'
    assert handoff['preparation'] == bind(PREP / 'handoff.json')
    assert handoff['target_sha256'] == prep_handoff['target']['sha256'] == 'cd59859d1dbcbc6c36961822b02daebc519a80a224ba8fa63d3235933b328101'
    assert handoff['maintained_adoption'] is False and handoff['independent_canonical_acceptance'] is False
    assert (BUILD / 'preparation-capture.json').read_bytes() == (PREP / 'capture.json').read_bytes()
    assert intake['preparation'] == bind(PREP / 'handoff.json')
    assert intake['preparation_manifest'] == bind(PREP / 'artifact-manifest.json')
    assert intake['sql_originals_pinned'] == capture['sql_originals']
    for name in ('common.py', 'prepare_and_build.py'):
        assert (BUILD / name).read_bytes() == (PREP / name).read_bytes()
    passed('accepted_input_lineage', 'Execution binds the exact accepted target, immutable preparation, helpers and source/SQL inventory.')
    commands = load(BUILD / 'commands.json')
    assert commands == load(PREP / 'root-build-commands.json')
    executions = load(BUILD / 'execution-ledger.json')['executions']
    assert len(executions) == 2 and {row['name'] for row in executions} == {'source_builder', 'ontology_builder'}
    for execution in executions:
        assert execution['exit_code'] == 0
        assert execution['command'] == commands[execution['name']]
        assert execution['cwd'] == commands['cwd'] and execution['environment'] == commands['environment']
        assert datetime.datetime.fromisoformat(execution['started_at']) <= datetime.datetime.fromisoformat(execution['finished_at'])
        assert execution['seconds'] >= 0
        assert execution == load(BUILD / (execution['name'] + '-execution.json'))
        verify(execution['stdout'])
        verify(execution['stderr'])
        assert Path(execution['stderr']['path']).read_bytes() == b''
    assert 'Retained 2474 independently sourced identifiers from 595 sources' in (BUILD / 'source_builder.stdout').read_text()
    assert 'Wrote 1963 authority terms to ' + str(BUILD / 'ontology.json') in (BUILD / 'ontology_builder.stdout').read_text()
    passed('two_actual_native_executions', {'executions': executions, 'no_builder_rerun_by_audit': True})
    expected_staged = set()
    for section, rows, destination in [('candidate_code', capture['candidate_code'], 'runtime'), ('records', capture['records'], 'corpus'), ('baseline', capture['baseline'], 'baseline')]:
        for row in rows:
            path = BUILD / destination / row['relative_path']
            assert path.read_bytes() == Path(row['captured']['path']).read_bytes(), (section, path)
            expected_staged.add(path)
    target = BUILD / 'corpus/sedoheptulose-bisphosphate-bypass.yaml'
    assert target.read_bytes() == Path(capture['target']['captured']['path']).read_bytes()
    expected_staged.add(target)
    retention_path = BUILD / 'corpus/_authority_retention/selector.yaml'
    retention = capture['retention_ids']
    assert yaml.safe_load(retention_path.read_bytes()) == {'id': retention[0], 'participants': [{'id': value} for value in retention[1:]]}
    expected_staged.add(retention_path)
    actual_staged = set().union(*(files(BUILD / name) for name in ('runtime', 'corpus', 'baseline')))
    assert actual_staged == expected_staged
    staged_rows = load(BUILD / 'staged-bindings.json')['files']
    assert {Path(row['path']) for row in staged_rows} == expected_staged
    assert len(staged_rows) == len(expected_staged)
    for row in staged_rows:
        verify(row)
    passed('all_staged_inputs_exact', {'runtime': 222, 'prior_pathways': 230, 'new_pathways': 1, 'retention_ids': 4411, 'staged_files': len(expected_staged), 'retention_contains_no_labels': True})
    old = {kind: load(BUILD / 'baseline' / (kind + '.json')) for kind in ('sources', 'ontology')}
    actual = {kind: load(BUILD / (kind + '.json')) for kind in old}
    expected = load(PREP / 'expected-native-comparison.json')['snapshots']
    assert (BUILD / 'expected-native-comparison.json').read_bytes() == (PREP / 'expected-native-comparison.json').read_bytes()
    selector_replay = load(PREP_AUDIT / 'native-selector-replay.json')

    assert (len(actual['sources']['terms']), len(actual['sources']['sources']), len(actual['ontology']['terms']), len(actual['ontology']['sources'])) == (2474, 595, 1963, 5)

    assert (BUILD / 'ontology.json').read_text() == json.dumps(actual['ontology'], indent=2, sort_keys=True) + '\n'
    delta = {kind: {section: {key: value for key, value in actual[kind][section].items() if key not in old[kind][section]} for section in ('terms', 'sources')} for kind in old}
    assert (len(delta['sources']['terms']), len(delta['sources']['sources']), len(delta['ontology']['terms']), len(delta['ontology']['sources'])) == (20, 8, 6, 0)
    write('actual-authority-additions.json', delta)
    passed('complete_actual_snapshots', {'source_terms': 2474, 'source_metadata': 595, 'ontology_terms': 1963, 'ontology_metadata': 5, 'all_prior_objects_exact': True, 'all_selected_identifiers_present': True, 'all_actual_objects_equal_accepted_comparison': True})
    manifest = load(BUILD / 'manifest.json')
    previous_manifest = load(BUILD / 'baseline/manifest.json')
    updates = load(BUILD / 'rhea-updates.json')
    previous_updates = load(BUILD / 'baseline/rhea-updates.json')
    assert manifest[:len(previous_manifest)] == previous_manifest and len(previous_manifest) == 750 and len(manifest) == 758
    assert len({entry['key'] for entry in manifest}) == len(manifest)
    assert updates['ids'][:len(previous_updates['ids'])] == previous_updates['ids'] and len(previous_updates['ids']) == 22 and len(updates['ids']) == 24
    assert {key: value for key, value in updates.items() if key != 'ids'} == {key: value for key, value in previous_updates.items() if key != 'ids'}
    for name in ('manifest.json', 'rhea-updates.json'):
        assert (BUILD / name).read_bytes() == (PREP / 'captured/proposals' / name).read_bytes()
    passed('native_input_order', {'manifest_old_prefix': 750, 'manifest_total': 758, 'Rhea_old_prefix': 22, 'Rhea_total': 24, 'prior_selection_order_and_metadata_exact': True})
    sql_proof = []
    for row in capture['sql_originals']:
        path = Path(row['path'])
        stat = path.stat()
        assert stat.st_size == row['bytes'] and stat.st_mtime_ns == row['mtime_ns']
        metadata = actual['ontology']['sources'][row['source_key']]
        assert metadata['sha256'] == row['sha256']
        assert metadata['sha256_scope'] == 'uncompressed_semantic_sql_database'
        with sqlite3.connect(path.resolve().as_uri() + '?mode=ro', uri=True) as connection:
            prefixes = dict(connection.execute('SELECT prefix, base FROM prefix'))
            declarations = list(connection.execute("SELECT subject, object FROM statements WHERE predicate='owl:versionIRI'"))
            assert len(declarations) == 1
            subject, release = declarations[0]
            values = {predicate: obj or literal for predicate, obj, literal in connection.execute("SELECT predicate, object, value FROM statements WHERE subject=? AND predicate IN ('owl:versionInfo', 'dcterms:license')", (subject,))}
        def expanded(value):
            value = value.strip('<>')
            prefix, separator, local = value.partition(':')
            return prefixes[prefix] + local if separator and prefix in prefixes else value
        assert metadata['url'] == expanded(release)
        assert metadata['license'] == expanded(values['dcterms:license'])
        assert metadata['version'] == values['owl:versionInfo']
        sql_proof.append({'input_pin': row, 'actual_native_full_hash_and_metadata': metadata, 'independent_readonly_metadata_query': 'passed', 'audit_full_SQL_rehash': False})
    write('actual-SQL-provenance.json', {'sources': sql_proof, 'scope': 'Full hashes were computed by unchanged actual native builder; audit verifies exact execution/code/output pins, current stats and independent read-only SQL metadata. Audit does not repeat large full-file hashes.'})
    passed('all_four_native_full_SQL_hashes', {'databases': 4, 'actual_native_hashes_match_pins': True, 'metadata_independently_queried': True, 'no_audit_SQL_full_rehash_claim': True})
    comparison = load(BUILD / 'proposal-comparison.json')
    assert comparison['against_accepted_authority_proposals'] == {kind: {'terms': [], 'sources': []} for kind in old}
    assert comparison['expectation_never_used_as_builder_input'] is True
    verification = load(BUILD / 'verification.json')
    assert verification['status'] == 'native_full_build_passed_external_only'
    assert verification['baseline_scope'] == capture['scope']
    assert verification['target_sha256'] == handoff['target_sha256']
    assert verification['native_full_SQL_hashes_match'] == 4 and verification['old_objects_exact'] is True
    for field in ('scientific_acceptance', 'maintained_writes', 'shared_cache_writes', 'git_github_mutations', 'environment_installs_or_sync', 'source_exhausted'):
        assert verification[field] is False
    assert capture['observed_head'] == 'ed6337d0adbcee81f962c80391e12c0dcc084908'
    assert capture['prospective_records_with_ATP'] == 231
    assert 'EG is not claimed merged' in capture['scope']
    for name, row in handoff['outputs'].items():
        assert row == bind(BUILD / name)
    passed('truthful_completion_scope', 'Actual canonical build passed; prospective baseline, root scientific acceptance and later maintained/merged integration retain separate scopes.')
    write('original-symlink-audit.json', {'freshly_verified_originals': original_proof, 'count': 756})

    write('checks.json', {'started_at': started, 'finished_at': now(), 'status': 'passed', 'checks': CHECKS, 'full_build_reexecuted': False, 'maintained_writes': False})
    print(json.dumps({'status': 'passed', 'checks': len(CHECKS), 'full_build_reexecuted': False}))

if __name__ == '__main__':
    try:
        main()
    except Exception:
        write('failure-' + datetime.datetime.now(datetime.UTC).strftime('%H%M%S') + '.json', {'at': now(), 'completed_checks': CHECKS, 'traceback': traceback.format_exc()})
        raise
