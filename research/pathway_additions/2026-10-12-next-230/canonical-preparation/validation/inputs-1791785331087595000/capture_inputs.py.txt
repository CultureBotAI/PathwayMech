def copy_binding(source, destination):
    before = bind(source)
    destination.parent.mkdir(parents=True, exist_ok=True)
    assert not destination.exists()
    shutil.copyfile(source, destination)
    after = bind(destination)
    assert (before["bytes"], before["sha256"]) == (after["bytes"], after["sha256"])
    return {"original": before, "captured": after}


def verify_packet(name, directory, digest):
    handoff = check({"path": str(directory / "handoff.json"), "sha256": digest})
    data = load(handoff["path"])
    manifest = check(data.get("manifest") or data["artifact_manifest"])
    rows = load(manifest["path"])["artifacts"]
    paths = set()
    for row in rows:
        path = Path(row.get("path") or row["relative_path"])
        if not path.is_absolute():
            assert ".." not in path.parts
            path = directory / path
        assert path.is_relative_to(directory), path
        assert path not in paths, path
        check({**row, "path": str(path)})
        paths.add(path)
    observed = set(inventory(directory))
    allowed = {directory / "handoff.json", Path(manifest["path"])}
    original_links = set()
    if name == "eg_build":
        intake = load(directory / "execution-intake.json")
        for row in intake["source_originals_freshly_hashed"]:
            path = directory / "originals" / row["cache_relative_path"]
            assert path.is_symlink() and path.resolve() == Path(row["path"]).resolve()
            check({**row, "path": str(path)})
            original_links.add(path)
    assert observed == paths | allowed | original_links, {
        "extra": sorted(str(p) for p in observed - paths - allowed - original_links),
        "missing": sorted(str(p) for p in paths - observed),
    }
    return {"name": name, "handoff": handoff, "manifest": manifest,
            "artifacts_verified": len(rows), "exact_original_symlinks_verified": len(original_links),
            "closed_hidden_ignored_followed_inventory": True}


def native_collectors():
    sys.path.insert(0, str(HERE / "captured/code/src"))
    # This directory was created after interpreter startup; invalidate cached missing entries.
    importlib.invalidate_caches()
    package = importlib.import_module("pathwaymech")
    assert Path(package.__file__).resolve() == (HERE / "captured/code/src/pathwaymech/__init__.py").resolve()
    modules = {}
    for kind in ("source", "ontology"):
        path = HERE / "captured/code/scripts" / ("build_" + kind + "_identifier_snapshot.py")
        spec = importlib.util.spec_from_file_location("atp_capture_" + kind, path)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        modules[kind] = module
    return modules


def selections(path, modules):
    # corpus_ids accepts a directory. Each compared packet directory holds one target YAML.
    assert list(path.parent.glob("*.yaml")) == [path]
    data = yaml.safe_load(path.read_text())
    return {"source_collector": sorted(modules["source"].corpus_ids(path.parent)),
            "ontology_collector": sorted(modules["ontology"].walk_identifiers(data, {"GO", "CHEBI", "NCBITaxon", "RHEA"}))}
