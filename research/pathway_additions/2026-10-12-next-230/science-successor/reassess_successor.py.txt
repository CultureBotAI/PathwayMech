def verify_packet(path, handoff_sha, manifest_sha, count):
    assert sha(path / 'handoff.json') == handoff_sha
    assert sha(path / 'artifact-manifest.json') == manifest_sha
    manifest = js(path / 'artifact-manifest.json')
    items = manifest['artifacts']
    assert len(items) == count
    inventory = subprocess.run(['rg', '--files', '--no-ignore', '--hidden', '--follow', '-0', str(path)], check=True, capture_output=True)
    actual = {str(Path(p.decode()).relative_to(path)) for p in inventory.stdout.split(b'\0') if p}
    expected = {row['path'] for row in items} | {'handoff.json', 'artifact-manifest.json'}
    assert actual == expected, {'extra': sorted(actual - expected), 'missing': sorted(expected - actual)}
    for item in items:
        assert sha(path / item['path']) == item['sha256'], item['path']
        assert (path / item['path']).stat().st_size == item['bytes'], item['path']
    return {'package': str(path), 'handoff': bind(path / 'handoff.json'), 'manifest': bind(path / 'artifact-manifest.json'),
            'artifacts_verified': count, 'closed_inventory_includes_ignored_hidden_followed': True, 'unchanged': True}

def diff(a, b, path='$'):
    if type(a) is not type(b): return [{'path': path, 'before': a, 'after': b}]
    if isinstance(a, dict):
        assert a.keys() == b.keys(), path
        return [entry for k in a for entry in diff(a[k], b[k], f'{path}.{k}')]
    if isinstance(a, list):
        assert len(a) == len(b), path
        return [entry for i, (x, y) in enumerate(zip(a, b)) for entry in diff(x, y, f'{path}[{i}]')]
    return [] if a == b else [{'path': path, 'before': a, 'after': b}]
