def now():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')


def bind(path):
    path = Path(path)
    data = path.read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}


def load(path):
    return json.loads(Path(path).read_text())


def write(path, value):
    with Path(path).open('x') as stream:
        stream.write(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def run(name, command, directory):
    start = now()
    result = subprocess.run(command, cwd=ROOT, env=ENV, capture_output=True)
    stdout, stderr = directory / (name + '.stdout'), directory / (name + '.stderr')
    stdout.write_bytes(result.stdout)
    stderr.write_bytes(result.stderr)
    receipt = {'name': name, 'command': command, 'cwd': str(ROOT),
               'started_at': start, 'finished_at': now(), 'exit_code': result.returncode,
               'stdout': bind(stdout), 'stderr': bind(stderr)}
    write(directory / (name + '.json'), receipt)
    print(name, result.returncode, flush=True)
    result.check_returncode()
    return receipt
