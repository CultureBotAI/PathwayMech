"""Root-only actual archive and native inspection/assembly. Never saves review/history.

Requires --apply-archive: this future root execution appends only the exact
new dossier/final-gates archive. Preparation never executes this command.
"""
import argparse
import copy
import gzip
import shlex
import subprocess
import yaml
from support import *


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('gates', 'preservation', 'preservation-assessment', 'qa', 'visual', 'history', 'root-intake', 'output'):
        parser.add_argument('--' + name, required=True, type=Path)
    parser.add_argument('--display-assessment', type=Path)
    parser.add_argument('--additional-receipt', action='append', default=[], type=Path)
    parser.add_argument('--apply-archive', action='store_true', required=True)
    args = parser.parse_args()
    verify_support()
    assert_head(); gates, inputs = verify_gates(args.gates)
    preserved = load(args.preservation)
    bridge = verify_final_bridge()
    assert bind(args.preservation) == bridge['historical_preservation']
    assert preserved['status'] == 'passed' and preserved['gate_receipt'] == bridge['historical_gates']
    assert preserved['target_sha256'] == HISTORICAL_SHA and not preserved['problems']
    assessment = load(args.preservation_assessment)
    assert assessment['status'] == 'passed' and assessment['preservation'] == bind(args.preservation)
    assert assessment['gate_receipt'] == bind(args.gates)
    qa = load(args.qa)
    assert qa['status'] == 'passed' and qa['render_receipt'] == gates['receipts'][0]
    verify(qa['gate_observation'])
    assert (qa['edges'], qa['qualifiers'], qa['evidence_uses'], qa['references']) == (67, 24, 145, 94)
    for key in ('target', 'page', 'css', 'overview', 'stdout', 'stderr', 'kgx_checks', 'nodes', 'kgx_edges', 'final_target_bridge'):
        verify(qa[key])
    assert qa['final_target_bridge'] == bind(BRIDGE_PATH)
    kgx_qa = load(qa['kgx_checks']['path'])
    assert kgx_qa['all_context_multiplicities_exact'] and kgx_qa['all_target_edge_metadata_evidence_qualifiers_exact']
    assert kgx_qa['node_display_delta'] == preserved['node_display_delta']
    visual = load(args.visual)
    assert visual['page'] == qa['page'] and visual['overview'] == qa['overview']
    assert visual['programmatic_receipt'] == bind(args.qa) and visual['observed'].strip()
    history = load(args.history)
    assert history['status'] == 'native_execution_complete' and history['exit_code'] == 0
    for key in ('history', 'stdout', 'stderr', 'command'):
        verify(history[key])
    assert datetime.fromisoformat(history['finished_at']) <= datetime.fromisoformat(gates['receipts'][0]['started_at'])
    assert Path(history['history']['path']).is_relative_to(ROOT / 'history')
    assert any(row['sha256'] == history['history']['sha256'] and
               row['relative_path'] == str(Path(history['history']['path']).relative_to(ROOT))
               for row in gates['render_history_inputs'])
    for row in load(HERE / 'review-evidence-bindings.json'):
        verify(row['original'])
        verify(row['original'], ROOT / row['maintained_reference'])
    # A root review of the sealed support must be separately documented; do not
    # infer acceptance solely from possession of the package.
    intake = load(args.root_intake)
    assert intake['status'] == 'accepted' and intake['handoff'] == bind(HERE / 'handoff.json')
    if preserved['node_display_delta']:
        assert args.display_assessment is not None
        display = load(args.display_assessment)
        assert display['status'] == 'accepted_exact_two_display_cells'
        assert display['delta'] == preserved['node_display_delta'] == kgx_qa['node_display_delta']
        assert display['target_sha256'] == HISTORICAL_SHA
        assert display['rationale'].strip()
        verify(display['observation'])
        verify(preserved['approved_node_display_delta'])
        assert load(preserved['approved_node_display_delta']['path']) == display['delta']
    out = external_new(args.output)
    archive = ROOT / DOSSIER_REL / 'final-gates'
    assert not archive.exists() and not archive.is_symlink()
    out.mkdir(); archive.mkdir()
    archive_map = []

    def archive_copy(path, name):
        path = Path(path); destination = archive / name
        assert destination.parent == archive and not destination.exists()
        body = path.read_bytes(); serialization = 'exact_bytes'
        if path.suffix in ('.tsv', '.html') and len(body) > 2 * 1024 * 1024:
            destination = destination.with_name(destination.name + '.gz')
            body = gzip.compress(body, mtime=0); serialization = 'gzip-exact-bytes'
            assert gzip.decompress(body) == path.read_bytes()
        elif path.suffix != '.png':
            text = body.decode('utf-8')
            if any(line.rstrip() != line for line in text.splitlines()) or text.endswith('\n\n'):
                destination = destination.with_name(destination.name + '.json')
                body = (json.dumps({'serialization': 'lossless-utf8-json', 'original': bind(path),
                                    'text': text}, indent=2, ensure_ascii=False) + '\n').encode()
                serialization = 'lossless-utf8-json'
        with destination.open('xb') as stream:
            stream.write(body)
        archive_map.append({'source': bind(path), 'archived': bind(destination), 'serialization': serialization})
        return destination

    local = {}
    for key, source in [('gates', args.gates), ('preservation', args.preservation),
        ('preservation-assessment', args.preservation_assessment), ('gate-inputs', Path(gates['inputs']['path'])),
        ('page-qa', args.qa), ('visual-observation', args.visual), ('history-execution', args.history),
        ('root-support-intake', args.root_intake)]:
        local[key] = archive_copy(source, key + '.json')
    for key in ('stdout', 'stderr', 'command'):
        archive_copy(history[key]['path'], 'history-' + key + '.json')
    for key, source in [('final-target-bridge', Path(BRIDGE_PATH)),
        ('historical-gates', Path(bridge['historical_gates']['path'])),
        ('metadata-correction-proof', Path(bridge['root_metadata_bridge']['path'])),
        ('independent-title-assessment', Path(bridge['independent_title_assessment']['path'])),
        ('kgx-qa', Path(qa['kgx_checks']['path'])),
        ('final-cache-preservation', Path(bridge['cache_preservation']['path']))]:
        local[key] = archive_copy(source, key + '.json')
    historical_gates = load(bridge['historical_gates']['path'])
    archive_copy(historical_gates['inputs']['path'], 'historical-gate-inputs.json')
    for row in historical_gates['receipts']:
        for key in ('stdout', 'stderr'):
            archive_copy(row[key]['path'], 'historical-' + row['name'] + '.' + key)
    correction_adoption = load(gates['adoption']['path']); verify(gates['adoption'])
    local['correction-adoption'] = archive_copy(gates['adoption']['path'], 'correction-adoption.json')
    local['correction-edit-execution'] = archive_copy(Path(gates['adoption']['path']).parent / 'history.json', 'correction-edit-execution.json')
    for key in ('stdout', 'stderr'):
        archive_copy(correction_adoption['history_execution'][key]['path'], 'correction-history-' + key)
    for key in ('before_inventory', 'proposal'):
        archive_copy(correction_adoption[key]['path'], 'correction-' + key + '.json')
    for name in ('original-page.html', 'original-edges.tsv', 'original-nodes.tsv'):
        archive_copy(Path(gates['adoption']['path']).parent / name, 'correction-' + name)
    archive_copy('/private/tmp/pathwaymech-next-230-correct-bibliography-20261012.py', 'correction-helper.py.txt')
    for key, source in [('issue460-body', '/private/tmp/pathwaymech-next-230-whitespace-issue-20261012.md'),
                        ('issue460-creation', '/private/tmp/pathwaymech-next-230-whitespace-issue-creation-20261012.json')]:
        local[key] = archive_copy(source, key + '.json')
    if args.display_assessment:
        local['display-assessment'] = archive_copy(args.display_assessment, 'display-assessment.json')
    for row in gates['receipts']:
        for key in ('stdout', 'stderr'):
            archive_copy(row[key]['path'], row['name'] + '.' + key)
    for key in ('approved_node_display_delta', 'approved_extra_files'):
        if preserved.get(key):
            verify(preserved[key]); local[key] = archive_copy(preserved[key]['path'], key + '.json')
    failures = []
    if 'original_failure' in gates:
        local['original-gate-failure'] = archive_copy(gates['original_failure']['path'], 'original-gate-failure.json')
        local['gate-recovery-inputs'] = archive_copy(gates['recovery_inputs']['path'], 'gate-recovery-inputs.json')
        recovery = load(gates['recovery_inputs']['path'])
        if recovery.get('helper'):
            verify(recovery['helper']); archive_copy(recovery['helper']['path'], 'recovery-helper.py.txt')
        for i, row in enumerate(load(gates['original_failure']['path'])['receipts']):
            if row['exit_code']:
                failures.append(row)
                for key in ('stdout', 'stderr'):
                    archive_copy(row[key]['path'], 'original-failure-' + str(i) + '-' + key)
    for i, source in enumerate(args.additional_receipt):
        local['additional-receipt-' + str(i)] = archive_copy(source, 'additional-receipt-' + str(i) + '.json')
    archive_copy(qa['gate_observation']['path'], 'page-qa-gate-observation.json')
    for name in ('original.svg', 'styled.svg', 'overview.png'):
        archive_copy(args.qa.parent / name, 'graph-' + name)
    for name in ('handoff.json', 'artifact-manifest.json', 'review-content-fragments.json',
                 'review-evidence-bindings.json', 'support.py', 'page_qa.py', 'preservation_assessment.py',
                 'assemble_review.py', 'save_review.py', 'archive_native.py', 'inspect_with_inputs.py',
                 'bind_final_outputs.py', 'title-trim-assessment.json', 'fragment-consistency.json'):
        archive_copy(HERE / name, 'support-' + name + ('.txt' if name.endswith('.py') else ''))
    write(archive / 'archive-map.json', archive_map)
    # Inspect AFTER archive; every archived observation has an exact current hash.
    known = ['schema/record_review.yaml', 'scripts/record_review.py', 'docs/record-reviews.md',
      'docs/record-review-profile.md', 'conf/record_review.yaml',
      '.claude/skills/review-yaml-record/SKILL.md', '.claude/skills/add-pathway/SKILL.md',
      'CLAUDE.md', 'justfile', 'docs/CURATION.md', 'docs/HARMONIZATION.md', 'docs/CAUSAL_GRAPHS.md',
      'history/README.md', 'scripts/new_history_record.py', 'src/pathwaymech/schema/history.yaml',
      'src/pathwaymech/schema.py', 'src/pathwaymech/schema/pathwaymech.yaml', 'src/pathwaymech/strict.py',
      'conf/identifier_policy.yaml', 'src/pathwaymech/kgx.py', 'src/pathwaymech/graph_view.py',
      'src/pathwaymech/assets/pathway-network.css', 'docs/IDENTIFIERS.md',
      'scripts/build_source_identifier_snapshot.py', 'scripts/build_ontology_identifier_snapshot.py',
      'tests/test_pathbank_identifier_authority.py', 'conf/identifier_sources.json',
      'conf/rhea_identifier_updates.json', 'data/identifier_authorities/sources.json',
      'data/identifier_authorities/ontology.json', PAGE_REL, 'pages/downloads/nodes.tsv', 'pages/downloads/edges.tsv']
    current_inputs = set(known) | set(files(ROOT, DOSSIER_REL))
    current_inputs.add(str(Path(history['history']['path']).relative_to(ROOT)))
    current_inputs.add(correction_adoption['history']['relative_path'])
    write(out / 'native-inspection-inputs.json', sorted(current_inputs))
    command = [PYTHON, '-B', str(HERE / 'inspect_with_inputs.py'), '--input-list',
               str(out / 'native-inspection-inputs.json'), '--targets', str(HERE / 'targets.yaml')]
    started = now(); result = subprocess.run(command, cwd=ROOT, env=ENV, capture_output=True)
    (out / 'inspect.stdout').write_bytes(result.stdout); (out / 'inspect.stderr').write_bytes(result.stderr)
    write(out / 'inspect-execution.json', {'command': command, 'cwd': str(ROOT), 'started_at': started,
       'finished_at': now(), 'exit_code': result.returncode, 'stdout': bind(out / 'inspect.stdout'),
       'stderr': bind(out / 'inspect.stderr')})
    result.check_returncode(); inspection = json.loads(result.stdout)
    assert inspection['source']['git_revision'] == BASE
    record = yaml.safe_load((ROOT / TARGET_REL).read_bytes())
    assert len(record['mechanistic_edges']) == 67 and len(record['references']) == 94
    document = copy.deepcopy(load(HERE / 'review-content-fragments.json')['review_fields'])
    finished = now()
    document.update({'review_id': datetime.fromisoformat(finished).strftime('%Y%m%dT%H%M%SZ') + '-sedoheptulose-bisphosphate-bypass',
       'started_at': started, 'finished_at': finished, 'skill': '.claude/skills/review-yaml-record/SKILL.md',
       'completion': 'completed', 'verdict': 'pass_with_limitations',
       'source': inspection['source'], 'targets': inspection['targets'],
       'reviewer': {'identity': 'Codex /root (maintained integrator; support preparation by /root/atp_canonical_preparation)',
          'kind': 'agent', 'independence': 'self_review',
          'independence_basis': 'Root integrated this record and reviews current execution/preservation/representation. Scientific content explicitly inherits separately independent original/successor assessments. Support preparation is not an independent literature review.'},
       'summary': 'Exact67-edge,94-reference successor retains accepted scientific fixes450–453 and the explicit issue460 bibliography trim bridge from the historical full native build. Historical full gates, exact bibliography-only bridge, current corrective checks, old-corpus preservation and static graph/text/KGX checks passed. Root reads this assembled content before native validation/save; no future lifecycle success is claimed.'})
    for key, path in local.items():
        document['evidence'].append({'evidence_id': key, 'kind': 'validation', 'reference': str(path.relative_to(ROOT)),
          'accessed_at': finished, 'support': 'supports', 'snapshot_sha256': bind(path)['sha256'],
          'summary': 'Actual exact receipt; original bindings, scope and results retained.'})
    document['checks'] = [{'check_id': 'corrective-' + row['name'], 'name': 'Actual corrected-target ' + row['name'],
       'status': 'passed', 'required': True, 'target_ids': [TARGET_ID],
       'summary': 'Actual corrected-target command passed; no full test rerun is inferred.',
       'command': shlex.join(row['command']), 'exit_code': row['exit_code'], 'evidence_ids': ['gates'],
       'scope_note': 'Current metadata-only correction checks include pre-render EDIT; original full tests remain separately historical. Later review/archive additions require final checks.'}
       for row in gates['receipts']]
    for row in historical_gates['receipts']:
        document['checks'].append({'check_id': 'historical-' + row['name'], 'name': 'Historical full maintained ' + row['name'],
          'status': 'passed', 'required': False, 'target_ids': [TARGET_ID],
          'summary': 'Actual originalcd59859 full gate before exact bibliography trim; preserved under original target and not relabeled as a full rerun.',
          'command': shlex.join(row['command']), 'exit_code': row['exit_code'], 'evidence_ids': ['historical-gates'],
          'scope_note': 'Current6ed1607a consumers and exact metadata-only bridge provide explicitly narrower successor coverage.'})
    document['assessments'].append({'assessment_id': 'external-issue-460', 'area': 'representation',
      'topic': 'sbp-reference-title-whitespace', 'outcome': 'supported', 'target_ids': [TARGET_ID],
      'summary': 'Actual PMID:31464375 title.rstrip removes only trailing space/newline; exact parsed/canonical-selector/KGX bridge and corrective checks preserve science. External issue460 closure remains separate.',
      'evidence_ids': ['issue460-body', 'issue460-creation', 'independent-title-assessment', 'metadata-correction-proof', 'final-target-bridge']})
    document['links'].append('https://github.com/CultureBotAI/PathwayMech/issues/460')
    for i, row in enumerate(failures):
        document['checks'].append({'check_id': 'original-failed-' + str(i), 'name': 'Retained original failed ' + row['name'],
          'status': 'failed', 'required': False, 'target_ids': [TARGET_ID], 'command': shlex.join(row['command']),
          'exit_code': row['exit_code'], 'summary': 'Original failed attempt preserved; successful required successor is separately bound. See exact streams and recovery receipt for cause, not an inferred failure class.',
          'evidence_ids': ['original-gate-failure', 'gate-recovery-inputs']})
    for key, area, summary, eids in [
      ('preservation', 'consistency', 'All230 prior records/pages, prior review/history bytes, old authority objects, old KGX edges and source-context multiplicities remain exact. Shared display deltas, if any, are separately byte-bound in preservation approval.', ['preservation', 'preservation-assessment']),
      ('representation', 'representation', visual['observed'] + ' Exact67 ordered graph triples,24 qualifier descriptions,145 evidence uses and94 reference identities are present.', ['page-qa', 'visual-observation'])]:
        document['checks'].append({'check_id': key, 'name': key, 'status': 'passed', 'required': True,
           'target_ids': [TARGET_ID], 'summary': summary, 'evidence_ids': eids})
        document['assessments'].append({'assessment_id': 'root-' + key, 'area': area, 'topic': key,
           'outcome': 'supported', 'target_ids': [TARGET_ID], 'summary': summary, 'evidence_ids': eids})
    assert all(row['accessed_at'].endswith('Z') for row in document['evidence'])
    assert_history(ROOT, gates['render_history_inputs'])
    write(out / 'completed-review.json', document)
    write(out / 'assembly.json', {'status': 'assembled_pending_root_read_native_serialization_save',
        'at': now(), 'target': bind(ROOT / TARGET_REL), 'review': bind(out / 'completed-review.json'),
        'inspection': bind(out / 'inspect.stdout'), 'archive_map': bind(archive / 'archive-map.json'),
        'gate_receipt': bind(args.gates), 'inputs': len(current_inputs), 'maintained_review_saved': False,
        'late_history_written': False})
    print(json.dumps(load(out / 'assembly.json'), indent=2))

if __name__ == '__main__':
    main()
