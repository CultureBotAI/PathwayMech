def digest(b):
    return hashlib.sha256(b).hexdigest()


def sha(p):
    h = hashlib.sha256()
    with Path(p).open("rb") as f:
        for chunk in iter(lambda: f.read(4 * 1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

    part = [e for e in record["mechanistic_edges"] if e["predicate"] == "part_of"]
    assert Counter(e["subject"] for e in part) == Counter("MetaCyc:" + x for x in members)
    affected = []
    for e in record["mechanistic_edges"]:
        if {e["subject"], e["object"]} & {
            "MetaCyc:RXN-23316",
            "MetaCyc:RXN-23312",
            "MetaCyc:N-23-DHB-Lysyl-MxcG",

# Native InChIs establish actual small-molecule atom/charge vectors.
def vector(ins):
    layers = ins.split("/")
    out = Counter()
    if not layers[1].startswith("p"):
        for atom, n in re.findall(r"([A-Z][a-z]?)(\d*)", layers[1]):
            out[atom] += int(n or 1)
    for layer in layers[1:]:
        if layer.startswith("p"):
            out["H"] += int(layer[1:])
            out["charge"] += int(layer[1:])
        elif layer.startswith("q"):
            out["charge"] += int(layer[1:])
    return out


def plus(a, b, n=1):
    c = a.copy()
    for k, v in b.items():
        c[k] += n * v
    return Counter({k: v for k, v in c.items() if v})
