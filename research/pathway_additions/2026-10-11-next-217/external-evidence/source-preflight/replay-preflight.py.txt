def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda: f.read(4 * 1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

sources = []
failures = []
for receipt in sorted(O.glob('acquisition*.json')):
    rows = json.loads(receipt.read_text())
    if isinstance(rows, dict):
        rows = rows.get('sources', rows.get('results', []))
    for row in rows:

def vector(inchi):
    assert inchi.startswith('InChI=1S/')
    layers = inchi.split('/')
    formula = layers[1]
    v = Counter()
    if not formula.startswith('p'):
        for element, count in re.findall(r'([A-Z][a-z]?)(\d*)', formula):
            v[element] += int(count or 1)
    for layer in layers[1:]:
        if layer.startswith('p'):
            v['H'] += int(layer[1:])
            v['charge'] += int(layer[1:])
        if layer.startswith('q'):
            v['charge'] += int(layer[1:])
    return v

def add(a, b, coefficient=1):
    c = a.copy()
    for key, val in b.items():
        c[key] += coefficient * val
    return Counter({k: v for k, v in c.items() if v})

carrier['23-DHB-MxcF'] = add(carrier['MxcF'], Counter({'C':7,'H':4,'O':3}))
carrier['Lysyl-MxcG'] = add(carrier['MxcG'], Counter({'C':6,'H':14,'N':2,'O':1,'charge':2}))
carrier['N-23-DHB-Lysyl-MxcG'] = add(carrier['MxcG'], Counter({'C':13,'H':17,'N':2,'O':4,'charge':1}))
carrier['N-N-di23-DHB-Lysyl-MxcG'] = add(carrier['MxcG'], Counter({'C':20,'H':20,'N':2,'O':7}))
vectors = {k: Counter(v['vector']) for k,v in chem.items()} | carrier
balances = {}
for frame, r in reactions.items():
    delta = Counter()
    for side, sign in [('left',-1),('right',1)]:
        for x in r['physiological_sides'][side]:
            delta = add(delta, vectors[x['id']], sign*x['coefficient'])
    assert not delta, (frame, delta)
    balances[frame] = {'residual':dict(delta),'kind':'exact small-molecule balance' if frame in {'DHBAMPLIG-RXN','RXN-22679','RXN-23314','RXN-23315'} else 'symbolic carrier conservation; residue deltas derived, not issued carrier formulae'}
nets = {}

    net = Counter()
    for frame in expected - ({'RXN-23315'} if name == 'A' else {'RXN-23314'}):
        n = 2 if frame in {'DHBAMPLIG-RXN','RXN-23310'} else 1
        for side, sign in [('left',-1),('right',1)]:
            for x in reactions[frame]['physiological_sides'][side]:
                net[x['id']] += n*sign*x['coefficient']
    net = {k:v for k,v in net.items() if v}
    assert not any(k in carrier for k in net)
    delta = Counter()
    for frame,n in net.items():delta = add(delta,vectors[frame],n)
    assert not delta
    nets[name] = {'coefficients_products_positive':net,'element_and_charge_residual':dict(delta)}
