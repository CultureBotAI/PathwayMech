from datetime import UTC, datetime
import hashlib
import json
from pathlib import Path

O = Path(__file__).resolve().parent

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

result = json.loads((O / 'replay-final.json').read_text())
execution = json.loads((O / 'replay-final-execution.json').read_text())
assert execution['exit_code'] == 0
assert execution['finished_at'] >= result['finished_at']
rows = []
failures = []
for receipt in sorted(O.glob('acquisition*.json')):
    data = json.loads(receipt.read_text())
    assert isinstance(data, list)
    for row in data:
        if 'path' not in row:
            failures.append({'receipt': str(receipt), **row})
            continue
        p = Path(row['path'])
        assert digest(p) == row['sha256'], p
        accepted = row.get('status') == 200 and p.name != 'metacyc-CPD-25916.xml'
        rows.append({**row, 'acquisition_receipt': str(receipt), 'accepted_source_body': accepted})
assert len(rows) == 66
assert len({x['path'] for x in rows}) == len(rows)
assert {str(p) for p in (O / 'originals').iterdir()} == {x['path'] for x in rows}
manifest = {'sealed_at': datetime.now(UTC).isoformat(), 'bodies': rows, 'failures_without_body': failures, 'scope': 'All66 original responses rehashed,59 accepted bodies. Failed response bodies are retained but excluded from evidence. Main scientific replay covered65 responses; the one later original Figure5 image is independently rehashed here and visually inspected. The ChEBI original SQL full digest is in replay-final.json.'}
with (O / 'source-manifest.json').open('x') as f:
    json.dump(manifest, f, indent=2)
    f.write('\n')
