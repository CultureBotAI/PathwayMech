"""Small fixtures exercise the actual driver guards; no scientific replay occurs."""
import importlib.util
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
OUT = ROOT / sys.argv[1]
OUT.mkdir(exist_ok=False)
spec = importlib.util.spec_from_file_location('integration_driver', ROOT / 'integration-replay.py')
d = importlib.util.module_from_spec(spec)
spec.loader.exec_module(d)
results = []


def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')


def entry(path, role='runtime'):
    return {'path': str(path), 'resolved_path': str(path.resolve()), 'sha256': d.file_hash(path), 'role': role}


def reject(name, function, message):
    try:
        function()
    except d.IntegrityError as error:
        d.require(message in str(error), 'Unexpected rejection for ' + name + ': ' + str(error))
        results.append({'name': name, 'passed': True, 'actual_exception': type(error).__name__, 'message': str(error)})
    else:
        raise RuntimeError('Guard failed to reject: ' + name)


def fixture(name, role='runtime'):
    root = OUT / name
    root.mkdir()
    runtime = root / 'runtime'
    runtime.mkdir()
    path = runtime / 'input.txt'
    path.write_text('accepted bytes\n')
    document = {'manifest_identity': {'path': str(root / 'manifest.json'), 'resolved_path': str(root / 'manifest.json')},
                'files': [entry(path, role)], 'inventories': [{'root': str(runtime), 'paths': d.inventory(runtime)}]}
    write(root / 'manifest.json', document)
    own = entry(root / 'manifest.json', 'manifest')
    d.capture_bindings(document['files'])
    d.final_recheck(document, own)
    return root, path, document, own


try:
    # Positive narrow fixture demonstrates the same final guard accepts intact inputs.
    fixture('positive-narrow')
    results.append({'name': 'positive-narrow-capture-and-final-recheck', 'passed': True})
    for role in ['authority', 'policy', 'runtime', 'manifest']:
        for phase in ['preflight', 'final']:
            root, path, document, own = fixture(phase + '-' + role, role)
            path.write_text('changed ' + role + '\n')
            operation = (lambda document=document: d.capture_bindings(document['files'])) if phase == 'preflight' else (lambda document=document, own=own: d.final_recheck(document, own))
            reject(phase + '-' + role + '-digest-change', operation, 'Input digest changed')
    root, path, document, own = fixture('late-pre-import-formal-change', 'authority')
    path.write_text('changed after initial capture\n')
    reject('late-pre-import-formal-recheck', lambda: d.recheck_formal_inputs(document, own), 'Input digest changed')
    for phase in ['preflight', 'final']:
        root, path, document, own = fixture(phase + '-own-manifest', 'runtime')
        manifest = Path(own['path'])
        manifest.write_bytes(manifest.read_bytes() + b' ')
        operation = (lambda: d.capture_manifest(manifest, own['sha256'])) if phase == 'preflight' else (lambda: d.final_recheck(document, own))
        reject(phase + '-binding-manifest-change', operation, 'manifest digest changed' if phase == 'preflight' else 'Input digest changed')
    for phase in ['preflight', 'final']:
        root, path, document, own = fixture(phase + '-symlink')
        destination = root / 'same-byte-destination.txt'
        destination.write_bytes(path.read_bytes())
        link = root / 'input-alias.txt'
        link.symlink_to(path)
        alias_entry = entry(link)
        document['files'].append(alias_entry)
        d.capture_bindings(document['files'])
        link.unlink()
        link.symlink_to(destination)
        d.require(d.file_hash(link) == alias_entry['sha256'], 'Same-byte fixture has unequal bytes')
        operation = (lambda: d.capture_bindings(document['files'])) if phase == 'preflight' else (lambda: d.final_recheck(document, own))
        reject(phase + '-same-byte-symlink-retarget', operation, 'Resolved input identity changed')
    for phase in ['preflight', 'final']:
        root, path, document, own = fixture(phase + '-manifest-symlink')
        original = root / 'manifest.json'
        identical = root / 'same-manifest.json'
        identical.write_bytes(original.read_bytes())
        original.rename(root / 'original-manifest.json')
        original.symlink_to(identical)
        operation = (lambda: d.capture_manifest(original, own['sha256'])) if phase == 'preflight' else (lambda: d.final_recheck(document, own))
        reject(phase + '-same-byte-manifest-symlink-retarget', operation, 'identity changed' if phase == 'preflight' else 'Resolved input identity changed')
    for phase in ['preflight', 'final']:
        for mutation in ['add', 'remove']:
            root, path, document, own = fixture(phase + '-inventory-' + mutation)
            if mutation == 'add':
                (path.parent / '.ignored-new-runtime.py').write_text('changed=True\n')
            else:
                path.unlink()
            operation = (lambda: d.check_inventories(document['inventories'], document['files'])) if phase == 'preflight' else (lambda: d.final_recheck(document, own))
            reject(phase + '-inventory-' + mutation, operation, 'Input inventory changed')
    root = OUT / 'output-fixtures'
    root.mkdir()
    for kind in ['file', 'directory', 'dangling-symlink']:
        path = root / kind
        if kind == 'file':
            path.write_text('existing\n')
        elif kind == 'directory':
            path.mkdir()
        else:
            path.symlink_to(root / 'absent')
        reject('existing-output-' + kind, lambda: d.check_output(path, [], []), 'Output already exists')
    protected = root / 'protected'
    protected.mkdir()
    reject('protected-output-tree', lambda: d.check_output(protected / 'new', [str(protected)], []), 'protected input tree')
    root, path, document, own = fixture('duplicate-bindings')
    reject('duplicate-binding-path', lambda: d.capture_bindings(document['files'] * 2), 'Duplicate binding path')
    reject('duplicate-json-key', lambda: d.read_json('{"input":1,"input":2}'), 'Duplicate JSON key')
    # Real CLI runs stop before main project/helper imports and before output creation.
    cli_out = OUT / 'must-not-exist'
    cases = [
        ('optimized-cli', [sys.executable, '-B', '-O', str(ROOT / 'integration-replay.py')], 'Optimized Python is forbidden'),
        ('bad-manifest-digest-cli', [sys.executable, '-B', str(ROOT / 'integration-replay.py'), '--repo', str(ROOT), '--helper', str(ROOT / 'integration-replay.py'), '--binding-manifest', own['path'], '--binding-sha256', '0' * 64, '--output-dir', str(cli_out)], 'Binding-manifest digest changed'),
    ]
    for name, command, message in cases:
        process = subprocess.run(command, capture_output=True)
        (OUT / (name + '.stdout')).write_bytes(process.stdout)
        (OUT / (name + '.stderr')).write_bytes(process.stderr)
        d.require(process.returncode != 0 and message in process.stderr.decode(), 'CLI rejection failed: ' + name)
        d.require(not cli_out.exists(), 'Rejected CLI created output')
        results.append({'name': name, 'passed': True, 'command': command, 'exit_code': process.returncode,
                        'stdout_sha256': d.digest(process.stdout), 'stderr_sha256': d.digest(process.stderr), 'output_created': False})
    report = {'status': 'passed_narrow_boundary_controls_only', 'at': d.now(),
              'driver': entry(ROOT / 'integration-replay.py'), 'controls': entry(Path(__file__).resolve()),
              'cases': results, 'count': len(results),
              'scope': 'Actual unchanged driver functions and two early-reject real CLI invocations. Small external fixtures only; no scientific helper, canonical project checks, 20GB source rereads, or full positive integration replay. No injection, monkeypatch, or production bypass is used.'}
    write(OUT / 'results.json', report)
    print(json.dumps({'status': report['status'], 'cases': len(results), 'receipt': str(OUT / 'results.json'), 'sha256': d.file_hash(OUT / 'results.json')}))
except Exception as error:
    write(OUT / 'failure.json', {'status': 'failed_control_attempt', 'error': repr(error), 'completed_cases': results})
    raise
