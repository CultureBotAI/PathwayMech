def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def write(path, document):
    with path.open('x') as stream:
        json.dump(document, stream, indent=2)
        stream.write('\n')

def package(path, relative):
    require(path.is_file() or path.is_symlink(), 'Unexpected package entry')
    declared = str(path)
    if path.is_symlink():
        original = str(path.readlink()).encode()
        document = {'format': 'immutable-symlink-observation-v1', 'original_path': declared,
                    'link_target': original.decode(), 'resolved_path': str(path.resolve()),
                    'target_exists': path.exists(), 'target_sha256': sha(path.read_bytes()) if path.is_file() else None}
        relative = Path(str(relative) + '.symlink.json')
        raw = (json.dumps(document, indent=2) + '\n').encode()
        mode = 'symlink_observation_no_maintained_symlink'
    else:
        original = path.read_bytes()
        fixture = 'controls-attempt-1' in path.parts and path.name != 'results.json'
        if path.suffix == '.py':
            relative = Path(str(relative) + '.txt')
            raw, mode = original, 'byte_exact_inert_script'
        elif fixture or path.suffix in {'.stdout', '.stderr', '.tsv'}:
            document = {'format': 'lossless-immutable-evidence-v1', 'original_sha256': sha(original),
                        'original_bytes': len(original), 'encoding': 'utf-8', 'content': original.decode('utf-8')}
            relative = Path(str(relative) + '.bytes.json')
            raw = (json.dumps(document, indent=2) + '\n').encode()
            require(json.loads(raw)['content'].encode('utf-8') == original, 'Wrapper round trip failed')
            mode = 'lossless_utf8_wrapper'
        else:
            raw, mode = original, 'byte_exact'
    destination = STAGE / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open('xb') as stream:
        stream.write(raw)
    require(destination.suffix != '.py', 'Active Python would enter maintained tree')
    proposed = DOSSIER / relative
    require(not (REPO / proposed).exists() and not (REPO / proposed).is_symlink(), 'Proposed maintained collision')
    rows.append({'source_path': declared, 'source_resolved_path': str(path.resolve()), 'source_sha256': sha(original),
                 'source_bytes': len(original), 'mode': mode, 'packaged_path': str(destination),
                 'packaged_sha256': sha(raw), 'maintained_relative_path': str(proposed)})
