 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def git(*a):return subprocess.check_output(['git',*a],cwd=root,text=True).strip()
review=Path('/private/tmp/pathwaymech-next-217-authority-root-review.json')
assert sha(review)=='2d3fd54c7f707b33e6e780f734c82e3813a6373d93f65c1b7cc4471408d27b51'

assert git('status','--porcelain=v1') in ('?? .venv','')
selection_path=prep/'selected/selection.json';assert sha(selection_path)==acceptance['selection_sha256']
s=json.loads(selection_path.read_text());lock=s['preparation_baseline']
