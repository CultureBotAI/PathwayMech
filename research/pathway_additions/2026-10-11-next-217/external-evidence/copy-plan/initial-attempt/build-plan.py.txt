def sha(p):
    with Path(p).open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

def capture(p, expected=None):
    p = Path(p).absolute()
    resolved = p.resolve(strict=True)
    b = p.read_bytes()
    h = hashlib.sha256(b).hexdigest()
    if expected is not None and h != expected:
        raise ValueError('source digest mismatch: ' + str(p))
    if str(p) in bindings and bindings[str(p)] != (str(resolved), h):
        raise ValueError('source changed during capture: ' + str(p))
    bindings[str(p)] = (str(resolved), h)
    return b

def read(p):
    return json.loads(capture(p))

def save(p, obj):
    p.parent.mkdir(parents=True, exist_ok=True)
    with p.open('x') as f:
        json.dump(obj, f, indent=2, ensure_ascii=False)
        f.write('\n')

def bad_whitespace(b):
    return bool(b) and (any(line.rstrip(b'\r\n').endswith((b' ', b'\t')) for line in b.splitlines(keepends=True)) or b.endswith(b'\n\n'))

def add(p, dest, role, *, wrap=False, active=False, unique=False, source_name=None):
    p = Path(p).absolute()
    if str(p) in paths:
        return paths[str(p)]
    b = capture(p)
    digest = hashlib.sha256(b).hexdigest()
    dest = Path(dest)
    if not active:
        if dest.suffix in {'.py', '.yaml', '.yml', '.log'}:
            dest = dest.with_name(dest.name + '.txt')
    # All fixtures are wrapped, including malformed JSON, exact YAML, Python,
    # output sentinels and no-final-newline bodies. Never parse/normalize them.
    wrap = wrap or p.suffix in {'.log', '.tsv', '.jsonl', '.diff'} or bad_whitespace(b)
    if active and wrap:
        raise ValueError('current active bytes require review, not normalization: ' + str(p))
    mode = 'lossless_json_wrapper' if wrap else 'exact_bytes'
    key = (digest, mode)
    if key in canon and not unique:
        item = dict(path=str(p), sha256=digest, bytes=len(b), disposition='exact-byte-alias', destination=canon[key], mode=mode, role=role)
        aliases.append(item)
        paths[str(p)] = item
        return item
    source = p
    if wrap:
        try:
            content = b.decode('utf-8')
            assert content.encode('utf-8') == b
            encoding = 'utf-8'
        except UnicodeDecodeError:
            content = base64.b64encode(b).decode('ascii')
            encoding = 'base64'
        source = HERE / 'payload' / (digest + '.json')
        if not source.exists():
            save(source, dict(format='lossless-immutable-evidence-v1', original_sha256=digest, original_bytes=len(b), encoding=encoding, content=content))
        dest = dest.with_name(dest.name + '.json')
    if dest.is_absolute() or '..' in dest.parts or any(x['destination'] == str(dest) for x in copies):
        raise ValueError('invalid or colliding destination: ' + str(dest))
    item = dict(path=str(p), sha256=digest, bytes=len(b), source_name=source_name or p.name, disposition='planned-copy', destination=str(dest), copy_source=str(source), destination_sha256=sha(source), mode=mode, role=role)
    copies.append(item)
    canon.setdefault(key, str(dest))
    paths[str(p)] = item
    return item

def archive(p, group, rel=None, **kw):
    return add(p, DOSSIER/'external-evidence'/group/(rel or Path(p).name), 'immutable external evidence; not a historical canonical review', **kw)

def external(p, reason):
    p = Path(p).absolute()
    if str(p) in paths:
        return
    if p.is_symlink():
        entry = dict(path=str(p), disposition='retained-external-symlink', target=os.readlink(p), reason=reason)
    else:
        b = capture(p)
        entry = dict(path=str(p), sha256=hashlib.sha256(b).hexdigest(), bytes=len(b), disposition='retained-external', reason=reason)
    paths[str(p)] = entry
    excluded.append(entry)
