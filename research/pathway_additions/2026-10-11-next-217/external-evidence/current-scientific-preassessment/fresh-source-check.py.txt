assertions = []
for edge in record['mechanistic_edges']:
    for n, e in enumerate(edge['evidence']):
        ref = refs[e['reference_id']]
        path = sources[ref['source_sha256']]
        locator = e['source_locator']
        matches = []
        if locator.startswith('/'):
            if path.suffix == '.json':
                body = json.loads(path.read_text())
                for selector in locator.split('; '):
                    value = body
                    for key in selector.strip('/').split('/'):
                        key = key.replace('~1', '/').replace('~0', '~')
                        value = value[int(key)] if isinstance(value, list) else value[key]
                    matches.append(1)
            else:
                body = html.fromstring(path.read_bytes()) if path.suffix == '.html' else etree.fromstring(path.read_bytes())
                for selector in locator.split('; '):
                    value = body.xpath(selector)
                    assert value, (edge['id'], n, selector)
                    matches.append(len(value))
        assertions.append({'edge':edge['id'],'evidence_index':n,'reference':ref['id'],'body_sha256':ref['source_sha256'],'locator':locator,'matches':matches})
assert len(assertions) == 239

    e = etree.parse(str(P/f'metacyc-{name}{suffix}.xml')).xpath(f'/*/*[@frameid="{name}"]')
    assert len(e) == 1
    return e[0]
