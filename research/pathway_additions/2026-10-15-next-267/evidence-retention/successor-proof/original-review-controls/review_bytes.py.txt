def sha(body):
    return hashlib.sha256(body).hexdigest()

def unique(pairs):
    d = {}
    for k, v in pairs:
        need(k not in d, ('duplicate JSON key', k))
        d[k] = v
    return d

def loads(body):
    return json.loads(body, object_pairs_hook=unique)

def load(path):
    return loads(path.read_bytes())

def save(name, obj):
    (OUT / name).write_text(json.dumps(obj, indent=2, sort_keys=True) + '\n')

def identity(path):
    h = hashlib.sha256()
    size = 0
    with path.open('rb') as f:
        while part := f.read(1024 * 1024):
            h.update(part)
            size += len(part)
    return {'sha256': h.hexdigest(), 'bytes': size}

def inventory(root):
    result = {}
    for dirname, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            path = Path(dirname) / name
            need(not path.is_symlink(), ('symlink in frozen inventory', str(path)))
        for name in files:
            path = Path(dirname) / name
            need(path.is_file(), ('not regular file', str(path)))
            result[str(path.relative_to(root))] = identity(path)
    return result

def path(root, relative):
    need(isinstance(relative, str) and relative and '\\' not in relative, ('path syntax', relative))
    rel = Path(relative)
    need(not rel.is_absolute() and '..' not in rel.parts and str(rel) == relative and relative != '.', ('unsafe path', relative))
    target = root / rel
    need(target.resolve().is_relative_to(root.resolve()), ('escape', relative))
    return target

def check(body, digest, size):
    need(type(size) is int and size >= 0 and len(body) == size and sha(body) == digest, ('identity', digest, size, len(body)))

def unzip(body, digest, size):
    # Independent stdlib gzip path validates the complete stream and trailer.
    result = gzip.decompress(body)
    check(result, digest, size)
    return result
