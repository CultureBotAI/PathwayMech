import datetime
import hashlib
import json
import os
from pathlib import Path

ROOT=Path(__file__).parent
def identity(path):
    h=hashlib.sha256();size=0
    with path.open('rb') as stream:
        while part:=stream.read(1024*1024):h.update(part);size+=len(part)
    return {'sha256':h.hexdigest(),'bytes':size}
def dump(path,value):path.write_text(json.dumps(value,indent=2,sort_keys=True)+'\n')
result=json.loads((ROOT/'complete-byte-review.json').read_text())
assert result['status']=='pass'
assert json.loads((ROOT/'frozen-before.json').read_text())==json.loads((ROOT/'frozen-after.json').read_text())
assert json.loads((ROOT/'source-owners-before.json').read_text())==json.loads((ROOT/'source-owners-after.json').read_text())
adversarial=json.loads((ROOT/'independent-adversarial-results.json').read_text())
assert adversarial['rejected_controls']==37 and adversarial['unexpected_acceptances']==7
assert len(json.loads((ROOT/'findings.json').read_text())['findings'])==2
handoff='''# Independent next267 retention review handoff

Verdict: **needs curation**. Complete independent recovery and input preservation passed. Two recovery-helper defects remain in the exact original payload: R01, moderate, historical completeness/provenance (#604); R02, minor, resolved materialization-root confinement (#605). - Reviewed proposed payload manifest: `9a81127803cb3e0688736a1aeab036d118d1e84f4f46a53f3e3a825d3eb5bcd7`, 5,896 physical files.
- Independent reconstruction: all 9,202 contents; all 13,577 physical occurrences from 35 owners; all 6,175 current compact declarations; all 477 historical pairs and 926 historical paths.
- Complete exact inherited provenance: 1,971 actual merged Git paths and 2,599 content rows; exact commit/tree/path/blob/bytes/decode checks.
- Original compressed representations: 4,534 distinct physically owned gzip identities at 5,955 paths, with every gzip layer decoded and exact original compressed bytes retained.
- All 64 reference originals and 100 edge bindings verified, including the three archived prior266 source members. Four external database originals remain explicitly out of cutoff.
- All 5,809 new storage endpoints are below 100 MiB; total 368,457,730 bytes and maximum 46,024,886 bytes. Four-chunk Noctua reconstruction passed.
- Independent adversarial controls: 37 expected rejections, three positive controls, seven unexpected acceptances supporting precisely two findings. Outside-scratch destination probes intercepted the first mkdir: zero out-of-scope writes.

Read `ASSESSMENT.md`, `findings.json`, `complete-byte-review.json`, `independent-adversarial-results.json`, and the exact before/after inventories. All review execution sessions terminated before freeze. `artifact-manifest.json` binds every regular review file, including hidden/ignored files and failed attempts, excluding only itself and `freeze-proof.json`.
'''
(ROOT/'HANDOFF.md').write_text(handoff)
rows=[]
for dirname,dirs,files in os.walk(ROOT,followlinks=False):
    for name in dirs+files:assert not (Path(dirname)/name).is_symlink()
    for name in files:
        path=Path(dirname)/name;rel=str(path.relative_to(ROOT))
        if rel in {'artifact-manifest.json','freeze-proof.json'}:continue
        rows.append({'path':rel,**identity(path)})
rows.sort(key=lambda r:r['path'])
dump(ROOT/'artifact-manifest.json',{'scope':'Complete frozen independent external review including actual reconstruction outputs and preserved failures; original scientific roles unchanged.','excluded_self_records':['artifact-manifest.json','freeze-proof.json'],'files':rows})
