 p=OUT/name;assert not p.exists(),p;p.write_text(json.dumps(data,indent=2,ensure_ascii=False)+'\n')

assert sha(ROOT/'conf/identifier_sources.json')==au['bindings']['/private/tmp/pathwaymech-next-260-authority-integration-successor/proposal/identifier_sources.json']
