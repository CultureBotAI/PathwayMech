def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def write_json(p, obj):
    assert not p.exists(), p
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(obj, indent=2, ensure_ascii=False) + '\n')

def differences(a, b, path=''):
    result = []
    if isinstance(a, dict) and isinstance(b, dict):
        for key in sorted(set(a) | set(b)):
            if key not in a or key not in b:
                result.append({'path': path + '/' + key, 'before': a.get(key), 'after': b.get(key)})
            else:
                result += differences(a[key], b[key], path + '/' + key)
    elif isinstance(a, list) and isinstance(b, list):
        if all(isinstance(v, dict) and 'id' in v for v in a + b):
            return differences({v['id']: v for v in a}, {v['id']: v for v in b}, path)
        assert len(a) == len(b), path
        for i, (left, right) in enumerate(zip(a, b)):
            result += differences(left, right, path + '/' + str(i))
    elif a != b:
        result.append({'path': path, 'before': a, 'after': b})
    return result
