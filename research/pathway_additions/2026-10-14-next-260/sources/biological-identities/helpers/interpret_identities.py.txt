    if item['object_type']!='Protein':continue
    frame=item['identifier'].split(':',1)[1];node=ET.parse(item['source']['path']).getroot().find('Protein')
    components=[]
    for slot in node.findall('component'):
        p=slot.find('Protein');components.append({'native_component':'MetaCyc:'+p.attrib['frameid'],'coefficient':slot.findtext('coefficient'),'coefficient_source_xpath':f"/ptools-xml/Protein[@frameid='{frame}']/component[{len(components)+1}]/coefficient"})

    rows.append({'native_identifier':item['identifier'],'native_source':item['source'],'native_genes':['MetaCyc:'+x['frame'] for x in item['native_refs']['gene']],'native_species':[x['frame'] for x in item['native_refs']['species']],'components':components,'reference_uniprot_ids':['UniProtKB:'+x for x in up],'reference_deposit_bindings':evidence,'scope_limit':limit,'preparation_scope_authority':'Native activity/preparation ledger and independent primary dossier control biological interpretation; this source author supplies identity only.'})
