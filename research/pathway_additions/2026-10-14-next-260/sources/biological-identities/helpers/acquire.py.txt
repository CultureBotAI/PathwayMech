"""Acquire exact public issuing objects; preserve attempts without overwrites."""
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
from pathlib import Path
import hashlib, json, sys, urllib.error, urllib.request

B = Path(__file__).resolve().parents[1]
attempt = sys.argv[1]
assert attempt in ['sandbox', 'public']
requests_file = B / 'requests.json'
requests = json.loads(requests_file.read_text())
if attempt == 'public':
    old = json.loads((B / 'receipts/acquisition-sandbox.json').read_text())
    successful = {x['key'] for x in old['observations'] if x.get('status') == 200}
    requests = [x for x in requests if x['key'] not in successful]
dest = B / 'attempts' / attempt
dest.mkdir(parents=True, exist_ok=False)

def fetch(row):
    result = {**row, 'started_utc': datetime.now(timezone.utc).isoformat()}
    path = dest / row['filename']
    try:
        req = urllib.request.Request(row['url'], headers={'User-Agent': 'PathwayMech source identity research; public read-only request'})
        with urllib.request.urlopen(req, timeout=45) as response:
            body = response.read()
            result.update(status=response.status, final_url=response.url, headers={k.lower(): v for k, v in response.headers.items() if k.lower() in ['content-type', 'content-length', 'location', 'x-uniprot-release', 'x-uniprot-release-date']})
        path.write_bytes(body)
        result.update(path=str(path), sha256=hashlib.sha256(body).hexdigest(), bytes=len(body))
    except urllib.error.HTTPError as error:
        body = error.read()
        path.write_bytes(body)
        result.update(status=error.code, final_url=error.url, path=str(path), sha256=hashlib.sha256(body).hexdigest(), bytes=len(body), error=str(error))
    except Exception as error:
        result.update(error=type(error).__name__ + ': ' + str(error))
    result['finished_utc'] = datetime.now(timezone.utc).isoformat()
    record = dest / (row['filename'] + '.receipt.json')
    record.write_text(json.dumps(result, indent=2) + '\n')
    return result

with ThreadPoolExecutor(max_workers=3) as pool:
    observations = list(pool.map(fetch, requests))
receipt = B / 'receipts' / ('acquisition-' + attempt + '.json')
assert not receipt.exists()
receipt.write_text(json.dumps({'attempt': attempt, 'request_plan_sha256': hashlib.sha256(requests_file.read_bytes()).hexdigest(), 'observations': observations, 'scope': 'Public unauthenticated requests; safe header allowlist; raw source bodies remain external; HTTP success still requires semantic/identity parsing.'}, indent=2) + '\n')
print(json.dumps({'attempt': attempt, 'requests': len(requests), 'http_200': sum(x.get('status') == 200 for x in observations), 'failures': [{k: x.get(k) for k in ['key', 'status', 'error']} for x in observations if x.get('status') != 200]}, indent=2))
