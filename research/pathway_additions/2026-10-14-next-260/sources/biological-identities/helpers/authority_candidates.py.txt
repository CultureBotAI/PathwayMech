def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def save(name,value):

profiles=[]
def probe(kind,path,url,version,provenance,requested=None):
    path=Path(path);identity=(kind,str(path),url,version)
    for old in profiles:
        if (old['kind'],old['original']['path'],old['url'],old['version'])==identity:return old
    row={'profile_id':'P'+str(len(profiles)+1),'kind':kind,'original':bind(path),'url':url,'version':version,'provenance':provenance,'requested_identity':requested}
    try:
        terms=list(source_terms(kind,path,url=url,version=version));row['terms']=terms;row['status']='accepted' if terms else 'no_authority_term'
    except Exception as error:row.update(status='rejected',error=type(error).__name__+': '+str(error),terms=[])
    profiles.append(row);return row
