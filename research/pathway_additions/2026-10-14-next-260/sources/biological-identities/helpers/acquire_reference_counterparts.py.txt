"""Exact native RefSeq and original GalD/GalC protein-accession counterparts."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib,json,urllib.request,urllib.error
B=Path(__file__).resolve().parents[1]
accessions=['NP_744662.1','NP_744663.1','AAN68125.1','AAN68126.1']
url='https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=protein&id='+','.join(accessions)+'&rettype=gb&retmode=xml'
target=B/'attempts/public-reference-counterparts';target.mkdir(parents=True,exist_ok=False)
row={'url':url,'requested_accessions':accessions,'started_utc':datetime.now(timezone.utc).isoformat()}
path=target/'reference-counterparts.xml'
try:
    with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech read-only public source identity research'}),timeout=60) as response:
        data=response.read();row.update(status=response.status,final_url=response.url,headers={k.lower():v for k,v in response.headers.items() if k.lower() in ['content-type','content-length']})
    path.write_bytes(data);row.update(path=str(path),bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
except urllib.error.HTTPError as error:
    data=error.read();path.write_bytes(data);row.update(status=error.code,error=str(error),path=str(path),bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
except Exception as error:row['error']=type(error).__name__+': '+str(error)
row['finished_utc']=datetime.now(timezone.utc).isoformat()
p=B/'receipts/acquisition-public-reference-counterparts.json';assert not p.exists();p.write_text(json.dumps(row,indent=2)+'\n')
print(json.dumps({'status':row.get('status'),'sha256':row.get('sha256'),'error':row.get('error')},indent=2))
