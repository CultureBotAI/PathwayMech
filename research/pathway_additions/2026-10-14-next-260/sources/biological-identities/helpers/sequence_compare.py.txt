"""Translate original INSDC CDS coordinates and compare exact reference sequences."""
from pathlib import Path
import hashlib, json, re, xml.etree.ElementTree as ET

B=Path(__file__).resolve().parents[1]
def bind(path):
    path=Path(path);data=path.read_bytes()
    return {'path':str(path),'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()}
def seqhash(sequence):return hashlib.sha256(sequence.encode()).hexdigest()
def parse_embl(path):
    raw=Path(path).read_text();header=raw.splitlines()[0]
    accession=re.fullmatch(r'ID\s+(\S+); SV (\d+);.* (\d+) BP\.',header)
    assert accession,header
    dna=''.join(re.findall('[a-z]+',raw.split('\nSQ   ')[1].split('\n',1)[1].split('//')[0])).upper()
    assert len(dna)==int(accession[3])
    features=[];current=None
    for line in raw.splitlines():
        if not line.startswith('FT   '):continue
        key=line[5:21].strip();value=line[21:].strip()
        if key:
            current={'type':key,'location':value,'qualifier_lines':[]};features.append(current)
        elif current is not None:current['qualifier_lines'].append(value)
    for feature in features:
        text=' '.join(feature['qualifier_lines']);qualifiers={}
        for match in re.finditer(r'/([a-zA-Z0-9_]+)(?:="([^"]*)"|=([^ /]+))?',text):
            key,value,unquoted=match.groups();value=value if value is not None else unquoted
            if key=='translation':value=re.sub(r'\s+','',value)
            qualifiers.setdefault(key,[]).append(value)
        feature['qualifiers']=qualifiers
    return {'accession':accession[1]+'.'+accession[2],'dna':dna,'features':features,'source':bind(path),'source_annotations':[f for f in features if f['type']=='source']}

code={a+b+c:aa for (a,b,c),aa in zip(((a,b,c)for a in 'TCAG' for b in 'TCAG' for c in 'TCAG'),'FFLLSSSSYY**CC*WLLLLPPPPHHQQRRRRIIIMTTTTNNKKSSRRVVVVAAAADDEEGGGG')}
def translate(feature,dna):
    match=re.fullmatch(r'(complement\()?([0-9]+)\.\.([0-9]+)\)?',feature['location'])
    assert match,feature['location']
    reverse,start,end=match.groups();nt=dna[int(start)-1:int(end)]
    if reverse:nt=nt.translate(str.maketrans('ACGT','TGCA'))[::-1]
    q=feature['qualifiers'];assert q['codon_start']==['1'] and q['transl_table']==['11'],q
    assert len(nt)%3==0 and nt[:3] in ['ATG','GTG','TTG','CTG','ATT','ATC','ATA']
    aa=''.join(code[nt[i:i+3]] for i in range(0,len(nt),3));assert aa.endswith('*') and '*' not in aa[:-1]
    aa='M'+aa[1:-1];assert aa==q['translation'][0]
    return nt,aa
