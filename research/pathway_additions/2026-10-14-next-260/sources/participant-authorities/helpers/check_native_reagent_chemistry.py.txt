def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''): h.update(block)
    return h.hexdigest()

def describe(m):
    if m is None: return {'parse_error': True}
    Chem.AssignStereochemistry(m, cleanIt=True, force=True)
    unstereo = Chem.Mol(m); Chem.RemoveStereochemistry(unstereo)
    return {'smiles': Chem.MolToSmiles(m, isomericSmiles=True),
            'connectivity_smiles': Chem.MolToSmiles(unstereo, isomericSmiles=False),
            'inchi': Chem.MolToInchi(m), 'key': Chem.MolToInchiKey(m),
            'formula': rdMolDescriptors.CalcMolFormula(m), 'charge': Chem.GetFormalCharge(m),
            'atoms': dict(collections.Counter(a.GetSymbol() for a in Chem.AddHs(m).GetAtoms())),
            'atom_formal_charges': sorted((a.GetSymbol(), a.GetFormalCharge()) for a in m.GetAtoms() if a.GetFormalCharge()),
            'chiral_centers': Chem.FindMolChiralCenters(m, includeUnassigned=True)}

def from_cml(e):
    atoms = e.findall('./atomArray/atom'); rw = Chem.RWMol(); amap = {}; changes = []
    conf = Chem.Conformer(len(atoms)); conf.Set3D(False)
    for a in atoms:
        raw = a.attrib['elementType']; symbol = raw[0].upper() + raw[1:].lower()
        if symbol != raw: changes.append({'atom': a.attrib['id'], 'original_elementType': raw, 'interpreted_symbol': symbol})
        at = Chem.Atom(symbol); at.SetFormalCharge(int(a.get('formalCharge', '0')))
        idx = rw.AddAtom(at); amap[a.attrib['id']] = idx
        conf.SetAtomPosition(idx, (float(a.get('x2', '0')), float(a.get('y2', '0')), 0.))
    for b in e.findall('./bondArray/bond'):
        a, z = b.get('atomRefs').split()
        rw.AddBond(amap[a], amap[z], {1: Chem.BondType.SINGLE, 2: Chem.BondType.DOUBLE, 3: Chem.BondType.TRIPLE}[int(b.get('order'))])
    m = rw.GetMol(); Chem.SanitizeMol(m); m.AddConformer(conf); Chem.DetectBondStereochemistry(m)
    return m, changes
