from pathlib import Path
import datetime,hashlib,json,urllib.request
ROOT=Path('/private/tmp/pathwaymech-next-260-participant-authorities')
OUT=ROOT/'retained/pubchem-additional-acquisition.json'
assert not OUT.exists()
rows=[]
for cid in ['3727','6098323']:
 p=ROOT/'originals'/('PubChem-'+cid+'.json');assert not p.exists()
 url='https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/cid/'+cid+'/JSON'
 row={'requested_cid':int(cid),'url':url,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech-source-research/1.0'}),timeout=45) as r:
   body=r.read();row.update({'status':r.status,'final_url':r.url,'content_type':r.headers.get('Content-Type'),'sha256':hashlib.sha256(body).hexdigest(),'bytes':len(body)})
  j=json.loads(body);row['returned_cid']=j['PC_Compounds'][0]['id']['id']['cid'];row['exact_id']=row['returned_cid']==int(cid)
  with p.open('xb') as f:f.write(body)
  row['path']=str(p)
 except Exception as e:row['error']=repr(e)
 rows.append(row)
with OUT.open('x') as f:json.dump(rows,f,indent=2);f.write('\n')
print(json.dumps(rows,indent=2))
