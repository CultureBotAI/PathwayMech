def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(8*1024*1024),b''):h.update(b)
 return h.hexdigest()

def formula_atoms(formula):
 result=collections.Counter()
 for component in formula.split('.'):
  match=re.match(r'^(\d+)(.*)$',component)
  scale=int(match[1]) if match else 1;component=match[2] if match else component
  tokens=re.findall(r'([A-Z][a-z]?)([0-9]*)',component)
  assert ''.join(a+b for a,b in tokens)==component,formula
  for atom,n in tokens:result[atom]+=scale*int(n or 1)
 return dict(result)
