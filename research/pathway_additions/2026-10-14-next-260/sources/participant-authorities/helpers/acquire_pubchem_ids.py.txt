from pathlib import Path
import datetime,hashlib,json,sys,urllib.request
ROOT=Path('/private/tmp/pathwaymech-next-260-participant-authorities')
out=ROOT/'retained'/('pubchem-ids-acquisition-'+sys.argv[1]+'.json');assert not out.exists();rows=[]
for cid in sys.argv[2:]:
 p=ROOT/'originals'/('PubChem-'+cid+'.json');assert not p.exists();url='https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/cid/'+cid+'/JSON'
 r={'requested_cid':int(cid),'url':url,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech-source-research/1.0'}),timeout=40) as response:
   body=response.read();r.update({'status':response.status,'final_url':response.url,'bytes':len(body),'sha256':hashlib.sha256(body).hexdigest()})
  j=json.loads(body);r['returned_cid']=j['PC_Compounds'][0]['id']['id']['cid'];r['identity_exact']=r['returned_cid']==int(cid)
  with p.open('xb') as f:f.write(body)
  r['path']=str(p)
 except Exception as exc:r['error']=repr(exc)
 rows.append(r)
with out.open('x') as f:json.dump(rows,f,indent=2);f.write('\n')
print(json.dumps(rows,indent=2))
