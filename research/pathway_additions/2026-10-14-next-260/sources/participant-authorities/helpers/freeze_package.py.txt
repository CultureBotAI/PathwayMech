from pathlib import Path
import datetime,hashlib,json

MANIFEST=ROOT/'artifact-manifest.json';SELECTION=ROOT/'compact-retention-selection.json'
assert not MANIFEST.exists() and not SELECTION.exists()
def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
 return h.hexdigest()
verification=ROOT/'retained/primary-binding-and-verification.json';v=json.loads(verification.read_text());assert v['all_bounded_checks_pass']
for binding in v['proof_bindings']:
 p=Path(binding['path']);assert sha(p)==binding['sha256']
raw=json.loads((ROOT/'retained/external-originals-manifest.json').read_text())
for binding in raw['files']:
 p=Path(binding['path']);assert p.stat().st_size==binding['bytes'] and sha(p)==binding['sha256']
files=[]
for p in sorted(ROOT.rglob('*')):

  assert not p.is_symlink(),p
  files.append({'path':str(p.relative_to(ROOT)),'bytes':p.stat().st_size,'sha256':sha(p)})
