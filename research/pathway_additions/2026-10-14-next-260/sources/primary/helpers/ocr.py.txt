from pathlib import Path
import subprocess,concurrent.futures,json,hashlib
root=Path('/private/tmp/pathwaymech-next-260-primary');out=root/'projections/ocr';out.mkdir(exist_ok=True)
def run(p):
 dest=out/p.stem
 proc=subprocess.run(['tesseract',str(p),str(dest),'--psm','3'],capture_output=True,text=True)
 q=dest.with_suffix('.txt')
 return {'original':str(p),'original_sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'output':str(q),'output_sha256':hashlib.sha256(q.read_bytes()).hexdigest() if q.exists() else None,'exit_code':proc.returncode,'stderr':proc.stderr,'status':'auxiliary OCR; original PNG visual reading controls'}
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as p:rows=list(p.map(run,sorted((root/'originals').glob('*page-*.png'))))
(root/'ocr-bindings.json').write_text(json.dumps(rows,indent=2)+'\n');print(len(rows),'OCR pages')
