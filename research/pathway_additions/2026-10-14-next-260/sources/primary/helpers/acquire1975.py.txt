def main():
    from pathlib import Path
    import urllib.request,urllib.error,hashlib,json,datetime,sys
    root=Path('/private/tmp/pathwaymech-next-260-primary')
    rows=[('PMC236050.html','https://pmc.ncbi.nlm.nih.gov/articles/PMC236050/'),('PMC236050-europe.xml','https://www.ebi.ac.uk/europepmc/webservices/rest/PMC236050/fullTextXML'),('PMID1194238-europe.json','https://www.ebi.ac.uk/europepmc/webservices/rest/search?query=EXT_ID:1194238%20AND%20SRC:MED&format=json&resultType=core')]
    receipts=[]
    for name,url in rows:
     row={'name':name,'url':url,'started_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
     try:
      req=urllib.request.Request(url,headers={'User-Agent':'PathwayMech research contact source-verification'})
      with urllib.request.urlopen(req,timeout=45) as r:
       body=r.read();row.update(status=r.status,final_url=r.url,content_type=r.headers.get('Content-Type'))
      path=root/'originals'/name
      if path.exists():raise RuntimeError('refusing to overwrite original')
      path.write_bytes(body);row.update(path=str(path),bytes=len(body),sha256=hashlib.sha256(body).hexdigest())
     except urllib.error.HTTPError as e:
      body=e.read();path=root/'originals'/(name+'.http-error');path.write_bytes(body);row.update(status=e.code,error=str(e),path=str(path),bytes=len(body),sha256=hashlib.sha256(body).hexdigest())
     except Exception as e:row['error']=repr(e)
     receipts.append(row)
    (root/('acquisition-'+sys.argv[1]+'.json')).write_text(json.dumps(receipts,indent=2)+'\n')
    print(json.dumps(receipts,indent=2))

if __name__ == "__main__":
    main()
