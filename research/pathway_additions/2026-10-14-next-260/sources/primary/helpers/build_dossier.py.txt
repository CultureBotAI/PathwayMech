from pathlib import Path

import hashlib

ROOT = Path('/private/tmp/pathwaymech-next-260-primary')
AUTHOR = ROOT / 'author'
AUTHOR.mkdir(exist_ok=True)

def save(name, value):
    (AUTHOR / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def binding(path):
    p = Path(path)
    data = p.read_bytes()
    return {'path': str(p), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

claims = []

def claim(paper, page, locus, topic, observation, scope, disposition, reason, evidence='primary prose; research paraphrase, not schema source_assertion'):
    serial = sum(x['reference'] == paper for x in claims) + 1
    pmc, first = ('PMC221622', 1154) if paper == 'PMID:7142106' else ('PMC236050', 1374)
    item = {'claim_id': ('K82' if first == 1154 else 'S75') + f'-{serial:03d}', 'reference': paper,
            'source': binding(ROOT / 'originals' / f'{pmc}-page-{page-first+1}.png'),
            'locator': f'printed p.{page}; {locus}', 'topic': topic,
            'observation': observation, 'experimental_scope': scope,
            'disposition': disposition, 'reason': reason, 'evidence_kind': evidence}
    claims.append(item)
    return item['claim_id']
