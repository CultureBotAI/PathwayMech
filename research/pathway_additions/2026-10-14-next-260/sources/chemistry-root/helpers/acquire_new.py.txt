from pathlib import Path
import datetime,hashlib,json,urllib.request
root=Path('/private/tmp/pathwaymech-next-260-chemistry-root')
receipt=root/'retained/acquisition.json';assert not receipt.exists()
rows=[]
for name,url in [('ChEBI-58304.json','https://www.ebi.ac.uk/chebi/backend/api/public/compound/58304/'),('KEGG-R03549.txt','https://rest.kegg.jp/get/rn:R03549')]:
 p=root/'originals'/name;assert not p.exists();started=datetime.datetime.now(datetime.timezone.utc).isoformat()
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech-source-research/1.0','Accept':'application/json' if name.endswith('.json') else 'text/plain'}),timeout=45) as response:
   body=response.read();row={'url':url,'utc':started,'status':response.status,'final_url':response.url,'content_type':response.headers.get('Content-Type'),'sha256':hashlib.sha256(body).hexdigest(),'bytes':len(body),'path':str(p)}
  p.write_bytes(body);rows.append(row)
 except Exception as exc:rows.append({'url':url,'utc':started,'error':repr(exc)})
 receipt.write_text(json.dumps(rows,indent=2)+'\n')
 print(json.dumps(rows[-1]),flush=True)
assert all(r.get('status')==200 for r in rows)
