def rootloc(frame):return f"/ptools-xml/{objects[frame].tag}[@frameid='{frame}']"
def binding(frame,paths):
    out=[]
    for rel in paths:
        if rel=='':hits=[objects[frame]]
        elif (match:=re.fullmatch(r"(left|right)\[Compound/@frameid='([^']+)'\]",rel)):
            hits=[x for x in objects[frame].findall(match[1]) if any(c.get('frameid')==match[2] for c in x.findall('Compound'))]
        else:hits=objects[frame].findall(rel)
        assert hits,(frame,rel)
        ordinals=collections.Counter();direct=[]
        for slot in objects[frame]:
            ordinals[slot.tag]+=1
            if any(hit is descendant for hit in hits for descendant in slot.iter()):
                direct.append(rootloc(frame)+f'/{slot.tag}[{ordinals[slot.tag]}]')
        out.append({'reference_id':'MetaCyc:'+frame,'source_path':manifest[frame]['path'],'source_sha256':manifest[frame]['sha256'],'source_locator':rootloc(frame)+('/'+rel if rel else ''),'matches':len(hits),'direct_slot_locators':direct})
    return out
