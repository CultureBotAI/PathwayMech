def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def save(name,x):
    p=ROOT/name;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(x,indent=2,ensure_ascii=False)+'\n')

    objects[frame]=next(x for x in ET.parse(row['path']).getroot() if x.get('frameid')==frame)
def met(frame):return 'MetaCyc:'+frame
acts=load(NATIVE/'projections/activity-preparation-dispositions.json')

chem={x['frame']:x['recommended_identity'] for x in load('/private/tmp/pathwaymech-next-260-participant-authorities/retained/identity-dispositions.json')['native_compounds']}

for n,f in enumerate(formal,1):

    check(e['id'],(e['subject'],e['predicate'],e['object'])==((f['exact_chebi'],'consumes',f['chosen_directional_reaction']) if inp else (f['chosen_directional_reaction'],'produces',f['exact_chebi'])))

        for i,slot in enumerate(obj.findall('component'),1):
            e=edges[f'native-composition-{frame}-{i}'];component=slot.find('Protein').get('frameid');count=slot.findtext('coefficient')
            check(e['id'],(e['subject'],e['predicate'],e['object'])==(met(frame),'has_part',met(component)) and f'{count} copies' in e['description'])
for a in acts:
    act=a['activity'];obj=objects[act];eid='native-catalysis-'+act
    if eid in edges:

        check(eid,(e['subject'],e['predicate'],e['object'])==(met(a['enzyme']),'catalyzes',chosen))
    for n,c in enumerate(obj.findall('cofactor'),1):
        e=edges[f'native-cofactor-{act}-{n}'];compound=c.find('Compound').get('frameid');pubs=[p.get('frameid') for p in c.findall('citation/Publication')]
        qe=e['endpoint_qualifiers'][0]['evidence'][0]
        actual_pub=[x.find('Publication').get('frameid') for x in obj.findall('evidence') if x.find('Publication').get('frameid') in pubs]
        check(e['id']+' chemical and same-study qualifier',e['subject']==met(a['enzyme']) and e['object']==chem[compound] and len(actual_pub)==1 and actual_pub[0] in qe['source_assertion'])

for r in regs:

    if r['frame'] in handed:check(eid+' reserved',eid not in edges);continue
    e=edges[eid];obj=objects[r['frame']]
    reg=obj.find('regulator/Compound').get('frameid');act=obj.find('regulated-entity/Enzymatic-Reaction').get('frameid');target=objects[act].find('enzyme/Protein').get('frameid');sign=obj.findtext('mode')

# Compact dossiers retain dispositions and exact locators while omitting full source prose/XML.
def compact(value):
    if isinstance(value,list):return [compact(v) for v in value]
    if isinstance(value,dict):return {k:compact(v) for k,v in value.items() if k not in {'xml_projection','direct_text','native_refs','direct_crossreferences','inline_organism_scopes','current_owner','candidate_options','source_slots','protein_context_slots','preparation_notes'}}
    return value
for name in ['activity-dispositions.json','regulator-dispositions.json','fact-dispositions.json','protein-dispositions.json','gene-context-dispositions.json']:
    raw=load(ROOT/name);slim=compact(raw)
    for original,projected in zip(raw,slim):
        if 'source_slots' in original:projected['original_slot_bindings']=[{k:s[k] for k in ['slot','source_locator','source_sha256','source_path']} for s in original['source_slots']]
        if name=='gene-context-dispositions.json':
            projected['gene_product_bindings']=original['native_refs']['product']
            projected['source_crossreferences_context']=original['direct_crossreferences']
    save('compact/'+name,slim)
