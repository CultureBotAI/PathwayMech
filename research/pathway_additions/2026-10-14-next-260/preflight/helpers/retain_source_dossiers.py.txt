from pathlib import Path
import datetime
import gzip
import hashlib
import json

WORKTREE = Path('/private/tmp/pathwaymech-next-260-20261014')
DEST = WORKTREE / 'research/pathway_additions/2026-10-14-next-260'
CI = Path('/private/tmp/pathwaymech-next-260-ci')
rows = []
pending = []


def sha(data):
    return hashlib.sha256(data).hexdigest()


def retain(source, relative, expected=None):
    source = Path(source)
    relative = Path(relative)
    assert not relative.is_absolute() and '..' not in relative.parts
    data = source.read_bytes()
    if expected:
        assert sha(data) == expected, source
    encoding = 'identity'
    # Preserve whitespace-bearing originals exactly; do not trim historic inputs.
    if any(line.endswith((b' ', b'\t')) for line in data.splitlines()):
        encoded = gzip.compress(data, mtime=0)
        relative = Path(str(relative) + '.gz')
        encoding = 'gzip'
        assert gzip.decompress(encoded) == data
    else:
        encoded = data
    target = DEST / relative
    assert not target.exists(), target
    assert all(existing != target for existing, _ in pending), target
    pending.append((target, encoded))
    rows.append({
        'source_path': str(source), 'source_sha256': sha(data),
        'source_bytes': len(data), 'retained_path': str(relative),
        'retained_sha256': sha(encoded), 'retained_bytes': len(encoded),
        'encoding': encoding,
    })


def package(name):
    return Path('/private/tmp/pathwaymech-next-260-' + name)


def verify_package(root):
    manifest = root / 'artifact-manifest.json'
    obj = json.loads(manifest.read_text())
    for row in obj['files']:
        data = (root / row['path']).read_bytes()
        assert len(data) == row['bytes'] and sha(data) == row['sha256'], row['path']
    return obj


assert not DEST.exists(), DEST
dedup = json.loads((CI / 'dedup-disposition.json').read_text())
assert dedup['base'] == 'b2590dfd2385c83fe4f028ecd5309a0407739838'
assert dedup['semantic_scan']['exact_candidate_or_complete_formal_matches'] == 0

for name in ['primary', 'chemistry-root']:
    root = package(name)
    manifest = verify_package(root)
    for item in manifest['files']:
        rel = item['path']
        selected = (
            rel == 'HANDOFF.md'
            or rel.startswith('author/')
            or rel.startswith('retained/')
            or rel.startswith('helpers/')
            or rel in ['external-originals-manifest.json', 'verification.json',
                       'reuse-bindings.json', 'ocr-bindings.json']
            or rel.startswith('acquisition-')
            or rel in ['projections/source-acquisition-decisive-manifest.json',
                       'projections/source-original-bindings.json']
        )
        if selected:
            target = rel + '.txt' if rel.startswith('helpers/') else rel
            retain(root / rel, Path('sources') / name / target, item['sha256'])
    retain(root / 'artifact-manifest.json', Path('sources') / name / 'external-artifact-manifest.json')
