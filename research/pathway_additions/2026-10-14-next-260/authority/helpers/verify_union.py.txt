"""Verify exact selected union and unchanged baseline; never promote candidates."""
from pathlib import Path
import argparse,datetime,hashlib,json,shutil,sys
import yaml
B=Path(__file__).resolve().parents[1]
def sha(p):
 h=hashlib.sha256()
 with Path(p).open('rb') as f:
  for block in iter(lambda:f.read(1024*1024),b''):h.update(block)
 return h.hexdigest()
def load(p):return json.loads(Path(p).read_text())
def save(name,obj):
 p=B/'retained'/name;assert not p.exists(),f'Preserve prior verification attempt: {p}'
 p.write_text(json.dumps(obj,indent=2,ensure_ascii=False)+'\n');return p
def main():
 parser=argparse.ArgumentParser();parser.add_argument('--record',type=Path,required=True);parser.add_argument('--record-sha256',required=True);args=parser.parse_args()

 assert preservation['command_file_sha256']==sha(B/'retained/canonical-build-commands.json')
 for key in ['source_builder','ontology_builder']:
  r=load(B/'retained'/(key+'-receipt.json'));assert r['exit_code']==0 and r['command']==commands[key] and r['command_file_sha256']==preservation['command_file_sha256'];assert sha(r['log_path'])==r['log_sha256']
 planned=set(load(B/'retained/planned-identifiers.json'));retained=set(load(B/'retained/retained-authority-identifiers.json'));freeze=load(B/'retained/selection-freeze.json')
 assert set(freeze['selected_identifiers'])==planned
 record=yaml.safe_load(args.record.read_text());actual={record['id']}|{n['id'] for field in ['taxa','participants','reactions','gene_clusters'] for n in record.get(field,[])}

 report=dict(status='author_verification_passed_pending_independent_review',role='Authority-integration author; no independent scientific verdict',utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),baseline_commit=base['baseline_commit'],baseline_identifier_count=len(retained),selected_identifier_count=len(planned),snapshots={},maintained_files_edited=False)
 oldterms={};newterms={}
 for name in ['sources','ontology']:
  oldpath=B/'baseline/data/identifier_authorities'/(name+'.json');newpath=B/'proposal'/(name+'.union.json');old=load(oldpath);new=load(newpath)
  assert sha(newpath)==preservation['outputs'][name+'.union.json'];assert old['version']==new['version']
  for section in ['terms','sources']:
   changed=[key for key,value in old[section].items() if new[section].get(key)!=value];assert not changed,(name,section,changed)
  assert not newterms.keys()&new['terms'].keys();newterms.update(new['terms']);oldterms.update(old['terms'])
  report['snapshots'][name]=dict(before_sha256=sha(oldpath),after_sha256=sha(newpath),baseline_terms=len(old['terms']),final_terms=len(new['terms']),baseline_owners=len(old['sources']),final_owners=len(new['sources']),all_baseline_objects_exact=True,new_terms=sorted(new['terms'].keys()-old['terms'].keys()),new_owners=sorted(new['sources'].keys()-old['sources'].keys()))
 assert set(oldterms)==retained and len(retained)==4989
 assert planned<=newterms.keys() and set(newterms)-retained==planned-retained
 oldmanifest=load(B/'baseline/conf/identifier_sources.json');newmanifest=load(B/'proposal/identifier_sources.json');adds=load(B/'retained/proposed-source-additions.json')
 assert newmanifest[:len(oldmanifest)]==oldmanifest and newmanifest[len(oldmanifest):]==adds
 assert {e['key'] for e in adds}==set(report['snapshots']['sources']['new_owners'])
 assert not report['snapshots']['ontology']['new_owners']
 report['manifest']=dict(baseline_entries=len(oldmanifest),new_entries=len(adds),final_entries=len(newmanifest),baseline_entry_prefix_exact=True,sha256=sha(B/'proposal/identifier_sources.json'))
 for row in commands['database_bindings']:
  p=Path(row['path']);stat=p.stat();assert stat.st_size==row['bytes'] and stat.st_mtime_ns==row['mtime_ns'];assert str(p.resolve())==row['resolved']
  assert load(B/'proposal/ontology.union.json')['sources'][row['owner_key']]['sha256']==row['sha256']
 report['pinned_databases']=dict(count=4,canonical_complete_database_hashes_match_baseline=True,size_and_mtime_unchanged=True)
 shadow=B/'identifier-verification-root';assert not shadow.exists();(shadow/'conf').mkdir(parents=True);(shadow/'data/identifier_authorities').mkdir(parents=True)
 shutil.copyfile(B/'baseline/conf/identifier_policy.yaml',shadow/'conf/identifier_policy.yaml')
 for name in ['sources','ontology']:shutil.copyfile(B/'proposal'/(name+'.union.json'),shadow/'data/identifier_authorities'/(name+'.json'))

 current_root=Path('/private/tmp/pathwaymech-next-260-20261014')

 for row in base['baseline_files']:assert sha(row['path'])==row['sha256']==sha(row['maintained_path'])

 print(json.dumps(dict(status=report['status'],final_terms=len(newterms),new_terms=len(planned-retained),new_owners=len(adds),records_checked=len(rows)+1)))
