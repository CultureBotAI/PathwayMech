def bind(path):
    path = Path(path)
    raw = path.read_bytes()
    return {'path': str(path), 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def write(name, value):
    (A / name).write_text(json.dumps(value, indent=2) + '\n')


def verify_packet(directory, handoff_sha):
    handoff = bind(directory / 'handoff.json')
    assert handoff['sha256'] == handoff_sha
    manifest = json.loads((directory / 'artifact-manifest.json').read_text())
    rows = manifest.get('files', manifest.get('artifacts'))
    for row in rows:
        actual = bind(directory / row['path'])
        assert (actual['bytes'], actual['sha256']) == (row['bytes'], row['sha256']), row['path']
    return {'handoff': handoff, 'manifest': bind(directory / 'artifact-manifest.json'),
            'artifacts_verified': len(rows)}
