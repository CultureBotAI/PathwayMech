import time,concurrent.futures,json,sys,pathlib,urllib.request,urllib.error,datetime,hashlib
D=pathlib.Path(__file__).parent
req=json.loads(pathlib.Path(sys.argv[1]).read_text())
def fetch(x):
 time.sleep(1); name,url=x; p=D/'raw'/name
 if p.exists(): raise RuntimeError('Refuse overwrite '+str(p))
 r={'key':name,'request_url':url,'requested_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech research source review'}),timeout=40) as h:
   b=h.read(); r.update(status=h.status,final_url=h.url,headers=dict(h.headers))
  p.write_bytes(b); r.update(path=str(p),bytes=len(b),sha256=hashlib.sha256(b).hexdigest(),payload_class='pdf' if b.startswith(b'%PDF') else 'xml' if b.lstrip().startswith(b'<?xml') else 'json' if b.lstrip().startswith((b'{',b'[')) else 'html_or_other')
 except Exception as e:r['error']=str(e)
 return r
with concurrent.futures.ThreadPoolExecutor(max_workers=1) as ex: out=list(ex.map(fetch,req))
p=pathlib.Path(sys.argv[1]).with_suffix('.receipts.json'); p.write_text(json.dumps(out,indent=2)+'\n')
for r in out:print(r['key'],r.get('status'),r.get('bytes'),r.get('error',''))
