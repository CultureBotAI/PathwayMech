        'bounds':'Inherited full scientific review covers the original record; this successor replays all locators/hashes and tests scoped corrections. It does not claim a fresh independent scientific review.'})
    def differences(before,after,pointer):
        if before==after:
            return []
        if isinstance(before,dict) and isinstance(after,dict):
            rows=[]
            for key in list(before)+[key for key in after if key not in before]:
                if key not in before:
                    rows.append({'pointer':pointer+'/'+key,'operation':'add','new':after[key]})
                elif key not in after:
                    rows.append({'pointer':pointer+'/'+key,'operation':'remove','old':before[key]})
                else:
                    rows.extend(differences(before[key],after[key],pointer+'/'+key))
            return rows
        if isinstance(before,list) and isinstance(after,list):
            rows=[]
            for index in range(max(len(before),len(after))):
                if index>=len(before):
                    rows.append({'pointer':pointer+'/'+str(index),'operation':'add','new':after[index]})
                elif index>=len(after):
                    rows.append({'pointer':pointer+'/'+str(index),'operation':'remove','old':before[index]})
                else:
                    rows.extend(differences(before[index],after[index],pointer+'/'+str(index)))
            return rows
        return [{'pointer':pointer,'operation':'replace','old':before,'new':after}]

        {'keys':['pathbank-root','opaque-pathbank-interface'],'disposition':'Separate root-accepted adapter now supports native typed root/opaque child identity. No scientific inference or child expansion changed.'}

        'Added four source-bounded entry evidence assertions and assay prose preserving all weak and negative uptake comparisons, internal mannose tension and separate 300 mM NaCl induction effect. Intact-cell MC4100 context is not assigned to the unchanged inverted-membrane assay or KM522 osmoprotection.',
        'Added experimental_construct subject qualifiers to both positive CdaR gene regulation claims. Exact historical/current reference distinction, paper-numbered P297S and Tn5 loss controls, MC4100/pRM1 rescue and reporter inference travel with the claims in YAML, page, graph and KGX.',
        'Replaced 15 evidence locators: PtsI catalytic His189 Active site, H-NS location plus SUBUNIT, and 13 exact original Rhea triples using contains1. All affected scientific assertions, endpoints, predicates, native sides and prior state qualifiers remain exact.'
    ]
    acceptance=[
        ['derived/primary-assay-replay.json','derived/scope-and-conflicts.json','derived/unchanged-object-proof.json','derived/semantic-negative-controls.json'],
        ['derived/primary-assay-replay.json','derived/graph-page-kgx-roundtrip.json','derived/unchanged-object-proof.json','derived/semantic-negative-controls.json'],
        ['derived/uniprot-assertion-replay.json','derived/rhea-exact-original-replay.json','derived/all-evidence-locator-replay.json','derived/unchanged-object-proof.json']
    ]

             'original_finding':finding,'original_review':bind(REVIEW/'handoff.json'),
             'disposition':'addressed_in_external_successor_pending_independent_review',
