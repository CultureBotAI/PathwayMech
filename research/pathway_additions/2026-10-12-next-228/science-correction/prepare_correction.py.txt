def bind(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size,
            'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}


def load(path):
    return json.loads(Path(path).read_bytes())


def write(name, value):
    path = OUT / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def packet(directory, digest, count):
    handoff = bind(directory / 'handoff.json')
    assert handoff['sha256'] == digest
    manifest = load(directory / 'artifact-manifest.json')
    rows = manifest.get('artifacts', manifest.get('files'))
    assert len(rows) == count
    for row in rows:
        path = directory / row.get('relative_path', row.get('path'))
        actual = bind(path)
        assert (actual['bytes'], actual['sha256']) == (row['bytes'], row['sha256']), path
    return {'handoff': handoff, 'manifest': bind(directory / 'artifact-manifest.json'),
            'artifacts_verified': count}
