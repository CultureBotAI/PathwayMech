def now():
    return datetime.now(timezone.utc).isoformat()


def load(path):
    return json.loads(Path(path).read_bytes())


def bind(path):
    path = Path(path)
    body = path.read_bytes()
    return {'path': str(path), 'bytes': len(body), 'sha256': hashlib.sha256(body).hexdigest()}


def write(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

        source = Path(source)
        destination = archive / name
        body = source.read_bytes()
        # Retain text byte-for-byte in a lossless wrapper when Git whitespace
        # checks would reject its original formatting. Never normalize evidence.
        if destination.suffix != '.png':
            text = body.decode('utf-8')
            if any(line.rstrip() != line for line in text.splitlines()) or text.endswith('\n\n'):
                destination = destination.with_name(destination.name + '.json')
                write(destination, {'serialization': 'lossless-utf8-json',
                                    'original': bind(source), 'text': text})
            else:
                with destination.open('xb') as stream:
                    stream.write(body)
        else:
            with destination.open('xb') as stream:
                stream.write(body)
        mappings.append({'source': bind(source), 'archived': bind(destination)})
        return destination
