from pathlib import Path
import urllib.request,json,hashlib,datetime,concurrent.futures,sys,time
out=Path(__file__).parent; raw=out/'raw';reqs=json.loads(Path(sys.argv[1]).read_text())
def fetch(item):
 k,url=item;p=raw/k;assert not p.exists(),p
 row={'key':k,'request_url':url,'requested_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech primary evidence curation'}),timeout=45) as r:
   b=r.read();p.write_bytes(b);row.update(status=r.status,final_url=r.url,headers=dict(r.headers),path=str(p),bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
 except Exception as e:row['error']=str(e)
 print(json.dumps({k:v for k,v in row.items() if k!='headers'}),flush=True);return row
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as e: rows=list(e.map(fetch,reqs.items()))
(out/(Path(sys.argv[1]).stem+'-receipts.json')).write_text(json.dumps(rows,indent=2)+'\n')
