def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
 return h.hexdigest()

   if r['id'] in ['MetaCyc:LYXISOM-RXN','RHEA:18870','RHEA:18499','RHEA:22369']:overlap.append({'path':str(p),'pathway':j['id'],'reaction':r['id']})
