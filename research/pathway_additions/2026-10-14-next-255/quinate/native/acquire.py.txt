from pathlib import Path
import concurrent.futures, datetime, hashlib, json, urllib.request, xml.etree.ElementTree as ET

ROOT=Path(__file__).resolve().parent
ORIG=ROOT/'originals'
ORIG.mkdir(exist_ok=True)
BASE='https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=META:'
def fetch(fid):
    out=ORIG/f'META-{fid}.xml'
    if out.exists():
        b=out.read_bytes();root=ET.fromstring(b)
        return {'frameid':fid,'path':str(out),'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b),'cached':True}
    url=BASE+fid
    rec={'frameid':fid,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech evidence curation/1.0'}),timeout=60) as r:
            b=r.read();rec.update(status=r.status,final_url=r.url,content_type=r.headers.get('Content-Type'))
        root=ET.fromstring(b)
        assert len([e for e in root if e.get('frameid')==fid])==1,(fid,root.tag)
        out.write_bytes(b);rec.update(path=str(out),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b),root_tag=root.tag,version=root.attrib)
    except Exception as e:rec['error']=f'{type(e).__name__}: {e}'
    return rec
def batch(ids,label):
    ids=sorted(set(ids))
    with concurrent.futures.ThreadPoolExecutor(max_workers=6) as ex:rows=list(ex.map(fetch,ids))
    (ROOT/f'acquisition-{label}.json').write_text(json.dumps(rows,indent=2)+'\n')
    print(label, len(rows),sum('error' not in r for r in rows),'success',flush=True)
    for r in rows:
        if 'error' in r:print(r,flush=True)
    return rows
def frame(fid):
    return next(e for e in ET.parse(ORIG/f'META-{fid}.xml').getroot() if e.get('frameid')==fid)
if __name__=='__main__':
    import sys
    batch(sys.argv[2:],sys.argv[1])
