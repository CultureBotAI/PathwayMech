from acquire import *
def acquire(fid):
    url='https://bf2i-artsymbiocyc.insa-lyon.fr/META/NEW-IMAGE?type=ENZYME&object='+fid
    rec={'frameid':fid,'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
    try:
        with urllib.request.urlopen(url,timeout=60) as r:b=r.read();rec.update(final_url=r.url,status=r.status)
        f=ORIG/f'META-{fid}.html';f.write_bytes(b);rec.update(path=str(f),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
    except Exception as e:rec['error']=str(e)
    return rec
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:rows=list(ex.map(acquire,['MONOMER-1','MONOMER-67','MONOMER-37','QUINATEDEG-PWY']))
(ROOT/'acquisition-html.json').write_text(json.dumps(rows,indent=2)+'\n')
print(json.dumps(rows,indent=2))
