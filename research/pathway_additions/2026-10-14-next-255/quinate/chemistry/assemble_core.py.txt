import yaml

B = Path(__file__).parent
N = Path('/private/tmp/pathwaymech-next-255-native/originals')

audit = json.loads((B / 'chemical-identity-and-reaction-audit.json').read_text())
pid = 'MetaCyc:QUINATEDEG-PWY'
record = dict(
    id=pid, label='quinate degradation I',
    description=('The three-step MetaCyc 27.0 quinate degradation I definition converts L-quinate '
                 'through 3-dehydroquinate and 3-dehydroshikimate to protocatechuate. The initial '
                 'oxidation retains the source-defined quinone/quinol classes; PQQ is an enzyme '
                 'cofactor, not a net reaction substrate. The shared dehydroquinate dehydration '
                 'is reversible although the pathway selects catabolic forward flux. Enzyme '
                 'evidence is qualified by organism, preparation and construct; the record does '
                 'not assert a purified three-enzyme reconstitution in one strain.'),
    pathway_type='aromatic-compound-catabolism', taxa=[], participants=[], reactions=[],
    mechanistic_edges=[], references=[])
def label(database, identifier):
    with sqlite3.connect('file:/private/tmp/pathwaymech-curation-authorities/'+database+'.db?mode=ro', uri=True) as db:
        return db.execute("select value from statements where subject=? and predicate='rdfs:label'", (identifier,)).fetchone()[0]
def ref(identifier, title, path, url, version='MetaCyc 27.0'):
    item = dict(id=identifier, title=title, url=url, source_version=version,
                source_sha256=hashlib.sha256(path.read_bytes()).hexdigest())
    existing = next((r for r in record['references'] if r['id']==identifier), None)
    if existing is None: record['references'].append(item)
    else: assert existing == item, (existing, item)
def native_ref(frame, folder=N):
    identifier='MetaCyc:'+frame
    ref(identifier, 'MetaCyc native object '+frame, folder/('META-'+frame+'.xml'),
        'https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=META:'+frame)
    return identifier
def evidence(identifier, assertion, locator):
    assert len(assertion)<=400, assertion
    return dict(reference_id=identifier, source_assertion=assertion, source_locator=locator)
def edge(subject, predicate, obj, description, items):
    record['mechanistic_edges'].append(dict(id='pending',subject=subject,predicate=predicate,
        object=obj,description=description,evidence=items))
native_ref('QUINATEDEG-PWY')
for number in ['62977','442']:
    identifier='NCBITaxon:'+number
    record['taxa'].append(dict(id=identifier,label=label('ncbitaxon',identifier)))
    edge(pid,'part_of',identifier,'Native pathway species context; individual enzyme evidence retains its strain and preparation scope.',
         [evidence(pid,'The pathway species slot lists TAX-'+number+'; native pathway evidence includes mutant-phenotype and direct-assay annotations.',
          '/ptools-xml/Pathway[@frameid="QUINATEDEG-PWY"]/species/Organism[@frameid="TAX-'+number+'"]')])

mapping = {x['native_id'].split(':',1)[1]:x['selected_id'] for x in audit['concrete_mappings']}
for frame in ['QUINATE','DEHYDROQUINATE','3-DEHYDRO-SHIKIMATE','3-4-DIHYDROXYBENZOATE','WATER']:
    identifier=mapping[frame]
    record['participants'].append(dict(id=identifier,label=label('chebi',identifier),category='small_molecule'))
for frame, name in [('Quinones','a quinone'),('Reduced-Quinones','a quinol')]:
    mapping[frame]='MetaCyc:'+frame
    record['participants'].append(dict(id='MetaCyc:'+frame,label=name,category='chemical_entity'))
    native_ref(frame, B/'originals-meta27')
rxmapping = {x['native_frame']:x['selected_id'] for x in audit['reactions']}
for row in audit['reactions']:
    frame=row['native_frame']; identifier=row['selected_id']; native_ref(frame)
    title = ('quinate oxidation with a quinone' if identifier.startswith('MetaCyc:') else
             label('rhea',identifier))
    record['reactions'].append(dict(id=identifier,label=title,category='molecular_activity',
        direction='reversible' if row['native_direction']=='REVERSIBLE' else 'left_to_right'))
    edge(identifier,'part_of',pid,'One of the three explicit native pathway members; the native reaction direction is preserved.',
         [evidence(pid,'The reaction-list includes '+frame+' and the selected pathway layout is L2R.',
                   '/ptools-xml/Pathway[@frameid="QUINATEDEG-PWY"]/reaction-list|reaction-layout[Reaction/@frameid="'+frame+'"]')])
    if identifier.startswith('RHEA:'):
        ref(identifier,'Rhea equation '+identifier,Path(audit['rhea_original']['path']),
            'https://www.rhea-db.org/rhea/'+identifier.split(':')[1],'Rhea 142')
    for side,predicate in [('left','consumes'),('right','produces')]:
        for compound in row['native_sides'][side]:
            source_frame=compound['frame']; chemical=mapping[source_frame]
            items=[evidence('MetaCyc:'+frame,
                'The native '+side+' side contains '+source_frame+'. No explicit coefficient is supplied.',
                '/ptools-xml/Reaction[@frameid="'+frame+'"]/'+side+'/Compound[@frameid="'+source_frame+'"]')]
            if chemical.startswith('CHEBI:'):
                native_ref(source_frame,B/'originals-meta27')
                items.append(evidence('MetaCyc:'+source_frame,
                    'The native compound links '+chemical+'; its complete InChI agrees with the selected ChEBI molecular identity.',
                    '/ptools-xml/Compound[@frameid="'+source_frame+'"] /dblink[dblink-db="CHEBI"]|inchi'))
            if identifier.startswith('RHEA:'):
                molecule=next(x for x in row['rhea_'+side] if x['id']==chemical)
                items.append(evidence(identifier,
                    identifier+' '+side+' side contains '+chemical+' with coefficient '+str(molecule['coefficient'])+'.',
                    '; '.join(molecule['locator'])))
            description = 'Selected catabolic pathway orientation.'
            if compound['class_marker']=='true':
                description += ' Native generic quinone class; this does not establish specificity for every class member or a measured physiological quinone.'
            if row['native_direction']=='REVERSIBLE':
                description += ' The reaction itself is reversible; this edge does not restrict reverse chemistry.'
            edge(chemical if predicate=='consumes' else identifier,predicate,
                 identifier if predicate=='consumes' else chemical,description,items)
