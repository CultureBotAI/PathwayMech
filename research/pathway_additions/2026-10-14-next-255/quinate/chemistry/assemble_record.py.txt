from pathlib import Path
import hashlib, json
import yaml
from pathwaymech.schema import validate_record

B=Path(__file__).parent
R=Path('/private/tmp/pathwaymech-next-255-20261014')
P=Path('/private/tmp/pathwaymech-next-255-primary/biological-graph-extension.json')
record=json.loads((B/'core-record.json').read_text())
extension=json.loads(P.read_text())
authority={}
for name in ['sources','ontology']:
    authority.update(json.loads((R/'data/identifier_authorities'/f'{name}.json').read_text())['terms'])
for field in ['taxa','participants','references']:
    for row in extension[field]:
        existing=next((n for n in record[field] if n['id']==row['id']),None)
        if existing is None:record[field].append(row)
        else:assert existing==row,(field,existing,row)
original_biological_ids=[e['id'] for e in extension['mechanistic_edges']]
record['mechanistic_edges'].extend(extension['mechanistic_edges'])
for field in ['taxa','participants']:
    for node in record[field]:
        node['label']=authority[node['id']]['label']
for i,e in enumerate(record['mechanistic_edges'],1):e['id']='quinate-'+str(i).zfill(3)
validate_record(record)
target=R/'data/pathways/quinate-degradation-i.yaml'
target.write_text(yaml.safe_dump(record,sort_keys=False,allow_unicode=True,width=100))
mapping={identifier:'quinate-'+str(len(record['mechanistic_edges'])-len(extension['mechanistic_edges'])+i).zfill(3)
         for i,identifier in enumerate(original_biological_ids,1)}
receipt=dict(record=str(target.relative_to(R)),sha256=hashlib.sha256(target.read_bytes()).hexdigest(),
    core_sha256=hashlib.sha256((B/'core-record.json').read_bytes()).hexdigest(),
    extension_sha256=hashlib.sha256(P.read_bytes()).hexdigest(),
    edges=len(record['mechanistic_edges']),participants=len(record['participants']),
    reactions=len(record['reactions']),references=len(record['references']),
    biological_proposal_to_final_edge_id=mapping,
    canonical_labels_applied=True)
(B/'assembly-check.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps(receipt))
