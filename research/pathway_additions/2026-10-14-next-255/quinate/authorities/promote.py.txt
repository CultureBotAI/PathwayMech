from pathlib import Path
import datetime,hashlib,json,subprocess
B=Path(__file__).resolve().parent
R=Path('/private/tmp/pathwaymech-next-255-20261014')
BASE='bfa3f5ed0e843472c0fd173ce91df2a90dd4ddb2'
def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        while b:=f.read(1024*1024):h.update(b)
    return h.hexdigest()
assert subprocess.check_output(['git','rev-parse','HEAD'],cwd=R,text=True).strip()==BASE
c=json.loads((B/'canonical-build-commands.json').read_text())
for name,h in c['builder_sha256'].items():assert sha(R/'scripts'/name)==h,(name,'builder changed')
for key in ['source_builder','ontology_builder']:
    receipt=json.loads((B/(key+'-receipt.json')).read_text());assert receipt['exit_code']==0
    assert sha(Path(receipt['log_path']))==receipt['log_sha256']
report={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'baseline_commit':BASE,'snapshots':{},'builder_sha256':c['builder_sha256']}
all_terms={}
destinations=[]
for name in ['sources','ontology']:
    old=json.loads((B/'baseline'/f'{name}.json').read_text());new=json.loads((B/f'{name}.union.json').read_text())
    assert old['version']==new['version']
    for section in ['terms','sources']:
        changed=[k for k,v in old[section].items() if new[section].get(k)!=v]
        assert not changed,(name,section,changed)
    report['snapshots'][name]={'before_sha256':sha(B/'baseline'/f'{name}.json'),'after_sha256':sha(B/f'{name}.union.json'),'baseline_terms':len(old['terms']),'final_terms':len(new['terms']),'new_terms':sorted(set(new['terms'])-set(old['terms'])),'baseline_source_owners':len(old['sources']),'final_source_owners':len(new['sources']),'baseline_objects_unchanged':True}
    for k,v in new['terms'].items():
        assert k not in all_terms
        all_terms[k]=v
    destinations.append((R/'data/identifier_authorities'/f'{name}.json',B/f'{name}.union.json',B/'baseline'/f'{name}.json'))
planned=set(json.loads((B/'planned-identifiers.json').read_text()));assert planned<=set(all_terms),planned-set(all_terms)
old=json.loads((B/'baseline/identifier_sources.json').read_text());new=json.loads((B/'identifier_sources.json').read_text());assert new[:len(old)]==old
digests={};descriptors=[]
def verify(obj,where):
    if isinstance(obj,dict):
        if 'path' in obj and 'sha256' in obj:
            p=B/'originals'/obj['path'];assert p.is_file(),str(p)
            if str(p) not in digests:digests[str(p)]=sha(p)
            assert digests[str(p)]==obj['sha256'],(where,str(p))
            descriptors.append({'context':where,'path':str(p),'sha256':digests[str(p)]})
        for k,v in obj.items():verify(v,where+'/'+k)
    elif isinstance(obj,list):
        for i,v in enumerate(obj):verify(v,where+f'/{i}')
verify(new,'identifier_sources');verify(json.loads((B/'baseline/rhea_identifier_updates.json').read_text()),'rhea_identifier_updates')
report['manifest']={'baseline_entries':len(old),'final_entries':len(new),'prior_prefix_exact':True,'all_original_hashes_match':True,'hash_bound_descriptors':len(descriptors),'distinct_original_paths':len(digests),'sha256':sha(B/'identifier_sources.json')}
report['planned_named_identifiers']=len(planned)
report['final_authority_terms']=len(all_terms)
report['baseline_authority_terms']=len(json.loads((B/'retained-authority-identifiers.json').read_text()))
destinations.append((R/'conf/identifier_sources.json',B/'identifier_sources.json',B/'baseline/identifier_sources.json'))
for dst,src,baseline in destinations:assert dst.read_bytes()==baseline.read_bytes(),f'Concurrent authority change: {dst}'
for dst,src,baseline in destinations:
    temp=dst.with_suffix(dst.suffix+'.next255.tmp');temp.write_bytes(src.read_bytes());temp.replace(dst)
    assert sha(dst)==sha(src)
report['promoted_paths']=[str(x[0]) for x in destinations]
(B/'preservation-and-promotion.json').write_text(json.dumps(report,indent=2)+'\n')
(B/'final-original-bindings.json').write_text(json.dumps(descriptors,indent=2)+'\n')
(B/'planned-authority-labels.json').write_text(json.dumps({k:all_terms[k] for k in sorted(planned)},indent=2)+'\n')
print(json.dumps(report,indent=2))
