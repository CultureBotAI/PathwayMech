import urllib.request,urllib.error,hashlib,json,sys,datetime
from pathlib import Path
root=Path('/private/tmp/pathwaymech-next-256-primary')
url,name=sys.argv[1:]
row={'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'name':name}
try:
 r=urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0 PathwayMech research'}),timeout=90)
 b=r.read(); p=root/'originals'/name; p.write_bytes(b)
 row.update(status=r.status,final_url=r.url,content_type=r.headers.get('Content-Type'),path=str(p),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
except urllib.error.HTTPError as e:
 b=e.read();p=root/'originals'/(name+'.error');p.write_bytes(b);row.update(status=e.code,error=str(e),path=str(p),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
except Exception as e: row['error']=repr(e)
with (root/'acquisition.jsonl').open('a') as f:f.write(json.dumps(row)+'\n')
print(json.dumps(row))
