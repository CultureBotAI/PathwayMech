class SectionText(HTMLParser):
    """Read text inside one exact original-HTML section for quote replay."""
    def __init__(self, section_id):
        super().__init__()
        self.section_id = section_id
        self.depth = 0
        self.parts = []

    def handle_starttag(self, tag, attrs):
        if tag == 'section':
            if self.depth:
                self.depth += 1
            elif dict(attrs).get('id') == self.section_id:
                self.depth = 1

    def handle_endtag(self, tag):
        if tag == 'section' and self.depth:
            self.depth -= 1

    def handle_data(self, data):
        if self.depth:
            self.parts.append(data)

quote_replay=[]
def quotation(text, section_id, edge_id):
    assert len(text) <= 400
    parser = SectionText(section_id)
    parser.feed((ROOT/'originals/15691962-pmc.html').read_text())
    section_text = re.sub(r'\s+', ' ', ''.join(parser.parts)).strip()
    assert text in section_text, (section_id, text)
    quote_replay.append({
        'edge_id': edge_id,
        'reference_id': 'PMID:15691962',
        'source_path': str(ROOT/'originals/15691962-pmc.html'),
        'source_sha256': digest(ROOT/'originals/15691962-pmc.html'),
        'section_id': section_id,
        'quote_sha256': hashlib.sha256(text.encode()).hexdigest(),
        'verbatim_word_count': len(text.split()),
        'exact_whitespace_normalized_match': True,
        'section_match_count': section_text.count(text),
    })
    return {'reference_id':'PMID:15691962','quote':text,'source_locator':f'PMC546756 #{section_id}, first paragraph; probe definitions in #sec6.'}
