import AppKit
import PDFKit
import Foundation

let args = CommandLine.arguments
guard args.count == 3, let document = PDFDocument(url: URL(fileURLWithPath: args[1])) else {
    fatalError("Usage: render_research_pdf.swift INPUT_PDF OUTPUT_PREFIX")
}
for number in [2,8]+Array(89...99) {
    guard let page = document.page(at: number) else { fatalError("Missing PDF page") }
    let bounds = page.bounds(for: .mediaBox)
    let scale: CGFloat = 2.0
    let width = Int(bounds.width * scale)
    let height = Int(bounds.height * scale)
    guard let context = CGContext(data: nil, width: width, height: height, bitsPerComponent: 8,
                                  bytesPerRow: 0, space: CGColorSpaceCreateDeviceRGB(),
                                  bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else {
        fatalError("Cannot allocate rendering context")
    }
    context.setFillColor(CGColor(gray: 1, alpha: 1))
    context.fill(CGRect(x: 0, y: 0, width: width, height: height))
    context.scaleBy(x: scale, y: scale)
    page.draw(with: .mediaBox, to: context)
    guard let image = context.makeImage(),
          let png = NSBitmapImageRep(cgImage: image).representation(using: .png, properties: [:]) else {
        fatalError("Cannot encode rendered page")
    }
    let output = URL(fileURLWithPath: args[2] + "-" + String(number + 1) + ".png")
    if FileManager.default.fileExists(atPath: output.path) { fatalError("Refusing overwrite") }
    try png.write(to: output, options: .atomic)
}
print("Rendered 13 selected PDF pages")
