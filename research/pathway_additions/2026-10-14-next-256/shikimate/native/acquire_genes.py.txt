from acquire import *
rows=[]
for gid in ['2879574','2879469']:
    url=f'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=gene&id={gid}&retmode=xml'
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    try:
        with urllib.request.urlopen(url,timeout=60) as r:b=r.read();meta={'status':r.status,'final_url':r.url}
        tree=ET.fromstring(b);f=ORIG/f'NCBI-Gene-{gid}.xml';f.write_bytes(b)
        rows.append({'gene_id':gid,'url':url,'retrieved_at':start,**meta,'path':str(f),'sha256':hashlib.sha256(b).hexdigest(),'bytes':len(b)})
        for g in tree.findall('Entrezgene'):
            print(gid, ET.tostring(g.find('Entrezgene_track-info'),encoding='unicode'),ET.tostring(g.find('Entrezgene_gene'),encoding='unicode'),ET.tostring(g.find('Entrezgene_source'),encoding='unicode'))
    except Exception as exc:rows.append({'gene_id':gid,'url':url,'retrieved_at':start,'error':repr(exc)})
(ROOT/'acquisition-genes.json').write_text(json.dumps(rows,indent=2)+'\n')
