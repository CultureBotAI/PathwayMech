def ids(fid,path):
    return [e.get('frameid') for e in frame(fid).findall(path)]
def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def loc(fid,slotname=None):
    out=f'/ptools-xml/{frame(fid).tag}[@frameid="{fid}"]'
    return ' | '.join(out+'/'+part for part in slotname.split('|')) if slotname else out
