def add_native_ref(ident,path):
    assert path.is_file(),path
    accession=ident.split(':',1)[1]
    if ident.startswith('MetaCyc:'):
        ref={'id':ident,'title':f'MetaCyc native object {accession}','url':f'https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=META:{accession}','source_version':'MetaCyc 27.0','source_sha256':sha(path)}
    else:
        raw=load(path)
        ref={'id':ident,'title':f'UniProtKB {accession}','url':f'https://rest.uniprot.org/uniprotkb/{accession}.json','source_version':f"retrieved 2026-10-14; entryVersion {raw['entryAudit']['entryVersion']}; sequenceVersion {raw['entryAudit']['sequenceVersion']}",'source_sha256':sha(path)}
    if ident in refs: assert refs[ident]['source_sha256']==ref['source_sha256'],ident
    refs[ident]=ref

def assertion(ref,text,locator):return {'reference_id':ref,'source_assertion':text,'source_locator':locator}
