def tally(items):
 out=collections.Counter()
 for item in items:
  for atom,n in re.findall(r'([A-Z][a-z]*)([0-9]*)',item['formula']):out[atom]+=int(n or 1)*item['coefficient']
  out['charge']+=item['charge']*item['coefficient']
 return dict(out)
