"""External author replay: original RDF direction/sides and fail-closed publication.

This checks an external author proposal, not maintained production authorities or
scientific-review approval. Every consumed semantic file is SHA-256 bound before
use and rehashed before the accepted bundle and success receipt appear together.
"""

def now(): return datetime.now(timezone.utc).isoformat()
def require(ok, code, detail=''):
    if not ok: raise ValueError(f'{code}: {detail}')
def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda: f.read(1024*1024), b''): h.update(block)
    return h.hexdigest()
def dump(path, value):
    Path(path).write_text(json.dumps(value, indent=2, ensure_ascii=False)+'\n')

class Bindings:
    def __init__(self, manifest):
        self.manifest = Path(manifest).resolve()
        self.manifest_hash = sha(self.manifest)
        self.data = json.loads(self.manifest.read_text())
        require(self.data['version'] == 1, 'INPUT_MANIFEST_VERSION')
        self.entries = {}
        self.start = {}
        for role, entry in self.data['inputs'].items(): self.add(entry, role)
    def add(self, entry, role):
        p = str(Path(entry['path']).absolute())
        require(re.fullmatch('[0-9a-f]{64}',entry['sha256']), 'INPUT_HASH_FORMAT', p)
        if p in self.entries:
            require(self.entries[p]['sha256'] == entry['sha256'], 'CONFLICTING_INPUT_HASH', p)
            self.entries[p]['roles'].append(role)
        else: self.entries[p] = {'path':p,'sha256':entry['sha256'],'roles':[role]}
    def path(self, role): return Path(self.data['inputs'][role]['path'])
    def verify(self, path):
        p = str(Path(path).absolute())
        expected = self.entries[p]['sha256']
        observed = sha(p)
        require(observed == expected, 'INPUT_HASH_MISMATCH', {'path':p,'expected':expected,'observed':observed})
        self.start[p] = observed
        return observed
    def bytes(self, path):
        p = str(Path(path).absolute()); raw = Path(p).read_bytes()
        observed = hashlib.sha256(raw).hexdigest()
        require(observed == self.entries[p]['sha256'], 'INPUT_HASH_MISMATCH', p)
        self.start[p] = observed
        return raw
    def text(self, role): return self.bytes(self.path(role)).decode()
    def json(self, role): return json.loads(self.text(role))
    def check_all(self):
        require(sha(self.manifest) == self.manifest_hash, 'MANIFEST_CHANGED')
        for p in self.entries: self.verify(p)
    def final(self):
        require(set(self.start) == set(self.entries), 'UNVERIFIED_SEMANTIC_INPUT')
        self.check_all()
        return list(self.entries.values())

RH = 'http://rdf.rhea-db.org/'
NS = {'rh':RH,'rdfs':'http://www.w3.org/2000/01/rdf-schema#','rdf':'http://www.w3.org/1999/02/22-rdf-syntax-ns#'}
WRAPPER = '<root '+' '.join(f'xmlns:{k}="{v}"' for k,v in NS.items())+'>'

def original_rhea(raw_gzip, record, projection):
    """Resolve exact selected objects, reverse master links and native side URIs.

    Neither numeric Rhea offsets nor stored orientations supply the direction.
    Every master is discovered by its rh:directionalReaction link in original RDF;
    direction is then determined by native master_L/master_R substrate/product
    resources. All projected properties are compared to freshly parsed originals.
    """
    raw = gzip.decompress(raw_gzip).decode()
    starts = {}
    for m in re.finditer(r'<rdf:Description rdf:about="([^"]+)">',raw):
        starts.setdefault(m[1],[]).append(m.start())
    nodes = {}
    def get(uri):
        if uri in nodes: return nodes[uri]
        require(uri in starts, 'RDF_OBJECT_MISSING', uri)
        props = {}
        for start in starts[uri]:
            end = raw.index('</rdf:Description>',start)+len('</rdf:Description>')
            node = ET.fromstring(WRAPPER+raw[start:end]+'</root>')[0]
            for child in node:
                key = child.tag.split('}',1)[1]
                value = child.get('{'+NS['rdf']+'}resource') or child.text
                if value not in props.setdefault(key,[]): props[key].append(value)
        nodes[uri] = props
        return props
    evidence = []
    for reaction in [r for r in record['reactions'] if r['id'].startswith('RHEA:')]:
        rid = reaction['id']; uri = RH+rid.split(':')[1]; n = get(uri)
        require(n.get('accession') == [rid] and n.get('id') == [rid.split(':')[1]] and n.get('status') == [RH+'Approved'] and RH+'DirectionalReaction' in n.get('subClassOf',[]), 'RDF_DIRECTIONAL_IDENTITY', rid)
        # Search every original link to this exact URI, then identify its enclosing
        # RDF subject. The selected record/projection never supplies the master ID.
        masters = []
        token = '<rh:directionalReaction rdf:resource="'+uri+'"/>'
        for match in re.finditer(re.escape(token),raw):
            start = raw.rfind('<rdf:Description rdf:about="',0,match.start())
            match_master = re.match(r'<rdf:Description rdf:about="([^"]+)">',raw[start:])
            require(match_master is not None,'RDF_MASTER_SUBJECT',rid)
            master_uri = match_master[1]
            if master_uri not in masters: masters.append(master_uri)
        require(len(masters) == 1,'RDF_UNIQUE_MASTER',{'id':rid,'masters':masters})
        master_uri = masters[0]; master = get(master_uri); mid = master_uri.removeprefix(RH)
        require(master.get('accession') == ['RHEA:'+mid] and master.get('id') == [mid] and master.get('status') == [RH+'Approved'] and RH+'Reaction' in master.get('subClassOf',[]) and uri in master.get('directionalReaction',[]), 'RDF_MASTER_IDENTITY', rid)
        left,right = master_uri+'_L',master_uri+'_R'
        require(set(master.get('side',[])) == {left,right} and len(master['side']) == 2, 'RDF_MASTER_SIDES',rid)
        pair = (n.get('substrates'),n.get('products'))
        if pair == ([left],[right]): direction = 'left_to_right'
        elif pair == ([right],[left]): direction = 'right_to_left'
        else: raise ValueError('RDF_ORIENTATION_UNRESOLVED: '+rid)
        require(reaction['direction'] == direction,'NATIVE_DIRECTION_MISMATCH',{'id':rid,'declared':reaction['direction'],'native':direction,'master':'RHEA:'+mid,'substrates':n['substrates'],'products':n['products']})
        require(n.get('equation') == [reaction['label']], 'NATIVE_EQUATION_MISMATCH', rid)
        sides = {}
        for side,predicate in [('substrates','consumes'),('products','produces')]:
            expected = {}
            for sid in n[side]:
                s = get(sid)
                for participant in s['contains']:
                    coefs = [int(key[8:]) for key,values in s.items() if re.fullmatch(r'contains\d+',key) and participant in values]
                    require(len(coefs) == 1,'RDF_COEFFICIENT',participant)
                    cu = get(participant)['compound'];require(len(cu)==1,'RDF_COMPOUND',participant)
                    compound = get(cu[0]); accession = compound['accession'][0]
                    cid = accession if accession.startswith('CHEBI:') else 'RHEA-COMP:'+cu[0].split('Compound_')[1]
                    require(cid not in expected,'DUPLICATE_RDF_CHEMICAL_SIDE',cid)
                    expected[cid] = coefs[0]
            actual = {}
            for edge in record['mechanistic_edges']:
                if edge['predicate'] == predicate and edge['object' if predicate == 'consumes' else 'subject'] == rid:
                    cid = edge['subject' if predicate == 'consumes' else 'object']
                    require(cid not in actual,'DUPLICATE_RECORD_CHEMICAL_SIDE',cid)
                    coefficient = re.search(r'^Stoichiometric coefficient (\d+)\.',edge['description'])
                    require(coefficient is not None,'COEFFICIENT_MISSING',edge['id'])
                    actual[cid] = int(coefficient[1])
            require(expected == actual,'NATIVE_CHEMICAL_SIDE_MISMATCH',{'id':rid,'side':side,'native':expected,'actual':actual})
            sides[side] = expected
        evidence.append({'reaction':rid,'master':'RHEA:'+mid,'native_direction':direction,'declared_direction':reaction['direction'],'native_substrate_resources':n['substrates'],'native_product_resources':n['products'],'chemical_sides':sides})
    require(len(evidence)==3 and sum(len(v) for e in evidence for v in e['chemical_sides'].values())==14,'FULL_CHEMISTRY_COUNTS')
    derived=[]
    for e in evidence:
        rid=e['reaction']; sides={}
        for side in ['substrates','products']:
            vals=[]
            for side_uri in nodes[RH+rid.split(':')[1]][side]:
                for part in get(side_uri)['contains']:
                    c_uri=get(part)['compound'][0]; c=get(c_uri)
                    vals.append({'participant':part,'compound_uri':c_uri,'id':c['accession'][0],'coefficient':next(int(k[8:]) for k,v in get(side_uri).items() if re.fullmatch(r'contains\d+',k) and part in v),'formula':c['formula'][0],'charge':int(c['charge'][0])})
            sides[side]=vals
        balance=collections.Counter()
        for side,sign in [('substrates',-1),('products',1)]:
            for c in sides[side]:
                for atom,num in re.findall(r'([A-Z][a-z]?)(\d*)',c['formula']):balance[atom]+=sign*c['coefficient']*int(num or 1)
                balance['charge']+=sign*c['coefficient']*c['charge']
        require(not any(balance.values()),'CHEMICAL_BALANCE',rid)
        derived.append({'id':rid,'master':e['master'],'direction':e['native_direction'],'equation':get(RH+rid.split(':')[1])['equation'][0],'sides':sides,'balance':dict(balance)})
    require(derived==projection,'DERIVED_CHEMISTRY_PROJECTION_MISMATCH')
    return nodes,evidence,len(starts)

def native_owl(raw):
 root=etree.fromstring(raw);out={}
 for n in root:
  key=n.get('{'+BP['rdf']+'}ID') or n.get('{'+BP['rdf']+'}about')
  if not key:continue
  p={}
  for c in n:p.setdefault(etree.QName(c).localname,[]).append(c.get('{'+BP['rdf']+'}resource') or c.text)
  require(key not in out,'DUPLICATE_BIOPAX_ID',key);out[key]={'type':etree.QName(n).localname,'properties':p}
 return root,out
