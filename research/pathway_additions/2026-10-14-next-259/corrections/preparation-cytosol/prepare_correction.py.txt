def bind(path):
    path = Path(path)
    h = hashlib.sha256()
    with path.open('rb') as handle:
        for block in iter(lambda: handle.read(1048576), b''):
            h.update(block)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': h.hexdigest()}

def strings(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, dict):
        for child in value.values():
            yield from strings(child)
    elif isinstance(value, list):
        for child in value:
            yield from strings(child)


def normalize(value):
    return ' '.join(value.split())


def extract(path):
    path = Path(path)
    if path.suffix == '.json':
        return normalize('\n'.join(strings(json.loads(path.read_text()))))
    if path.suffix in ['.html', '.htm']:
        return normalize(html.fromstring(path.read_bytes()).text_content())
    if path.suffix == '.xml':
        return normalize(' '.join(etree.fromstring(path.read_bytes()).itertext()))
    return normalize(path.read_text())

def diff(a, b, path=''):
    result = []
    if isinstance(a, dict) and isinstance(b, dict):
        for key in a.keys() | b.keys():
            loc = path + '/' + str(key)
            if key not in a:
                result.append({'path': loc, 'before': None, 'after': b[key]})
            elif key not in b:
                result.append({'path': loc, 'before': a[key], 'after': None})
            else:
                result.extend(diff(a[key], b[key], loc))
    elif isinstance(a, list) and isinstance(b, list):
        for i in range(max(len(a), len(b))):
            loc = path + '/' + str(i)
            if i >= len(a):
                result.append({'path': loc, 'before': None, 'after': b[i]})
            elif i >= len(b):
                result.append({'path': loc, 'before': a[i], 'after': None})
            else:
                result.extend(diff(a[i], b[i], loc))
    elif a != b:
        result.append({'path': path, 'before': a, 'after': b})
    return result
