changes = []
def compare(a, b, path=''):
    if a == b:
        return
    if isinstance(a, dict) and isinstance(b, dict):
        for key in sorted(a.keys() | b.keys()):
            if key not in a or key not in b:
                changes.append({'path': path + '/' + key, 'before': a.get(key), 'after': b.get(key)})
            else:
                compare(a[key], b[key], path + '/' + key)
    elif isinstance(a, list) and isinstance(b, list):
        if all(isinstance(v, dict) and 'id' in v for v in a + b):
            compare({v['id']: v for v in a}, {v['id']: v for v in b}, path)
        else:
            for index in range(max(len(a), len(b))):
                if index >= len(a) or index >= len(b):
                    changes.append({'path': f'{path}/{index}', 'before': a[index] if index < len(a) else None, 'after': b[index] if index < len(b) else None})
                else:
                    compare(a[index], b[index], f'{path}/{index}')
    else:
        changes.append({'path': path, 'before': a, 'after': b})

quotes = collections.defaultdict(list)
def quote_walk(value, path=''):
    if isinstance(value, dict):
        if 'quote' in value:
            quotes[value['reference_id']].append({'path': path, 'quote': value['quote'], 'words': len(value['quote'].split())})
        for key, child in value.items():
            quote_walk(child, path + '/' + key)
    elif isinstance(value, list):
        for index, child in enumerate(value):
            quote_walk(child, path + '/' + str(index))

def strings(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, dict):
        for child in value.values():
            yield from strings(child)
    elif isinstance(value, list):
        for child in value:
            yield from strings(child)

class Blocks(HTMLParser):
    def __init__(self):
        super().__init__(); self.kind = None; self.buffer = []; self.rows = []; self.all_text = []
    def handle_starttag(self, tag, attrs):
        if tag in ['p', 'tr'] and self.kind is None:
            self.kind = tag; self.buffer = []
    def handle_endtag(self, tag):
        if tag == self.kind:
            self.rows.append((tag, norm(''.join(self.buffer)).strip())); self.kind = None
    def handle_data(self, text):
        self.all_text.append(text)
        if self.kind:
            self.buffer.append(text)
