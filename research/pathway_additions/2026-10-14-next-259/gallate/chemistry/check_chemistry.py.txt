def chem(m):
    if m is None:return {'parsed':False}
    Chem.AssignStereochemistry(m,cleanIt=True,force=True)
    return {'parsed':True,'smiles':Chem.MolToSmiles(m,isomericSmiles=True),'hydrogen_normalized_smiles':Chem.MolToSmiles(Chem.RemoveHs(m),isomericSmiles=True),'inchi':Chem.MolToInchi(m),'key':Chem.MolToInchiKey(m),'formula':rdMolDescriptors.CalcMolFormula(m),'charge':Chem.GetFormalCharge(m),'double_bond_stereo':[str(b.GetStereo()) for b in m.GetBonds() if b.GetBondType()==Chem.BondType.DOUBLE],'chiral_centers':Chem.FindMolChiralCenters(m,includeUnassigned=True),'atoms':dict(collections.Counter(a.GetSymbol() for a in Chem.AddHs(m).GetAtoms()))}
def cml_mol(e):
    rw=Chem.RWMol(); amap={}; atoms=e.findall('./atomArray/atom'); conf=Chem.Conformer(len(atoms));conf.Set3D(False)
    for a in atoms:
        at=Chem.Atom(a.attrib['elementType']);at.SetFormalCharge(int(a.get('formalCharge','0')));i=rw.AddAtom(at);amap[a.attrib['id']]=i
        conf.SetAtomPosition(i,(float(a.get('x2','0')),float(a.get('y2','0')),0.))
    for b in e.findall('./bondArray/bond'):
        u,v=b.get('atomRefs').split();rw.AddBond(amap[u],amap[v],{1:Chem.BondType.SINGLE,2:Chem.BondType.DOUBLE,3:Chem.BondType.TRIPLE}[int(b.get('order'))])
    m=rw.GetMol();Chem.SanitizeMol(m);m.AddConformer(conf);Chem.DetectBondStereochemistry(m);Chem.AssignStereochemistry(m,cleanIt=True,force=True);return m
