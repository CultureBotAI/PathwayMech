def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for block in iter(lambda:f.read(1024*1024),b''):h.update(block)
    return h.hexdigest()
def bind(p):return {'path':str(p),'bytes':p.stat().st_size,'sha256':sha(p)}
def write(n,d):
    p=R/n
    if p.exists():raise FileExistsError(p)
    p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')

def drawn_graph(subs):
    rw=Chem.RWMol()
    for i in range(6):
        atom=Chem.Atom('C');atom.SetIsAromatic(True);rw.AddAtom(atom)
    for i in range(6):rw.AddBond(i,(i+1)%6,Chem.BondType.AROMATIC)
    def atom(element,charge=0):
        x=Chem.Atom(element);x.SetFormalCharge(charge);return rw.AddAtom(x)
    def bond(a,b,double=False):rw.AddBond(a,b,Chem.BondType.DOUBLE if double else Chem.BondType.SINGLE)
    for pos,group in subs.items():
        root=pos-1
        if group=='OH':bond(root,atom('O'))
        elif group=='CH3':bond(root,atom('C'))
        elif group=='OCH3':
            o=atom('O');bond(root,o);bond(o,atom('C'))
        elif group=='COOH':
            c=atom('C');bond(root,c);bond(c,atom('O'),True);bond(c,atom('O'))
        elif group=='NO2':
            n=atom('N',1);bond(root,n);bond(n,atom('O'),True);bond(n,atom('O',-1))
        else:raise ValueError(group)
    m=rw.GetMol();Chem.SanitizeMol(m);return m
