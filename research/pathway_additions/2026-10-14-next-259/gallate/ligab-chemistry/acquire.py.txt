"""Download named issuing records only when explicitly invoked."""
from pathlib import Path
import concurrent.futures,datetime,hashlib,json,urllib.request,urllib.error
ROOT=Path(__file__).resolve().parent
IDS=['88714','18026','16318','17254','141700']
def get(cid):
    url=f'https://www.ebi.ac.uk/chebi/backend/api/public/compound/{cid}/'
    target=ROOT/'originals'/f'CHEBI-{cid}.json'
    if target.exists():raise FileExistsError(target)
    receipt={'requested_url':url,'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech bounded chemical identity research'}),timeout=45) as response:
            data=response.read();receipt.update(status=response.status,final_url=response.url,headers=dict(response.headers))
    except urllib.error.HTTPError as e:
        data=e.read();receipt.update(status=e.code,error=str(e));target=target.with_suffix('.http-error.txt')
    except Exception as e:return dict(receipt,error=repr(e))
    target.write_bytes(data)
    receipt.update(path=str(target.relative_to(ROOT)),bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
    return receipt
if __name__=='__main__':
    import sys
    receipt=ROOT/('acquisition-public.json' if '--public' in sys.argv else 'acquisition-initial.json')
    if receipt.exists():raise FileExistsError(receipt)
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:rows=list(pool.map(get,IDS))
    receipt.write_text(json.dumps(rows,indent=2)+'\n')
    print(json.dumps([{k:v for k,v in row.items() if k in ['requested_url','status','bytes','error']} for row in rows],indent=2))
