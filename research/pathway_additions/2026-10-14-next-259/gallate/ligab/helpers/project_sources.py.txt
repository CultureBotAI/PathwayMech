"""Offline projections; no network and no source mutation."""
from pathlib import Path
from html.parser import HTMLParser
from xml.etree import ElementTree as ET
import hashlib, json, shutil, zipfile

ROOT = Path(__file__).resolve().parent
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def write(name, obj):
    p=ROOT/name
    if p.exists(): raise FileExistsError(p)
    p.write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n')

class Node:
    def __init__(self, tag, attrs=(), parent=None):
        self.tag,self.attrs,self.parent,self.children=tag,dict(attrs),parent,[]
    def text(self):
        return ''.join(x if isinstance(x,str) else x.text() for x in self.children)
    def find(self, tag=None, id=None):
        out=[]
        for c in self.children:
            if isinstance(c,Node):
                if (tag is None or c.tag==tag) and (id is None or c.attrs.get('id')==id): out.append(c)
                out.extend(c.find(tag,id))
        return out
class Parse(HTMLParser):
    def __init__(self): super().__init__();self.root=Node('root');self.at=self.root
    def handle_starttag(self,t,a):
        n=Node(t,a,self.at);self.at.children.append(n)
        if t not in ('img','meta','link','input','br','hr','source','wbr','area','base','embed','param','track','col'):self.at=n
    def handle_endtag(self,t):
        p=self.at
        while p.parent and p.tag!=t:p=p.parent
        if p.parent:self.at=p.parent
    def handle_data(self,s):self.at.children.append(s)
def norm(s):return ' '.join(s.split())
html=ROOT/'originals/PMC3926093.html'
p=Parse();p.feed(html.read_text())
tables=[]
for tid in ['T1','T2','T3']:
    n=p.root.find(id=tid)[0]
    tables.append({'id':tid,'source_sha256':sha(html),'locator':f'#{tid}',
                   'title_and_footnotes':norm(n.text()),
                   'rows':[[norm(c.text()) for c in r.children if isinstance(c,Node) and c.tag in ('th','td')] for r in n.find('tr')]})
write('projections/2013-structured-tables.json',tables)
paragraphs=[{'id':n.attrs['id'],'text':norm(n.text())} for n in p.root.find('p') if n.attrs.get('id','').startswith('P')]
sections=[{'id':n.attrs.get('id'),'heading':norm(n.find()[0].text()) if n.find() else ''} for n in p.root.find('section') if n.attrs.get('id','').startswith('S')]
write('projections/2013-paragraph-locators.json',{'source_sha256':sha(html),'paragraphs':paragraphs,'sections':sections})
