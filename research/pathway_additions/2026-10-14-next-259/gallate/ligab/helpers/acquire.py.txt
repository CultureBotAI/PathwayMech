B=Path(__file__).parent
def acquire(name,url):
 p=B/'originals'/name
 if p.exists(): raise ValueError('Refuse overwrite: '+str(p))
 r={'name':name,'url':url,'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech author source research'}),timeout=40) as x:
   data=x.read(); r.update(status=x.status,final_url=x.url,content_type=x.headers.get('Content-Type'))
