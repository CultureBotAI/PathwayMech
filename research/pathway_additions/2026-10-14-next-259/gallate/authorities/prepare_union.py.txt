B = Path(__file__).resolve().parent
R = Path('/private/tmp/pathwaymech-next-259-20261014')

PREFIXES = {'GO', 'CHEBI', 'NCBITaxon', 'RHEA'}
inputs = {}


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def bind(path, digest=None):
    path = Path(path)
    row = dict(path=str(path), resolved=str(path.resolve()), sha256=digest or sha(path))
    assert str(path) not in inputs or inputs[str(path)] == row
    inputs[str(path)] = row
    return row


def load(path):
    bind(path)
    return json.loads(Path(path).read_text())


def save(name, value):
    path = B / name
    assert not path.exists(), f'Preserve prior attempt before replacement: {path}'
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')
    return path


def git(*args):
    return subprocess.check_output(['git', *args], cwd=R)


assert git('rev-parse', 'HEAD').decode().strip() == BASE
assert not (B / 'canonical-build-commands.json').exists()
baseline = {}
for rel, name in [('conf/identifier_sources.json', 'identifier_sources.json'),

    assert (R / rel).read_bytes() == (B / 'baseline' / name).read_bytes() == git('show', BASE + ':' + rel)

    bind(R / rel)

draft_path = A / 'gallate-degradation-i.yaml'

record = yaml.safe_load(draft_bytes)
actual = {record['id']} | {n['id'] for section in ['taxa', 'participants', 'reactions'] for n in record.get(section, [])}
assert record['id'] == 'MetaCyc:GALLATE-DEGRADATION-II-PWY' and actual == planned

    path = B / name

candidate_paths = [B / 'authority-candidates.json', B / 'supplemental-authority-candidates.json']
candidates = {}
for path in candidate_paths:
    for row in load(path):
        candidates.setdefault(row['identifier'], []).append(dict(row=row, inventory=str(path), inventory_sha256=sha(path)))
adds, new_bindings, previews, staged = [], [], [], {}
for ident in sorted(new_ids):
    if ident.split(':', 1)[0] in PREFIXES:
        continue
    rows = candidates.get(ident, [])
    assert len(rows) == 1, (ident, len(rows), 'Resolve exact original candidate before selection')
    wrapper = rows[0]
    candidate = wrapper['row']['candidate']
    source = Path(candidate['path'])
    assert sha(source) == candidate['sha256']
    parsed = list(source_terms(candidate['kind'], source, url=candidate['url'], version=candidate['version']))
    matched = [term for term in parsed if term[0] == ident]
    assert len(matched) == 1 and list(matched[0]) == wrapper['row']['parsed_term'], (ident, parsed)
    rel = 'meta259/' + source.name
    assert rel not in staged or staged[rel] == str(source), ('cache filename collision', rel)
    staged[rel] = str(source)
    license_text = {'uniprot': 'CC-BY-4.0', 'ncbi-protein-xml': 'NCBI data usage policies; source attribution retained', 'kegg-gene-flatfile': 'KEGG database use terms; redistribution not assessed', 'kegg-pseudogene-flatfile': 'KEGG database use terms; redistribution not assessed'}.get(candidate['kind'], 'BioCyc license; redistribution not assessed')
    entry = dict(key='meta259-' + ident.replace(':', '-'), kind=candidate['kind'], path=rel,
                 url=candidate['url'], version=candidate['version'], license=license_text, sha256=candidate['sha256'])
    adds.append(entry)
    new_bindings.append(dict(identifier=ident, entry=entry, original=str(source), **wrapper))
    previews.append(dict(identifier=ident, term=matched[0], scope='Canonical parser applied to exact issuing original; identity and name only.'))
    bind(source, candidate['sha256'])

ontology_previews = []

    assert db.stat().st_size == prior['database']['bytes']
    owner_key, owner = prior['owner']
    assert baseline['ontology.json']['sources'][owner_key] == owner
    requested = sorted(i for i in new_ids if i.startswith(prior['prefix'] + ':'))
    with sqlite3.connect(db.resolve().as_uri() + '?mode=ro', uri=True) as connection:
        terms, missing = extract_terms(connection, requested, owner_key)
    assert not missing, (prior['prefix'], missing)
    ontology_previews.append(dict(prefix=prior['prefix'], owner=owner_key, terms=terms))
    stat = db.stat()
    database_bindings.append(dict(path=str(db), resolved=str(db.resolve()), sha256=prior['database']['sha256'], bytes=stat.st_size, mtime_ns=stat.st_mtime_ns, owner_key=owner_key,
                                  verification='Prior inventory verified full database hash. Canonical builder will independently hash complete database and final verifier requires unchanged owner. Runner also checks size and mtime before/after; no redundant preparer database hashing.'))
assert sum(len(row['terms']) for row in ontology_previews) == 20
