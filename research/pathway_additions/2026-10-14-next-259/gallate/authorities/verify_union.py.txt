"""Verify canonical union preservation externally; this script cannot promote files."""
from pathlib import Path
import argparse
import datetime
import hashlib
import json
import shutil
import subprocess
import sys

import yaml

B = Path(__file__).resolve().parent
R = Path('/private/tmp/pathwaymech-next-259-20261014')
BASE = '7e18d0e2b7dd066ea5176b3f679624f7e50a0dc3'


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def load(path):
    return json.loads(Path(path).read_text())


def save(name, value):
    p = B / name
    assert not p.exists(), f'Preserve prior verification attempt: {p}'
    p.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--record', required=True, type=Path)
    parser.add_argument('--record-sha256', required=True)
    args = parser.parse_args()
    assert sha(args.record) == args.record_sha256
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=R, text=True).strip() == BASE
    commands = load(B / 'canonical-build-commands.json')
    preservation = load(B / 'build-input-preservation.json')
    assert preservation['status'] == 'passed' and preservation['all_inputs_unchanged']
    assert preservation['command_file_sha256'] == sha(B / 'canonical-build-commands.json')
    for relative, digest in commands['runtime_sha256'].items():
        assert sha(R / relative) == digest
    for key in ['source_builder', 'ontology_builder']:
        receipt = load(B / (key + '-receipt.json'))
        assert receipt['exit_code'] == 0 and receipt['command'] == commands[key]
        assert receipt['command_file_sha256'] == sha(B / 'canonical-build-commands.json')
        assert sha(receipt['log_path']) == receipt['log_sha256']
    retained = set(load(B / 'retained-authority-identifiers.json'))
    planned = set(load(B / 'planned-identifiers.json'))
    frozen = load(B / 'selection-freeze.json')
    assert set(frozen['selected_identifiers']) == planned
    record = yaml.safe_load(args.record.read_text())
    actual = {record['id']} | {n['id'] for field in ['taxa', 'participants', 'reactions'] for n in record.get(field, [])}
    assert actual == planned and len(planned) == 87
    report = dict(status='author_verification_passed_pending_independent_review',
                  utc=datetime.datetime.now(datetime.timezone.utc).isoformat(), baseline_commit=BASE,
                  frozen_identifier_count=len(planned), baseline_identifier_count=len(retained),
                  snapshots={}, final_record=dict(path=str(args.record), sha256=args.record_sha256),
                  selection_checkpoint=frozen, maintained_files_edited=False)
    old_terms, all_terms = {}, {}
    for name in ['sources', 'ontology']:
        old = load(B / 'baseline' / (name + '.json'))
        new_path = B / (name + '.union.json')
        assert sha(new_path) == preservation['outputs'][name + '.union.json']
        new = load(new_path)
        assert old['version'] == new['version']
        for section in ['terms', 'sources']:
            changed = [key for key, value in old[section].items() if new[section].get(key) != value]
            assert not changed, (name, section, changed)
        assert not all_terms.keys() & new['terms'].keys()
        all_terms.update(new['terms'])
        old_terms.update(old['terms'])
        report['snapshots'][name] = dict(before_sha256=sha(B / 'baseline' / (name + '.json')),
                                        after_sha256=sha(new_path), baseline_terms=len(old['terms']),
                                        final_terms=len(new['terms']), baseline_source_owners=len(old['sources']),
                                        final_source_owners=len(new['sources']),
                                        baseline_objects_unchanged=True,
                                        new_terms=sorted(new['terms'].keys() - old['terms'].keys()),
                                        new_source_owners=sorted(new['sources'].keys() - old['sources'].keys()))
    assert len(retained) == 4954 and retained == set(old_terms)
    assert planned <= all_terms.keys() and set(all_terms) - retained == planned - retained
    assert len(all_terms) == 4989
    old_manifest = load(B / 'baseline/identifier_sources.json')
    new_manifest = load(B / 'identifier_sources.json')
    additions = load(B / 'proposed-source-additions.json')
    assert len(old_manifest) == 1047 and len(additions) == 15 and len(new_manifest) == 1062
    assert new_manifest[:len(old_manifest)] == old_manifest
    assert new_manifest[len(old_manifest):] == additions
    assert {row['key'] for row in additions} == set(report['snapshots']['sources']['new_source_owners'])
    assert not report['snapshots']['ontology']['new_source_owners']
    report['manifest'] = dict(baseline_entries=len(old_manifest), final_entries=len(new_manifest),
                              baseline_entry_prefix_exact=True, new_entries=len(additions), sha256=sha(B / 'identifier_sources.json'))
    for row in commands['database_bindings']:
        stat = Path(row['path']).stat()
        assert stat.st_size == row['bytes'] and stat.st_mtime_ns == row['mtime_ns']
        owner = load(B / 'ontology.union.json')['sources'][row['owner_key']]
        assert owner['sha256'] == row['sha256']
    report['pinned_databases'] = dict(count=4, canonical_builder_complete_hashes_match_previous_owners=True,
                                      size_and_mtime_unchanged=True, redundant_verifier_full_database_hashing=False)
    report['original_input_hash_verification'] = preservation
    report['exact_selected_new_ids'] = sorted(planned - retained)
    report['final_authority_terms'] = len(all_terms)
    # Test unchanged canonical policy and native entity typing on the whole copied corpus.
    shadow = B / 'identifier-verification-root'
    assert not shadow.exists()
    (shadow / 'conf').mkdir(parents=True)
    (shadow / 'data/identifier_authorities').mkdir(parents=True)
    (shadow / 'data/pathways').mkdir(parents=True)
    policy = R / 'conf/identifier_policy.yaml'
    shutil.copyfile(policy, shadow / 'conf/identifier_policy.yaml')
    for name in ['sources', 'ontology']:
        shutil.copyfile(B / (name + '.union.json'), shadow / 'data/identifier_authorities' / (name + '.json'))
    for row in load(B / 'baseline-record-bindings.json'):
        assert sha(row['path']) == row['sha256'] == sha(R / row['git_path'])
        destination = shadow / row['git_path']
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(row['path'], destination)
    shutil.copyfile(args.record, shadow / 'data/pathways/gallate-degradation-i.yaml')
    assert len(list((shadow / 'data/pathways').rglob('*.yaml'))) == 256
    sys.path.insert(0, str(R / 'src'))
    from pathwaymech.identifiers import identifier_errors
    errors = identifier_errors(shadow)
    save('identifier-policy-verification.json', dict(status='passed' if not errors else 'failed',
        errors=errors, records_checked=256, authority_terms=len(all_terms),
        policy_path=str(policy), policy_sha256=sha(policy), record_sha256=args.record_sha256,
        scope='Canonical offline identifier/name/native entity placement gate only; not scientific approval or full repository validation.'))
    assert not errors, errors
    # Verify maintained authority files remain at exact baseline; root owns promotion.
    for rel, name in [('conf/identifier_sources.json', 'identifier_sources.json'),
                      ('conf/rhea_identifier_updates.json', 'rhea_identifier_updates.json'),
                      ('data/identifier_authorities/sources.json', 'sources.json'),
                      ('data/identifier_authorities/ontology.json', 'ontology.json')]:
        assert (R / rel).read_bytes() == (B / 'baseline' / name).read_bytes()
    assert sha(args.record) == args.record_sha256
    save('final-record-selection-binding.json', dict(final_record_path=str(args.record),
        final_record_sha256=args.record_sha256, selected_identifiers=sorted(planned), count=len(planned),
        previous_record_observation=frozen['record_observation'], previous_selection_observation=frozen['selection_observation'],
        exact_identifier_set_unchanged=True, authority_build_inputs_unchanged=True,
        note='Scientific author successor changes are separate from frozen identifier selection; this proves selection equality only.'))
    save('planned-authority-labels.json', {key: all_terms[key] for key in sorted(planned)})
    save('union-preservation-verification.json', report)
    print(json.dumps(dict(status=report['status'], final_authority_terms=len(all_terms), new_terms=len(planned-retained),
                         manifest_entries=len(new_manifest), records_checked=256, maintained_files_edited=False)))


if __name__ == '__main__':
    main()
