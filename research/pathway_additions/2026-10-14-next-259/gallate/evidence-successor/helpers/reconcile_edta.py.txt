from pathlib import Path
import datetime,hashlib,json,sqlite3
R=Path(__file__).resolve().parent
P=Path('/private/tmp/pathwaymech-curation-authorities/chebi.db')
def bind(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
 return {'path':str(p),'bytes':p.stat().st_size,'sha256':h.hexdigest()}

query="""select subject,predicate,value from statements where predicate in ('rdfs:label','oio:hasExactSynonym','oio:hasRelatedSynonym','oio:hasBroadSynonym','oio:hasNarrowSynonym') and (lower(value) like '%edta%' or lower(value) like '%edetate%' or lower(value) like '%edetic%' or lower(value) like '%ethylenediaminetetraacet%' or lower(value) like '%ethylenedinitrilotetraacet%') order by subject,predicate,value"""
rows=db.execute(query).fetchall()
