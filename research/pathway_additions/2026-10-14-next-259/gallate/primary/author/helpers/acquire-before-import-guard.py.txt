import concurrent.futures,datetime,hashlib,json,urllib.request,urllib.error
from pathlib import Path
out=Path('/private/tmp/pathwaymech-next-259-primary')
jobs=[('Nogales2011-supp.pdf','https://onlinelibrary.wiley.com/action/downloadSupplement?doi=10.1111%2Fj.1365-2958.2010.07448.x&file=MMI_7448_sm_fS1-fS8-tS1.pdf'),('PMC9942251.xml','https://www.ebi.ac.uk/europepmc/webservices/rest/PMC9942251/fullTextXML'),('PMC9942251.html','https://pmc.ncbi.nlm.nih.gov/articles/PMC9942251/'),('Nogales2005.pdf','https://www.jbc.org/article/S0021925819482112/pdf'),('Nogales2011-csic.html','https://digital.csic.es/handle/10261/51769'),('FN669140.xml','https://www.ebi.ac.uk/ena/browser/api/xml/FN669140')]
def run(job):
 name,url=job;r={'name':name,'url':url,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech evidence research'}),timeout=45) as s:b=s.read();r.update(status=s.status,final_url=s.url,content_type=s.headers.get('Content-Type'))
 except urllib.error.HTTPError as e:b=e.read();r.update(status=e.code,error=str(e),content_type=e.headers.get('Content-Type'));name=name+'.failed'
 except Exception as e:r.update(error=str(e));return r
 dest=out/'originals'/name;dest.write_bytes(b);r.update(path=str(dest),bytes=len(b),sha256=hashlib.sha256(b).hexdigest());return r
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:rows=list(pool.map(run,jobs))
(out/'acquisition-initial.json').write_text(json.dumps(rows,indent=2)+'\n');print(json.dumps(rows,indent=2))
