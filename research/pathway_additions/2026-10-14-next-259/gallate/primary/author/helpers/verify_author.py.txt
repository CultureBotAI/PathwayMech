import json,hashlib

def h(f):return hashlib.sha256(f.read_bytes()).hexdigest()
def strings(v):
 if isinstance(v,str):return [v]
 if isinstance(v,dict):return sum((strings(x) for x in v.values()),[])
 if isinstance(v,list):return sum((strings(x) for x in v),[])
 return []
sources={'PMID:21219457':['candidate_wiley-web-result.json','candidate_wiley_part2-web-result.json'],'DOI:10.1016/j.procbio.2021.01.019':['desb2021_rg_mid-web.json']}
checks=[]
def walk(v):
 if isinstance(v,dict):

   matches=[]
   for n in sources[v['reference_id']]:
    f=P/'originals'/n
    if v['quote'] in '\n'.join(strings(json.loads(f.read_text()))):matches.append({'path':str(f),'sha256':h(f)})

  for x in v.values():walk(x)
 elif isinstance(v,list):
  for x in v:walk(x)
