def sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
 return h.hexdigest()
def binding(p):return {'path':str(p),'realpath':str(p.resolve()),'bytes':p.stat().st_size,'sha256':sha(p)}
def write(n,d):
 p=R/n
 if p.exists():raise FileExistsError(p)
 p.write_text(json.dumps(d,ensure_ascii=False,indent=2)+'\n')
def triples(db,id):return [{'predicate':p,'object':o,'value':v} for p,o,v in db.execute('select predicate,object,value from statements where subject=? order by predicate,object,value',(id,))]
def values(rows,p):return [x['object'] or x['value'] for x in rows if x['predicate']==p]
def val(rows,p):
 v=values(rows,p);return v[0] if v else None
def strip(s):return html.unescape(re.sub('<[^>]*>','',s))

 while todo:
