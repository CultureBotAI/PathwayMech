from pathlib import Path
import urllib.request,urllib.error,datetime,hashlib,json,time
B=Path(__file__).parent;rows=[]
for n,u in {'AB110976.embl':'https://www.ebi.ac.uk/ena/browser/api/embl/AB110976?download=false','BAC79261.1.xml':'https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=protein&id=BAC79261.1&rettype=gb&retmode=xml'}.items():
 p=B/'originals'/n;assert not p.exists();r={'path':str(p),'url':u,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}

  with urllib.request.urlopen(u,timeout=45)as f:b=f.read();r.update(status=f.status,final_url=f.url)
