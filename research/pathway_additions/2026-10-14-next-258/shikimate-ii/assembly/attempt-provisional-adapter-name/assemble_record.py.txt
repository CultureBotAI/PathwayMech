"""Integrate frozen proposals and verified native identities outside maintained data."""
from pathlib import Path
import copy
import hashlib
import json
import re
import sqlite3
import sys

B = Path(__file__).parent
R = Path('/private/tmp/pathwaymech-next-258-20261014')
N = Path('/private/tmp/pathwaymech-next-258-native')
P = Path('/private/tmp/pathwaymech-next-258-primary/author/package')
A = Path('/private/tmp/pathwaymech-next-258-authorities')
sys.path.insert(0, str(R / 'scripts'))
from build_source_identifier_snapshot import source_terms
from pathwaymech.schema import validate_record

record = json.loads((B / 'core-record.json').read_text())
native = json.loads((N / 'native-graph-proposal.json').read_text())
primary = json.loads((P / 'edge-proposals.json').read_text())
index = json.loads((P / 'package-index.json').read_text())
for item in index['files']:
    assert hashlib.sha256((P / item['file']).read_bytes()).hexdigest() == item['sha256']
candidates = {item['identifier']: item for item in json.loads((N / 'authority-candidates.json').read_text())}
baseline_terms = {}
for name in ['sources', 'ontology']:
    baseline_terms.update(json.loads((A / 'baseline' / (name + '.json')).read_text())['terms'])
changes = []
proposal_map = []
previews = []


def normalize_reference(identifier):
    return 'UniProtKB:' + identifier.removeprefix('primary258-UniProt-') if identifier.startswith('primary258-UniProt-') else identifier


def prose(text):
    # Author drafts preserve source text; normalize only presentation fields.
    replacements = {'C.glutamicum': 'C. glutamicum', 'E.coli': 'E. coli', 'ATCC13032': 'ATCC 13032',
        'ATCC13869': 'ATCC 13869', 'glutamicum2256': 'glutamicum 2256', 'coliBL21': 'coli BL21',
        'His6-QsuB': 'His6-QsuB', 'C-terminalHis6': 'C-terminal His6', 'Sourcegene': 'Source gene',
        'in50mM': 'in 50 mM', '10mM': '10 mM', 'pH7.5': 'pH 7.5', 'Co>Mg>Mn': 'Co > Mg > Mn',
        'about3mM': 'about 3 mM', 'above4mM': 'above 4 mM', 'Fig7caption': 'Fig. 7 caption'}
    for old, new in replacements.items():
        text = text.replace(old, new)
    return text


for item in native['references'] + json.loads((P / 'record-references.json').read_text()):
    item = copy.deepcopy(item)
    item['id'] = normalize_reference(item['id'])
    existing = next((r for r in record['references'] if r['id'] == item['id']), None)
    if existing is None:
        record['references'].append(item)
    else:
        assert existing['source_sha256'] == item['source_sha256'], (existing, item)
        changes.append(dict(reference_id=item['id'], disposition='same complete source hash; first citation metadata retained'))

for proposal in native['edges'] + primary:
    edge = copy.deepcopy(proposal)
    proposal_id = edge.pop('proposal_id')
    edge.pop('id', None)
    edge['description'] = prose(edge['description'])
    for evidence in edge['evidence']:
        evidence['reference_id'] = normalize_reference(evidence['reference_id'])
    for qualifier in edge.get('endpoint_qualifiers', []):
        qualifier['description'] = prose(qualifier['description'])
        for evidence in qualifier['evidence']:
            evidence['reference_id'] = normalize_reference(evidence['reference_id'])
    record['mechanistic_edges'].append(edge)
    proposal_map.append(dict(proposal_id=proposal_id, edge_index=len(record['mechanistic_edges']) - 1))

supplied = {item['id']: item for item in native['nodes'] + native.get('reactions', [])}
activity_ids = {item['id'] for item in native.get('reactions', [])}
connected = {edge[role] for edge in record['mechanistic_edges'] for role in ['subject', 'object']}
existing = {record['id']} | {item['id'] for field in ['taxa', 'participants', 'reactions'] for item in record[field]}
for identifier in sorted(connected - existing):
    if identifier in baseline_terms:
        label = baseline_terms[identifier]['label']
    elif identifier in candidates:
        candidate = candidates[identifier]
        path = Path(candidate['path'])
        assert hashlib.sha256(path.read_bytes()).hexdigest() == candidate['sha256']
        terms = list(source_terms(candidate['kind'], path, url=candidate['url'], version=candidate['version']))
        matching = [item for item in terms if item[0] == identifier]
        assert len(matching) == 1, (identifier, terms)
        label = matching[0][1]
        previews.append(dict(identifier=identifier, term=matching[0], candidate=candidate))
    elif identifier.startswith(('CHEBI:', 'GO:')):
        dbname = 'chebi' if identifier.startswith('CHEBI:') else 'go'
        with sqlite3.connect('file:/private/tmp/pathwaymech-curation-authorities/' + dbname + '.db?mode=ro', uri=True) as db:
            rows = db.execute("select value from statements where subject=? and predicate='rdfs:label'", (identifier,)).fetchall()
        assert len(rows) == 1, (identifier, rows)
        label = rows[0][0]
    else:
        raise AssertionError(('No issuing identity available', identifier))
    if identifier in supplied:
        category = supplied[identifier]['category']
    elif identifier.startswith(('KEGG:cgt:', 'MetaCyc:G-')):
        category = 'gene'
    elif identifier.startswith('CHEBI:'):
        category = 'cofactor' if identifier in {'CHEBI:18420', 'CHEBI:29035', 'CHEBI:48828'} else 'small_molecule'
    elif identifier.startswith('GO:'):
        category = 'cellular_component'
    else:
        assert identifier.startswith(('UniProtKB:', 'NCBIProtein:'))
        category = 'protein'
    destination = 'reactions' if identifier in activity_ids else 'participants'
    record[destination].append(dict(id=identifier, label=label, category=category))

for index, edge in enumerate(record['mechanistic_edges'], 1):
    edge['id'] = 'shikimate-ii-' + str(index).zfill(3)
for row in proposal_map:
    row['edge_id'] = record['mechanistic_edges'][row.pop('edge_index')]['id']
used = {item['reference_id'] for edge in record['mechanistic_edges']
        for item in edge['evidence'] + [e for q in edge.get('endpoint_qualifiers', []) for e in q['evidence']]}
assert used <= {r['id'] for r in record['references']}
record['references'] = [r for r in record['references'] if r['id'] in used]
validate_record(record)
quoted = {}
for edge in record['mechanistic_edges']:
    for item in edge['evidence'] + [e for q in edge.get('endpoint_qualifiers', []) for e in q['evidence']]:
        if 'quote' in item:
            quoted[item['reference_id']] = quoted.get(item['reference_id'], 0) + len(item['quote'].split())
assert all(value <= 25 for value in quoted.values()), quoted
(B / 'assembled-record.json').write_text(json.dumps(record, indent=2) + '\n')
(B / 'integration-dispositions.json').write_text(json.dumps(dict(
    scope='Author integration; not independent review.', references=changes, proposal_edges=proposal_map,
    same_endpoint_assays='Native 2008 and primary 2013 QsuD catalysis edges retain distinct experimental construct qualifiers and NADP assay claims.',
    quote_counts_including_repeats=quoted, new_identity_previews=previews,
    record_sha256=hashlib.sha256((B / 'assembled-record.json').read_bytes()).hexdigest()), indent=2) + '\n')
print(json.dumps({'edges': len(record['mechanistic_edges']), 'participants': len(record['participants']), 'references': len(record['references'])}))
