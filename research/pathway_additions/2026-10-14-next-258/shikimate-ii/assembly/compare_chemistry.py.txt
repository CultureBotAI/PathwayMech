def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        while block := stream.read(1024 * 1024):
            h.update(block)
    return h.hexdigest()

def bind(path):
    return dict(path=str(path), bytes=path.stat().st_size, sha256=sha(path))

authority = json.loads((R / 'data/identifier_authorities/ontology.json').read_text())

db = sqlite3.connect('file:' + str(A / 'chebi.db') + '?mode=ro', uri=True)
def term(identifier):
    keys = {'rdfs:label', 'chemrof:inchi_string', 'chemrof:inchi_key_string',
            'chemrof:charge', 'chemrof:generalized_empirical_formula', 'chemrof:smiles_string'}
    return {key: value or obj for key, obj, value in db.execute(
        'select predicate,object,value from statements where subject=?', (identifier,)) if key in keys}

for frame, number in mapping.items():
    source = NEW / 'originals' / ('META-' + frame + '.xml') if frame == 'SHIKIMATE' else OLD / 'originals' / ('META-' + frame + '.xml')

    native = doc.find('Compound')
    assert native.get('frameid') == frame and native.get('detail') == 'full'
    assert doc.find('metadata/PGDB').get('version') == '27.0'
    identifier = 'CHEBI:' + number
    pinned = term(identifier)
    inchi = native.findtext('inchi')
    mol = Chem.MolFromInchi(inchi)
    assert mol is not None
    checks = dict(direct_chebi_xref=number in [x.findtext('dblink-oid') for x in native.findall('dblink') if x.findtext('dblink-db') == 'CHEBI'],
                  exact_inchi=inchi == pinned['chemrof:inchi_string'],
                  exact_inchi_key=native.findtext('inchi-key').removeprefix('InChIKey=') == pinned['chemrof:inchi_key_string'],
                  charge=Chem.GetFormalCharge(mol) == int(pinned['chemrof:charge']))
    assert all(checks.values()), (frame, checks)
    chemical_rows.append(dict(native_id='MetaCyc:' + frame, selected_id=identifier,

                                               charge=Chem.GetFormalCharge(mol),
                                               isomeric_smiles=Chem.MolToSmiles(mol, isomericSmiles=True))))

nodes = rhea['nodes']; NS = 'http://rdf.rhea-db.org/'
def side(uri):
    result = []
    weighted_members = set()
    for key, values in nodes[uri].items():
        if key.startswith('contains') and key != 'contains':
            assert re.fullmatch(r'contains[1-9][0-9]*', key), (uri, key)
            for participant in values:
                assert participant not in weighted_members, (uri, participant)
                weighted_members.add(participant)
                compound_uri = nodes[participant]['compound'][0]
                compound = nodes[compound_uri]
                result.append(dict(id=compound['accession'][0], coefficient=int(key[8:]),
                                   formula=compound['formula'][0], charge=int(compound['charge'][0]),
                                   locator=[uri + ' rh:' + key + ' ' + participant,
                                            participant + ' rh:compound ' + compound_uri]))
    assert result
    assert set(nodes[uri]['contains']) == weighted_members
    return result

def totals(items):
    counter = collections.Counter()
    for item in items:
        atoms = re.findall(r'([A-Z][a-z]*)([0-9]*)', item['formula'])
        assert ''.join(a + n for a, n in atoms) == item['formula']
        for atom, count in atoms:
            counter[atom] += int(count or 1) * item['coefficient']
        counter['charge'] += item['charge'] * item['coefficient']
    return dict(counter)

reactions = []
for frame, master, selected in [('RXN-20484', '17741', '17742'), ('DHSHIKIMATE-DEHYDRO-RXN', '24848', '24849')]:
    source = N / ('META-' + frame + '.xml')
    assert sha(source) == native_bindings[str(source)]['sha256']
    native = E.parse(source).getroot().find('Reaction')
    assert native.get('frameid') == frame and native.get('detail') == 'full'
    native_sides = {name: [dict(frame=x.find('Compound').get('frameid'),
                               explicit_coefficient=x.findtext('coefficient'),
                               interpreted_coefficient=int(x.findtext('coefficient') or 1)) for x in native.findall(name)] for name in ['left', 'right']}
    assert all(len(rows) == len({x['frame'] for x in rows}) for rows in native_sides.values())
    assert all(x['interpreted_coefficient'] > 0 for rows in native_sides.values() for x in rows)
    selected_node = nodes[NS + selected]
    assert selected_node['substrates'] == [NS + master + '_L'] and selected_node['products'] == [NS + master + '_R']
    left, right = side(NS + master + '_L'), side(NS + master + '_R')
    assert totals(left) == totals(right)
    for key, items in [('left', left), ('right', right)]:
        assert collections.Counter({'CHEBI:' + mapping[x['frame']]: x['interpreted_coefficient'] for x in native_sides[key]}) == collections.Counter({x['id']: x['coefficient'] for x in items})
    assert native.findtext('reaction-direction') == {
        'RXN-20484': 'PHYSIOL-LEFT-TO-RIGHT',
        'DHSHIKIMATE-DEHYDRO-RXN': 'LEFT-TO-RIGHT',
    }[frame]
    reactions.append(dict(native_frame=frame, native_source=bind(source), native_sides=native_sides,
                          native_direction=native.findtext('reaction-direction'),
                          native_ec=[x.text for x in native.findall('ec-number')],
                          selected='RHEA:' + selected, family=rhea['directions'][master],
                          left=left, right=right, atom_charge_totals=totals(left),
                          whole_sides_equal=True, balanced=True,
                          decision='Complete molecular sides and direction agree. Pathway membership and enzyme specificity require their own source evidence.'))
