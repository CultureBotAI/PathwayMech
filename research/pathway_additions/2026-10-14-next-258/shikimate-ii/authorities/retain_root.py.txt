"""Retain compact author assembly and authority reproduction artifacts exactly."""

def bind(path):
    data = path.read_bytes()
    return dict(path=str(path), sha256=hashlib.sha256(data).hexdigest(), bytes=len(data))


def retain(root, names, destination):
    destination.mkdir(parents=True, exist_ok=True)
    rows = []
    for name in names:
        source = root / name
        data = source.read_bytes()
        if source.suffix == '.py':
            ast.parse(data)
        if source.suffix == '.json':
            json.loads(data)
        rel = name + '.txt' if source.suffix == '.py' else name
        text = data.decode('utf-8')
        wrapper = any(line != line.rstrip() for line in text.splitlines())
        if wrapper:
            rel += '.bytes.json'
            output = (json.dumps(dict(original=bind(source), encoding='base64', data=base64.b64encode(data).decode()), indent=2) + '\n').encode()
        else:
            output = data
        target = destination / rel
        target.parent.mkdir(parents=True, exist_ok=True)
        assert not target.exists() or target.read_bytes() == output, str(target)
        target.write_bytes(output)
        rows.append(dict(original=bind(source), retained=bind(target),
                         repository_path=str(target.relative_to(R)), encoding='base64' if wrapper else 'byte-identical'))
    manifest = dict(retained_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
        scope='Author evidence and reproduction; independent review is separate.', files=rows)
    target = destination / 'retention.json'
    assert not target.exists()
    target.write_text(json.dumps(manifest, indent=2) + '\n')
    return rows
