from acquire import *
rows=[]
items=[('KEGG-genome-T00501.txt','https://rest.kegg.jp/get/T00501'),('META-PROTOCATECHUATE-ORTHO-CLEAVAGE-PWY.html','https://bf2i-artsymbiocyc.insa-lyon.fr/META/NEW-IMAGE?type=PATHWAY&object=PROTOCATECHUATE-ORTHO-CLEAVAGE-PWY')]
for name,url in items:
 row={'url':url,'retrieved_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  with urllib.request.urlopen(url,timeout=60) as r:b=r.read();row.update(status=r.status,final_url=r.url,content_type=r.headers.get('Content-Type'))
  if name.endswith('.txt'):assert b.startswith(b'ENTRY')
  else:assert b'<html' in b.lower() and b'protocatechuate' in b.lower()
  p=ORIG/name;p.write_bytes(b);row.update(path=str(p),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
 except Exception as e:row['error']=str(e)
 rows.append(row)
(ROOT/'acquisition-genome-interface.json').write_text(json.dumps(rows,indent=2)+'\n');print(json.dumps(rows,indent=2))
