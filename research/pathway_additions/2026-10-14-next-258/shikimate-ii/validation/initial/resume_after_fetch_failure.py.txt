"""Finish only the failed public-network gate and the remaining whitespace gate."""
from pathlib import Path
import datetime
import hashlib
import json
import os
import subprocess
import time

B = Path(__file__).parent
R = Path('/private/tmp/pathwaymech-next-258-20261014')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


original = (B / 'receipts.json').read_bytes()
rows = json.loads(original)
assert len(rows) == 5 and all(x['exit_code'] == 0 for x in rows[:4])
assert rows[-1]['command'] == ['just', 'vendored-check'] and rows[-1]['exit_code'] == 1
failed_log = Path(rows[-1]['log_path'])
assert sha(failed_log) == rows[-1]['log_sha256']
assert 'nodename nor servname provided, or not known' in failed_log.read_text()
assert not (B / 'initial-attempt-receipts.json').exists()
(B / 'initial-attempt-receipts.json').write_bytes(original)
failure = dict(observed_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
               reason='The sandbox could not resolve raw.githubusercontent.com. Only the public-network gate is retried with network access; no input, command, or policy waiver is changed.',
               failed_attempts=[rows[-1]], original_receipts_sha256=sha(B / 'initial-attempt-receipts.json'))
(B / 'failed-attempts.json').write_text(json.dumps(failure, indent=2) + '\n')
inputs = json.loads((B / 'tested-core-input-hashes.json').read_text())
assert not [name for name, digest in inputs.items() if sha(R / name) != digest]
env = dict(os.environ, **json.loads((B / 'environment.json').read_text()))
completed = rows[:4]
for command, name in [(['just', 'vendored-check'], 'just-vendored-check-network-retry.log'),
                      (['git', 'diff', '--check'], 'git-diff-check.log')]:
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    then = time.monotonic()
    log = B / name
    with log.open('xb') as output:
        result = subprocess.run(command, cwd=R, env=env, stdout=output, stderr=subprocess.STDOUT)
    receipt = dict(command=command, cwd=str(R), started_utc=started,
                   elapsed_seconds=round(time.monotonic() - then, 2), exit_code=result.returncode,
                   log_path=str(log), log_sha256=sha(log))
    completed.append(receipt)
    (B / 'resumed-attempt-receipts.json').write_text(json.dumps(completed, indent=2) + '\n')
    print(json.dumps(receipt), flush=True)
    assert result.returncode == 0, log.read_text()
changed = [name for name, digest in inputs.items() if sha(R / name) != digest]
assert not changed, changed
(B / 'receipts.json').write_text(json.dumps(completed, indent=2) + '\n')
(B / 'final-input-preservation.json').write_text(json.dumps(dict(
    tested_core_files=len(inputs), changed_after_checks=changed,
    final_record_sha256=sha(R / 'data/pathways/shikimate-degradation-ii.yaml'),
    initial_failed_network_attempt_retained='initial-attempt-receipts.json',
    completion='Four original successful commands plus the successful network retry and remaining whitespace check.'), indent=2) + '\n')
print('All six effective gates passed; original network failure retained.', flush=True)
