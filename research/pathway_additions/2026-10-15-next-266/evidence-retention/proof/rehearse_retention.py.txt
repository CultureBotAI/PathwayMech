"""Guarded external copy and exact source-layout reconstruction rehearsal."""

import hashlib
import json
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path("/private/tmp/pathwaymech-next-266-retention-preparation")
PAYLOAD = ROOT / "payload"
REPO = Path("/private/tmp/pathwaymech-next-266-20261015")
COPY = ROOT / "guarded-copy"
RECONSTRUCTION = ROOT / "reconstructed-sources"


def sha(body):
    return hashlib.sha256(body).hexdigest()


def dump(path, value):
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n")


def files(root):
    paths = list(root.rglob("*"))
    assert not any(p.is_symlink() for p in paths)
    return {str(p.relative_to(root)): p for p in paths if p.is_file()}


def run(name, args, expected=0):
    result = subprocess.run([sys.executable, "-B", *args], capture_output=True)
    (ROOT / (name + ".stdout")).write_bytes(result.stdout)
    (ROOT / (name + ".stderr")).write_bytes(result.stderr)
    if expected == 0:
        assert result.returncode == 0, (name, result.stderr.decode())
    else:
        assert result.returncode != 0, name
    return {
        "name": name,
        "argv": [sys.executable, "-B", *args],
        "returncode": result.returncode,
        "stdout_sha256": sha(result.stdout),
        "stderr_sha256": sha(result.stderr),
    }


assert not COPY.exists() and not RECONSTRUCTION.exists()
manifest_bytes = (PAYLOAD / "artifact-manifest.json").read_bytes()
manifest = json.loads(manifest_bytes)
expected = {r["path"]: r for r in manifest["files"]}
expected["artifact-manifest.json"] = {
    "path": "artifact-manifest.json",
    "sha256": sha(manifest_bytes),
    "bytes": len(manifest_bytes),
}
assert set(files(PAYLOAD)) == set(expected)
COPY.mkdir()
for relative, row in expected.items():
    body = (PAYLOAD / relative).read_bytes()
    assert sha(body) == row["sha256"] and len(body) == row["bytes"]
    destination = COPY / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open("xb") as stream:
        stream.write(body)
    assert destination.read_bytes() == body
assert set(files(COPY)) == set(expected)

        run(

                str(COPY / "decode_retained.py"),

                name,
                "--materialize",

    )
    rows = json.loads(source_map.read_bytes())["source_occurrences"]
    assert set(files(target)) == {r["path"] for r in rows}

        body = (target / row["path"]).read_bytes()

        {
            "package": name,
            "physical_files": len(rows),
            "physical_bytes": sum(r["bytes"] for r in rows),
