def now():
    return datetime.datetime.now(datetime.UTC).isoformat().replace('+00:00', 'Z')

def load(path):
    return json.loads(Path(path).read_bytes())

def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()

def bind(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}

def verify(row):
    actual = bind(row['path'])
    assert actual['sha256'] == row['sha256'], row['path']
    assert actual['bytes'] == row.get('bytes', actual['bytes']), row['path']
    return actual

def write(name, value):
    with (HERE / name).open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

def files(path):
    return {Path(value) for value in subprocess.check_output(['rg', '--files', '-uu', '-L', str(path)], text=True).splitlines()}

def passed(name, detail):
    CHECKS.append({'name': name, 'status': 'passed', 'detail': detail})
    print(name + ': passed', flush=True)

def packet(directory, digest, originals=None):
    handoff = verify({'path': str(directory / 'handoff.json'), 'sha256': digest})
    data = load(handoff['path'])
    manifest = verify(data['manifest'])
    rows = load(manifest['path'])['artifacts']
    assert len({row['path'] for row in rows}) == len(rows)
    for row in rows:
        assert Path(row['path']).is_relative_to(directory), row['path']
        verify(row)
    declared = {Path(row['path']) for row in rows}
    assert files(directory) == declared | {directory / 'handoff.json', directory / 'artifact-manifest.json'} | set(originals or []), 'closed inventory ' + str(directory)
    return data, rows

def key_order_differences(left, right, path='$'):
    result = []
    if isinstance(left, dict):
        if list(left) != list(right):
            result.append(path)
        for key in left:
            result.extend(key_order_differences(left[key], right[key], path + '.' + key))
    elif isinstance(left, list):
        for index, (a, b) in enumerate(zip(left, right)):
            result.extend(key_order_differences(a, b, path + f'[{index}]'))
    return result
