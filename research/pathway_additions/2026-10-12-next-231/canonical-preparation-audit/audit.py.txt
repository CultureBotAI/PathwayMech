def now():
    return datetime.datetime.now(datetime.UTC).isoformat().replace('+00:00', 'Z')


def load(path):
    return json.loads(Path(path).read_bytes())


def binding(path):
    path = Path(path)
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': digest.hexdigest()}


def verify(row):
    actual = binding(row['path'])
    assert actual['sha256'] == row['sha256'], row['path']
    assert actual['bytes'] == row.get('bytes', actual['bytes']), row['path']
    return actual


def write(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')


def files(path):
    return {Path(p) for p in subprocess.check_output(['rg', '--files', '-uu', '-L', str(path)], text=True).splitlines()}


def passed(name, detail):
    checks.append({'name': name, 'status': 'passed', 'detail': detail})
    print(name + ': passed', flush=True)


def rejected(function):
    try:
        function()
    except AssertionError as error:
        return {'rejected': True, 'error': str(error)}
    raise AssertionError('Unexpected acceptance of bounded negative control')


def packet(row):
    verify(row['handoff'])
    verify(row['manifest'])
    directory = Path(row['handoff']['path']).parent
    artifacts = load(row['manifest']['path'])['artifacts']
    paths = set()
    for item in artifacts:
        path = Path(item.get('path') or item['relative_path'])
        if not path.is_absolute():
            assert '..' not in path.parts
            path = directory / path
        assert path.is_relative_to(directory) and path not in paths
        verify({**item, 'path': str(path)})
        paths.add(path)
    links = set()
    if directory == PRIOR:
        for item in load(PRIOR / 'execution-intake.json')['source_originals_freshly_hashed']:
            path = PRIOR / 'originals' / item['cache_relative_path']
            assert path.is_symlink() and path.resolve() == Path(item['path']).resolve()
            verify({**item, 'path': str(path)})
            links.add(path)
    assert files(directory) == paths | links | {Path(row['handoff']['path']), Path(row['manifest']['path'])}
    return {'directory': str(directory), 'artifacts': len(paths), 'original_links': len(links)}
