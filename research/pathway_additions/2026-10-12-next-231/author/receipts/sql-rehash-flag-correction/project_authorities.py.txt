def bind(p):
    p = Path(p)
    with p.open('rb') as f:
        sha = hashlib.file_digest(f, 'sha256').hexdigest()
    return {'path': str(p), 'bytes': p.stat().st_size, 'sha256': sha}

def save(n, v):
    p = OUT/'authority'/n
    with p.open('x') as f:
        json.dump(v, f, indent=2, ensure_ascii=False)
        f.write('\n')

def add(kind, p, url, key, ids, version='27.0', existing=None):
    p = Path(p)
    entry = copy.deepcopy(existing) if existing else {
        'key': key, 'kind': kind, 'path': 'ppi231-'+p.name, 'url': url,
        'version': version, 'license': 'Upstream terms; public-release licensing deferred',
        'sha256': bind(p)['sha256']}
    assert entry['sha256'] == bind(p)['sha256']
    q = cache/entry['path']
    q.parent.mkdir(parents=True, exist_ok=True)
    q.symlink_to(p.resolve())
    ts = list(sb.source_terms(kind, p, url=url, version=version))
    selected = [t for t in ts if t[0] in ids]
    assert {t[0] for t in selected} == set(ids), (key, ids, selected)
    entries.append(entry)
    originals.append({'entry': entry, 'original': bind(p), 'external_cache': str(q)})
    replays.append({'entry_key': key, 'selected_terms': selected, 'original': bind(p)})
    wanted.update(ids)

for prefix, db in [('CHEBI','chebi'),('NCBITaxon','ncbitaxon'),('GO','go')]:
    ids = sorted(i for i in selected_o if i.startswith(prefix+':'))
    p = SQL/(db+'.db')
    before = {'size':p.stat().st_size, 'mtime_ns':p.stat().st_mtime_ns}
    with sqlite3.connect(p.resolve().as_uri()+'?mode=ro&immutable=1', uri=True) as conn:
        meta = ob.read_source(conn,p)
        key = db+'-'+meta['version']
        ts, missing = ob.extract_terms(conn,ids,key)
        assert not missing, missing
        raw = {i:[{'predicate':a,'object':b,'value':c} for a,b,c in conn.execute(
            'SELECT predicate,object,value FROM statements WHERE subject=?',(i,))] for i in ids}
    assert before == {'size':p.stat().st_size, 'mtime_ns':p.stat().st_mtime_ns}
    ontology['sources'][key] = meta
    ontology['terms'].update(ts)
    oreplay.append({'original_path':str(p),'stat_before_after':before,'metadata':meta,
                    'selected_terms':ts,'native_statements':raw,
                    'full_sql_rehash':False,'binding_owner':str(BASE/'handoff.json')})

decisions = []
for name, current, addition in [('sources',composed_s,snap),('ontology',composed_o,ontology)]:
    for ident, term in addition['terms'].items():
        if ident in current['terms']:
            old = current['terms'][ident]
            assert old['label'] == term['label'] and set(old.get('synonyms',[])) == set(term.get('synonyms',[])), (ident,old,term)
            decisions.append({'id':ident,'disposition':'preserve existing exact object','proposal_source':term['source'],'retained_source':old['source']})
        else:
            current['terms'][ident] = term
            decisions.append({'id':ident,'disposition':'new native term','source':term['source']})
    used = {t['source'] for t in current['terms'].values()}
    for key, meta in addition['sources'].items():
        if key not in used:
            continue
        if key in current['sources']:
            assert current['sources'][key] == meta, key
        else:
            current['sources'][key] = meta
    assert all(current['terms'][i] == t for i,t in (base_s if name=='sources' else base_o)['terms'].items())
