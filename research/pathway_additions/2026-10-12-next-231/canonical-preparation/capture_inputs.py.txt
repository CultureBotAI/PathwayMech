def copy_binding(source, destination):
    before = bind(source)
    destination.parent.mkdir(parents=True, exist_ok=True)
    assert not destination.exists()
    shutil.copyfile(source, destination)
    after = bind(destination)
    assert (before['bytes'], before['sha256']) == (after['bytes'], after['sha256'])
    return {'original': before, 'captured': after}


def verify_packet(name, directory, digest):
    handoff = check({'path': str(directory/'handoff.json'), 'sha256': digest})
    data = load(handoff['path'])
    manifest = check(data.get('manifest') or data['artifact_manifest'])
    rows = load(manifest['path'])['artifacts']
    paths = set()
    for row in rows:
        path = Path(row.get('path') or row['relative_path'])
        if not path.is_absolute():
            assert '..' not in path.parts
            path = directory/path
        assert path.is_relative_to(directory) and path not in paths
        check({**row, 'path': str(path)})
        paths.add(path)
    links = set()
    if name == 'atp_build':
        for row in load(directory/'execution-intake.json')['source_originals_freshly_hashed']:
            p = directory/'originals'/row['cache_relative_path']
            assert p.is_symlink() and p.resolve() == Path(row['path']).resolve()
            check({**row, 'path': str(p)})
            links.add(p)
    observed = set(inventory(directory))
    expected = paths | {directory/'handoff.json', Path(manifest['path'])} | links
    assert observed == expected, {'packet': name, 'extra': sorted(str(p) for p in observed-expected), 'missing': sorted(str(p) for p in expected-observed)}
    return {'name': name, 'handoff': handoff, 'manifest': manifest, 'artifacts_verified': len(rows), 'exact_original_symlinks_verified': len(links), 'closed_hidden_ignored_followed_inventory': True}


def native_collectors():
    sys.path.insert(0, str(HERE/'captured/code/src'))
    importlib.invalidate_caches()
    package = importlib.import_module('pathwaymech')
    assert Path(package.__file__).resolve() == (HERE/'captured/code/src/pathwaymech/__init__.py').resolve()
    modules = {}
    for kind in ['source', 'ontology']:
        p = HERE/'captured/code/scripts'/('build_'+kind+'_identifier_snapshot.py')
        spec = importlib.util.spec_from_file_location('ppi_capture_'+kind, p)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        modules[kind] = module
    return modules


def selections(path, modules):
    assert list(path.parent.glob('*.yaml')) == [path]
    data = yaml.safe_load(path.read_bytes())
    return {'source_collector': sorted(modules['source'].corpus_ids(path.parent)), 'ontology_collector': sorted(modules['ontology'].walk_identifiers(data, {'GO', 'CHEBI', 'NCBITaxon', 'RHEA'}))}
