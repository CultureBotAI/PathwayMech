def now():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def binding(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}


def load(path):
    return json.loads(Path(path).read_bytes())


def verify(path, item, allow_symlink=False):
    path = Path(path)
    assert path.is_file() and (allow_symlink or not path.is_symlink()), path
    assert sha(path) == item['sha256'], path
    if 'bytes' in item:
        assert path.stat().st_size == item['bytes'], path


def write_new(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')


def git(root, *args):
    return subprocess.check_output(['git', *args], cwd=root, text=True, env={**os.environ, 'GIT_OPTIONAL_LOCKS': '0'}).strip()


def safe_relative(value):
    path = Path(value)
    assert not path.is_absolute() and '..' not in path.parts
    assert str(path) not in ('', '.')
    return path


def inventory(root, relative='.', exclude_runtime=False):
    """Use rg's hidden/ignored/followed inventory and fail on traversal errors."""
    command = ['rg', '--files', '-uu', '-L']
    if exclude_runtime:
        command += ['-g', '!.git', '-g', '!.git/**', '-g', '!.venv', '-g', '!.venv/**']
    command.append(relative)
    result = subprocess.run(command, cwd=root, text=True, capture_output=True)
    assert result.returncode in (0, 1) and not result.stderr, result.stderr
    paths = sorted(str(Path(p)) for p in result.stdout.splitlines())
    assert len(paths) == len(set(paths))
    return [{'relative_path': p, **binding(root / p),
             'symlink': (root / p).is_symlink()} for p in paths]


def external_new(path, root):
    path = Path(path).absolute()
    assert path.parent.is_dir() and path.parent.resolve().is_relative_to(Path('/private/tmp'))
    resolved = path.resolve()

def validate_base(root, base, clean=True):
    assert BASE is not None and TREE is not None, 'Actual merged baseline is not yet bound'
    assert BASE_INFO.get('status') == 'actual_merged_base_verified'
    verify(PACKAGE / 'handoff.json', BASE_INFO['preparation'])
    root = Path(root).absolute()
    assert root == ROOT and root.is_dir()
    assert base == BASE and git(root, 'rev-parse', 'HEAD') == BASE
    assert git(root, 'rev-parse', 'HEAD^{tree}') == TREE
    assert git(root, 'branch', '--show-current') == BRANCH
    subprocess.run(['git', 'merge-base', '--is-ancestor', base,
                    'refs/remotes/origin/main'], cwd=root, check=True)
    if clean:
        assert not git(root, 'diff', '--name-only')
        assert not git(root, 'diff', '--cached', '--name-only')
        paths = {r['relative_path'] for r in inventory(root, exclude_runtime=True)}
        assert paths == set(git(root, 'ls-files').splitlines()), (
            'Unexpected untracked/ignored/missing maintained input', paths.symmetric_difference(
                set(git(root, 'ls-files').splitlines())))
    return root


def verify_packet(item):
    verify(item['handoff']['path'], item['handoff'])
    verify(item['manifest']['path'], item['manifest'])
    directory = Path(item['handoff']['path']).parent
    manifest = load(item['manifest']['path'])
    rows = manifest if isinstance(manifest, list) else manifest.get('artifacts', manifest.get('files'))
    assert len(rows) == item['artifact_count']
    assert not item.get('allowed_external_bindings')
    declared = set()
    for row in rows:
        relative = row.get('relative_path', row.get('path'))
        path = Path(relative)
        if not path.is_absolute():
            path = directory / safe_relative(relative)
        if not path.is_relative_to(directory):
            raise AssertionError('PPi packets have no permitted outside-manifest artifact: ' + str(path))
        verify(path, row, allow_symlink=True)
        declared.add(str(path.relative_to(directory)))
    original_links = []
    if item.get('original_links_receipt'):
        verify(item['original_links_receipt']['path'], item['original_links_receipt'])
        for row in load(item['original_links_receipt']['path'])['source_originals_freshly_hashed']:
            path = directory / 'originals' / safe_relative(row['cache_relative_path'])
            assert path.is_symlink() and path.resolve() == Path(row['path']).resolve()
            verify(path, row, allow_symlink=True); original_links.append(str(path.relative_to(directory)))
    assert len(original_links) == item.get('original_symlink_count', 0)
    supplementary = set()
    for row in item.get('supplemental_observed_files', []):
        relative = str(safe_relative(row['relative_path']))
        assert relative not in declared and relative not in supplementary
        verify(directory / relative, row, allow_symlink=True)
        supplementary.add(relative)
    assert {r['relative_path'] for r in inventory(directory)} == declared | {'handoff.json', 'artifact-manifest.json'} | set(original_links) | supplementary, 'Packet inventory changed'

def verify_authority_conservation(root, adopted=False):
    expected = load(PACKAGE / 'expected-changes.json')
    for kind in ('sources', 'ontology'):
        baseline = load(BUILD / 'baseline' / (kind + '.json'))
        successor = load(BUILD / (kind + '.json'))
        assert baseline['version'] == successor['version']
        for section in ('terms', 'sources'):
            assert all(successor[section].get(k) == v for k, v in baseline[section].items())
            assert [k for k in successor[section] if k in baseline[section]] == list(baseline[section]), 'Prior authority relative order changed'
            additions = {k: v for k, v in successor[section].items() if k not in baseline[section]}
            assert additions == expected['authorities'][kind][section]
        verify(root / AUTHORITY_FILES[kind], binding(BUILD / (kind + '.json')) if adopted
               else binding(BUILD / 'baseline' / (kind + '.json')))
    before = load(BUILD / 'baseline/manifest.json')
    after = load(BUILD / 'manifest.json')
    assert after[:len(before)] == before
    assert after[len(before):] == expected['manifest_additions']
    prior_rhea = load(BUILD / 'baseline/rhea-updates.json')
    next_rhea = load(BUILD / 'rhea-updates.json')
    assert next_rhea['ids'] == prior_rhea['ids'] + expected['rhea_additions']
    assert {k: v for k, v in prior_rhea.items() if k != 'ids'} == {
        k: v for k, v in next_rhea.items() if k != 'ids'}
    for kind in ('manifest', 'rhea-updates'):
        verify(root / AUTHORITY_FILES[kind], binding(
            BUILD / (kind + '.json') if adopted else BUILD / 'baseline' / (kind + '.json')))

def verify_cache_union(cache, previous, new, adopted=False):
    """Check every old body, including unselected originals, and exact new paths."""
    assert not (set(previous) & set(new)), 'New cache path collides with old cache'
    current = {r['relative_path']: r for r in inventory(cache)}
    assert set(current) == set(previous) | (set(new) if adopted else set()), 'Cache inventory differs from exact original union'
    for relative, item in previous.items(): verify(cache / safe_relative(relative), item, allow_symlink=True)
    for relative, item in new.items():
        verify(item['original']['path'], item['original'], allow_symlink=True)
        destination = cache / safe_relative(relative)
        assert destination.parent.resolve().is_relative_to(cache.resolve())
        if adopted: verify(destination, item['original'])
        else: assert not destination.exists() and not destination.is_symlink(), 'PPi original destination already occupied'

def history_ready_for_render(root, baseline):
    import yaml
    current = inventory(root, 'history')
    old = {r['relative_path']: r for r in baseline['history']}
    for relative, item in old.items():
        verify(root / relative, item, allow_symlink=True)
    additions = [r for r in current if r['relative_path'] not in old]
    target_history = []
    for item in additions:
        if item['relative_path'].endswith('.yaml'):
            document = yaml.safe_load((root / item['relative_path']).read_bytes())
            if document.get('target', {}).get('path') == TARGET_REL:
                target_history.append(item)
    assert target_history, 'Scaffold the native PPi curation history before rendering and gates'
    return current

def verify_gate_input_scope(root, gates, adoption_receipt, allowed_additions=None):
    """Full hidden/ignored input closure; additions require exact new-file bindings."""
    verify(gates['inputs']['path'], gates['inputs'])
    inputs = load(gates['inputs']['path'])
    verify(adoption_receipt, inputs['adoption_receipt'])
    verify(root / TARGET_REL, inputs['target'])
    assert inputs['render_history_inputs'] == gates['render_history_inputs']
    verify_render_history(root, inputs['render_history_inputs'])
    generated = set(load(PACKAGE / 'expected-changes.json')['allowed_generated_paths'])
    rows = inputs['all_maintained_inputs']; previous = {r['relative_path']: r for r in rows}
    assert len(previous) == len(rows)
    additions = {}
    for row in allowed_additions or []:
        relative = str(safe_relative(row['relative_path']))
        assert relative.startswith(('reviews/structured/', 'history/', DOSSIER_REL + '/'))
        assert relative not in previous and relative not in additions, 'Approval cannot replace gate input'
        verify(root / relative, row); additions[relative] = row
    current = {r['relative_path']: r for r in inventory(root, exclude_runtime=True)}
    assert set(current) - generated == (set(previous) | set(additions)) - generated, 'Unbound nongenerated input inventory change'
    for relative, row in previous.items():
        if relative not in generated: verify(root / relative, row, allow_symlink=True)
    return {'input_receipt': binding(gates['inputs']['path']),
        'nongenerated_inputs_checked': sum(r not in generated for r in previous),
        'exact_new_additions_checked': len(additions), 'whole_render_history_exact': True,
        'complete_current_inventory_verified': True}
