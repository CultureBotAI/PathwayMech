def bind(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size,
            'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}


def write(name, value):
    with (OUT / name).open('x') as f:
        json.dump(value, f, indent=2); f.write('\n')


def rows(root):
    result = subprocess.run(['rg', '--files', '-uu', '-L', str(root)], text=True, capture_output=True)
    assert result.returncode in (0, 1) and not result.stderr
    return set(map(Path, result.stdout.splitlines()))

def actual_receipts():
    summary = guard.load(PACKAGE / 'validation-summary.json'); results = []
    for item in summary['receipts']:
        guard.verify(item['receipt']['path'], item['receipt'])
        receipt = guard.load(item['receipt']['path'])
        assert receipt['exit_code'] == item['exit_code']
        for key in ['stdout', 'stderr']: guard.verify(receipt[key]['path'], receipt[key])
        snapshot = Path(item['receipt']['path']).parent
        changes = []
        for helper in PACKAGE.glob('*.py'):
            old = snapshot / (helper.name + '.txt')
            if old.exists() and old.read_bytes() != helper.read_bytes(): changes.append(helper.name)
        results.append({'name': receipt['name'], 'exit_code': receipt['exit_code'],
                        'receipt': item['receipt'], 'helper_delta_from_final': changes})
    final_guard = guard.load(summary['latest']['guards']['path'])
    assert '42 passed' in Path(final_guard['stdout']['path']).read_text()
    assert sum(item['exit_code'] != 0 for item in results) == 4
    write('actual-receipts.json', results)
    return {'commands': len(results), 'preserved_failures': 4, 'guard_tests': 42,
            'latest_guard_helper_deltas': next(r['helper_delta_from_final'] for r in results if r['name'] == 'guards')}


def dossier():
    data = guard.load(PACKAGE / 'dossier-plan.json'); destinations = set(); kinds = {}
    for row in data['mappings']:
        dest = str(guard.safe_relative(row['destination']))
        assert dest not in destinations and not dest.endswith('.py')
        assert dest not in {'archive-map.json', 'adoption.json'}
        destinations.add(dest); guard.verify(row['source']['path'], row['source'])
        kind = row['serialization']; kinds[kind] = kinds.get(kind, 0) + 1
        if row.get('original'): guard.verify(row['original']['path'], row['original'])
        if kind == 'lossless_utf8_json_wrapper':
            wrapped = guard.load(row['source']['path'])
            assert wrapped['text'].encode() == Path(row['original']['path']).read_bytes()
        elif kind == 'exact_bytes' and Path(row['source']['path']).suffix == '.py':
            assert dest.endswith('.py.txt')
    assert len(destinations) == 385
    return {'mappings': 385, 'serializations': kinds, 'reserved_paths_absent': True}

def external_path_guard():
    samples = [PACKAGE / 'audit-must-not-write.json', guard.BUILD / 'audit-must-not-write.json',
               guard.CACHE / 'audit-must-not-write.json', guard.CORRECTION / 'audit-must-not-write.json']
    for path in samples:
        try: guard.external_new(path, OUT / 'fictional-root')
        except AssertionError: continue
        raise AssertionError('Protected output accepted: ' + str(path))
    existing = OUT / 'existing.txt'; existing.write_text('fixture\n')
    try: guard.external_new(existing, OUT / 'fictional-root')
    except AssertionError: pass
    else: raise AssertionError('Existing output accepted')
    return {'protected_locations_rejected': len(samples), 'existing_output_rejected': True}


def cache_retarget_probe():
    directory = OUT / 'cache-retarget-fixture'; directory.mkdir()
    cache = directory / 'cache'; cache.mkdir()
    first, second = directory / 'first', directory / 'second'
    first.write_bytes(b'identical bounded fixture\n'); second.write_bytes(first.read_bytes())
    link = cache / 'old'; link.symlink_to(first)
    original = guard.inventory(cache)[0]
    previous = {'old': original}
    guard.verify_cache_union(cache, previous, {}, adopted=False)
    link.unlink(); link.symlink_to(second)
    accepted = False
    try:
        guard.verify_cache_union(cache, previous, {}, adopted=False)
        accepted = True
    except AssertionError: pass
    return {'retargeted_link_accepted': accepted, 'old_target': str(first), 'new_target': str(second),
            'same_bytes': first.read_bytes() == second.read_bytes(),
            'complete_inventory_equality_also_passes': guard.inventory(cache) == [original],
            'baseline_row': original, 'current_row': guard.inventory(cache)[0]}

        dest = OUT / 'helpers-read' / (p.name + '.txt'); dest.parent.mkdir(exist_ok=True)
        dest.write_bytes(p.read_bytes())
    checks = []
    for name, function in [('closed_seal', seal_inputs), ('accepted_actual_inputs', actual_inputs),
                           ('actual_receipts', actual_receipts), ('dossier', dossier),
                           ('exact_handle_and_runtime', implementation), ('eg_only_base_guard', old_base_guard),
                           ('external_output_guards', external_path_guard), ('cache_retarget_probe', cache_retarget_probe)]:
        start = datetime.datetime.now(datetime.UTC).isoformat()
        try:
            detail = function(); row = {'name': name, 'status': 'completed', 'detail': detail}
        except Exception as error:
            row = {'name': name, 'status': 'unexpected_failure', 'type': type(error).__name__, 'message': str(error)}
        row['started_at'] = start; row['finished_at'] = datetime.datetime.now(datetime.UTC).isoformat()
        checks.append(row); print(json.dumps(row), flush=True)
        write(name + '.json', row)
    write('checks.json', checks)
    return 1 if any(r['status'] == 'unexpected_failure' for r in checks) else 0
