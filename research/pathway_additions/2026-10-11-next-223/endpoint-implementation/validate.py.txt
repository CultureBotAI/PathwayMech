def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def capture(path, expected=None):
    path = Path(path)
    digest = sha(path)
    assert expected is None or digest == expected, str(path)
    row = {'sha256': digest, 'bytes': path.stat().st_size, 'resolved_path': str(path.resolve())}
    assert str(path) not in bindings or bindings[str(path)] == row
    bindings[str(path)] = row
    return digest

def save(name, value):
    with (CHECKS / name).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

def command(name, argv, expected=0, cwd=REPO):
    before = datetime.now(UTC).isoformat()
    result = subprocess.run(argv, cwd=cwd, env=env, capture_output=True, check=False)
    stdout = CHECKS / (name + '.stdout.log')
    stderr = CHECKS / (name + '.stderr.log')
    stdout.write_bytes(result.stdout)
    stderr.write_bytes(result.stderr)
    receipt = {'argv': argv, 'cwd': str(cwd), 'started': before,
               'finished': datetime.now(UTC).isoformat(), 'exit_code': result.returncode,
               'stdout_sha256': sha(stdout), 'stderr_sha256': sha(stderr)}
    save(name + '.receipt.json', receipt)
    assert result.returncode == expected, receipt
    print(name + ': ' + result.stdout.decode().strip(), flush=True)
    return receipt

# Exact source-example replay is intentionally separate from generic grammar.
# A deletion, scientifically wrong role or changed construct wording can remain
# syntactically valid; the fixed source-qualified claim must still differ.
def replay(candidate):
    observed = {edge['id']: edge for edge in candidate['mechanistic_edges']}
    for original in examples['edges']:
        assert observed[original['id']] == original, original['id']

replay(fixture)
controls = []
def negative(name, mutate, generic_failure=False):
    candidate = copy.deepcopy(fixture)
    mutate(candidate)
    (CHECKS / ('negative-' + name + '.yaml')).write_text(yaml.safe_dump(candidate, sort_keys=False))
    try:
        replay(candidate)
    except AssertionError:
        pass
    else:
        raise AssertionError('Source-specific control accepted: ' + name)
    try:
        validate_record(candidate)
        generic = 'valid grammar; rejected by exact source replay'
        assert not generic_failure, name
    except ValidationError as error:
        generic = str(error)
    controls.append({'case': name, 'source_replay_rejected': True, 'native_result': generic})
