 ensure(g.findtext('strain')=='K-12 substr. MG1655' and g.findtext('dblink/dblink-oid')=='511145','EcoCyc native taxon mismatch')
 xs=[x for x in t if x.tag!='metadata'];ensure(len(xs)==1,'EcoCyc top-level cardinality mismatch');x=xs[0]
 ensure(x.get('ID')=='ECOLI:'+fid and x.get('orgid')=='ECOLI' and x.get('frameid')==fid,'EcoCyc native identity mismatch')
 expected='RNA' if fid=='RNA0-241' else 'Compound' if fid=='CAMP' else 'Protein';ensure(x.tag==expected,'EcoCyc native biological type mismatch')
