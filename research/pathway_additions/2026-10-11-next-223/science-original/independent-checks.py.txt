def digest(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def save(name, value):
    with (DEST / name).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

rdf = {}
def fetch(uris):
    wanted = set(uris) - rdf.keys()
    if not wanted:
        return
    found = collections.defaultdict(lambda: collections.defaultdict(list))
    with gzip.open('/private/tmp/pathwaymech-curation-authorities/rhea.rdf.gz', 'rb') as stream:
        for _, element in etree.iterparse(stream, events=('end',), tag='{'+RDF+'}Description',
                                         resolve_entities=False, no_network=True):
            uri = element.get('{'+RDF+'}about')
            if uri in wanted:
                for child in element:
                    key = etree.QName(child).localname
                    value = child.get('{'+RDF+'}resource') or child.text
                    if value not in found[uri][key]:
                        found[uri][key].append(value)
            element.clear()
            while element.getprevious() is not None:
                del element.getparent()[0]
    assert wanted == found.keys(), wanted - found.keys()
    rdf.update({uri: dict(fields) for uri, fields in found.items()})

chemistry = {}
triples = []
for identifier, node in directions.items():
    uri = RH + identifier
    fields = rdf[uri]
    substrate, product = fields['substrates'][0], fields['products'][0]
    master = substrate.rsplit('_', 1)[0]
    assert master == product.rsplit('_', 1)[0]
    assert rdf[master]['status'] == [RH + 'Approved'] and uri in rdf[master]['directionalReaction']
    orientation = 'left_to_right' if substrate.endswith('_L') else 'right_to_left'
    assert node['direction'] == orientation and node['label'] == fields['equation'][0]
    sides = {}
    balance = collections.Counter()
    complete_balance = True
    for role, predicate, sign in [('substrates', 'consumes', -1), ('products', 'produces', 1)]:
        side = rdf[fields[role][0]]
        rows = []
        for participant in side['contains']:
            coefficient = [int(key[8:]) for key, values in side.items()
                           if re.fullmatch('contains[0-9]+', key) and participant in values]
            assert len(coefficient) == 1
            compound = rdf[rdf[participant]['compound'][0]]
            curie = compound['accession'][0].replace('GENERIC:', 'RHEA-COMP:')
            n = coefficient[0]
            assert n == 1, 'Record evidence must retain every native coefficient'
            triples.append((curie, predicate, node['id']) if sign < 0 else (node['id'], predicate, curie))
            effective = rdf[compound['reactivePart'][0]] if compound.get('reactivePart') else compound
            formula = effective.get('formula', [''])[0]
            if not re.fullmatch(r'(?:[A-Z][a-z]?[0-9]*)+', formula) or 'charge' not in effective:
                complete_balance = False
            else:
                for atom, count in re.findall(r'([A-Z][a-z]?)([0-9]*)', formula):
                    balance[atom] += sign * n * int(count or 1)
                balance['charge'] += sign * n * int(effective['charge'][0])
            rows.append({'id': curie, 'coefficient': n, 'participant_uri': participant,
                         'native_compound': compound, 'reactive_part': effective if effective != compound else None})
        sides[role] = rows
    if complete_balance:
        assert not any(balance.values()), (identifier, balance)
    chemistry[node['id']] = {'orientation': orientation, 'equation': fields['equation'][0],
        'master': master, 'sides': sides, 'balance_calculated': complete_balance,
        'nonzero_atom_charge_delta': {k: v for k, v in balance.items() if v} if complete_balance else None,
        'native_balance_flag': rdf[master].get('isChemicallyBalanced')}
save('fresh-rhea.json', chemistry)

def check_source_shape(candidate):
    edges = candidate['mechanistic_edges']
    observed = collections.Counter((e['subject'], e['predicate'], e['object']) for e in edges if e['predicate'] in ['consumes', 'produces'])
    assert observed == collections.Counter(triples), 'source chemical sides'
    pairs = {(e['subject'], e['predicate'], e['object']) for e in edges}
    assert set(native_memberships + native_ordering) <= pairs, 'complete native children/order'
    for child in children:
        assert (candidate['id'], 'has_part', 'MetaCyc:' + child) in pairs, 'complete native children'
    for row in assembly:
        edge, = [e for e in edges if (e['subject'], e['predicate'], e['object']) == (row['subject'], 'has_part', row['object'])]
        assert any(f"component {row['component']} coefficient {row['coefficient']}." in item.get('source_assertion', '')
                   for item in edge['evidence']), 'native assembly coefficient'
    reactions = {r['id']: r for r in candidate['reactions']}
    for identifier, data in chemistry.items():
        assert reactions[identifier]['direction'] == data['orientation'], 'selected physiological direction'
    assert reactions['BiGG:ACMANAptspp']['direction'] == 'left_to_right', 'model direction'
    # Reviewer-selected critical scope rules derive from independently read
    # primary methods and results; they are not a whole author-payload equality.
    for number, subject, subject_terms in [(107,'MetaCyc:URIDYLYL-PII',['His-tagged','refolded','uridylylated','not repurified']),
        (108,'UniProtKB:P0A761',['His-tagged','ASKA']), (109,'MetaCyc:PTSH-MONOMER',['nonphosphorylated','His-tagged'])]:
        edge = next(e for e in edges if e['id'] == f'amino-{number}')
        assert edge['subject'] == subject and edge['object'] == 'UniProtKB:P0A759' and edge['predicate'] == 'activates', 'exact assay endpoints'
        q = {item['endpoint']: item for item in edge.get('endpoint_qualifiers', [])}
        assert set(q) == {'subject', 'object'}, 'both experimental endpoints qualified'
        assert all(word in q['subject']['description'] for word in subject_terms), 'subject preparation scope'
        assert all(word in q['object']['description'] for word in ['Ser1-NagB','BW25113','first nagB codon omitted','SUMO','Ulp1','native Met','No sequence equivalence elsewhere']), 'NagB construct scope'
        assert all(item['kind'] == 'experimental_construct' and item['evidence'] for item in q.values()), 'construct evidence'
    crp = next(e for e in edges if e['id'] == 'amino-105')
    assert crp['predicate'] == 'regulates', 'unsigned CRP promoter regulation'
    q, = crp['endpoint_qualifiers']
    assert q['endpoint'] == 'subject' and q['kind'] == 'molecular_state' and 'cAMP-bound CRP homodimer' in q['description'], 'CRP state scope'
    assert next(e for e in edges if e['id'] == 'amino-106')['predicate'] == 'regulates', 'unsigned NagC promoter regulation'
    assert 'substrate InChIKey conflicts' in candidate['description'], 'structural conflict retained'
    nanr = next(e for e in edges if e['id'] == 'amino-101')['evidence'][0]['source_assertion']
    assert 'remained dimeric' in nanr and 'older ligand-induced monomerization' in nanr, 'NanR conflict retained'
    return True

controls = []
def reject(name, mutate):
    candidate = copy.deepcopy(record)
    mutate(candidate)
    try:
        check_source_shape(candidate)
    except (AssertionError, KeyError, ValueError) as error:
        controls.append({'name': name, 'rejected': True, 'reason': str(error)})
    else:
        raise AssertionError('Independent false-pass: ' + name)
    (DEST / ('control-' + name + '.yaml')).write_text(yaml.safe_dump(candidate, sort_keys=False))
