"""Root-owned read-only preservation check after next223 integration and rendering."""
from pathlib import Path
from collections import Counter
from dataclasses import asdict
import argparse,csv,hashlib,json,sys


def digest(obj):
    return hashlib.sha256(json.dumps(obj,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode()).hexdigest()


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def table(path):
    with path.open(newline='') as stream:
        reader=csv.DictReader(stream,delimiter='\t');rows=list(reader);columns=reader.fieldnames
    if len({row['id'] for row in rows})!=len(rows) or any(None in r for r in rows):
        raise ValueError('Malformed or duplicate KGX rows: '+str(path))
    return columns,{r['id']:r for r in rows}


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root',required=True,type=Path)
    parser.add_argument('--baseline',required=True,type=Path)
    parser.add_argument('--baseline-kgx',required=True,type=Path)
    parser.add_argument('--output',required=True,type=Path)
    parser.add_argument('--approved-node-display-delta',type=Path)
    args=parser.parse_args();root=args.root.resolve()
    if args.output.exists():raise ValueError('Refusing to replace a previous preservation report')
    base=json.loads(args.baseline.read_bytes());old=json.loads(args.baseline_kgx.read_bytes())
    assert sha(args.baseline_kgx)==base['kgx_baseline_sha256']
    assert root==Path(base['root']).resolve()
    problems=[];retained={}
    for field in ['records','pages','reviews','history']:
        rows=base[field];bad=[]
        for row in rows:
            path=root/row['path']
            if not path.is_file() or sha(path)!=row['sha256']:bad.append(row['path'])
        retained[field]={'checked':len(rows),'changed_or_missing':bad}
        problems.extend(field+': '+p for p in bad)
    target=root/'data/pathways/amino-sugar-superpathway.yaml'
    assert sha(target)=='ebbc3eb8334b28b975444285a6e0aab7315a3153a73492270a083cd87fb4f5ff'
    sys.path[:0]=[str(root/'src'),str(root)]
    import yaml
    from pathwaymech.schema import validate_record
    from pathwaymech.kgx import kgx_edges,kgx_nodes
    from scripts.record_review import review_paths
    model=validate_record(yaml.load(target.read_bytes(),Loader=yaml.CSafeLoader))
    actual_records={str(p.relative_to(root)) for p in (root/'data/pathways').rglob('*.yaml') if p.is_file()}
    expected_records={r['path'] for r in base['records'] if r['path'].endswith('.yaml')}|{str(target.relative_to(root))}
    if actual_records!=expected_records:problems.append('Unexpected maintained record set')
    reviews=review_paths(root)
    old_reviews={r['path'] for r in base['reviews'] if r['path'].endswith('/review.yaml')}
    if not old_reviews<=set(reviews):problems.append('Old structured review bundle missing')
    node_columns,nodes=table(root/'pages/downloads/nodes.tsv')
    edge_columns,edges=table(root/'pages/downloads/edges.tsv')
    if node_columns!=old['node_columns']:problems.append('KGX node columns changed')
    if edge_columns!=old['edge_columns']+['endpoint_qualifiers']:problems.append('Unexpected KGX edge columns')
    lost_edges=[];changed_edges=[];nonblank_legacy=[]
    for identity,expected in old['edges'].items():
        row=edges.get(identity)
        if row is None:lost_edges.append(identity);continue
        if digest({k:row[k] for k in old['edge_columns']})!=expected:changed_edges.append(identity)
        if row.get('endpoint_qualifiers')!='':nonblank_legacy.append(identity)
    for name,values in [('lost edges',lost_edges),('changed old edge cells',changed_edges),('qualified legacy edges',nonblank_legacy)]:
        if values:problems.append(name+': '+str(len(values)))
    expected_new_edges={r.id:asdict(r) for r in kgx_edges([model])}
    actual_new_edges={k:v for k,v in edges.items() if k not in old['edges']}
    if actual_new_edges!=expected_new_edges:problems.append('New edge rows do not exactly match the accepted target')
    new_node_rows={n.id:asdict(n) for n in kgx_nodes([model])}
    display_delta=[];lost_contexts=[];unexpected_contexts=[];added_contexts=0
    for identity in set(old['nodes'])|set(nodes):
        row=nodes.get(identity);prior=old['nodes'].get(identity)
        if row is None:problems.append('Lost node: '+identity);continue
        current=Counter(digest(c) for c in json.loads(row['source_contexts'] or '[]'))
        previous=Counter(prior['context_hash_counts']) if prior else Counter()
        missing=previous-current
        if missing:lost_contexts.append({'id':identity,'missing':dict(missing)})
        expected=Counter(digest(c) for c in json.loads(new_node_rows.get(identity,{}).get('source_contexts') or '[]'))
        additions=current-previous;added_contexts+=sum(additions.values())
        if additions!=expected:unexpected_contexts.append(identity)
        if prior:
            for key,value in prior['display'].items():
                if row[key]!=value:display_delta.append({'id':identity,'column':key,'before':value,'after':row[key]})
    display_delta.sort(key=lambda r:(r['id'],r['column']))
    approved=json.loads(args.approved_node_display_delta.read_bytes()) if args.approved_node_display_delta else []
    if display_delta!=approved:problems.append('Unreviewed global node display differences')
    if lost_contexts:problems.append('Old node source contexts lost')
    if unexpected_contexts:problems.append('Unexpected added node source contexts')
    for name in ['nodes.tsv','edges.tsv']:
        if sha(root/'pages/downloads'/name)!=sha(root/'output/kgx'/name):problems.append('Final KGX downloads/output mismatch: '+name)
    page=root/'pages/records/MetaCyc_GLCMANNANAUT-PWY.html'
    if not page.is_file():problems.append('New record page missing')
    result={'status':'passed' if not problems else 'needs_review','baseline_sha256':sha(args.baseline),
        'baseline_kgx_sha256':sha(args.baseline_kgx),'target_sha256':sha(target),'retained_files':retained,
        'old_record_count':base['record_count'],'old_page_count':base['page_count'],
        'old_review_bundles':base['structured_review_bundles'],'current_review_bundles':len(reviews),
        'old_edges':old['edge_count'],'old_edge_cells_exact':not lost_edges and not changed_edges,
        'new_blank_legacy_endpoint_column':not nonblank_legacy,'old_source_contexts':old['source_context_count'],
        'lost_contexts':lost_contexts,'unexpected_context_nodes':unexpected_contexts,
        'new_edges':len(actual_new_edges),'new_source_contexts':added_contexts,
        'node_display_delta':display_delta,'approved_node_display_delta_sha256':sha(args.approved_node_display_delta) if args.approved_node_display_delta else None,
        'new_page_sha256':sha(page) if page.is_file() else None,'problems':problems,
        'scope':'No broad waiver: all old edge cells and identities, contexts, record pages and review/history bytes preserved. Only the added blank legacy endpoint_qualifiers column is preauthorized; any global display change requires a separately bound exact review.'}
    args.output.parent.mkdir(parents=True,exist_ok=True)
    args.output.write_text(json.dumps(result,indent=2,ensure_ascii=False)+'\n')
    print(json.dumps({k:v for k,v in result.items() if k not in ['retained_files','lost_contexts','node_display_delta']},indent=2))
    return 0 if not problems else 1


if __name__=='__main__':raise SystemExit(main())
