"""Decode a preserved lossless wrapper to a new external file."""
from pathlib import Path
import argparse,base64,gzip,hashlib,json
p=argparse.ArgumentParser(description=__doc__)
p.add_argument('wrapper',type=Path);p.add_argument('output',type=Path)
a=p.parse_args();o=json.loads(a.wrapper.read_bytes())
if o['encoding']=='utf-8':b=o['text'].encode('utf-8')
elif o['encoding']=='gzip+base64':b=gzip.decompress(base64.b64decode(o['data'],validate=True))
else:raise ValueError('Unsupported wrapper encoding')
assert len(b)==o['source_bytes'] and hashlib.sha256(b).hexdigest()==o['source_sha256']
a.output.parent.mkdir(parents=True,exist_ok=True)
with a.output.open('xb') as stream:stream.write(b)
print(o['source_sha256'],len(b),a.output)
