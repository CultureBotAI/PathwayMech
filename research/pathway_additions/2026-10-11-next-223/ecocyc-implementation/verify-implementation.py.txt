"""Bounded external EcoCyc verification against intact sources and retained authority."""
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys

import yaml

OUT = Path(__file__).parent
ROOT = OUT / 'repo'
CACHE = Path('/private/tmp/pathwaymech-curation-authorities/sources')
PYTHON = '/private/tmp/pathwaymech-next-195-20261011/.venv/bin/python'
RUN = OUT / 'checks-final'
RUN.mkdir()
OVERLAY = OUT / 'originals-overlay'
OVERLAY.mkdir()
assert sys.dont_write_bytecode
assert Path(sys.prefix).resolve() == Path(PYTHON).parent.parent.resolve()

def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def save(path, data):
    with Path(path).open('x') as stream:
        stream.write(json.dumps(data, indent=2, ensure_ascii=False) + '\n')

def capture(path):
    path = Path(path)
    return {'path': str(path), 'resolved_path': str(path.resolve()),
            'bytes': path.stat().st_size, 'sha256': sha(path)}

original_manifest = json.loads((ROOT / 'conf/identifier_sources.json').read_bytes())
before = json.loads((ROOT / 'data/identifier_authorities/sources.json').read_bytes())
assert len(original_manifest) == 629
assert len(before['terms']) == 2265 and len(before['sources']) == 476
entries = json.loads((OUT / 'proposal/new-source-entries.json').read_bytes())
positives = json.loads((OUT / 'proposal/authentic-positive.json').read_bytes())
bindings = json.loads((OUT / 'proposal/input-bindings.json').read_bytes())
for row, entry in zip(positives, entries):
    (OVERLAY / entry['path']).symlink_to(row['original_path'])
originals = []
for entry in original_manifest:
    origin = CACHE / entry['path']
    destination = OVERLAY / entry['path']
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.symlink_to(origin)
    originals.append({'key': entry['key'], 'path': str(origin),
                      'resolved_path': str(origin.resolve()), 'sha256': entry['sha256']})

def verify_original(row):
    if 'resolved_path' in row:
        assert str(Path(row['path']).resolve()) == row['resolved_path'], row['path']
    assert sha(row['path']) == row['sha256'], row['path']
    return {**row, 'bytes': Path(row['path']).stat().st_size}

with ThreadPoolExecutor(max_workers=4) as pool:
    originals = list(pool.map(verify_original, originals))
save(OUT / 'proposal/identifier_sources.json', original_manifest + entries)
save(RUN / 'source-inputs-before.json', originals + bindings)
save(RUN / 'runtime.json', {'executable': sys.executable, 'prefix': sys.prefix,
    'version': sys.version, 'executable_sha256': sha(sys.executable),
    'yaml': capture(yaml.__file__), 'cwd': str(ROOT), 'uv_sync_executed': False,
    'builder': capture(ROOT / 'scripts/build_source_identifier_snapshot.py'),
    'lock': capture(ROOT / 'uv.lock')})
changed = ['scripts/build_source_identifier_snapshot.py', 'tests/test_ecocyc_identifier_authority.py',
    'conf/identifier_policy.yaml', 'docs/IDENTIFIERS.md', 'src/pathwaymech/schema.py',
    'src/pathwaymech/schema/pathwaymech.yaml', 'src/pathwaymech/cli.py']
for rel in changed:
    target = RUN / 'inputs' / rel
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes((ROOT / rel).read_bytes())
save(RUN / 'code-inputs.json', [capture(ROOT / rel) for rel in changed])
spec = importlib.util.spec_from_file_location('external_source_builder', ROOT / 'scripts/build_source_identifier_snapshot.py')
builder = importlib.util.module_from_spec(spec)
spec.loader.exec_module(builder)
for row, entry in zip(positives, entries):
    parsed = list(builder.source_terms(entry['kind'], Path(row['original_path']),
        url=entry['url'], version=entry['version']))
    assert parsed == [(row['id'], row['label'], row['synonyms'])]

# IDs-only selectors keep authority grounded in originals, never curated labels.
retained = OUT / 'retained-authority-selector'
retained.mkdir()
(retained / 'ids.yaml').write_text(yaml.safe_dump({'id': 'GO:0008150',
    'participants': [{'id': i} for i in sorted(before['terms'])]}))
records = OUT / 'selection-corpus'
shutil.copytree(ROOT / 'data/pathways', records)
current_ids = {i for i in builder.corpus_ids(ROOT / 'data/pathways')
               if i.split(':', 1)[0] in builder.SOURCE_NAMESPACES}
assert current_ids <= before['terms'].keys()
unused = sorted(before['terms'].keys() - current_ids)
assert len(unused) == 21
(records / 'authority-selector.yaml').write_text(yaml.safe_dump({'id': 'GO:0008150',
    'participants': [{'id': i} for i in unused + [r['id'] for r in positives]]}))
save(RUN / 'selection-scope.json', {'baseline': 'All 2265 historical authority IDs only',
    'augmented': '223 unchanged records, three native EcoCyc IDs and 21 retained-unused prior IDs',
    'unused_retained_ids': unused, 'curated_labels_as_authority': False})
focused = sorted(set([
    *[str(p.relative_to(ROOT)) for p in (ROOT / 'tests').glob('test_*authorit*.py')],
    'tests/test_metacyc_complex_naming.py', 'tests/test_source_identifier_snapshot.py',
    'tests/test_identifiers.py', 'tests/test_identifier_snapshot_builders.py',
    'tests/test_endpoint_qualifiers.py', 'tests/test_graph_context.py', 'tests/test_site_metadata.py',
]))
commands = [
    ('lint', [PYTHON, '-B', '-m', 'ruff', 'check', *[r for r in changed if r.endswith('.py')]]),
    ('focused', [PYTHON, '-B', '-m', 'pytest', '-q', *focused]),
    ('baseline-cli', [PYTHON, '-B', 'scripts/build_source_identifier_snapshot.py', '--manifest',
        str(ROOT / 'conf/identifier_sources.json'), '--cache-dir', str(OVERLAY), '--records',
        str(retained), '--output', str(RUN / 'baseline-sources.json')]),
    ('augmented-cli', [PYTHON, '-B', 'scripts/build_source_identifier_snapshot.py', '--manifest',
        str(OUT / 'proposal/identifier_sources.json'), '--cache-dir', str(OVERLAY), '--records',
        str(records), '--output', str(RUN / 'augmented-sources.json')]),
]
receipts = []
for name, command in commands:
    start = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(command, cwd=ROOT, capture_output=True, env={**os.environ,
        'PYTHONDONTWRITEBYTECODE': '1', 'UV_NO_SYNC': '1',
        'PYTHONPATH': str(ROOT / 'src') + os.pathsep + str(ROOT)})
    (RUN / (name + '.stdout')).write_bytes(result.stdout)
    (RUN / (name + '.stderr')).write_bytes(result.stderr)
    receipts.append({'name': name, 'command': command, 'started_at': start,
        'finished_at': datetime.now(timezone.utc).isoformat(), 'exit_code': result.returncode})
    (RUN / 'execution.json').write_text(json.dumps(receipts, indent=2) + '\n')
    print(name, result.returncode, result.stdout.decode()[-5000:], result.stderr.decode()[-1500:], flush=True)
    assert result.returncode == 0, name
baseline = json.loads((RUN / 'baseline-sources.json').read_bytes())
after = json.loads((RUN / 'augmented-sources.json').read_bytes())
assert baseline == before
assert len(after['terms']) == 2268 and len(after['sources']) == 479
assert all(after['terms'][k] == v for k, v in before['terms'].items())
assert all(after['sources'][k] == v for k, v in before['sources'].items())
for row, entry in zip(positives, entries):
    assert after['terms'][row['id']] == {'label': row['label'], 'synonyms': row['synonyms'], 'source': entry['key']}
save(RUN / 'preservation.json', {'status': 'passed', 'baseline_exact': True,
    'old_terms': 2265, 'old_source_metadata': 476, 'new_terms': 3, 'new_metadata': 3,
    'all_old_values_exact': True, 'manifest_before': 629, 'manifest_after': 632,
    'new_original_bodies': 3, 'baseline_records': 223, 'labels_not_authority': True,
    'baseline_snapshot_sha256': sha(RUN / 'baseline-sources.json'),
    'augmented_snapshot_sha256': sha(RUN / 'augmented-sources.json')})
with ThreadPoolExecutor(max_workers=4) as pool:
    verified_after = list(pool.map(verify_original, originals + bindings))
save(RUN / 'source-inputs-after.json', verified_after)
baseline_bindings = json.loads((OUT / 'baseline.json').read_bytes())['files']
for row in baseline_bindings:
    assert sha(row['path']) == row['sha256'], row['path']
save(RUN / 'maintained-preservation.json', {'status': 'passed',
    'unchanged_original_copied_files': len(baseline_bindings), 'maintained_writes': False,
    'shared_cache_writes': False, 'git_mutations': False})
print('full_source_and_maintained_preservation_passed', flush=True)
