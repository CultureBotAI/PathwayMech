bindings = {}


def capture(path, expected=None):
    path = Path(path).absolute()
    body = path.read_bytes()
    digest = hashlib.sha256(body).hexdigest()
    assert expected is None or digest == expected, str(path)
    row = {'path': str(path), 'resolved_path': str(path.resolve(strict=True)), 'sha256': digest, 'bytes': len(body)}
    assert str(path) not in bindings or bindings[str(path)] == row
    bindings[str(path)] = row
    return body


def save(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

def mutate(case, change, source_url=url, version='27.0'):
    root = copy.deepcopy(original_root)
    change(root)
    path = fixtures / (case + '.xml')
    with path.open('xb') as stream:
        stream.write(ElementTree.tostring(root, encoding='utf-8', xml_declaration=True))
    try:
        terms(path, url=source_url, version=version)
    except (ValueError, ElementTree.ParseError) as error:
        return {'case': case, 'fixture': str(path), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
                'url': source_url, 'version': version, 'rejected': str(error)}
    raise AssertionError('negative control accepted: ' + case)

    if case == 'assembly-suffix-rejected':
        record['participants'][0]['label'] += ' tetramer'
    elif case == 'context-only-rejected':
        for identifier, row in snapshot['terms'].items():
            if identifier != 'MetaCyc:CPLX0-8588':
                row['label_kind'] = 'source_context'
    elif case == 'missing-complex-rejected':
        snapshot['terms'].pop('MetaCyc:ACNEULY-CPLX')

    save(sandbox / 'data/identifier_authorities/ontology.json', ontology_fixture)

        yaml.safe_dump(record, stream)
