def get(t):

 try:
  with urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'PathwayMech research contact public literature'}),timeout=60) as h: b=h.read();r.update(status=h.status,final_url=h.url,content_type=h.headers.get('Content-Type'))
