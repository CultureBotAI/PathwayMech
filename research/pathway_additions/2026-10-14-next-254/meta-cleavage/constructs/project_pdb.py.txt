from pathlib import Path
import json, hashlib, shlex
root=Path(__file__).parent
prior=Path('/private/tmp/pathwaymech-next-253-primary/originals')
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def loop_rows(text, category):
    lines=text.splitlines(); out=[]
    for i,line in enumerate(lines):
        if line.strip()!='loop_':continue
        j=i+1; cols=[]
        while j<len(lines) and lines[j].strip().startswith('_'):
            cols.append(lines[j].strip());j+=1
        if not cols or not cols[0].startswith(category+'.'):continue
        tokens=[]
        while j<len(lines) and lines[j].strip()!='#':
            tokens.extend(shlex.split(lines[j]));j+=1
        assert len(tokens)%len(cols)==0,(category,len(tokens),len(cols))
        out.extend(dict(zip(cols,tokens[k:k+len(cols)])) for k in range(0,len(tokens),len(cols)))
    return out
def spans(items):
    ranges=[]
    for n in sorted(set(items)):
        if ranges and n==ranges[-1][1]+1:ranges[-1][1]=n
        else:ranges.append([n,n])
    return ranges
