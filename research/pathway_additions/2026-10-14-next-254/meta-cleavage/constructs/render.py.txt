import fitz
from pathlib import Path
root=Path(__file__).parent
source=Path('/private/tmp/pathwaymech-next-253-primary-continuation/originals')
for name in ('bi8b00295_si_001','bi8b00713_si_001','bi9b00549_si_001'):
    doc=fitz.open(source/(name+'.pdf'))
    for i,page in enumerate(doc):
        if i==0: continue
        page.get_pixmap(matrix=fitz.Matrix(1.5,1.5)).save(root/'rendered'/f'{name}-p{i+1}.png')
    print(name,len(doc),flush=True)
