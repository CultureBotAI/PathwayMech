from pathlib import Path
from html.parser import HTMLParser
class Text(HTMLParser):
    def __init__(self):super().__init__();self.out=[];self.suppressed=0
    def handle_starttag(self,tag,attrs):
        if tag in ('script','style'):self.suppressed+=1
        if tag in ('p','h1','h2','h3','h4','tr','li'):self.out.append('\n')
    def handle_endtag(self,tag):
        if tag in ('script','style'):self.suppressed-=1
        if tag in ('p','h1','h2','h3','h4','tr','li'):self.out.append('\n')
    def handle_data(self,text):
        if not self.suppressed:self.out.append(text)
root=Path(__file__).parent
for name in ('PMC3416963','PMC3926093','PMC151511','PMC141877'):
    parser=Text();parser.feed((root/'originals'/(name+'.html')).read_text())
    (root/(name+'.txt')).write_text(''.join(parser.out))
