from pathlib import Path
import datetime
import gzip
import hashlib
import json
import shutil
import subprocess

ROOT = Path('/private/tmp/pathwaymech-next-264-20261015/research/pathway_additions/2026-10-15-next-264')
CI = Path('/private/tmp/pathwaymech-next-264-ci')
OUT = ROOT / 'root-verification/native-xml-storage-successor.json'
assert not OUT.exists()
def sha(data):
    return hashlib.sha256(data).hexdigest()

rows = []
for frame in ['CPLX-7699', 'CPLX-7700']:
    rel = 'source-evidence/native/compact/originals/META-' + frame + '.xml'
    old = ROOT / rel
    external = Path('/private/tmp/pathwaymech-next-264-native/compact/originals/META-' + frame + '.xml')
    raw = old.read_bytes()
    assert raw == external.read_bytes()
    target = old.with_name(old.name + '.gz')
    assert not target.exists()
    packed = gzip.compress(raw, mtime=0)
    target.write_bytes(packed)
    assert gzip.decompress(target.read_bytes()) == raw
    old.unlink()
    rows.append({'historical_retained_path': rel, 'new_retained_path': target.relative_to(ROOT).as_posix(),
                 'original_sha256': sha(raw), 'original_bytes': len(raw),
                 'stored_sha256': sha(packed), 'stored_bytes': len(packed),
                 'encoding': 'gzip_lossless', 'external_source_unchanged': str(external)})

failed = CI / 'initial-retained-whitespace-check.json'
assert json.loads(failed.read_bytes())['exit_code'] == 3
shutil.copyfile(failed, ROOT / 'root-verification/initial-retained-whitespace-check.json')
shutil.copyfile(Path(__file__), ROOT / 'root-verification/helpers/store_native_xml_losslessly.py.txt')
receipt = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
           'reason': 'Two native original XML bodies contain trailing whitespace in a curator biography. Store their exact bytes as gzip to satisfy Git whitespace checks without editing source content.',
           'supersedes_storage_only': 'The two retained paths in initial-source-retention.json and the historical native compact manifest; all frozen external originals/manifests and scientific assertions remain unchanged.',
           'files': rows}
OUT.write_text(json.dumps(receipt, indent=2) + '\n')
empty = CI / 'empty-for-whitespace-check'
proc = subprocess.run(['git', 'diff', '--no-index', '--check', str(empty), str(ROOT)], capture_output=True, text=True)
checked = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'command': proc.args,
           'exit_code': proc.returncode, 'stdout': proc.stdout, 'stderr': proc.stderr,
           'scope': 'Every currently retained research file, including untracked files, after exact gzip storage relocation.'}
(CI / 'retained-whitespace-check.json').write_text(json.dumps(checked, indent=2) + '\n')
assert proc.returncode == 0, checked
shutil.copyfile(CI / 'retained-whitespace-check.json', ROOT / 'root-verification/retained-whitespace-check.json')
print(json.dumps({'storage_successor': str(OUT), 'source_bodies_preserved': len(rows), 'whitespace_exit_code': proc.returncode}, indent=2))
