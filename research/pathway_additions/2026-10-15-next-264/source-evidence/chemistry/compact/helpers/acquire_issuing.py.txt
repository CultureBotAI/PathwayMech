import concurrent.futures,datetime,hashlib,json,pathlib,urllib.request,urllib.error
O=pathlib.Path('/private/tmp/pathwaymech-next-264-chemistry')
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
spec=[(f'ChEBI-{i}.json',f'https://www.ebi.ac.uk/chebi/backend/api/public/compound/{i}/') for i in [60129,16150,18135,15379,15378,57945,57540,16526]]
spec += [(f'META-{i}.xml',f'https://bf2i-artsymbiocyc.insa-lyon.fr/getxml?id=META:{i}') for i in ['NAD','NADH','CARBON-DIOXIDE']]
def fetch(row):
 name,url=row;p=O/'originals'/name;assert not p.exists();started=now()
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'PathwayMech bounded chemistry curation contact source via public endpoint'}),timeout=90) as r:
   b=r.read();record={'status':r.status,'final_url':r.url,'headers':dict(r.headers)}
 except urllib.error.HTTPError as e:b=e.read();record={'status':e.code,'error':str(e),'headers':dict(e.headers)}
 except Exception as e:b=b'';record={'status':None,'error':repr(e)}
 p.write_bytes(b);return {'file':str(p.relative_to(O)),'requested_url':url,'started_at':started,'finished_at':now(),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest(),**record}
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:rows=list(pool.map(fetch,spec))
(O/'acquisition.json').write_text(json.dumps({'scope':'Eight exact ChEBI compounds and three complete native compound originals. No broad source discovery.','requests':rows},indent=2)+'\n')
print(json.dumps([{k:r.get(k) for k in ['file','status','bytes','error']} for r in rows]))
