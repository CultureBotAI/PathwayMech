def bind(path):
    data = Path(path).read_bytes()
    return {'path': str(path), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}


def load(path):
    return json.loads(Path(path).read_text())


def write(path, value):
    with Path(path).open('x') as stream:
        json.dump(value, stream, indent=2, ensure_ascii=False)
        stream.write('\n')

def archive_one(source, relative):
    data = source.read_bytes()
    destination = archive / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    text = data.decode('utf-8')
    if any(line.rstrip() != line for line in text.splitlines()) or text.endswith('\n\n'):
        destination = destination.with_name(destination.name + '.json')
        data = (json.dumps({'serialization': 'lossless-utf8-json', 'original': bind(source), 'text': text}, ensure_ascii=False, indent=2) + '\n').encode()
    with destination.open('xb') as stream:
        stream.write(data)
    mappings.append({'source': bind(source), 'archived': bind(destination)})
