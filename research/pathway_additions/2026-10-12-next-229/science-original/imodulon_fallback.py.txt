def transport(url):
    p=OUT/('imodulon-api-'+hashlib.sha256(url.encode()).hexdigest()+'.json')
    r={'url':url,'started':datetime.now(timezone.utc).isoformat(),'path':str(p)}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'kg-microbe-sources/1','Accept':'application/json'}),timeout=20) as response:
            raw=response.read(); p.write_bytes(raw)
            r.update(status=response.status,final_url=response.url,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())
        return json.loads(raw)
    except Exception as e:
        r['error']=repr(e); raise
    finally:
        receipts.append(r); (OUT/'imodulon-api-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
