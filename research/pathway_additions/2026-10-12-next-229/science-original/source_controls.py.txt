def protein_id(frame):
    r=native['EcoCyc:'+frame]
    if frame=='PHOSPHO-ARCA':r=native['EcoCyc:'+r['modified_form'][0]]
    u=[x['id'] for x in r['direct_unifications'] if x['db']=='UNIPROT' and x['relationship']=='unification']
    return 'UniProtKB:'+u[0] if u else 'EcoCyc:'+frame
regjoins=[]
for e in t['mechanistic_edges']:
    if e['predicate'] not in ['activates','inhibits'] or not e['object'].startswith('Entrez:'):continue
    for ev in e['evidence']:
        if not ev['reference_id'].startswith('EcoCyc:REG'):continue
        n=native[ev['reference_id']]; gene=native[e['object']]['gene'][0]
        frame='EG10035' if gene=='aldA' else 'EG10351'
        if n['target'][0].startswith('PM'):
            assert len(n['TU'])==1; assert frame in native['EcoCyc:'+n['TU'][0]]['components']
        elif n['target'][0].startswith('TU'):assert frame in native['EcoCyc:'+n['target'][0]]['components']
        else:assert n['target']==[frame]
        expected=(protein_id(n['regulator'][0]),'activates' if n['mode']==['+'] else 'inhibits',e['object'])
        assert expected==(e['subject'],e['predicate'],e['object']),(e['id'],expected)
        regjoins.append({'edge':e['id'],'regulation':ev['reference_id'],'native_regulator':n['regulator'],'native_target':n['target'],'native_TU':n['TU'],'gene_frame':frame,'expected':expected})
(O/'regulatory-join-proof.json').write_text(json.dumps(regjoins,indent=2)+'\n')
def checks(record):
    errors=[]; es={e['id']:e for e in record['mechanistic_edges']}
    for reaction in chem:
        rx=reaction['reaction']
        expected={(s['predicate'],s['chemical']):s['coefficient'] for s in reaction['sides']}
        actual={}
        for e in es.values():
            if (e['predicate']=='consumes' and e['object']==rx) or (e['predicate']=='produces' and e['subject']==rx):
                m=re.search(r'coefficient (\d+)',e.get('description',''))
                actual[(e['predicate'],e['subject'] if e['predicate']=='consumes' else e['object'])]=int(m.group(1)) if m else None
        if actual!=expected:errors.append('issuing-rhea-sides:'+rx)
    for j in regjoins:
        e=es.get(j['edge'],{})
        if (e.get('subject'),e.get('predicate'),e.get('object'))!=tuple(j['expected']):errors.append('issuing-regulation-join:'+j['edge'])
    # These source-specific endpoint roles are derived from assay design: enzyme is
    # subject in catalysis/cofactor assays and object in inhibitor assays.
    for eid in ['eg-021','eg-022','eg-023','eg-024','eg-025','eg-026','eg-029','eg-030','eg-036','eg-037','eg-038','eg-039','eg-040','eg-054','eg-055','eg-056','eg-057','eg-058','eg-059','eg-061','eg-067']:
        e=es.get(eid,{})
        expected='subject' if e.get('predicate') in ['catalyzes','has_cofactor'] or eid in ['eg-056','eg-057','eg-058','eg-059','eg-061','eg-067'] else 'object'
        if not any(q['endpoint']==expected for q in e.get('endpoint_qualifiers',[])):errors.append('assay-endpoint-role:'+eid)
    # Exact scoped evidence roles matter even when both source objects are valid.
    for eid in ['eg-056','eg-057','eg-058','eg-059','eg-061','eg-067']:
        if es.get(eid,{}).get('evidence')!=baseline_edges[eid]['evidence']:errors.append('source-claim-selector-binding:'+eid)
    for eid in ['eg-037','eg-038']:
        if 'propionaldehyde' not in json.dumps(es[eid]):errors.append('assay-substrate-lost:'+eid)
    e=es['eg-039']
    if e['predicate']!='activates' or 'esterase' not in e['description'].lower() or not any('esterase' in q['description'].lower() for q in e.get('endpoint_qualifiers',[])):errors.append('esterase-specific-activation')
    for eid in ['eg-079','eg-080','eg-081','eg-082']:
        if es.get(eid,{}).get('predicate')!='has_part':errors.append('unsigned-pathbank-attachment:'+eid)
    if not any(q['endpoint']=='subject' and 'E15' in q['description'] and 'MG1655' in q['description'] for q in es['eg-021'].get('endpoint_qualifiers',[])):errors.append('selected-E15-assay-scope')
    for eid in ['eg-057','eg-058','eg-059']:
        if not any(q['endpoint']=='subject' and q['kind']=='molecular_state' and 'phosph' in q['description'].lower() for q in es[eid].get('endpoint_qualifiers',[])):errors.append('phosphorylated-ArcA-state:'+eid)
    return errors
