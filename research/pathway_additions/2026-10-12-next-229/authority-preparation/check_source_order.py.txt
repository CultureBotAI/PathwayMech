"""Check only selected EG identities against original bodies in accepted manifest order."""
from pathlib import Path
import hashlib
import json
import sys

O = Path(__file__).parent
B = Path('/private/tmp/pathwaymech-next-228-canonical-build-20261012')
sys.path[:0] = [str(O / 'runtime/scripts'), str(O / 'runtime/src')]
import build_source_identifier_snapshot as sb

base = json.loads((B / 'manifest.json').read_bytes())
incremental = json.loads((O / 'incremental-source-manifest.json').read_bytes())
expected = json.loads((O / 'incremental-source-build.json').read_bytes())
selected = set(expected['terms'])
known = {e['key'] for e in base}
entries = base + [e for e in incremental if e['key'] not in known]
bindings = {r['entry']['key']: r['original']['path']
            for r in json.loads((O / 'source-original-bindings.json').read_bytes())}
# These adapters emit selected namespaces. GO-CAM permits source-native individual
# IDs, so inspect its actual original too rather than assume a prefix exclusion.
kinds = {'uniprot', 'metacyc-html', 'metacyc-protein-xml',
         'metacyc-reaction-xml', 'ecocyc-protein-xml', 'ecocyc-rna-xml',
         'ncbi-entrezgene-xml', 'pathbank-biopax', 'pathbank-biopax-zip', 'gocam-tar'}
first = {}
matches = []
scanned = []
excluded = []
for position, entry in enumerate(entries):
    if entry['kind'] not in kinds:
        excluded.append({'position': position, 'key': entry['key'], 'kind': entry['kind'],
                         'reason': 'Native adapter emits only other identifier namespaces'})
        continue
    path = B / 'originals' / entry['path'] if entry['key'] in known else Path(bindings[entry['key']])
    with path.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    assert digest == entry['sha256'], entry['key']
    hits = [t for t in sb.source_terms(entry['kind'], path, url=entry['url'],
            version=entry['version'], archive=entry.get('archive'), sha256=entry['sha256'])
            if t[0] in selected]
    scanned.append({'position': position, 'key': entry['key'], 'kind': entry['kind'],
                    'path': str(path), 'sha256': digest, 'selected_hits': len(hits)})
    for identifier, label, synonyms in hits:
        term = {'label': label, 'synonyms': sorted(set(synonyms) - {label}), 'source': entry['key']}
        matches.append({'id': identifier, 'position': position, 'native_term': term})
        if identifier not in first:
            first[identifier] = term
        elif first[identifier]['source'] == entry['key']:
            first[identifier]['synonyms'] = sorted(
                set(first[identifier]['synonyms'] + [label] + synonyms) - {first[identifier]['label']})
assert first == expected['terms'], {'unexpected': {i: t for i, t in first.items() if t != expected['terms'].get(i)}}
result = {'status': 'passed', 'scope': 'Selected identities and first-source order only; not a full canonical build',
          'base_manifest_entries': len(base), 'proposed_entries': len(entries),
          'selected_identities': len(selected), 'inspected_originals': scanned,
          'excluded_by_native_namespace_contract': excluded, 'matches': matches,
          'first_native_terms': first}
(O / 'source-order-check.json').write_text(json.dumps(result, indent=2, ensure_ascii=False) + '\n')
print(json.dumps({'status': 'passed', 'selected': len(first), 'inspected': len(scanned),
                  'excluded': len(excluded), 'matches': len(matches)}))
