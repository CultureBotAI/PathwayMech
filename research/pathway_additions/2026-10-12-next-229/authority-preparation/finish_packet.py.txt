"""Record evidence bindings and the actual bounded validation history before sealing."""
from pathlib import Path
import datetime
import hashlib
import json

O = Path(__file__).parent


def bind(path):
    path = Path(path)
    with path.open('rb') as stream:
        digest = hashlib.file_digest(stream, 'sha256').hexdigest()
    return {'path': str(path), 'resolved_path': str(path.resolve()),
            'bytes': path.stat().st_size, 'sha256': digest}


def save(name, data):
    (O / name).write_text(json.dumps(data, indent=2, ensure_ascii=False) + '\n')


def main():
    original_bindings = {}
    containers = ['source-original-bindings.json', 'gene-identity-bridges.json',
                  'protein-identity-bridges.json', 'assembly-native-bridges.json',
                  'ArcA-state-adapter-disposition.json', 'effector-chemical-bridges.json',
                  'native-superpathway-bridge.json', 'uniprot-source-projections.json',
                  'pathbank-original-probe.json', 'ontology-native-replay.json']
    def visit(value, owner):
        if isinstance(value, dict):
            if {'path', 'bytes', 'sha256'} <= value.keys():
                path = value['path']
                binding = {k: value[k] for k in ('path', 'bytes', 'sha256')}
                if path in original_bindings:
                    assert original_bindings[path]['binding'] == binding
                else:
                    original_bindings[path] = {'binding': binding, 'used_by': []}
                original_bindings[path]['used_by'].append(owner)
            for child in value.values():
                visit(child, owner)
        elif isinstance(value, list):
            for child in value:
                visit(child, owner)
    for name in containers:
        visit(json.loads((O / name).read_bytes()), name)
    for path in sorted((O / 'raw').iterdir()):
        observed = bind(path)
        key = str(path)
        if key not in original_bindings:
            original_bindings[key] = {'binding': {k: observed[k] for k in ('path', 'bytes', 'sha256')}, 'used_by': ['raw acquisition; complete issuing body retained']}
    for path, row in original_bindings.items():
        row['used_by'] = sorted(set(row['used_by']))
        if Path(path).suffix == '.db':
            row['verification'] = 'Original SQL SHA/size captured during completed read-only native projection; exact source metadata agrees with accepted MG base. Large database not rehashed again during final sealing.'
            continue
        observed = bind(path)
        assert all(observed[k] == row['binding'][k] for k in ('path', 'bytes', 'sha256'))
        row['verification'] = 'Full bytes rehashed at packet finalization'
