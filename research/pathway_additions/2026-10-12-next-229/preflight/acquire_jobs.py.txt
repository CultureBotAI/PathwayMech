from pathlib import Path
import urllib.request,urllib.error,json,hashlib,datetime,sys
from concurrent.futures import ThreadPoolExecutor
O=Path(__file__).parent
J=Path(sys.argv[1]); jobs=json.loads(J.read_text())
def run(job):
 name,url=job;p=O/'raw'/name;assert not p.exists();r={'path':str(p),'url':url,'at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 try:
  req=urllib.request.Request(url,headers={'User-Agent':'PathwayMech evidence research/1.0'})
  with urllib.request.urlopen(req,timeout=50) as resp:raw=resp.read();r.update(status=resp.status,final_url=resp.url,content_type=resp.headers.get('Content-Type'))
 except urllib.error.HTTPError as e:raw=e.read();r.update(status=e.code,error=str(e))

 p.write_bytes(raw);r.update(bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest(),pdf_signature=raw.startswith(b'%PDF'),zip_signature=raw.startswith(b'PK'));return r
with ThreadPoolExecutor(max_workers=3) as ex:rows=list(ex.map(run,jobs))
(O/'derived'/(J.stem+'-receipts.json')).write_text(json.dumps(rows,indent=2)+'\n');print(json.dumps([{k:v for k,v in x.items() if k in ['path','status','bytes','error','pdf_signature','zip_signature']} for x in rows],indent=2))
