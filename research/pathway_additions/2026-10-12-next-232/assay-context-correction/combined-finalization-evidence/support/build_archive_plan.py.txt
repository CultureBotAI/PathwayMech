def add(source, relative, compress=False):
    source = Path(source)
    body = r.read(source)
    if compress:
        payload, encoding = gzip.compress(body, mtime=0), 'gzip-exact-bytes'
        relative += '.gz'
    else:
        payload, encoding = body, 'exact-bytes'
        if source.suffix not in ('.pdf', '.png'):
            r.need(all(line.rstrip(' \t') == line for line in body.decode().splitlines()),
                   'archive has trailing whitespace: ' + str(source))
    destination, payload_path = A + '/' + relative, P / 'payloads' / relative
    r.need(not any(row['destination'] == destination for row in rows), 'duplicate destination')
    r.new(payload_path, payload)
    rows.append({'source': r.binding(source), 'destination': destination,
                 'payload': r.binding(payload_path), 'encoding': encoding})
