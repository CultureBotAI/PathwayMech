def now():
    return datetime.now(timezone.utc).isoformat().replace('+00:00', 'Z')


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def binding(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': sha(path)}


def load(path):
    return json.loads(Path(path).read_bytes())


def verify(path, item, allow_symlink=False):
    path = Path(path)
    assert path.is_file() and (allow_symlink or not path.is_symlink()), path
    assert sha(path) == item['sha256'], path
    if 'bytes' in item:
        assert path.stat().st_size == item['bytes'], path


def write_new(path, value):

        stream.write('\n')


def git(root, *args):

def safe_relative(value):

IDENTITY_FIELDS = ('file_type', 'symlink', 'link_text', 'resolved_path')


def path_identity(path):
    """Fresh lexical link identity plus ultimate destination; no acquisition-time claim."""
    path = Path(path)
    mode = path.lstat().st_mode
    assert stat.S_ISREG(mode) or stat.S_ISLNK(mode), 'Only regular files and file symlinks are supported'
    linked = stat.S_ISLNK(mode)
    return {'file_type': 'symlink' if linked else 'regular', 'symlink': linked,
            'link_text': os.readlink(path) if linked else None,
            'resolved_path': str(path.resolve(strict=True))}


def verify_path_identity(path, item):
    assert all(key in item for key in IDENTITY_FIELDS), 'Fresh path identity binding required'
    assert path_identity(path) == {key: item[key] for key in IDENTITY_FIELDS}, 'Path link identity changed'
    verify(path, item, allow_symlink=True)


def inventory(root, relative='.', exclude_runtime=False):
    """Use rg's hidden/ignored/followed inventory and fail on traversal errors."""
    command = ['rg', '--files', '-uu', '-L']
    if exclude_runtime:
        command += ['-g', '!.git', '-g', '!.git/**', '-g', '!.venv', '-g', '!.venv/**']
    command.append(relative)
    result = subprocess.run(command, cwd=root, text=True, capture_output=True)
    assert result.returncode in (0, 1) and not result.stderr, result.stderr
    for base, dirs, files in os.walk(root / relative, followlinks=True):
        if exclude_runtime:
            dirs[:] = [d for d in dirs if d not in {'.git', '.venv'}]
            files = [f for f in files if f != '.git']
        for name in dirs + files:
            entry = Path(base) / name
            assert entry.exists() and (entry.is_file() or entry.is_dir()), ('broken or special inventory entry', str(entry))
    paths = sorted(str(Path(p)) for p in result.stdout.splitlines())
    assert len(paths) == len(set(paths))
    return [{'relative_path': p, **binding(root / p),
             **path_identity(root / p)} for p in paths]

def verify_packet(item):
    verify(item['handoff']['path'], item['handoff'])
    verify(item['manifest']['path'], item['manifest'])
    directory = Path(item['handoff']['path']).parent
    manifest = load(item['manifest']['path'])
    rows = manifest if isinstance(manifest, list) else manifest.get('artifacts', manifest.get('files'))
    assert len(rows) == item['artifact_count'] and not item.get('allowed_external_bindings')
    declared = set()
    for row in rows:
        relative = row.get('relative_path', row.get('path'))
        path = Path(relative)
        if not path.is_absolute(): path = directory / safe_relative(relative)
        assert path.is_relative_to(directory), 'Outside packet artifact'
        rel = str(path.relative_to(directory)); assert rel not in declared
        verify_packet_file(path, row, item.get('link_identities', {}).get(rel))
        declared.add(rel)
    links = manifest.get('original_links', []) if isinstance(manifest, dict) else []
    assert len(links) == item.get('original_symlink_count', 0)
    for row in links:
        path = Path(row['path']); assert path.is_relative_to(directory)
        relative = str(path.relative_to(directory)); assert relative not in declared
        verify_packet_file(path, row); declared.add(relative)
    supplementary = set()
    if item.get('supplemental_inventory'): verify(item['supplemental_inventory']['path'], item['supplemental_inventory'])
    for row in item.get('supplemental_observed_files', []):
        relative = str(safe_relative(row['relative_path']))
        assert relative not in declared and relative not in supplementary
        verify_packet_file(directory / relative, row, item.get('link_identities', {}).get(relative))
        supplementary.add(relative)
    assert {r['relative_path'] for r in inventory(directory)} == declared | {'handoff.json', 'artifact-manifest.json'} | supplementary, 'Packet inventory changed'

    identity = {**identity, 'symlink': identity['file_type'] == 'symlink'}
    verify_path_identity(path, {**row, **identity})

def ordered_equal(left, right):
    if type(left) is not type(right): return False
    if isinstance(left, dict): return list(left) == list(right) and all(ordered_equal(v, right[k]) for k, v in left.items())
    if isinstance(left, list): return len(left) == len(right) and all(ordered_equal(x, y) for x, y in zip(left, right, strict=True))
    return left == right

def verify_cache_union(cache, previous, new, adopted=False):
    """Check every old body, including unselected originals, and exact new paths."""
    assert not (set(previous) & set(new)), 'New cache path collides with old cache'
    current = {r['relative_path']: r for r in inventory(cache)}
    assert set(current) == set(previous) | (set(new) if adopted else set()), 'Cache inventory differs from exact original union'
    for relative, item in previous.items(): verify_path_identity(cache / safe_relative(relative), item)
    for relative, item in new.items():
        verify(item['original']['path'], item['original'], allow_symlink=True)
        destination = cache / safe_relative(relative)
        assert destination.parent.resolve().is_relative_to(cache.resolve())
        if adopted:
            verify(destination, item['original'])
            identity = path_identity(destination)
            assert identity == {'file_type': 'regular', 'symlink': False, 'link_text': None,
                'resolved_path': str(cache.resolve(strict=True) / safe_relative(relative))}, 'New copy must remain a regular file at its exact destination'
        else: assert not destination.exists() and not destination.is_symlink(), 'NAD original destination already occupied'

            if document.get('target', {}).get('path') == IMPLEMENTATION_HISTORY_REL:
                implementation_history.append(item)
    assert implementation_history, 'Scaffold native complex-adapter implementation history before rendering and gates'
    assert target_history, 'Scaffold the native NAD curation history before rendering and gates'
    return current


def verify_render_history(root, expected):
    assert inventory(root, 'history') == expected, 'Late history invalidates rendered pages and gate scope'


def verify_gate_input_scope(root, gates, adoption_receipt, allowed_additions=None):
    """Full hidden/ignored input closure; additions require exact new-file bindings."""
    verify(gates['inputs']['path'], gates['inputs'])
    inputs = load(gates['inputs']['path'])
    verify(adoption_receipt, inputs['adoption_receipt'])
    verify(root / TARGET_REL, inputs['target'])
    assert inputs['render_history_inputs'] == gates['render_history_inputs']
    verify_render_history(root, inputs['render_history_inputs'])
    generated = set(load(PACKAGE / 'expected-changes.json')['allowed_generated_paths'])
    rows = inputs['all_maintained_inputs']; previous = {r['relative_path']: r for r in rows}
    assert len(previous) == len(rows)
    additions = {}
    for row in allowed_additions or []:
        relative = str(safe_relative(row['relative_path']))
        assert relative.startswith(('reviews/structured/', 'history/', DOSSIER_REL + '/'))
        assert relative not in previous and relative not in additions, 'Approval cannot replace gate input'
        verify(root / relative, row); additions[relative] = row
    current = {r['relative_path']: r for r in inventory(root, exclude_runtime=True)}
    assert set(current) - generated == (set(previous) | set(additions)) - generated, 'Unbound nongenerated input inventory change'
    for relative, row in previous.items():
        if relative not in generated: verify_path_identity(root / relative, row)
    return {'input_receipt': binding(gates['inputs']['path']),
        'nongenerated_inputs_checked': sum(r not in generated for r in previous),
        'exact_new_additions_checked': len(additions), 'whole_render_history_exact': True,
        'complete_current_inventory_verified': True}
