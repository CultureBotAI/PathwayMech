 sources=[];local={}

 for key,path in [('gates',args.gates),('gate-inputs',gates['inputs']['path']),('preservation',args.preservation),('root-support-intake',args.root_intake),('adoption',ADOPTION),('merged-binding',MERGED),('history-execution',HISTORY),('page-qa',QA),('visual-observation',VISUAL),('display-approval',DISPLAY),('history-file-approval',preserved['approved_extra_files']['path'])]:add(key,path)

  for relative in files(packet,exclude_runtime=False):sources.append((packet/relative,key+'/'+relative))

  for key in ['stdout','stderr']:sources.append((Path(row[key]['path']),'native-history/'+str(index)+'.'+key))

  assert packet.is_dir() and packet.resolve().parent==Path('/private/tmp') and packet.name.startswith('pathwaymech-next-232-') and not packet.resolve().is_relative_to(ROOT)
  for relative in files(packet,exclude_runtime=False):sources.append((packet/relative,'failed-attempts/'+str(i)+'/'+relative))
 for row in load(HERE/'review-evidence-bindings.json'):
  if row['requires_archive']:
   destination,body,encoding=encoded_archive(row['original']['path'],ROOT/row['maintained_reference'])
   assert encoding=='exact_bytes' and str(destination.relative_to(ROOT))==row['maintained_reference']
   sources.append((Path(row['original']['path']),'evidence/'+Path(row['maintained_reference']).name))
  else:verify(row['original'],ROOT/row['maintained_reference'])

  verify(row);sources.append((Path(row['path']),'root-integration/'+row['relative_path']))

 assert len(destinations)==len(set(destinations))
 if args.check_inputs:
  print(json.dumps({'status':'inputs_verified_no_writes','files_planned':len(sources),'target':bind(ROOT/TARGET_REL),'gates':bind(args.gates)}));return
 out.mkdir();archive.mkdir()
 mapping=archive_files(sources,archive)
 by_source={r['source']['path']:r for r in mapping}
 # Current execution README supersedes no historical prose; both remain explicit.
 (archive/'README.md').write_text('''# Actual NAD finalization evidence\n\nThese receipts were archived after the nine maintained gates and exact preservation passed.  extras=inventory(ROOT,DOSSIER_REL+'/final-gates');verify_gates(args.gates,extras)
 known={r['relative_path'] for r in load(HERE/'preparation-input-bindings.json')['maintained_contract']}

 assert inspection['source']['git_revision']==BASE
 document=copy.deepcopy(load(HERE/'review-content-fragments.json')['review_fields']);finished=now();started=inspection_receipt['started_at']
 document.update({'review_id':datetime.fromisoformat(finished).strftime('%Y%m%dT%H%M%SZ')+'-nad-phosphorylation-and-dephosphorylation','started_at':started,'finished_at':finished,'skill':'.claude/skills/review-yaml-record/SKILL.md','completion':'completed','verdict':'pass_with_limitations','source':inspection['source'],'targets':inspection['targets'],'reviewer':{'identity':'Codex /root; support preparation by /root/atp_build_preparation_audit','kind':'agent','independence':'self_review','independence_basis':'Root integrated this record. Scientific judgments explicitly inherit separately independent complete original and bounded successor reviews; current execution/preservation/representation are assessed separately. No fresh full literature rereview.'},'summary':'Exact 102-edge, 48-qualifier, 97-reference NAD successor retains the accepted external corrections for issues 465–469. Actual full native authorities, two pre-render histories, nine maintained gates, complete bibliography-owner/static graph/KGX QA and exact prior-corpus/cache preservation are bound. Root must read this assembled content before native validation/save; no future publication success is claimed.'})
 for eid,relative in local.items():
  original=next(source for source,rel in sources if rel==relative);archived=by_source[str(original)]['archived'];reference=str(Path(archived['path']).relative_to(ROOT))
  document['evidence'].append({'evidence_id':eid,'kind':'validation','reference':reference,'accessed_at':finished,'snapshot_sha256':archived['sha256'],'support':'supports','summary':'Actual current execution receipt; exact archived bytes and original scope retained.'})
 for row in load(HERE/'review-evidence-bindings.json'):verify(row['original'],ROOT/row['maintained_reference'])
 document['checks']=[{'check_id':'maintained-'+r['name'],'name':'Actual maintained '+r['name'],'status':'passed','required':True,'target_ids':[TARGET_ID],'command':shlex.join(r['command']),'exit_code':r['exit_code'],'summary':'Actual full maintained command passed for exact target96e493; later archives/review still need final checks.','evidence_ids':['gates']} for r in gates['receipts']]
 for aid,area,summary,eids in [('preservation','consistency','All 232 old records/pages and 108 prior review bundles/history remain exact; all 11587 old edges and 7718 old contexts preserved. Exactly 102 new edges, 68 contexts and 28 nodes; no shared display-cell change. Old 789 cache identities unchanged and 11 new regular original copies exact.',['preservation','display-approval']),('representation','representation',visual['visual_observation']+' Complete 102-edge, 48-qualifier, 266-evidence, 97-owner-reference and four-direction page/KGX checks passed.',['page-qa','visual-observation'])]:
  document['checks'].append({'check_id':aid,'name':aid,'status':'passed','required':True,'target_ids':[TARGET_ID],'summary':summary,'evidence_ids':eids})
  document['assessments'].append({'assessment_id':'root-'+aid,'area':area,'topic':aid,'outcome':'supported','target_ids':[TARGET_ID],'summary':summary,'evidence_ids':eids})
 assert_history(ROOT,gates['render_history_inputs']);write(out/'completed-review.json',document)
 write(out/'archive-extra-files.json',extras)
 write(out/'assembly.json',{'status':'assembled_pending_root_read_native_serialization_save','at':now(),'target':bind(ROOT/TARGET_REL),'review':bind(out/'completed-review.json'),'inspection':inspection_receipt,'archive_map':bind(archive/'archive-map.json'),'archive_extras':bind(out/'archive-extra-files.json'),'gate_receipt':bind(args.gates),'preservation':bind(args.preservation),'support_intake':bind(args.root_intake),'maintained_review_saved':False,'late_history_written':False})
 print(json.dumps(load(out/'assembly.json'),indent=2))
