def preserve(old,new):
    assert old['version']==new['version']==1
    for section in ['sources','terms']:
        assert list(new[section])[:len(old[section])]==list(old[section]),section
        assert all(new[section][key]==value for key,value in old[section].items()),section


def original_paths(manifest,rhea):
    result={}
    for entry in manifest:
        for role,value in [('top_level',entry),('version_source',entry.get('version_source')),('class_source',entry.get('class_source'))]:
            if value is None:continue
            path=str(relative(value['path']));item=result.setdefault(path,{'sha256':value['sha256'],'roles':[]})
            assert item['sha256']==value['sha256']
            item['roles'].append({'key':entry['key'],'role':role})
    path=str(relative(rhea['path']));item=result.setdefault(path,{'sha256':rhea['sha256'],'roles':[]})
    assert item['sha256']==rhea['sha256'];item['roles'].append({'key':rhea['key'],'role':'rhea_update'})
    return result


def metadata_without_hash(connection):
    prefixes=dict(connection.execute('SELECT prefix, base FROM prefix'))
    declarations=list(connection.execute("SELECT subject, object FROM statements WHERE predicate='owl:versionIRI'"))
    assert len(declarations)==1 and declarations[0][1]
    subject,release=declarations[0]
    metadata={pred:obj or value for pred,obj,value in connection.execute("SELECT predicate,object,value FROM statements WHERE subject=? AND predicate IN ('owl:versionInfo','dcterms:license')",(subject,))}
    assert metadata['owl:versionInfo'] and metadata['dcterms:license']
    return {'url':ob.expand_iri(release,prefixes),'version':metadata['owl:versionInfo'],'license':ob.expand_iri(metadata['dcterms:license'],prefixes),'sha256_scope':'uncompressed_semantic_sql_database'}
