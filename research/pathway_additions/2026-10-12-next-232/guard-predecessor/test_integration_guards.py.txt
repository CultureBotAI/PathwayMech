@pytest.mark.parametrize('change', ['same-byte-retarget', 'same-target-link-text', 'symlink-to-regular', 'regular-to-symlink', 'indirect-retarget'])
def test_cache_identity_cannot_drift_with_identical_bodies(tmp_path, change):
    cache = tmp_path / 'cache'; cache.mkdir()
    first = tmp_path / 'first'; first.write_bytes(b'identical original body\n')
    second = tmp_path / 'second'; second.write_bytes(first.read_bytes())
    link = cache / 'old'
    intermediary = tmp_path / 'intermediary'
    if change == 'regular-to-symlink': link.write_bytes(first.read_bytes())
    elif change == 'indirect-retarget':
        intermediary.symlink_to(first); link.symlink_to(intermediary)
    else: link.symlink_to(first)
    rows = guard.inventory(cache); previous = {'old': rows[0]}
    guard.verify_cache_union(cache, previous, {})
    if change == 'indirect-retarget': intermediary.unlink(); intermediary.symlink_to(second)
    else:
        link.unlink()
        if change == 'symlink-to-regular': link.write_bytes(first.read_bytes())
        elif change == 'same-target-link-text': link.symlink_to('../first')
        else: link.symlink_to(second)
    assert link.read_bytes() == first.read_bytes() == second.read_bytes()
    assert guard.inventory(cache) != rows, 'Full baseline equality must also reject identity change'
    with pytest.raises(AssertionError, match='identity changed'):
        guard.verify_cache_union(cache, previous, {})
