def save(name, value):
    write_new(O / name, value)


def manifest_rows(item):
    directory = Path(item['handoff']['path']).parent
    data = read(item['manifest']['path'])
    rows = data if isinstance(data, list) else data.get('artifacts', data.get('files'))
    result = []
    for row in rows:
        candidate = Path(row.get('relative_path') or row['path'])
        path = candidate if candidate.is_absolute() else directory / candidate
        assert path.is_relative_to(directory) and '..' not in path.parts
        result.append({'relative_path': str(path.relative_to(directory)), **bind(path)})
    return result


def packet(label, suffix):
    directory = TMP / ('pathwaymech-next-231-' + suffix + '-20261012')
    handoff = bind(directory / 'handoff.json')
    if label in PINNED: assert handoff['sha256'] == PINNED[label]
    data = read(handoff['path']); manifest = data.get('manifest') or data['artifact_manifest']
    path = Path(manifest['path']); path = path if path.is_absolute() else directory / path
    verify(path, manifest); manifest = bind(path)
    item = {'handoff': handoff, 'manifest': manifest, 'artifact_count': 0, 'original_links_receipt': None}
    raw = read(path); raw_rows = raw if isinstance(raw, list) else raw.get('artifacts', raw.get('files'))
    rows = manifest_rows(item); assert len(rows) == len(raw_rows)
    for row, original in zip(rows, raw_rows, strict=True): verify(row['path'], original, allow_symlink=True)
    item['artifact_count'] = len(rows)
    links = []
    if label == 'canonical-build':
        receipt = directory / 'execution-intake.json'
        item['original_links_receipt'] = bind(receipt)
        for original in read(receipt)['source_originals_freshly_hashed']:
            path = directory / 'originals' / original['cache_relative_path']
            assert path.is_symlink() and path.resolve() == Path(original['path']).resolve()
            verify(path, original, allow_symlink=True); links.append(str(path.relative_to(directory)))
    observed_rows = inventory(directory)
    observed = {r['relative_path'] for r in observed_rows}
    declared = {r['relative_path'] for r in rows} | {'handoff.json', 'artifact-manifest.json'} | set(links)
    assert declared <= observed
    extra = observed - declared
    # Known author/science build caches are separately frozen observations, not science-seal artifacts.
    assert not extra or ((label == 'author' and len(extra) == 107 and all(r.startswith(('swift-cache/', 'uv-cache/')) for r in extra)) or
        (label == 'science-original' and len(extra) == 101 and all(r.startswith('swift-cache/') for r in extra))), (label, sorted(extra))
    item['supplemental_observed_files'] = [r for r in observed_rows if r['relative_path'] in extra]
    item['supplemental_scope'] = 'Exact newly observed build-cache bytes outside original author/science manifest; no scientific acceptance is attributed to these files.' if extra else None
    item['allowed_external_bindings'] = []
    item.update({'closed_inventory_verified': True, 'original_symlink_count': len(links),
                 'scope': 'Full closed manifest freshly verified; original artifact rows are retained once in their immutable manifest, not duplicated into this compact packet index.'})
    return item
