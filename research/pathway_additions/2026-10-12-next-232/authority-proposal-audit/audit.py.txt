def read(p):return json.loads(Path(p).read_bytes())
def bind(p):
    p=Path(p)
    with p.open('rb') as f:d=hashlib.file_digest(f,'sha256').hexdigest()
    return {'path':str(p),'bytes':p.stat().st_size,'sha256':d}
def check(p,r):
    actual=bind(p);assert actual['sha256']==r['sha256'] and actual['bytes']==r['bytes'],str(p)
def identity(p):
    p=Path(p);return {'file_type':'symlink' if p.is_symlink() else 'regular','link_text':os.readlink(p) if p.is_symlink() else None,'resolved_path':str(p.resolve(strict=True))}
def paths(root):
    result=subprocess.run(['rg','--files','-uu','-L',str(root)],text=True,capture_output=True)
    assert result.returncode in (0,1) and not result.stderr
    return set(map(Path,result.stdout.splitlines()))
def write(name,data):
    with (OUT/name).open('x') as f:json.dump(data,f,indent=2);f.write('\n')
def ordered(v):return json.dumps(v,ensure_ascii=False,separators=(',',':'))

def main():
    for p in PACK.glob('*.py'):
        dest=OUT/'helpers-read'/(p.name+'.txt');dest.parent.mkdir(exist_ok=True);dest.write_bytes(p.read_bytes())
    checks=[]
    for name,fn in [('closed_proposal',closure),('named_scope_and_runtime',scope_and_runtime),('originals_companion_and_cache_bridge',originals_and_cache),('old_objects_and_order',objects_and_order),('bounded_native_source_replay',native_selected_replay),('bounded_SQL_selection_replay',ontology_selected_replay),('actual_receipts_and_lineage',receipts_and_lineage)]:
        at=datetime.datetime.now(datetime.UTC).isoformat()
        try:row={'name':name,'status':'passed','detail':fn()}
        except Exception as e:row={'name':name,'status':'failed','type':type(e).__name__,'message':str(e)}
        row['started_at']=at;row['finished_at']=datetime.datetime.now(datetime.UTC).isoformat();checks.append(row);write(name+'.json',row);print(json.dumps(row),flush=True)
    write('checks.json',checks)
    return 0 if all(r['status']=='passed' for r in checks) else 1
