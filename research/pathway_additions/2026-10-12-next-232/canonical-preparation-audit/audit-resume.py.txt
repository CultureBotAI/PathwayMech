def bind(path):
    path = Path(path)
    with path.open('rb') as f:
        digest = hashlib.file_digest(f, 'sha256').hexdigest()
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': digest}


def verify(row):
    got = bind(row['path'])
    assert got == {k: row[k] for k in got}, row['path']
    return got


def ident(path):
    path = Path(path)
    mode = path.lstat().st_mode
    assert stat.S_ISREG(mode) or stat.S_ISLNK(mode)
    return {'file_type': 'symlink' if stat.S_ISLNK(mode) else 'regular',
            'link_text': os.readlink(path) if stat.S_ISLNK(mode) else None,
            'resolved_path': str(path.resolve(strict=True))}
