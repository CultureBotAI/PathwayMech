def load(p):
    return json.loads(Path(p).read_text())

def bind(p):
    p = Path(p)
    return dict(path=str(p), bytes=p.stat().st_size, sha256=hashlib.sha256(p.read_bytes()).hexdigest())

def save(n, value):
    p = A / n
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def verify(root, expected):
    assert bind(root / 'handoff.json')['sha256'] == expected
    manifest = load(root / 'artifact-manifest.json')
    rows = manifest['artifacts']
    for row in rows:
        relative = row.get('relative_path', row['path'])
        p = root / relative
        assert bind(p)['sha256'] == row['sha256'] and p.stat().st_size == row['bytes'], p
    return dict(handoff=bind(root / 'handoff.json'), manifest=bind(root / 'artifact-manifest.json'), verified_artifacts=len(rows))
