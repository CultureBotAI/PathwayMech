"""Complete the unsealed #466 source locator for K12 strain provenance."""
from pathlib import Path
import hashlib
import json
import shutil
import yaml

A=Path(__file__).parent;L=A/'working-lineage/first-candidate'
assert not L.exists();L.mkdir(parents=True)
preserve=['draft/nad-phosphorylation-and-dephosphorylation.yaml','candidate-checks',
          'ledgers/working-evidence-selector-ledger.json','authority-successor-bridge.json',
          'semantic-deltas.json','bounded-scientific-checks.json','provisional-identifier-consumer.json',
          'original-replay.json','edge-source-crosswalk.json','bidirectional-source-coverage.json',
          'bibliography-checks.json','helper-lineage.json','check_successor.py','prepare_checks.py',
          'check_identifiers_provisional.py','check_bibliography.py']
def bind(p):
    p=Path(p);return dict(path=str(p),bytes=p.stat().st_size,sha256=hashlib.sha256(p.read_bytes()).hexdigest())
for n in preserve:
    src=A/n;dst=L/n;dst.parent.mkdir(parents=True,exist_ok=True)
    if src.is_dir():shutil.copytree(src,dst)
    else:shutil.copyfile(src,dst)
target=A/'draft/nad-phosphorylation-and-dephosphorylation.yaml';before=bind(target)
d=yaml.safe_load(target.read_text());ev=d['mechanistic_edges'][40]['evidence'][1]
old=json.loads(json.dumps(ev));assert ev['reference_id']=='PMID:3025169'
ev['source_locator']='PDF pages 1–2 (printed pp. 184–185): p. 1 Materials and methods, Growth conditions (K12); p. 2 Results, Properties of NAD kinase (93-fold, nonhomogeneous), Table 2 NADP rows and adjacent Regulation of NAD kinase'
ledger_path=A/'ledgers/working-evidence-selector-ledger.json';ledger=json.loads(ledger_path.read_text());row=ledger[-1]
assert row['evidence']==old and row['selector']['pages']==[2]
row['evidence']=ev;row['selector']['pages']=[1,2]
j=json.loads(Path(row['selector']['projection']).read_text());pages=j['pages']if isinstance(j,dict)else j
row['observed']=[r for r in pages if r['page_1based']in [1,2]]
assert len(row['observed'])==2 and 'E. coli K-12 was grown' in row['observed'][0]['text']
target.write_text(yaml.safe_dump(d,sort_keys=False,allow_unicode=True,width=100))
ledger_path.write_text(json.dumps(ledger,indent=2,ensure_ascii=False)+'\n')
(A/'working-locator-refinement.json').write_text(json.dumps(dict(before=before,after=bind(target),
     source_assertion_unchanged=ev['source_assertion'],old_locator=old['source_locator'],new_locator=ev['source_locator'],
     selector_pages=[1,2],reason='Pre-seal independent reviewer requested the exact page 1 K12 growth-condition locator in addition to page 2 preparation/table evidence.',
     old_candidate_artifacts_preserved=str(L)),indent=2,ensure_ascii=False)+'\n')
print(json.dumps(bind(target)))
