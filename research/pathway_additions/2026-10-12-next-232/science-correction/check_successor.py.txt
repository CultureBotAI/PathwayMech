def diff(x,y,p=''):
    if type(x) is not type(y): return [dict(path=p,before=x,after=y)]
    if isinstance(x,dict):
        out=[]
        for k in x.keys()|y.keys():
            if k not in x or k not in y: out.append(dict(path=p+'/'+k,before=x.get(k),after=y.get(k)))
            else: out+=diff(x[k],y[k],p+'/'+k)
        return out
    if isinstance(x,list):
        out=[]
        for i in range(max(len(x),len(y))):
            if i>=len(x) or i>=len(y):out.append(dict(path=p+'/'+str(i),before=x[i] if i<len(x) else None,after=y[i] if i<len(y) else None))
            else:out+=diff(x[i],y[i],p+'/'+str(i))
        return out
    return [] if x==y else [dict(path=p,before=x,after=y)]
