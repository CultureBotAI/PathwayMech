import json
import os
from pathlib import Path

A=Path(__file__).resolve().parent
O=Path('/private/tmp/pathwaymech-next-232-canonical-build-20261012')


def bind(p):
    p=Path(p)
    with p.open('rb') as f:digest=hashlib.file_digest(f,'sha256').hexdigest()
    return {'path':str(p),'bytes':p.stat().st_size,'sha256':digest}


def load(p):return json.loads(Path(p).read_bytes())


def write(name,obj):
    with (A/name).open('x') as f:json.dump(obj,f,indent=2,ensure_ascii=False);f.write('\n')

manifest=load(O/'artifact-manifest.json')

    assert {'file_type':'symlink' if p.is_symlink() else 'regular','link_text':os.readlink(p) if p.is_symlink() else None,'resolved_path':str(p.resolve(strict=True))}=={k:row[k] for k in ('file_type','link_text','resolved_path')}
