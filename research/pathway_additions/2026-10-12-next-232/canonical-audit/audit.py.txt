def load(p):
    return json.loads(Path(p).read_bytes())


def save(name,value):
    with (A/name).open('x') as f:
        json.dump(value,f,indent=2,ensure_ascii=False);f.write('\n')


def bind(p):
    p=Path(p)
    with p.open('rb') as f:digest=hashlib.file_digest(f,'sha256').hexdigest()
    return {'path':str(p),'bytes':p.stat().st_size,'sha256':digest}


def identity(p):
    p=Path(p);mode=p.lstat().st_mode
    assert stat.S_ISREG(mode) or stat.S_ISLNK(mode),str(p)
    target=p.resolve(strict=True);assert target.is_file()
    return {'file_type':'symlink' if stat.S_ISLNK(mode) else 'regular','link_text':os.readlink(p) if stat.S_ISLNK(mode) else None,'resolved_path':str(target)}


def check(r,ident=None):
    got=bind(r['path']);assert got=={k:r[k] for k in got},r['path']
    if ident is None and 'file_type' in r:ident={k:r[k] for k in ('file_type','link_text','resolved_path')}
    if ident is not None:assert identity(r['path'])==ident,r['path']
    return got


def inventory(directory):
    r=subprocess.run(['rg','--files','--no-ignore','--hidden','--follow',str(directory)],capture_output=True,text=True,check=True)
    assert not r.stderr
    for parent,dirs,files in os.walk(directory,followlinks=True):
        for name in dirs+files:
            p=Path(parent)/name;assert p.exists() and (p.is_file() or p.is_dir())
    return {Path(x) for x in r.stdout.splitlines()}


def ordered_equal(a,b):
    if type(a) is not type(b):return False
    if isinstance(a,dict):return list(a)==list(b) and all(ordered_equal(v,b[k]) for k,v in a.items())
    if isinstance(a,list):return len(a)==len(b) and all(ordered_equal(x,y) for x,y in zip(a,b,strict=True))
    return a==b

        for k,v in old[section].items():assert k in new[section] and ordered_equal(v,new[section][k]),(section,k)
