def binding(path):
    path = Path(path)
    return {'path': str(path), 'bytes': path.stat().st_size,
            'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}


def write(name, value):
    with (HERE / name).open('x') as f:
        json.dump(value, f, indent=2); f.write('\n')

    for row in actual:

def helper_scope():
    changed, unchanged, delta = [], [], []
    assert {p.name for p in PACK.glob('*.py')} == {p.name for p in OLD.glob('*.py')}
    for p in sorted(PACK.glob('*.py')):
        before = OLD / p.name
        destination = HERE / 'helpers-read' / (p.name + '.txt'); destination.parent.mkdir(exist_ok=True)
        destination.write_bytes(p.read_bytes())
        if before.read_bytes() == p.read_bytes(): unchanged.append(p.name)
        else:
            changed.append(p.name)
            delta += difflib.unified_diff(before.read_text().splitlines(keepends=True), p.read_text().splitlines(keepends=True),
                                         fromfile='original/' + p.name, tofile='successor/' + p.name)
    (HERE / 'independent-helper-delta.patch').write_text(''.join(delta))
    assert unchanged == ['run-gates.py']
    # Confirm corrected cache verification remains on every relevant execution path.
    calls = {}
    for name in ['bind_merged_base.py', 'capture-baseline.py', 'adopt.py', 'run-gates.py', 'preserve-maintained.py']:
        tree = ast.parse((PACK / name).read_text())
        calls[name] = sorted({node.func.id for node in ast.walk(tree) if isinstance(node, ast.Call) and isinstance(node.func, ast.Name)})
    for name in ['bind_merged_base.py', 'capture-baseline.py', 'adopt.py']: assert 'verify_cache_inputs' in calls[name]
    for name in ['adopt.py', 'run-gates.py', 'preserve-maintained.py']: assert 'verify_adopted' in calls[name]
    for name in ['adopt.py', 'preserve-maintained.py']: assert 'verify_path_identity' in calls[name]
    assert 'inventory(CACHE)' in (PACK / 'capture-baseline.py').read_text()
    assert "assert inventory(CACHE) == baseline['source_cache_files']" in (PACK / 'adopt.py').read_text()
    assert g.BASE_CONFIG is None and g.BASE is None and g.TREE is None
    return {'changed_helpers': changed, 'unchanged_helpers': unchanged, 'execution_path_checks': calls,
            'actual_merged_base_bound': False}

def dossier():
    plan = g.load(PACK / 'dossier-plan.json'); destinations = set(); counts = {}
    for row in plan['mappings']:
        destination = str(g.safe_relative(row['destination']))
        assert destination not in destinations and not destination.endswith('.py')
        assert destination not in {'archive-map.json', 'adoption.json'}
        destinations.add(destination); g.verify(row['source']['path'], row['source'])
        kind = row['serialization']; counts[kind] = counts.get(kind, 0) + 1
        if row.get('original'): g.verify(row['original']['path'], row['original'])
        if kind == 'lossless_utf8_json_wrapper':
            assert g.load(row['source']['path'])['text'].encode() == Path(row['original']['path']).read_bytes()
    assert len(destinations) == 476
    required = {'integration-remediation/original_preparation/handoff.json',
        'integration-remediation/original_preparation/artifact-manifest.json',
        'integration-remediation/independent_audit/handoff.json',
        'integration-remediation/independent_audit/artifact-manifest.json',
        'integration-remediation/independent_audit/cache_retarget_probe.json',
        'integration-remediation/issue-463/receipt.json', 'integration-remediation/initial-cache-identity.json'}
    assert required <= destinations
    old_receipts = list((OLD / 'validation').glob('*/receipt.json'))
    for p in old_receipts: assert 'integration-remediation/original_preparation/' + str(p.relative_to(OLD)) in destinations
    return {'mappings': 476, 'serializations': counts, 'required_original_audit_issue_and_fresh_identity_evidence': sorted(required),
            'predecessor_command_receipts_retained': len(old_receipts)}


def reject(call):
    try: call()
    except (AssertionError, FileNotFoundError) as error: return {'type': type(error).__name__, 'message': str(error)}
    raise AssertionError('Invalid mutation unexpectedly accepted')


def independent_controls():
    results = []
    for mutation in ['alternate-destination', 'alternate-link-text', 'indirect-destination',
                     'link-to-regular', 'regular-to-link', 'parent-link-destination']:
        root = HERE / 'fixtures' / mutation; root.mkdir(parents=True)
        first, second = root / 'first', root / 'second'
        first.write_bytes(b'independent preserved body\n'); second.write_bytes(first.read_bytes())
        cache = root / 'cache'; cache.mkdir()
        path = cache / 'old'
        if mutation == 'regular-to-link': path.write_bytes(first.read_bytes())
        elif mutation == 'indirect-destination':
            (root / 'middle').symlink_to(first); path.symlink_to(root / 'middle')
        elif mutation == 'parent-link-destination':
            for name in ['directory-a', 'directory-b']:
                (root / name).mkdir(); (root / name / 'old').write_bytes(first.read_bytes())
            (cache / 'group').symlink_to(root / 'directory-a', target_is_directory=True)
            path = cache / 'group/old'
        else: path.symlink_to(first)
        before = g.inventory(cache); previous = {r['relative_path']: r for r in before}
        g.verify_cache_union(cache, previous, {})
        if mutation == 'indirect-destination':
            (root / 'middle').unlink(); (root / 'middle').symlink_to(second)
        elif mutation == 'parent-link-destination':
            (cache / 'group').unlink(); (cache / 'group').symlink_to(root / 'directory-b', target_is_directory=True)
        else:
            path.unlink()
            if mutation == 'link-to-regular': path.write_bytes(first.read_bytes())
            elif mutation == 'alternate-link-text': path.symlink_to('../first')
            else: path.symlink_to(second)
        assert path.read_bytes() == first.read_bytes() == second.read_bytes()
        assert g.inventory(cache) != before
        results.append({'mutation': mutation, 'direct_identity': reject(lambda: g.verify_path_identity(path, previous[str(path.relative_to(cache))])),
                        'cache_guard': reject(lambda: g.verify_cache_union(cache, previous, {})),
                        'baseline_inventory_rejected': True})
    root = HERE / 'fixtures/new-copy'; root.mkdir(); cache = root / 'cache'; cache.mkdir()
    source = root / 'original'; source.write_bytes(b'independent new body\n')
    plan = {'new': {'original': g.binding(source)}}; path = cache / 'new'
    path.write_bytes(source.read_bytes()); g.verify_cache_union(cache, {}, plan, adopted=True)
    path.unlink(); path.symlink_to(source)
    results.append({'mutation': 'new-copy-link', 'cache_guard': reject(lambda: g.verify_cache_union(cache, {}, plan, adopted=True))})
    results.append({'mutation': 'missing-fresh-fields', 'direct_identity': reject(lambda: g.verify_path_identity(path, g.binding(path)))})
    write('independent-controls.json', results)
    return {'negative_cases': len(results), 'same_bytes_preserved_in_each_retarget_or_type_case': True,
            'unchanged_old_identity_positive_controls': 6, 'regular_new_copy_positive_control': True,
            'additional_parent_directory_retarget_control': True}

def main():
    assert not os.environ.get('PATHWAYMECH_PPI_MERGED_BASE')
    checks = []
    for name, fn in [('closed_lineage', seals), ('fresh_cache_and_accepted_inputs', fresh_cache),
                     ('helper_delta_and_execution_paths', helper_scope), ('actual_50_test_receipts', receipts),
                     ('dossier476', dossier), ('independent_identity_controls', independent_controls),
                     ('final_input_preservation', final_preservation)]:
        at = datetime.datetime.now(datetime.UTC).isoformat()
        try: row = {'name': name, 'status': 'passed', 'details': fn()}
        except Exception as error: row = {'name': name, 'status': 'failed', 'type': type(error).__name__, 'message': str(error)}
        row['started_at'] = at; row['finished_at'] = datetime.datetime.now(datetime.UTC).isoformat()
        checks.append(row); write(name + '.json', row); print(json.dumps(row), flush=True)
    write('checks.json', checks)
    return 0 if all(r['status'] == 'passed' for r in checks) else 1
